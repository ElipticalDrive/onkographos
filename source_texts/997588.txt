概要
HDRからSDRへの変換（トーンマッピング）を実装する。
詳細タスク
Reinhard
ACES
Exposure制御
シェーダー実装 (HLSL)
```
cbuffer TonemapParams : register(b0) {
    float exposure;
    uint tonemapOperator;  // 0: Reinhard, 1: ACES, 2: Filmic
    float gamma;
};

Texture2D<float4> hdrInput : register(t0);

float3 ReinhardTonemap(float3 color) {
    return color / (color + 1.0);
}

float3 ACESFilm(float3 x) {
    float a = 2.51;
    float b = 0.03;
    float c = 2.43;
    float d = 0.59;
    float e = 0.14;
    return saturate((x * (a * x + b)) / (x * (c * x + d) + e));
}

float3 Uncharted2Tonemap(float3 x) {
    float A = 0.15;
    float B = 0.50;
    float C = 0.10;
    float D = 0.20;
    float E = 0.02;
    float F = 0.30;
    return ((x * (A * x + C * B) + D * E) / (x * (A * x + B) + D * F)) - E / F;
}

float4 main(float2 texCoord : TEXCOORD0) : SV_TARGET {
    float3 hdrColor = hdrInput.Sample(linearSampler, texCoord).rgb;
    
    // Exposure
    hdrColor *= exposure;
    
    // Tone mapping
    float3 mapped;
    switch (tonemapOperator) {
        case 0: mapped = ReinhardTonemap(hdrColor); break;
        case 1: mapped = ACESFilm(hdrColor); break;
        case 2: 
            float3 curr = Uncharted2Tonemap(hdrColor * 2.0);
            float3 whiteScale = 1.0 / Uncharted2Tonemap(float3(11.2, 11.2, 11.2));
            mapped = curr * whiteScale;
            break;
    }
    
    // Gamma correction
    mapped = pow(mapped, 1.0 / gamma);
    
    return float4(mapped, 1.0);
}
```
