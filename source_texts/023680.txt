Is your feature request related to a problem? Please describe.
As per the voxelization tutorial, I can visualize the voxels which are intersecting with elements. How can I visualize voxels which do not have any intersecting elements (i.e. voxels whose values are set to 0)
Below is the snippet taken from the documentation:
Open3D provides the method create_from_triangle_mesh that creates a voxel grid from a triangle mesh. It returns a voxel grid where all voxels that are intersected by a triangle are set to 1, all others are set to 0. The argument voxel_size defines the resolution of the voxel grid.

import open3d as o3d
import numpy as np

def visualize_non_filled_voxels(mesh, voxel_size):
    # Create voxel grid from the mesh
    voxel_grid = o3d.geometry.VoxelGrid.create_from_triangle_mesh(mesh, voxel_size=voxel_size)

    # Get dimensions of the voxel grid
    min_bound = voxel_grid.get_min_bound()
    max_bound = voxel_grid.get_max_bound()
    dims = np.ceil((max_bound - min_bound) / voxel_size).astype(int)
    
    # Create a new voxel grid with all voxels filled
    full_grid = o3d.geometry.VoxelGrid()
    for x in range(dims[0]):
        for y in range(dims[1]):
            for z in range(dims[2]):
                grid_index = np.array([x, y, z], dtype=np.int32)
                full_grid.add_voxel(o3d.utility.Voxel(grid_index))

    # Subtract the original grid from the full grid to get non-filled voxels
    non_filled_voxels = o3d.geometry.VoxelGrid()
    for voxel in full_grid.get_voxels():
        if not voxel_grid.check_if_included(voxel.grid_index):
            non_filled_voxels.add_voxel(voxel)

    # Visualize the non-filled voxels
    print("Visualizing non-filled voxels")
    o3d.visualization.draw_geometries([non_filled_voxels], window_name="Non-Filled Voxels")

# Example usage
if __name__ == "__main__":
    mesh = o3dtut.get_bunny_mesh()
    mesh.scale(1 / np.max(mesh.get_max_bound() - mesh.get_min_bound()), center=mesh.get_center())
    
    voxel_size = 0.05
    visualize_non_filled_voxels(mesh, voxel_size)
