```wgsl
/// Calcula la normal de una superficie SDF usando el gradiente
fn calc_normal(p: vec3) -> vec3 {
// Usar epsilon pequeño para precisión
let eps = 0.0001;
```
// Gradiente central (más preciso pero más caro)
let n = vec3<f32>(
    eval_scene(p + vec3<f32>(eps, 0.0, 0.0)) - eval_scene(p - vec3<f32>(eps, 0.0, 0.0)),
    eval_scene(p + vec3<f32>(0.0, eps, 0.0)) - eval_scene(p - vec3<f32>(0.0, eps, 0.0)),
    eval_scene(p + vec3<f32>(0.0, 0.0, eps)) - eval_scene(p - vec3<f32>(0.0, 0.0, eps))
);

return normalize(n);
```
}
/// Versión optimizada con tetrahedro (4 samples en vez de 6)
fn calc_normal_tetrahedron(p: vec3) -> vec3 {
let eps = 0.0001;
```
let k = vec2<f32>(1.0, -1.0);
return normalize(
    k.xyy * eval_scene(p + k.xyy * eps) +
    k.yyx * eval_scene(p + k.yyx * eps) +
    k.yxy * eval_scene(p + k.yxy * eps) +
    k.xxx * eval_scene(p + k.xxx * eps)
);
```
}
```

```
/// Luz direccional (sol)
struct DirectionalLight {
direction: vec3,
color: vec3,
intensity: f32,
}
/// Luz puntual
struct PointLight {
position: vec3,
color: vec3,
intensity: f32,
range: f32,
}
/// Componente difusa (Lambert)
fn diffuse_lambert(normal: vec3, light_dir: vec3) -> f32 {
return max(dot(normal, light_dir), 0.0);
}
/// Componente especular (Blinn-Phong)
fn specular_blinn_phong(
normal: vec3,
light_dir: vec3,
view_dir: vec3,
roughness: f32
) -> f32 {
let half_dir = normalize(light_dir + view_dir);
let shininess = (1.0 - roughness) * 128.0 + 1.0;
return pow(max(dot(normal, half_dir), 0.0), shininess);
}
/// Atenuación de luz puntual
fn attenuation(distance: f32, range: f32) -> f32 {
// Atenuación física (inverse square) con falloff suave
let d = distance / range;
return saturate(1.0 - d * d) / (1.0 + distance * distance);
}
/// Calcula iluminación completa para un punto
fn calculate_lighting(
position: vec3,
normal: vec3,
view_dir: vec3,
material: Material
) -> vec3 {
var color = vec3(0.0);
```
// Luz ambiental
let ambient = vec3<f32>(0.1, 0.12, 0.15) * material.albedo;
color += ambient;

// Luz direccional (sol)
let sun = DirectionalLight(
    normalize(vec3<f32>(1.0, 1.0, 0.5)),
    vec3<f32>(1.0, 0.95, 0.9),
    1.0
);

let sun_diffuse = diffuse_lambert(normal, sun.direction);
let sun_specular = specular_blinn_phong(normal, sun.direction, view_dir, material.roughness);

color += material.albedo * sun.color * sun.intensity * sun_diffuse;
color += sun.color * sun.intensity * sun_specular * (1.0 - material.roughness);

// Emisión
color += material.emission;

return color;
```
}

/// Fresnel Schlick
fn fresnel_schlick(cos_theta: f32, f0: vec3) -> vec3 {
return f0 + (1.0 - f0) * pow(1.0 - cos_theta, 5.0);
}
/// F0 basado en metallic/albedo
fn calc_f0(albedo: vec3, metallic: f32) -> vec3 {
return mix(vec3(0.04), albedo, metallic);
}
```

```
fn sky_color(direction: vec3) -> vec4 {
let t = direction.y * 0.5 + 0.5;
let horizon = vec3(0.7, 0.8, 0.95);
let zenith = vec3(0.3, 0.5, 0.9);
let ground = vec3(0.3, 0.3, 0.35);
```
var color: vec3<f32>;
if direction.y > 0.0 {

} else {
    color = mix(horizon, ground, pow(-direction.y, 0.3));
