Component
HIP runtime / clr -- surface/texture over a layered hipMalloc3DArray.
Summary
On gfx90a (CDNA2, ROCm 7.2.1), a layered cudaArray (hipMalloc3DArray with hipArrayLayered | hipArraySurfaceLoadStore) written one layer at a time via surf2DLayeredwrite in separate kernel launches returns the last-written layer's data for every layer index when read back in a later launch -- via surf2DLayeredread, tex2DLayered, or host hipMemcpy3D. Per-layer contents are lost; the array behaves as if it holds one layer. A non-layered 3D array of the same dimensions (no hipArrayLayered; surf3D/tex3D with the layer as the z coordinate) is correct. CUDA preserves the per-layer contents.
Environment
AMD Instinct MI250X, gfx90a (CDNA2)
ROCm 7.2.1
Reproducer (the bug)
Build: hipcc -O2 multilayer_check.cpp -o mlc Run: ./mlc
```
#include <cstring>
#include <hip/hip_runtime.h>
#include <cstdio>
#define CHECK(cmd) do{ hipError_t e=(cmd); if(e!=hipSuccess){ \
  printf("HIP error %s:%d: %s\n",__FILE__,__LINE__,hipGetErrorString(e)); return -1;}}while(0)
static const int W=32,H=32,L=6;
__global__ void writeOneLayer(hipSurfaceObject_t surf,int layer){
  int x=blockIdx.x*blockDim.x+threadIdx.x, y=blockIdx.y*blockDim.y+threadIdx.y;
  if(x>=W||y>=H) return;
  surf2DLayeredwrite((float)(layer*100+7), surf, x*4, y, layer);
}
__global__ void readAll(hipSurfaceObject_t surf,float* out,int cx,int cy){
  for(int l=0;l<L;l++){ float v; surf2DLayeredread(&v,surf,cx*4,cy,l); out[l]=v; }
}
int main(){
  hipChannelFormatDesc d=hipCreateChannelDesc(32,0,0,0,hipChannelFormatKindFloat);
  hipArray_t arr; hipExtent ext=make_hipExtent(W,H,L);
  CHECK(hipMalloc3DArray(&arr,&d,ext,hipArrayLayered|hipArraySurfaceLoadStore));
  hipResourceDesc rd; memset(&rd,0,sizeof(rd)); rd.resType=hipResourceTypeArray; rd.res.array.array=arr;
  hipSurfaceObject_t surf; CHECK(hipCreateSurfaceObject(&surf,&rd));
  dim3 b(8,8), g((W+7)/8,(H+7)/8);
  for(int l=0;l<L;l++) hipLaunchKernelGGL(writeOneLayer,g,b,0,0,surf,l);  // one layer per launch
  float* dout; CHECK(hipMalloc(&dout,L*sizeof(float)));
  hipLaunchKernelGGL(readAll,dim3(1),dim3(1),0,0,surf,dout,16,16);        // separate launch
  CHECK(hipDeviceSynchronize());
  float h[L]; CHECK(hipMemcpy(h,dout,L*sizeof(float),hipMemcpyDeviceToHost));
  for(int l=0;l<L;l++){ int exp=l*100+7; printf("  layer %d: got %.1f exp %d %s\n",l,h[l],exp,(int)h[l]==exp?"OK":"STALE"); }
  return 0;
}
```
Observed (each layer L is written L*100+7, then all layers read at one (x,y) in a separate launch)
```
  layer 0: got 507.0 exp 7   STALE
  layer 1: got 507.0 exp 107 STALE
  layer 2: got 507.0 exp 207 STALE
  layer 3: got 507.0 exp 307 STALE
  layer 4: got 507.0 exp 407 STALE
  layer 5: got 507.0 exp 507 OK     <- 507 is the value written to the LAST layer
```
Every layer returns 507.0 = the last-written layer's value. tex2DLayered and a host hipMemcpy3D of the array read back the same collapsed result.
Expected
Each layer returns its own value (7, 107, 207, 307, 407, 507), as on CUDA.
Control: the same data in a NON-layered 3D array is correct on ROCm
Dropping hipArrayLayered and using surf3Dwrite/surf3Dread/tex3D with the layer as the z coordinate returns the right per-slice value through both surface and texture -- so the data path itself is fine; only the layered addressing collapses. Build: hipcc -O2 array3d_check.cpp -o a3c Run: ./a3c
```
#include <cstring>
#include <hip/hip_runtime.h>
#include <cstdio>
#define CHECK(cmd) do{ hipError_t e=(cmd); if(e!=hipSuccess){ \
  printf("HIP error %s:%d: %s\n",__FILE__,__LINE__,hipGetErrorString(e)); return -1;}}while(0)
static const int W=32,H=32,D=6;
__global__ void writeSlice(hipSurfaceObject_t surf, int z){
  int x=blockIdx.x*blockDim.x+threadIdx.x, y=blockIdx.y*blockDim.y+threadIdx.y;
  if(x>=W||y>=H) return;
  surf3Dwrite((float)(z*100+7), surf, x*4, y, z);
}
__global__ void readSlicesSurf(hipSurfaceObject_t surf, float* out, int cx, int cy){
  for(int z=0;z<D;z++){ float v; surf3Dread(&v, surf, cx*4, cy, z); out[z]=v; }
}
__global__ void readSlicesTex(hipTextureObject_t tex, float* out, int cx, int cy){
  for(int z=0;z<D;z++){ out[z]=tex3D<float>(tex, cx+0.5f, cy+0.5f, z+0.5f); }
}
int main(){
  hipChannelFormatDesc d=hipCreateChannelDesc(32,0,0,0,hipChannelFormatKindFloat);
  hipExtent ext=make_hipExtent(W,H,D);
  hipArray_t arr; CHECK(hipMalloc3DArray(&arr,&d,ext,hipArraySurfaceLoadStore)); // NO Layered
  hipResourceDesc rd; memset(&rd,0,sizeof(rd)); rd.resType=hipResourceTypeArray; rd.res.array.array=arr;
  hipSurfaceObject_t s; CHECK(hipCreateSurfaceObject(&s,&rd));
  hipTextureDesc td; memset(&td,0,sizeof(td)); td.normalizedCoords=0;
  td.addressMode[0]=hipAddressModeClamp; td.addressMode[1]=hipAddressModeClamp; td.addressMode[2]=hipAddressModeClamp;
  td.readMode=hipReadModeElementType; td.filterMode=hipFilterModePoint;
  hipTextureObject_t tex; CHECK(hipCreateTextureObject(&tex,&rd,&td,nullptr));
  dim3 b(8,8), g((W+7)/8,(H+7)/8);
  for(int z=0;z<D;z++) hipLaunchKernelGGL(writeSlice,g,b,0,0,s,z);
  CHECK(hipDeviceSynchronize());
  float* dout; CHECK(hipMalloc(&dout,D*sizeof(float)));
  hipLaunchKernelGGL(readSlicesSurf,dim3(1),dim3(1),0,0,s,dout,16,16);
  CHECK(hipDeviceSynchronize());
  float h[D]; CHECK(hipMemcpy(h,dout,D*sizeof(float),hipMemcpyDeviceToHost));
  printf("3D-array SURFACE read per slice:\n");
  for(int z=0;z<D;z++){ int e=z*100+7; printf("  z%d got %.1f exp %d %s\n",z,h[z],e,(int)h[z]==e?"OK":"STALE"); }
  hipLaunchKernelGGL(readSlicesTex,dim3(1),dim3(1),0,0,tex,dout,16,16);
  CHECK(hipDeviceSynchronize());
  CHECK(hipMemcpy(h,dout,D*sizeof(float),hipMemcpyDeviceToHost));
  printf("3D-array TEXTURE read per slice:\n");
  for(int z=0;z<D;z++){ int e=z*100+7; printf("  z%d got %.1f exp %d %s\n",z,h[z],e,(int)h[z]==e?"OK":"STALE"); }
  return 0;
}
```
Output: every slice is correct via both paths --
```
3D-array SURFACE read per slice:  z0..z5 -> 7,107,207,307,407,507  (all OK)
3D-array TEXTURE read per slice:  z0..z5 -> 7,107,207,307,407,507  (all OK)
```
Notes
All layers are written before any read, with hipDeviceSynchronize between the writes and the read; recreating the surface/texture object after the writes does not help. The single discriminator is the hipArrayLayered flag.
How this was found
While porting popsift (a CUDA GPU-SIFT library) to ROCm/HIP as part of MOAT (https://github.com/jeffdaily/moat). Its Gaussian-pyramid and DoG octaves are layered arrays written per-level via surface and read per-level via texture; the layer collapse made the DoG all-zero, yielding 0 detected features. Workaround in the port: make the octave arrays non-layered 3D (surf3D/tex3D, level as z).

It corrects surf1D/2DLayered{read,write} in amd_surface_functions.h, which were wrongly bound to the _lod_ ockl builtins (the layer index was passed into the mip-LOD slot instead of the array coordinate) — that matches the surf2DLayered collapse you see.

One caveat: tex2DLayered already uses the correct _2Da builtin and isn't touched by my PR, and hipMemcpy3D is runtime-side — so if those still collapse after the patch, there's likely an additional layered-array issue in clr beyond the surface header. Curious what your test shows.

On the stock header the per-layer writes collapse onto a single slot (the last write, layer 5 = 507), so the layered reads return stale data rather than each layer's own value; with the patch all six layers read back their own values through every path.
Worth noting for the issue: although the PR only changes surf1D/2DLayered{read,write} and not tex2DLayered/hipMemcpy3D, the collapse was entirely a write-side problem -- surf2DLayeredwrite was routing layer into the mipmap-LOD argument of __ockl_image_store_lod_2D, so every write targeted the same slot. Once the writes land in the correct array slots (__ockl_image_store_2Da), all read paths -- including the untouched tex2DLayered and the runtime-side hipMemcpy3D -- return correct data. So there does not appear to be a separate tex2DLayered defect; fixing the write resolves the symptom on all paths. From our side this fully resolves the layered-array collapse in #275.
Reproducer used (compile hipcc -O2 --offload-arch=gfx90a clr275_repro.cpp), in case it's useful as a regression test -- prints COLLAPSED/ALL-FRESH per read path:
```
// Reproducer for ROCm/clr#275: layered cudaArray collapse on gfx90a.
// Writes 6 layers via surf2DLayeredwrite, then reads them back three ways
// (surf2DLayeredread, tex2DLayered, host hipMemcpy3D), plus a non-layered 3D
// array as a control. On a stock ROCm 7.2.x the three layered reads COLLAPSE
// (every layer returns the last write); the non-layered 3D path is correct.
// With ROCm/rocm-systems#6683 all three layered reads return their own values.
//
// Build/run:  hipcc -O2 --offload-arch=gfx90a clr275_repro.cpp -o repro && ./repro

#include <hip/hip_runtime.h>
#include <cstdio>
#include <cstring>
#include <cstdlib>

#define CHECK(cmd) do { hipError_t e=(cmd); if(e!=hipSuccess){ \
  printf("HIP error %s:%d: %s\n",__FILE__,__LINE__,hipGetErrorString(e)); return 2;}}while(0)

static const int W = 32, H = 32, D = 6;

__global__ void writeLayered(hipSurfaceObject_t surf, int layer) {
    int x = blockIdx.x*blockDim.x + threadIdx.x;
    int y = blockIdx.y*blockDim.y + threadIdx.y;
    if (x >= W || y >= H) return;
    surf2DLayeredwrite((float)(layer*100 + 7), surf, x*4, y, layer);
}
__global__ void readLayeredSurf(hipSurfaceObject_t surf, float* out, int cx, int cy) {
    for (int z = 0; z < D; z++) { float v; surf2DLayeredread(&v, surf, cx*4, cy, z); out[z] = v; }
}
__global__ void readLayeredTex(hipTextureObject_t tex, float* out, int cx, int cy) {
    for (int z = 0; z < D; z++) out[z] = tex2DLayered<float>(tex, cx+0.5f, cy+0.5f, z);
}
__global__ void writeSlice3D(hipSurfaceObject_t surf, int z) {
    int x = blockIdx.x*blockDim.x + threadIdx.x;
    int y = blockIdx.y*blockDim.y + threadIdx.y;
    if (x >= W || y >= H) return;
    surf3Dwrite((float)(z*100 + 7), surf, x*4, y, z);
}
__global__ void readSlice3DSurf(hipSurfaceObject_t surf, float* out, int cx, int cy) {
    for (int z = 0; z < D; z++) { float v; surf3Dread(&v, surf, cx*4, cy, z); out[z]=v; }
}

static int checkFresh(const char* label, const float* h) {
    int ok = 1;
    printf("%s:\n", label);
    for (int z = 0; z < D; z++) {
        int exp = z*100 + 7, got = (int)h[z];
        printf("  layer %d got %.1f exp %d %s\n", z, h[z], exp, got==exp?"OK":"STALE");
        if (got != exp) ok = 0;
    }
    printf("  => %s\n\n", ok ? "ALL-FRESH" : "COLLAPSED");
    return ok;
}

int main()
{
    hipDeviceProp_t prop; CHECK(hipGetDeviceProperties(&prop, 0));
    printf("device: %s  warpSize: %d\n\n", prop.name, prop.warpSize);

    hipChannelFormatDesc ch = hipCreateChannelDesc(32,0,0,0,hipChannelFormatKindFloat);
    hipExtent ext = make_hipExtent(W,H,D);
    dim3 b(8,8), g((W+7)/8,(H+7)/8);
    float* dout; CHECK(hipMalloc(&dout, D*sizeof(float)));
    float h[D];

    // ---- Layered array ----
    hipArray_t larr;
    CHECK(hipMalloc3DArray(&larr, &ch, ext, hipArrayLayered | hipArraySurfaceLoadStore));
    hipResourceDesc lrd; memset(&lrd,0,sizeof(lrd));
    lrd.resType = hipResourceTypeArray; lrd.res.array.array = larr;
    hipSurfaceObject_t ls; CHECK(hipCreateSurfaceObject(&ls, &lrd));
    hipTextureDesc td; memset(&td,0,sizeof(td)); td.normalizedCoords=0;
    td.addressMode[0]=td.addressMode[1]=td.addressMode[2]=hipAddressModeClamp;
    td.readMode=hipReadModeElementType; td.filterMode=hipFilterModePoint;
    hipTextureObject_t lt; CHECK(hipCreateTextureObject(&lt, &lrd, &td, nullptr));

    for (int z = 0; z < D; z++) hipLaunchKernelGGL(writeLayered, g, b, 0, 0, ls, z);
    CHECK(hipDeviceSynchronize());

    hipLaunchKernelGGL(readLayeredSurf, dim3(1), dim3(1), 0, 0, ls, dout, 16, 16);
    CHECK(hipDeviceSynchronize());
    CHECK(hipMemcpy(h, dout, D*sizeof(float), hipMemcpyDeviceToHost));
    checkFresh("(a) layered array, surf2DLayeredread", h);

    hipLaunchKernelGGL(readLayeredTex, dim3(1), dim3(1), 0, 0, lt, dout, 16, 16);
    CHECK(hipDeviceSynchronize());
    CHECK(hipMemcpy(h, dout, D*sizeof(float), hipMemcpyDeviceToHost));
    checkFresh("(b) layered array, tex2DLayered", h);

    {   // (c) host hipMemcpy3D from the layered array
        hipMemcpy3DParms p; memset(&p,0,sizeof(p));
        p.srcArray = larr;
        float* hbuf = (float*)calloc((size_t)W*H*D, sizeof(float));
        p.dstPtr = make_hipPitchedPtr(hbuf, W*sizeof(float), W, H);
        p.extent = make_hipExtent(W, H, D);
        p.kind   = hipMemcpyDeviceToHost;
        CHECK(hipMemcpy3D(&p));
        for (int z = 0; z < D; z++) h[z] = hbuf[(size_t)z*W*H + 16*W + 16];
        free(hbuf);
        checkFresh("(c) layered array, hipMemcpy3D (host readback @ [16,16])", h);
    }

    // ---- Non-layered 3D array (control) ----
    hipArray_t arr3;
    CHECK(hipMalloc3DArray(&arr3, &ch, ext, hipArraySurfaceLoadStore));
    hipResourceDesc rd3; memset(&rd3,0,sizeof(rd3));
    rd3.resType = hipResourceTypeArray; rd3.res.array.array = arr3;
    hipSurfaceObject_t s3; CHECK(hipCreateSurfaceObject(&s3, &rd3));
    for (int z = 0; z < D; z++) hipLaunchKernelGGL(writeSlice3D, g, b, 0, 0, s3, z);
    CHECK(hipDeviceSynchronize());
    hipLaunchKernelGGL(readSlice3DSurf, dim3(1), dim3(1), 0, 0, s3, dout, 16, 16);
    CHECK(hipDeviceSynchronize());
    CHECK(hipMemcpy(h, dout, D*sizeof(float), hipMemcpyDeviceToHost));
    checkFresh("(ctrl) non-layered 3D array, surf3Dread", h);

    return 0;
}
```

Confirming this on a different architecture family, and with what looks like the cause in the headers.
Same defect on gfx1201 (RDNA4, Radeon RX 9070), ROCm 7.2.1, Windows. So it is not specific to CDNA2.
In include/hip/amd_detail/amd_surface_functions.h, ROCm 7.2.1, lines 271-298, both layered accessors pass the array layer to the LOD parameter of a plain 2D image:
```
// surf2DLayeredread, ~line 276
auto tmp = __ockl_image_load_lod_2D(i, get_native_vector(coords), layer);

// surf2DLayeredwrite, ~line 298
__ockl_image_store_lod_2D(i, get_native_vector(coords), layer, tmp);
```
The 2D-array entry points are __ockl_image_load_2Da / __ockl_image_store_2Da, which take the layer as coords.z. As written, any access to a layer above 0 addresses a mip level that does not exist and is dropped silently. Both functions also compute the pixel address with __ockl_image_channel_data_type_2D / _order_2D rather than the _2Da variants.
Not a recent regression. The HIP SDK 6.2 toolchain has the identical misuse at amd_surface_functions.h:197:
```
__ockl_image_store_lod_2D(i, int2(x, y).data, layer, tmp);
```
Isolating read from write. A 32x4x4 layered float array, filled with hipMemcpy3D so the read accessors are tested on their own, then rewritten with the surface accessor and read back through the texture path that part A has just validated:
```
A. array filled with hipMemcpy3D, read back two ways
   tex2DLayered           0 of 512 wrong  (per layer: 0 0 0 0)
   surf2DLayeredread    384 of 512 wrong  (per layer: 0 128 128 128)

B. array zeroed, written with surf2DLayeredwrite, read with tex2DLayered
   tex2DLayered         416 of 512 wrong  (per layer: 32 128 128 128)
```
So on this device tex2DLayered reads correctly when the array is populated by other means, and both surface directions are correct on layer 0 and wrong on every layer above it — which is what routing the layer into the LOD parameter would produce. Part B additionally loses 32 of 128 cells on layer 0, consistent with the _2D vs _2Da channel accessors above, though the reproducer does not prove that part.
Calling __ockl_image_store_2Da directly, with the layer in coords.z, makes the write correct on all layers here, which is the workaround we are using rather than converting to non-layered 3D arrays.
A separate problem the same reproducer shows. hipGetDeviceProperties reports maxTexture2DLayered and maxSurface2DLayered as 2048 x 2048 x 2048 on this device while maxTexture2D is 16384 x 16384, and the layered limit is not real:
```
reported maxSurface2DLayered 2048 x 2048 x 2048, maxTexture2D 16384 x 16384
C. 8192 x 8192 x 3 layered array, far past the reported limit: no error
```
This matters for CUDA ports, which check those fields before choosing a path. In our case a library read the reported limit, concluded a 4032 x 3024 image could not be handled, and returned a null job that the caller then dereferenced. Happy to open that as its own issue if you would prefer it kept separate from the layer addressing.
Full reproducer (single file, no dependencies): https://github.com/dspl1236/Cheshire/blob/main/hip/port/popsift/bugreport_layered_surface.hip
