```
def test_fdk_scaling_invariance():
    vg = ts.volume(shape=32, size=1)
    pg = ts.cone(
        angles=32, shape=(48, 48), size=(2, 2), src_det_dist=2, src_orig_dist=2
    ).to_vec()

    # Create sinogram and reconstruct at the origin
    A0 = ts.operator(vg, pg)
    x = make_box_phantom()

    rec0 = fdk(A0, A0(x))
    bp0 = A0.T(A0(x))

    s = 2.0
    S = ts.scale(s)
    A = ts.operator(S * vg, S * pg)

    rec = fdk(A, A(x))
    bp = A.T(A(x))

    # Check that backprojection still matches (up to scaling)
    assert torch.allclose(bp / s**2, bp0)
    # Check that fdk still matches (fails!)
    assert torch.allclose(rec, rec0, atol=0.5, rtol=0.01)
```
