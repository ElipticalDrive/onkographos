Before you start, we recommend doing the previous SDF challenges first, so that you already know how to draw a circle. The donut is created by subtracting a smaller circle from a larger circle.

precision mediump float;
in vec2 v_uv;
uniform vec2 u_resolution;
out vec4 fragColor;
// SDF for a flat circle
float sdCircle(vec2 p, float r) {
return length(p) - r;
}
void main() {
// Normalize pixel coordinates to range [-1, 1]
vec2 uv = v_uv.xy * 2.0 - 1.0;
vec3 blueColor = vec3(0.0, 0.0, 1.0);
vec3 redColor = vec3(1.0, 0.0, 0.0);
vec3 finalColor = redColor;
// Calculate the SDF distance for a circle
