Hi,
Thanks for the question -- please see the comment below about the descriptions of each portion of the dataset -- feel to ask more questions if its not clear:
For the pouring dataset r_x.png and r_x_flow.png is a RGB image of the scene at a particular viewpoint at timestep t and at a subsequent timestep (t + delta) These images are generated using blender -- I can share the code if its helpful.
For the Gibson dataset r_x.png and r_x_flow.png are the same as the pouring setting. d_x.npy and d_x_flow.npy are the depths of the images in the same setting. flow_x.npy is the optical flow between the r_x.png and r_x_flow.png as computed by raft -- I can share the code also if its helpful.

def generate_dataset(ts2data, examples, depth_node):
    scene = bpy.context.scene
    out_dir = scene.render.filepath
    # assert not os.path.exists(out_dir)


    # Extract the scene objects.
    cam = scene.objects["Camera"]

    for i in range(20):
        bpy.data.worlds['World'].cycles.sample_as_light = True
        bpy.context.scene.cycles.blur_glossy = 2.0
        # bpy.context.scene.cycles.samples = 512
        # bpy.context.scene.cycles.transparent_min_bounces = 6
        # bpy.context.scene.cycles.transparent_max_bounces = 6
        # bpy.context.scene.cycles.samples = 16

        # Low resolution version
        bpy.context.scene.cycles.samples = 2
        bpy.context.scene.cycles.transparent_min_bounces = 1
        bpy.context.scene.cycles.transparent_max_bounces = 2
        bpy.context.scene.cycles.film_transparent = True

        bpy.ops.object.lamp_add(type='SUN')
        lamp2 = bpy.data.lamps['Sun']
        lamp2.use_specular = True
        lamp2.energy = 100.0
        bpy.ops.object.lamp_add(type='POINT', radius=30.0, location=(0, 0.0, 10.0))

    # Set up the camera to track an imaginary object at the origin.
    dummy = parent_obj_to_camera(cam)
    setup_cam(cam, dummy)

    idx, metadata = load_metadata(out_dir)

    if metadata is None:
        metadata = {
            "camera_angle_x": cam.data.angle_x,
            "frames": []
        }

    flow_metadata = {
        "camera_angle_x": cam.data.angle_x,
        "frames": []
    }


    for counter, (frame_id, ts, flow_ts, rot) in enumerate(examples):
        counter = counter + idx + 1
        # Set the frame to render.
        scene.frame_set(ts)

        # Set the rotation of the center object, which updates the camera pose.
        dummy.rotation_euler = rot

        # Render and save to path.
        path = os.path.join(out_dir, "r_{}".format(counter))
        scene.render.filepath = path
        depth_node.file_slots[0].path = "r_{}".format(counter)
        depth_path = os.path.join("depth", "r_{}".format(counter) + "{:04}".format(ts) + ".hdr")

        bpy.ops.render.render(write_still=True)

        # Store frame metadata.
        meta = construct_metadata(path, depth_path, cam, ts, ts2data, rot)
        metadata["frames"].append(meta)
        save_metadata(metadata, out_dir)

        # Store flow metadata
        if flow_ts is not None:
            scene.frame_set(flow_ts)

            path = os.path.join(out_dir, "r_{}_flow".format(counter))
            scene.render.filepath = path
            depth_node.file_slots[0].path = "r_{}_flow".format(counter)
            depth_path = os.path.join("depth", "r_{}_flow".format(counter) + "{:04}".format(ts) + ".hdr")
            flow_meta = construct_metadata(path, depth_path, cam, flow_ts, ts2data, rot)
            metadata["frames"].append(flow_meta)

            bpy.ops.render.render(write_still=True)
            save_metadata(metadata, out_dir)

        if counter > N_RANDOM_EXAMPLES:
            break


def construct_metadata(path, depth_path, cam, ts, ts2data, rot):
    frame_data = {
        'file_path': path + ".png",
        'depth_path': depth_path,
        'transform_matrix': listify_matrix(cam.matrix_world),
        'timestep': ts,
        'rot': list(rot),
        'offset': ts2data[ts]["offset"],
        'dimensions': ts2data[ts]["dimensions"]
    }
    return frame_data

def load_metadata(out_dir):
    path = os.path.join(out_dir, "transforms.json")

    if os.path.exists(path):
        metadata = load_json(path)
        idx = len(metadata['frames']) // 2
    else:
        idx = 0
        metadata = None

    return idx, metadata

def save_metadata(metadata, out_dir, flow=False):
    if flow:
        fname = "transforms_flow.json"
    else:
        fname = "transforms.json"
    path = os.path.join(out_dir, fname)
    save_json(path=path, data=metadata)


def parent_obj_to_camera(b_camera):
    b_empty = bpy.data.objects.new("Empty", None)
    b_empty.location = (0, 0, 0)
    b_camera.parent = b_empty  # setup parenting
    return b_empty


def setup_cam(cam, dummy):
    cam.location = CAM_LOCATION
    cam.data.clip_end = CAM_CLIP_END
    cam_constraint = cam.constraints.new(type='TRACK_TO')
    cam_constraint.track_axis = 'TRACK_NEGATIVE_Z'
    cam_constraint.up_axis = 'UP_Y'
    cam_constraint.target = dummy



def listify_matrix(matrix):
    matrix_list = []
    for row in matrix:
        matrix_list.append(list(row))
    return matrix_list
