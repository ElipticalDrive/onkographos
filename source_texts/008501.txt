package org.l2junity.gameserver;
import java.awt.*;
import java.awt.event.MouseEvent;
import java.awt.event.MouseListener;
import java.util.ArrayList;
import java.util.List;
import javax.swing.JFrame;
public class LineOnGrid extends JFrame
{
private static final int SIZE = 640;
private static final int CELL_SIZE = 32;
private static final int CELLS_LENGTH = SIZE / CELL_SIZE;
private CellNode[][] _cellsArray = new CellNode[CELLS_LENGTH][CELLS_LENGTH];
private int _lineStartX;
private int _lineStartY;
private int _lineEndX;
private int _lineEndY;
private int _currentVisitedCell;
public LineOnGrid()
{
setUndecorated(true);
final Dimension dim = Toolkit.getDefaultToolkit().getScreenSize();
setLocation(dim.width / 2 - SIZE / 2, dim.height / 2 - SIZE / 2);
pack();
setSize(SIZE, SIZE);
for (int nodeX = 0; nodeX < (getSize().width / CELL_SIZE); nodeX++)
{
for (int nodeY = 0; nodeY < (getSize().height / CELL_SIZE); nodeY++)
{
_cellsArray[nodeX][nodeY] = new CellNode(nodeX, nodeY);
}
}
addMouseListener(new MouseListener()
{
@Override
public void mouseClicked(MouseEvent e)
{
_lineStartX = CELL_SIZE / 2;
_lineStartY = CELL_SIZE / 2;
_lineEndX = e.getX();
_lineEndY = e.getY();
LineOnGrid.this.repaint();
}
@Override
public void mousePressed(MouseEvent e)
{
_lineStartX = e.getX();
_lineStartY = e.getY();
LineOnGrid.this.getGraphics().fillArc(e.getX(), e.getY(), 3, 3, 0, 360);
}
@Override
public void mouseReleased(MouseEvent e)
{
_lineEndX = e.getX();
_lineEndY = e.getY();
LineOnGrid.this.repaint();
}
@Override
public void mouseEntered(MouseEvent e)
{
}
@Override
public void mouseExited(MouseEvent e)
{
}
});
}
public void paint(Graphics g)
{
super.paint(g);
_currentVisitedCell = 0;
for (CellNode[] cellsArray1D : _cellsArray)
{
for (CellNode cellNode : cellsArray1D)
{
cellNode.draw(g);
}
}
VoxelIterator.iterate(_lineStartX, _lineStartY, 0, _lineEndX, _lineEndY, 0, (x, y, z) -> visitCell(x, y));
LineOnGrid.this.getGraphics().drawLine(_lineStartX, _lineStartY, _lineEndX, _lineEndY);
}
private void visitCell(int geoX, int geoY)
{
_cellsArray[geoX][geoY].drawGreen(LineOnGrid.this.getGraphics());
_cellsArray[geoX][geoY].drawNumber(LineOnGrid.this.getGraphics(), ++_currentVisitedCell);
}
private static class CellNode
{
int nodeX;
int nodeY;
public CellNode(int nodeX, int nodeY)
{
this.nodeX = nodeX;
this.nodeY = nodeY;
}
private int getX()
{
return nodeX * CELL_SIZE;
}
private int getY()
{
return nodeY * CELL_SIZE;
}
private void draw(Graphics g)
{
g.drawRect(getX(), getY(), CELL_SIZE, CELL_SIZE);
}
private void drawGreen(Graphics g)
{
final Color oldColor = g.getColor();
g.setColor(Color.GREEN);
g.fillRect(getX() + 1, getY() + 1, CELL_SIZE - 1, CELL_SIZE - 1);
g.setColor(oldColor);
}
private void drawNumber(Graphics g, int number)
{
g.drawString(String.valueOf(number), getX() + CELL_SIZE / 2, getY() + CELL_SIZE / 2);
}
}
private static class VoxelIterator
{
private static final double GRID_SIZE = CELL_SIZE;
private static final int MIN_GRID_X = 0;
private static final int MIN_GRID_Y = 0;
private static final int MIN_GRID_Z = 0;
private static final int MAX_GRID_X = CELLS_LENGTH - 1;
private static final int MAX_GRID_Y = CELLS_LENGTH - 1;
private static final int MAX_GRID_Z = CELLS_LENGTH - 1;
static void iterate(double startX, double startY, double startZ, double targetX, double targetY, double targetZ, VoxelConsumer consumer)
{
// The traversal algorithm breaks down the ray into intervals of 't' (or lazily said "time"), each of which spans one voxel.
// We start at the ray origin and visit each of these voxels in interval order.
// The integer variables X and Y are initialized to the starting voxel coordinates. In addition, the variables stepX and
// stepY are initialized to either 1 or -1 indicating whether X and Y are incremented or decremented as the ray crosses
// voxel boundaries (this is determined by the sign of the x and y components of →v: the vector that is the difference
// between the end and the start points).
int currVoxelX = (int)Math.floor(startX / GRID_SIZE);
int currVoxelY = (int)Math.floor(startY / GRID_SIZE);
int currVoxelZ = (int)Math.floor(startZ / GRID_SIZE);
final int lastVoxelX = (int)Math.floor(targetX / GRID_SIZE);
final int lastVoxelY = (int)Math.floor(targetY / GRID_SIZE);
final int lastVoxelZ = (int)Math.floor(targetZ / GRID_SIZE);
final double diffX = targetX - startX;
final double diffY = targetY - startY;
final double diffZ = targetZ - startZ;
final int stepX = sign(diffX);
final int stepY = sign(diffY);
final int stepZ = sign(diffZ);
// Compute tDeltaX and tDeltaY. tDeltaX indicates how far along the ray we must move (in units of 't') for the horizontal
// component of such a movement to equal the width of a voxel. Similarly, we store in tDeltaY the amount of movement
// along the ray which has a vertical component equal to the height of a voxel.
final double tDeltaX = stepX == 0 ? Double.MAX_VALUE : (stepX * GRID_SIZE / diffX);
final double tDeltaY = stepY == 0 ? Double.MAX_VALUE : (stepY * GRID_SIZE / diffY);
final double tDeltaZ = stepZ == 0 ? Double.MAX_VALUE : (stepZ * GRID_SIZE / diffZ);
// We determine the value of 't' at which the ray crosses the first vertical voxel boundary and store it in variable tMaxX.
// We perform a similar computation in y and store the result in tMaxY. The minimum of these two values will indicate how
// much we can travel along the ray and still remain in the current voxel.
double tMaxX = stepX > 0 ? (tDeltaX * frac1(startX / GRID_SIZE)) : (tDeltaX * frac0(startX / GRID_SIZE));
double tMaxY = stepY > 0 ? (tDeltaY * frac1(startY / GRID_SIZE)) : (tDeltaY * frac0(startY / GRID_SIZE));
double tMaxZ = stepZ > 0 ? (tDeltaZ * frac1(startZ / GRID_SIZE)) : (tDeltaZ * frac0(startZ / GRID_SIZE));
// Calculate how many times we need to iterate and loop as many times or stop if we have reached the max "time" of the ray.
int iterations = Math.abs(lastVoxelX - currVoxelX) + Math.abs(lastVoxelY - currVoxelY) + Math.abs(lastVoxelZ - currVoxelZ);
while (iterations-- >= 0 && (tMaxX <= 1 || tMaxY <= 1 || tMaxZ <= 1))
{
if (tMaxX < tMaxY)
{
if (tMaxX < tMaxZ)
{
currVoxelX += stepX;
if (currVoxelX < MIN_GRID_X || currVoxelX > MAX_GRID_X) // Boundry check
{
break;
}
tMaxX += tDeltaX;
}
else
{
currVoxelZ += stepZ;
if (currVoxelZ < MIN_GRID_Z || currVoxelZ > MAX_GRID_Z) // Boundry check
{
break;
}
tMaxZ += tDeltaZ;
}
}
else
{
if (tMaxY < tMaxZ)
{
currVoxelY += stepY;
if (currVoxelY < MIN_GRID_Y || currVoxelY > MAX_GRID_Y) // Boundry check
{
break;
}
tMaxY += tDeltaY;
}
else
{
currVoxelZ += stepZ;
if (currVoxelZ < MIN_GRID_Z || currVoxelZ > MAX_GRID_Z) // Boundry check
{
break;
}
tMaxZ += tDeltaZ;
}
}
consumer.accept(currVoxelX, currVoxelY, currVoxelZ);
}
}
/**
* @param number
* @return fractional part of float, for example, Frac(1.3) = 0.3, Frac(-1.7) = 0.3<br>
* Some float rounding inaccuracy may exist.
*/
private static double frac0(double number)
{
return number - Math.floor(number);
}
/**
* @param number
* @return remaining fractional part of float, for example, 1.3 = 0.7, Frac(-1.7) = 0.7<br>
* Some float rounding inaccuracy may exist.
*/
private static double frac1(double number)
{
return 1 - number + Math.floor(number);
}
/**
* @param number
* @return {@code 0} if the given number is 0, {@code -1} if the given number is negative, {@code 1} if the given number is positive.
*/
private static int sign(double number)
{
return number == 0 ? 0 : (number > 0 ? 1 : -1);
}
}
@FunctionalInterface
private interface VoxelConsumer
{
void accept(int voxelX, int voxelY, int voxelZ);
}
public static void main(String[] a)
{
new LineOnGrid().setVisible(true);
}
}
