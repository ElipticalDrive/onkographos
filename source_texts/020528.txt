But as expected, when I render the world that is around (100000 x 100000) with diffrent colors i get low frames/crash. How can I instance the 2d voxels/ get better performance? Or is this simply not possible in a normal canvas, in that case what's the best alternative for doing this.
```
//demo code
 
// import https://cdn.jsdelivr.net/gh/emilefokkema/infinite-canvas@v0.3-alpha/dist/infinite-canvas.js
 
var canvasElement = document.getElementById("canvas");
var rect = canvasElement.getBoundingClientRect();
canvasElement.width = rect.width;
canvasElement.height = rect.height;
var canvas = new InfiniteCanvas(canvasElement);
var ctx = canvas.getContext("2d", { alpha: false });
 
function create_block(position, color){

}
```
