Line Intersection
geometric.lineIntersection(a, b) returns the point where line a intersects line b. If the lines do not intersect, it returns null.
Drag either point to see how the intersection changes with the line.

<script>
	// @ts-nocheck
	import ExampleFrame from '$lib/components/ExampleFrame.svelte';
	import { pointerToSvgPoint } from '../utils/svg.js';
	import { clampScenePoint } from '../utils/relationExamples.js';
	import { createLineIntersectionState, getLineIntersectionGeometry } from './index.js';
	import Controls from './Controls.svelte';
	import Scene from './Scene.svelte';
	const BASE_WIDTH = 720,
		BASE_HEIGHT = 300;
	let svgElement = $state(undefined),
		stageWidth = $state(0),
		dragIndex = $state(null);
	let state = $state(createLineIntersectionState(BASE_WIDTH, BASE_HEIGHT));
	let previousWidth = $state(BASE_WIDTH),
		previousHeight = $state(BASE_HEIGHT);
	let stageHeight = $derived(BASE_HEIGHT);
	let geometry = $derived(getLineIntersectionGeometry(state));
	function movePoint(index, point) {
		state.line = state.line.map((entry, i) =>
			i === index ? clampScenePoint(point, state.width, state.height, 8) : entry
		);
	}
	function startDrag(index, event) {
		dragIndex = index;
		svgElement?.setPointerCapture(event.pointerId);
		movePoint(index, pointerToSvgPoint(event, svgElement));
	}
	function moveDrag(event) {
		if (dragIndex === null || !svgElement) return;
		movePoint(dragIndex, pointerToSvgPoint(event, svgElement));
	}
	function stopDrag(event) {
		if (event && svgElement?.hasPointerCapture(event.pointerId))
			svgElement.releasePointerCapture(event.pointerId);
		dragIndex = null;
	}
	function movePointFromKeyboard(index, event) {
		const step = event.shiftKey ? 10 : 1;
		const deltas = {
			ArrowLeft: [-step, 0],
			ArrowRight: [step, 0],
			ArrowUp: [0, -step],
			ArrowDown: [0, step]
		};
		const delta = deltas[event.key];
		if (!delta) return;
		event.preventDefault();
		movePoint(index, [state.line[index][0] + delta[0], state.line[index][1] + delta[1]]);
	}
	$effect(() => {
		if (
			!stageWidth ||
			!stageHeight ||
			(stageWidth === previousWidth && stageHeight === previousHeight)
		)
			return;
		const scaleX = stageWidth / previousWidth,
			scaleY = stageHeight / previousHeight;
		state.width = stageWidth;
		state.height = stageHeight;
		state.line = state.line.map((point) => [point[0] * scaleX, point[1] * scaleY]);
		previousWidth = stageWidth;
		previousHeight = stageHeight;
	});
</script>

{#snippet controls()}<Controls intersects={geometry.intersects} />{/snippet}
{#snippet scene()}<Scene
		bind:svgElement
		bind:stageWidth
		{state}
		{geometry}
		{startDrag}
		{moveDrag}
		{stopDrag}
		{movePointFromKeyboard}
	/>{/snippet}
<ExampleFrame {scene} {controls} />
