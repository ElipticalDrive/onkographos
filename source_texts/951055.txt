Implements a Wolfenstein 3D-style raycasting game engine that renders 3D maze environments from 2D maps, supporting player movement and configurable scene descriptions.
Core Implementation

Raycasting Engine (src/raycast.c): DDA algorithm for wall detection and height calculation, renders ceiling/floor/walls per vertical screen stripe
Scene Parser (src/parse.c): Reads .cub files for RGB floor/ceiling colors (F/C lines) and validates map grid (characters: 0 empty, 1 wall, N/S/E/W spawn)
Player Controls (src/controls.c): WASD movement with collision detection, arrow keys for camera rotation
Window Management (src/main.c): ESC key and X button handlers via mlx_hook events

Example Scene File

C 0,128,255

1000001
1111111

Player spawns at N facing north. Floor renders as orange (220,100,0), ceiling as blue (0,128,255).

#include "../includes/cub3d.h"
void move_forward(t_game *game)
{
double new_x;
double new_y;
double move_speed;
move_speed = 0.1;
new_x = game->player.pos_x + game->player.dir_x * move_speed;
new_y = game->player.pos_y + game->player.dir_y * move_speed;
if ((int)new_y >= 0 && (int)new_y < game->map.height
&& (int)new_x >= 0 && (int)new_x < (int)strlen(game->map.grid[(int)new_y])
&& game->map.grid[(int)new_y][(int)new_x] == '0')
{
game->player.pos_x = new_x;
game->player.pos_y = new_y;
}
}
void move_backward(t_game *game)
{
double new_x;
double new_y;
double move_speed;
move_speed = 0.1;
new_x = game->player.pos_x - game->player.dir_x * move_speed;
new_y = game->player.pos_y - game->player.dir_y * move_speed;
if ((int)new_y >= 0 && (int)new_y < game->map.height
&& (int)new_x >= 0 && (int)new_x < (int)strlen(game->map.grid[(int)new_y])
&& game->map.grid[(int)new_y][(int)new_x] == '0')
{
game->player.pos_x = new_x;
game->player.pos_y = new_y;
}
}
void move_left(t_game *game)
{
double new_x;
double new_y;
double move_speed;
move_speed = 0.1;
new_x = game->player.pos_x - game->player.plane_x * move_speed;
new_y = game->player.pos_y - game->player.plane_y * move_speed;
if ((int)new_y >= 0 && (int)new_y < game->map.height
&& (int)new_x >= 0 && (int)new_x < (int)strlen(game->map.grid[(int)new_y])
&& game->map.grid[(int)new_y][(int)new_x] == '0')

}
}

{
double new_x;
double new_y;
double move_speed;
move_speed = 0.1;
new_x = game->player.pos_x + game->player.plane_x * move_speed;
new_y = game->player.pos_y + game->player.plane_y * move_speed;

void rotate_left(t_game *game)
{
double old_dir_x;
double old_plane_x;
double rot_speed;
rot_speed = 0.05;
old_dir_x = game->player.dir_x;
game->player.dir_x = game->player.dir_x * cos(rot_speed)
- game->player.dir_y * sin(rot_speed);
game->player.dir_y = old_dir_x * sin(rot_speed)
+ game->player.dir_y * cos(rot_speed);
old_plane_x = game->player.plane_x;
game->player.plane_x = game->player.plane_x * cos(rot_speed)
- game->player.plane_y * sin(rot_speed);
game->player.plane_y = old_plane_x * sin(rot_speed)
+ game->player.plane_y * cos(rot_speed);
}
void rotate_right(t_game *game)
{
double old_dir_x;
double old_plane_x;
double rot_speed;
rot_speed = -0.05;
old_dir_x = game->player.dir_x;
game->player.dir_x = game->player.dir_x * cos(rot_speed)
- game->player.dir_y * sin(rot_speed);
game->player.dir_y = old_dir_x * sin(rot_speed)
+ game->player.dir_y * cos(rot_speed);
old_plane_x = game->player.plane_x;
game->player.plane_x = game->player.plane_x * cos(rot_speed)
- game->player.plane_y * sin(rot_speed);
game->player.plane_y = old_plane_x * sin(rot_speed)
+ game->player.plane_y * cos(rot_speed);
}
int handle_keypress(int keycode, t_game *game)
{
if (keycode == KEY_ESC)
cleanup_exit(game);
else if (keycode == KEY_W)
move_forward(game);
else if (keycode == KEY_S)
move_backward(game);
else if (keycode == KEY_A)
move_left(game);
else if (keycode == KEY_D)
move_right(game);
else if (keycode == KEY_LEFT)
rotate_left(game);
else if (keycode == KEY_RIGHT)
rotate_right(game);
return (0);
}

#include "../includes/cub3d.h"
void init_player(t_game *game, int x, int y, char dir)
{
game->player.pos_x = x + 0.5;
game->player.pos_y = y + 0.5;
if (dir == 'N')
{
game->player.dir_x = 0;
game->player.dir_y = -1;
game->player.plane_x = 0.66;
game->player.plane_y = 0;
}
else if (dir == 'S')
{
game->player.dir_x = 0;
game->player.dir_y = 1;
game->player.plane_x = -0.66;
game->player.plane_y = 0;
}
else if (dir == 'E')
{
game->player.dir_x = 1;
game->player.dir_y = 0;
game->player.plane_x = 0;
game->player.plane_y = 0.66;
}
else if (dir == 'W')
{
game->player.dir_x = -1;
game->player.dir_y = 0;
game->player.plane_x = 0;
game->player.plane_y = -0.66;
}
}
int init_game(t_game *game)
{
game->mlx = mlx_init();
if (!game->mlx)
return (0);
game->win = mlx_new_window(game->mlx, WIN_WIDTH, WIN_HEIGHT, "cub3D");
if (!game->win)
return (0);
game->img.img = mlx_new_image(game->mlx, WIN_WIDTH, WIN_HEIGHT);
if (!game->img.img)
return (0);
game->img.addr = mlx_get_data_addr(game->img.img,
&game->img.bits_per_pixel,
&game->img.line_length,
&game->img.endian);
return (1);
}
