data.mat
load data
voxelsize=0.5;
for i=1:3
tmp=xyz(:,i);
edges=min(tmp):voxelsize:max(tmp)+voxelsize;
xyz(:,i)=discretize( tmp , edges);
end
V=accumarray(xyz,1)>0;
volumeViewer(V);
