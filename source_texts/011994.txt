Fast, Exact, Stable algorithm ? Could be interesting (not yet sure if it will overflow, maybe not). I spent 5 minutes on verification. Perhaps in future I will investigate further until then you can beat me to it. This will be usefull for exact bullit hits instead of passing through on certain cases...
```
// Inputs: 
// startX, startY, startZ (Integer Grid Coordinates)
// originX, originY, originZ (Ray Origin in world space)
// vx, vy, vz (Ray direction vector, scaled to integers)
// cellSize (Width of a voxel)

// 1. Initialization Phase
int X = startX;
int Y = startY;
int Z = startZ;

int stepX = (vx >= 0) ? 1 : -1;
int stepY = (vy >= 0) ? 1 : -1;
int stepZ = (vz >= 0) ? 1 : -1;

// Calculate distance to the first boundary in world units
// If step is positive, boundary is (X+1)*CellSize. If negative, it is X*CellSize.
int nextBoundaryX = (stepX > 0) ? (X + 1) * cellSize : X * cellSize;
int nextBoundaryY = (stepY > 0) ? (Y + 1) * cellSize : Y * cellSize;
int nextBoundaryZ = (stepZ > 0) ? (Z + 1) * cellSize : Z * cellSize;

// Compute scale factors (Absolute values of cross-products)
// Note: Use 'long long' to prevent overflow if the grid is large
long long iDeltaX = cellSize * abs(vy * vz);
long long iDeltaY = cellSize * abs(vx * vz);
long long iDeltaZ = cellSize * abs(vx * vy);

// Initialize iMax (Scaled t to next boundary)
// Original logic: tMaxX = (nextBoundaryX - originX) / vx
// Scaled logic: Multiply by S = |vx * vy * vz|
// Result: (nextBoundaryX - originX) * vy * vz * sign_correction
long long iMaxX = (nextBoundaryX - originX) * vy * vz; 
if (vx < 0) iMaxX = -iMaxX; // Correct for negative direction divisions
if (vx == 0) iMaxX = LLONG_MAX; // Handle parallel rays

long long iMaxY = (nextBoundaryY - originY) * vx * vz;
if (vy < 0) iMaxY = -iMaxY;
if (vy == 0) iMaxY = LLONG_MAX;

long long iMaxZ = (nextBoundaryZ - originZ) * vx * vy;
if (vz < 0) iMaxZ = -iMaxZ;
if (vz == 0) iMaxZ = LLONG_MAX;


// 2. Incremental Phase [cite: 76]
// This loop mirrors the original paper's logic exactly 
// but uses integers instead of floats.

Object* hit = NULL;

do {
    if (iMaxX < iMaxY) {
        if (iMaxX < iMaxZ) {
            // X is the smallest, step in X
            X = X + stepX;
            if (X == justOutX) return NULL; // Outside grid [cite: 98]
            iMaxX = iMaxX + iDeltaX;        // Integer Addition
        } else {
            // Z is the smallest, step in Z
            Z = Z + stepZ;
            if (Z == justOutZ) return NULL;
            iMaxZ = iMaxZ + iDeltaZ;        // Integer Addition
        }
    } else {
        if (iMaxY < iMaxZ) {
            // Y is the smallest, step in Y
            Y = Y + stepY;
            if (Y == justOutY) return NULL;
            iMaxY = iMaxY + iDeltaY;        // Integer Addition
        } else {
            // Z is the smallest, step in Z
            Z = Z + stepZ;
            if (Z == justOutZ) return NULL;
            iMaxZ = iMaxZ + iDeltaZ;        // Integer Addition
        }
    }

    // Check voxel list [cite: 126]
    hit = CheckVoxelObjectList(X, Y, Z);

} while (hit == NULL);

return hit;
```
