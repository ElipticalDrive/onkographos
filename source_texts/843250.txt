In the current implementation, points within the same DepthSample are processed sequentially: each point is unprojected, its ray is carved, and then its cell is incremented. This introduces an order-dependence bug: if a deeper point's ray passes through a shallower point's cell in the same sample, the shallower point will be carved and lost if it was processed first, but will survive if it was processed second.
To ensure order-independence and prevent valid current-frame observations from being carved by neighboring rays in the same sample, we should perform all carving operations first, and then apply all increments.
```
    let added = 0;
    const validCells: GridCell[] = [];
    for (const point of sample.points) {
      const world = unprojectDepthPoint(
        point,
        sample.cameraPos,
        sample.cameraRot,
        sample.projectionMatrix
      );
      if (world) {
        validCells.push(this.cellForPosition(world));
      }
    }

    for (const cell of validCells) {
      if (!cellsEqual(cameraCell, cell)) {
        this.carve(cameraCell, cell);
      }
    }

    for (const cell of validCells) {
      this.increment(cell);
      added++;
    }
```
