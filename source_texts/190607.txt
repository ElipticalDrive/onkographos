```
from skimage.measure import marching_cubes

def compute_surface_area(binary_mask, spacing=(1.0, 1.0, 1.0)):
    mask_float = binary_mask.astype(float)
    verts, faces, _, _ = marching_cubes(mask_float, level=0.5, spacing=spacing)
    
    v0 = verts[faces[:, 0]]
    v1 = verts[faces[:, 1]]
    v2 = verts[faces[:, 2]]
    
    cross = np.cross(v1 - v0, v2 - v0)
    triangle_areas = 0.5 * np.linalg.norm(cross, axis=1)
    surface_area = float(np.sum(triangle_areas))
    return surface_area
```
