## 노이즈 알고리즘 함수

**보간 함수**
```bash
import numpy as np

def smoothstep(t: float) -> float:
    # 0~1 사이 값을 더 자연스럽게 변화시킨다.
    return t * t * (3.0 - 2.0 * t)

def lerp(a: float, b: float, t: float) -> float: 
# 선형 보간
    return a + (b - a) * t
```

**기본 노이즈 (Value)**

```bash
def value_noise_2d(width: int, height: int, grid: int, rng):
    gw = width // grid + 2
    gh = height // grid + 2

    # 격자점 랜덤값 생성
    lattice = rng.random((gh, gw), dtype=np.float32)

    noise = np.zeros((height, width), dtype=np.float32)

    for y in range(height):
        gy = y / grid
        y0 = int(gy)
        ty = smoothstep(gy - y0)

        for x in range(width):
            gx = x / grid
            x0 = int(gx)
            tx = smoothstep(gx - x0)

            # 격자 네 점
            v00 = lattice[y0,     x0]
            v10 = lattice[y0,     x0 + 1]
            v01 = lattice[y0 + 1, x0]
            v11 = lattice[y0 + 1, x0 + 1]

            a = lerp(v00, v10, tx)
            b = lerp(v01, v11, tx)
            noise[y, x] = lerp(a, b, ty)

    return noise
```

**Fractal 노이즈**

```bash
def fractal_noise_2d(width, height, rng,
                     base_grid=64,
                     octaves=5,
                     lacunarity=2.0,
                     gain=0.5):

    total = np.zeros((height, width), dtype=np.float32)
    amplitude = 1.0
    frequency = 1.0
    amp_sum = 0.0

    for _ in range(octaves):
        grid = max(1, int(base_grid / frequency))
        layer = value_noise_2d(width, height, grid, rng)

        total += layer * amplitude
        amp_sum += amplitude

        amplitude *= gain        # 점점 작아지는 진폭
        frequency *= lacunarity  # 점점 촘촘해지는 패턴

    total /= amp_sum

    # 0~1 정규화
    total = (total - total.min()) / (total.max() - total.min() + 1e-9)

    return total
```
