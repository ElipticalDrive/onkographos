    Sphere obj2(Vector(0,5,0),5,material2);
    Sphere Objects[] = {obj1,obj2};

    Light point_light(Vector(1.5,-0.5,-3.0),light_col);

    RenderEngine engine;

    Image img =  engine.render(scene);
