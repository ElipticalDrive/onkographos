Я пишу шейдер на GLSL для отрисовки вокселей из 3D текстуры.
Я не очень хорошо понимаю концепцию трассировки лучей через DDA, и не могу понять ошибку в моем коде, которая не позволяет корректно отображать воксели из 3D-текстуры. Я как-то пытался сделать это из туториала по созданию DDA для 2D-графики, и пытался перенести в 3D, но не получается.
Возможно, и скорее всего, что я неправильно реализовал DDA, и неправильно получаю значение вокселя из 3D-текстуры. Я бы еще хотел применить к расчетам 3 вектора - смещение модели в мировых координатах, вращение по Pitch -> Yaw -> Roll и масштабирование модели по трем осям.

 vec3 trace_dda_voxel(Ray cam_ray, vec3 pos, vec3 rot, vec3 size, sampler3D tex_voxel, vec3 tex_size, float max_dist) { 
    vec3 ro = cam_ray.pos; 
    vec3 rd = cam_ray.dir; 
 
    vec3 ray_unit_step_size = vec3( 
        sqrt(1.0 + (rd.y / rd.x) * (rd.y / rd.x) + (rd.z / rd.x) * (rd.z / rd.x)), 
        sqrt(1.0 + (rd.x / rd.y) * (rd.x / rd.y) + (rd.z / rd.y) * (rd.z / rd.y)), 
        sqrt(1.0 + (rd.x / rd.z) * (rd.x / rd.z) + (rd.y / rd.z) * (rd.y / rd.z)) 
    ); 
 
    vec3 map_check = floor(ro); 
    vec3 ray_length_1d; 
    vec3 step; 
    float traveled_distance = 0.0; 
 
    if (rd.x < 0.0) { 
        step.x = -1.0; 
        ray_length_1d.x = (ro.x - float(map_check.x)) * ray_unit_step_size.x; 
    } else { 
        step.x = 1.0; 
        ray_length_1d.x = (float(map_check.x + 1) - ro.x) * ray_unit_step_size.x; 
    } 
 
    if (rd.y < 0.0) { 
        step.y = -1.0; 
        ray_length_1d.y = (ro.y - float(map_check.y)) * ray_unit_step_size.y; 
    } else { 
        step.y = 1.0; 
        ray_length_1d.y = (float(map_check.y + 1) - ro.y) * ray_unit_step_size.y; 
    } 
 
    if (rd.z < 0.0) { 
        step.z = -1.0; 
        ray_length_1d.z = (ro.z - float(map_check.z)) * ray_unit_step_size.z; 
    } else { 
        step.z = 1.0; 
        ray_length_1d.z = (float(map_check.z + 1) - ro.z) * ray_unit_step_size.z; 
    } 
 
    while(traveled_distance < max_dist) { 
        if (ray_length_1d.x < ray_length_1d.y && ray_length_1d.x < ray_length_1d.z) { 
            map_check.x += step.x; 
            traveled_distance = ray_length_1d.x; 
            ray_length_1d.x += ray_unit_step_size.x; 
        } else if (ray_length_1d.y < ray_length_1d.z) { 
            map_check.y += step.y; 
            traveled_distance = ray_length_1d.y; 
            ray_length_1d.y += ray_unit_step_size.y; 
        } else { 
            map_check.z += step.z; 
            traveled_distance = ray_length_1d.z; 
            ray_length_1d.z += ray_unit_step_size.z; 
        } 
 
        int vox_id = int(texture(u_tex_voxel, map_check/tex_size).r * 255.0); 
        if (vox_id == 255) return vec3(1.0); 
    } 
 
    return vec3(0.0); 
}

Ошибки и замечания
Неправильное использование float:
В GLSL нет необходимости использовать float() для приведения типов, когда речь идет о целых числах. Вместо float(map_check.x) можно использовать просто map_check.x.
Проблемы с делением на ноль:
Если направление луча (rd) имеет компоненты, равные нулю, то может произойти деление на ноль при вычислении ray_unit_step_size. Необходимо добавить проверку на ноль перед делением.
Инициализация переменных:
Переменные ray_length_1d и step должны быть инициализированы в случае, если направление луча равно нулю. В противном случае это может привести к неопределенному поведению.
Проверка выхода за границы текстуры:
В коде не предусмотрена проверка, выходит ли map_check за пределы текстуры. Это может привести к ошибкам при доступе к текстуре.
Использование texture():
В функции используется texture(u_tex_voxel, map_check/tex_size), но переменная u_tex_voxel не определена в коде. Убедитесь, что она объявлена и доступна.
Логика выхода из цикла:
Условие выхода из цикла может быть не совсем корректным, так как оно обновляет traveled_distance только при каждом шаге, а не накапливает общее расстояние.
