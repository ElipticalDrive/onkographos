 #include <iostream>
#include <fstream>
#include <vector>
#include <cmath>
#define recursion_limit 1 

using namespace std;

class TGAManager : ofstream {
    public:
        TGAManager (int _width, int _height, int _bpp) : width(_width), height(_height), bpp (_bpp), _const(255) {}
        void setHeader ();
        void fillTGA ();  
        ofstream ofs;
    private:
        int width,height,bpp,_const;
};

void TGAManager::setHeader () {  
    ofs.open ("z.tga", ios::binary);
    ofs.put (0), ofs.put(0), ofs.put(2), ofs.put(0), ofs.put(0), ofs.put(0);
    ofs.put (0), ofs.put(0), ofs.put(0), ofs.put(0), ofs.put(0), ofs.put(0);
    ofs.put (width & _const), ofs.put(width >> 8);
    ofs.put (height & _const), ofs.put (height >> 8);
    ofs.put (bpp), ofs.put(0);
}

void TGAManager::fillTGA () {
    for (int i=0; i<width; i++)
        for (int j=0; j<height; j++)
            ofs.put (0), ofs.put (0), ofs.put (0); 
}

class Image {
    public:
        Image (int _width, int _height, int _bpp) : width (_width), height (_height), bpp (_bpp) {}
        int getHeight () { return height; }
        int getWidth () { return width; }
        int getBpp () { return bpp; }
    private:
        int width,height,bpp;
};

class Vec3f {
    public:
        Vec3f (float _x, float _y, float _z) : x (_x), y (_y), z (_z) {}
        float operator * (Vec3f vec) {float tmp (x*vec.x + y*vec.y + z*vec.z); return tmp; }
        Vec3f operator * (float t) {Vec3f tmp (x*t, y*t, z*t); return tmp; }
        Vec3f operator + (Vec3f vec) {Vec3f tmp (x+vec.x, y+vec.y, z+vec.z); return tmp;}
        Vec3f operator - (Vec3f vec) {Vec3f tmp (x-vec.x, y-vec.y, z-vec.z); return tmp;}
        void normalize () { x/= sqrtf (x*x + y*y + z*z), y/= sqrtf (x*x + y*y + z*z), z/= sqrtf (x*x + y*y + z*z); }
        float getX () {return x;}
        float getY () {return y;}
        float getZ () {return z;}
    private:
        float x,y,z;
};

struct Color {
    public:
        Color (int blue_, int green_, int red_) : blue (blue_), green (green_), red (red_) {}
        float blue,green,red;
};

class Light {
    public:
        Light (Vec3f position, Color _color) : pos (position), color (_color) {} 
        Vec3f getPos () { return pos; }
    private:
        Vec3f pos; 
        Color color;
};

class Sphere {
    public:
        Sphere (Vec3f _center, float _squared_radius, int _color) : center (_center), squared_radius (_squared_radius), color (_color) {}
        bool intersect (Vec3f ray_dir, Vec3f ray_ori, float& t);
        bool intersectDebug (Vec3f ray_dir, Vec3f ray_ori, float& t, float x, float y);
        Vec3f getCenter () { return center; }
        int getColor () { return color; }
    private:
        Vec3f center;
        float squared_radius;
        int color;
};


bool Sphere::intersectDebug (Vec3f ray_ori, Vec3f ray_dir, float& t, float x, float y) {
    Vec3f hipt = center - ray_ori;
    float cat_adj_len = hipt * ray_dir;
    if ((x==424 && y==424) || (x==670 && y==285)) {
        cout << endl;
        cout << "center: " << center.getX() << ' ' << center.getY() << ' '   << center.getZ() << endl;
        cout << "ray_ori: " << ray_ori.getX() << ' ' << ray_ori.getY() << ' '  << ray_ori.getZ() << endl;
        cout << "radius_squared: " << squared_radius << endl;
        cout << "hipt: " << hipt.getX() << ' ' << hipt.getY() << ' '  << hipt.getZ() << endl;
        cout << "cat_adj_len: " << cat_adj_len << endl;
    }
    if (cat_adj_len < 0)
        return false;
    float cad_op_len_sqr = hipt * hipt - cat_adj_len * cat_adj_len;
    if (x==670 && y==285) 
        cout << "cad_op_len_sqr = hipt*hipt - cat_adj_len*cat_adj_len: " << cad_op_len_sqr << endl;
    if (x==424 && y==424) 
        cout << "cad_op_len_sqr = hipt*hipt - cat_adj_len*cat_adj_len: " << cad_op_len_sqr << endl;
    if (cad_op_len_sqr > squared_radius)
        return false;
    float cat_adj_sml_len_sqr = squared_radius - cad_op_len_sqr;
    float t0 = cat_adj_len - sqrt (cat_adj_sml_len_sqr);
    if ((x==670 && y==285) || (x==424 && y==424)) {
        cout << "cat_adj_sml_len_sqr (= squared_radius - cad_op_len_sqr): " << cat_adj_sml_len_sqr << endl;
        cout << "t0 (= cat_adj_len - sqrt (cat_adj_sml_len_sqr)): " << t0 << endl;
    }
    if (t0 < t)  {
        t = t0;
//      cout << "t: " << t << endl;
        return true;
    }
    return false;
}

class Scene {
    public:
        vector <Sphere>& getSphereContainer () { return sphere_container; }
        vector <Color>& getMaterialContainer () { return material_container; }
        vector <Light>& getLightContainer () { return light_container; }
    private:
        vector <Sphere> sphere_container;
        vector <Color> material_container;
        vector <Light> light_container;
};

void trace (Scene scene, Image image) {
    TGAManager tga (image.getWidth(),image.getHeight(),image.getBpp());
    tga.setHeader ();
    vector <Sphere>& sphere_container = scene.getSphereContainer ();
    vector <Color>& material_container = scene.getMaterialContainer ();
    vector <Light>& light_container = scene.getLightContainer ();
    int count=0;
    for (int y=0; y<image.getHeight(); y++) {
        for (int x=0; x<image.getWidth(); x++) {
            float blue=0,green=0,red=0;


            int depth_recursion=0;
            bool intersection=false;
            float coef = 1;
            Vec3f ray_ori (x,y,0);
            Vec3f ray_dir (0,0,1);
            do {
                float closest_distance = 1000000;
                int closest_sphere=-1;
                for (int z=0; z<sphere_container.size(); z++) {
                    if (sphere_container[z].intersectDebug (ray_ori,ray_dir,closest_distance,x,y)) {
                        closest_sphere=z;
                        count++;
                        intersection=true;
                        if ((x==670 && y==285) || (x==424 && y==424))
                            cout << "closest_sphere: " << closest_sphere << endl;   
                    }
                }


                if (intersection) {
                    ray_ori = ray_ori + ray_dir * closest_distance;             
                    int color_num = sphere_container [closest_sphere].getColor();
                    
                    Color color = material_container [color_num];
                    if (x==670 && y==285) 
                            cout << "Color: " <<  color.blue << ' ' << color.green << ' ' << color.red << endl;
                    else if (x==424 && y==424)
                            cout << "Color: " << color.blue << ' ' << color.green << ' ' << color.red << endl;
                    
                        tga.ofs.put (color.blue), tga.ofs.put (color.green), tga.ofs.put (red);
                }
                
                    else
                        tga.ofs.put (0), tga.ofs.put (0), tga.ofs.put (0);
                depth_recursion++;
            } while (depth_recursion < recursion_limit && coef > 0);
        }
    }   
    cout << count << endl;
}
