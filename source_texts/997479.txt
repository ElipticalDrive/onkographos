﻿using System;
using UnityEngine;
using UnityEngine.Rendering;
[CreateAssetMenu(fileName = "New ConeTracingGlobals", menuName = "ScriptableObjects/VCT/ConeTracingGlobals")]
public class ConeTracingGlobals : ScriptableObject
{
[Header("Settings"), Space]
[SerializeField] private VoxelsRenderFeature voxelsRenderFeature;
[Header("Ambient Occlusion"), Space]
[SerializeField] private ConeParametersSet ambientOcclusionSet;
[Header("Global Illumination"), Space]
[SerializeField] private ConeParametersSet globalIlluminationSet;
private MaterialPropertyCache propertyCache = new();
private void OnValidate()
{
if (IsCurrentSettings())
UpdateGlobals();
}
private void UpdateGlobals()
{
ambientOcclusionSet.Update(propertyCache);
globalIlluminationSet.Update(propertyCache);
}
private bool IsCurrentSettings()
{
if (voxelsRenderFeature == null)
return false;
return voxelsRenderFeature.VoxelFeatureSettings.ConeTracingGlobals == this;
}
}
[Serializable]
public class ConeParametersSet
{
[SerializeField] private string tagID = "AO";
[SerializeField, Range(0, 20)] private int hemisphereConesCount = 6;
[SerializeField, Range(0, 20)] private int maxConeSteps = 8;
[SerializeField] private float absorption = 0.96f;
[SerializeField] private float baseAdd = -0.15f;
[SerializeField] private float spread = 0.5f;
[SerializeField] private float baseOffset = 0.2f;
[SerializeField, Range(0, 2)] private float noiseIntensity = 0.45f;
[SerializeField, Range(0, 8)] private int minMipLevel = 0;
[SerializeField] private float distanceSteepness = 1.79f;
[SerializeField] private float distanceScale = 0.67f;
[SerializeField] private float coneSize = 6.07f;
[SerializeField] private uint noiseSeed = 0;
public void Update(MaterialPropertyCache cache)
{
int hemisphereConesCountId = cache.Get(GetName("_HemisphereConesCount"));
Shader.SetGlobalInt(hemisphereConesCountId, hemisphereConesCount);
int maxConeStepsId = cache.Get(GetName("_MaxConeSteps"));
Shader.SetGlobalInt(maxConeStepsId, maxConeSteps);
int absorptionId = cache.Get(GetName("_ConeAbsorption"));
Shader.SetGlobalFloat(absorptionId, absorption);
int baseAddId = cache.Get(GetName("_ConeBaseAdd"));
Shader.SetGlobalFloat(baseAddId, baseAdd);
int spreadId = cache.Get(GetName("_ConeSpread"));
Shader.SetGlobalFloat(spreadId, spread);
int baseOffsetId = cache.Get(GetName("_ConeBaseOffset"));
Shader.SetGlobalFloat(baseOffsetId, baseOffset);
int noiseIntensityId = cache.Get(GetName("_ConeNoiseIntensity"));
Shader.SetGlobalFloat(noiseIntensityId, noiseIntensity);
int minMipLevelId = cache.Get(GetName("_ConeMinMipLevel"));
Shader.SetGlobalInt(minMipLevelId, minMipLevel);
int distanceSteepnessId = cache.Get(GetName("_ConeSteepness"));
Shader.SetGlobalFloat(distanceSteepnessId, distanceSteepness);
int distanceScaleId = cache.Get(GetName("_ConeDistScale"));
Shader.SetGlobalFloat(distanceScaleId, distanceScale);
int coneSizeId = cache.Get(GetName("_ConeSize"));
Shader.SetGlobalFloat(coneSizeId, coneSize);
int noiseSeedId = cache.Get(GetName("_ConeSeed"));
Shader.SetGlobalInt(noiseSeedId, (int)noiseSeed);
}
private string GetName(string baseName)
{
return baseName + "_" + tagID;
}
}
