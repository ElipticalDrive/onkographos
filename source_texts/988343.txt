Motivation
hard_voxelize(deterministic=True) is the default voxelization in
MMDetection3D-style pipelines. Its deterministic CUDA path is dominated by two
legacy bottlenecks in voxelization_cuda_kernel.cuh:

Complexity
Issue




point_to_voxelidx_kernel
parallel over points
O(N^2) worst
every thread re-scans all previous points to count same-voxel predecessors (for (int i = 0; i < index; ++i))


determin_voxel_num
<<<1, 1>>>
O(N), fully serial
the whole per-voxel numbering loop runs on a single thread



Measured on an NVIDIA L4 (200k points, max_points=35, max_voxels=20000):
73.85 ms per call, which makes voxelization a dominant preprocessing cost.
Modification
Replaces those two kernels with a hash + stable-sort pipeline, orchestrated
inside HardVoxelizeForwardCUDAKernelLauncher with ATen ops, so the Python API
and the pybind registration stay untouched:

voxel_index_kernel: per-point 1-D voxel id lin = (z * grid_y + y) * grid_x + x
(the same float32 arithmetic as dynamic_voxelize_kernel); out-of-range -> -1.
stable argsort by lin: equal ids become adjacent segments and keep their
original relative order, so each group head is the first point of that voxel in
input order -- exactly what the legacy first-occurrence numbering relies on.
group_meta_lead_kernel: group_start[G + 1] (sentinel group_start[G] = N)
plus each group leader's original row id.
voxels are ranked by leader order (second stable sort + index_copy_), then one
scatter_kernel writes features / coors / counts, deriving the in-group slot and
the group size on the fly (no materialized rank arrays).

The legacy launcher is kept verbatim as a static V1 function and is used whenever
NDim != 3 or points is not float32.

// ---------------------------------------------------------------------------
// The kernels below (voxel_index_kernel / group_meta_lead_kernel /
// scatter_kernel) implement a hash + stable-sort voxelization for the
// deterministic hard_voxelize path. They produce byte-identical output to
// the legacy point_to_voxelidx_kernel + determin_voxel_num pipeline while
// replacing the O(N^2) back-scan and the single-threaded voxel numbering with
// O(N log N) sort-based grouping.
// ---------------------------------------------------------------------------
// K1 voxel_index_kernel: map each point to a 1-D linear voxel id (perfect
// hash, row-major over (z, y, x)). Float32 arithmetic replicates mmcv's
// dynamic_voxelize_kernel exactly; out-of-range points get lin = -1.
template <typename T_int>
__global__ void voxel_index_kernel(const float* points, const int num_points,
const int num_features, const float vx,
const float vy, const float vz,
const float min_x, const float min_y,
const float min_z, const int grid_x,
const int grid_y, const int grid_z,
T_int* lin) {
CUDA_1D_KERNEL_LOOP(i, num_points) {
const float* p = points + (size_t)i * num_features;
const int c_x = (int)floorf((p[0] - min_x) / vx);
if (c_x < 0 || c_x >= grid_x) {
lin[i] = -1;
continue;
}
const int c_y = (int)floorf((p[1] - min_y) / vy);
if (c_y < 0 || c_y >= grid_y) {
lin[i] = -1;
continue;
}
const int c_z = (int)floorf((p[2] - min_z) / vz);
if (c_z < 0 || c_z >= grid_z) {
lin[i] = -1;
continue;
}
// row-major perfect hash; coors order is (z, y, x) as in mmcv
lin[i] = (c_z * grid_y + c_y) * grid_x + c_x;
}
}
// K2 group_meta_lead_kernel: record group_start[g] and leader_orig[g] for
// every group head. The loop iterates num_valid + 1 positions; position
// i == num_valid writes the sentinel group_start[G] = num_valid so that the
// scatter kernel can derive group sizes by difference. Stable sorting
// guarantees the group head is the point that appears first in the original
// order, which keeps voxel numbering identical to the legacy implementation.
template <typename T_int>
__global__ void group_meta_lead_kernel(const int64_t* group_id,
const bool* is_start,
const int64_t* orig_idx,
const int num_valid,
const int num_groups, T_int* group_start,
int64_t* leader_orig) {
CUDA_1D_KERNEL_LOOP(i, num_valid + 1) {
if (i == num_valid) {
// sentinel: virtual group G starts at position num_valid
group_start[num_groups] = num_valid;
continue;
}
if (is_start[i]) {
const int g = (int)group_id[i];
group_start[g] = i;
leader_orig[g] = orig_idx[i];
}
}
}
// K3 scatter_kernel: write point features into voxels; the group-head thread
// fills coors / num_points_per_voxel. rank / group_size / lin are derived
// on the fly instead of being materialized:
// r = i - group_start[g] (slot inside the group)
// size = group_start[g + 1] - group_start[g] (sentinel difference)
// lin = lin_sorted[i] (shared by the whole group)
// Only r == 0 writes the meta info, so there is no data race.
template <typename T_int>
__global__ void scatter_kernel(
const float* points, const int num_features, const int64_t* orig_idx,
const int64_t* group_id, const T_int* lin_sorted,
const T_int* group_start, const int64_t* voxel_id, const int num_valid,
const int max_points, const int max_voxels, const int grid_x,
const int grid_y, float* voxels, T_int* coors,
T_int* num_points_per_voxel) {
CUDA_1D_KERNEL_LOOP(i, num_valid) {
const int g = (int)group_id[i];
const int r = i - group_start[g];
if (r >= max_points) continue;
const int64_t sv = voxel_id[g];
if (sv >= max_voxels) continue;
const float* src = points + (size_t)orig_idx[i] * num_features;
float* dst = voxels + ((size_t)sv * max_points + r) * num_features;
for (int k = 0; k < num_features; ++k) dst[k] = src[k];
if (r == 0) {
const int size = group_start[g + 1] - group_start[g];
num_points_per_voxel[sv] = size < max_points ? size : max_points;
const T_int lin_id = lin_sorted[i];
const int cz = (int)(lin_id / (grid_y * grid_x));
const int cy = (int)((lin_id % (grid_y * grid_x)) / grid_x);
const int cx = (int)(lin_id % grid_x);
coors[sv * 3 + 0] = cz;
coors[sv * 3 + 1] = cy;
coors[sv * 3 + 2] = cx;
}
}
}

// Threads per block and block cap for the optimized kernels.
constexpr int kV2Threads = 256;
constexpr int kV2MaxBlocks = 4096;
int HardVoxelizeForwardCUDAKernelLauncher(
const at::Tensor &points, at::Tensor &voxels, at::Tensor &coors,
at::Tensor &num_points_per_voxel, const std::vector<float> voxel_size,
const std::vector<float> coors_range, const int max_points,
const int max_voxels, const int NDim = 3) {
at::cuda::CUDAGuard device_guard(points.device());
cudaStream_t stream = at::cuda::getCurrentCUDAStream();
const int num_points = points.size(0);
const int num_features = points.size(1);
if (num_points == 0) return 0;
const float voxel_x = voxel_size[0];
const float voxel_y = voxel_size[1];
const float voxel_z = voxel_size[2];
const float coors_x_min = coors_range[0];
const float coors_y_min = coors_range[1];
const float coors_z_min = coors_range[2];
const float coors_x_max = coors_range[3];
const float coors_y_max = coors_range[4];
const float coors_z_max = coors_range[5];
const int grid_x = round((coors_x_max - coors_x_min) / voxel_x);
const int grid_y = round((coors_y_max - coors_y_min) / voxel_y);
const int grid_z = round((coors_z_max - coors_z_min) / voxel_z);
// The optimized path is a 3-D perfect hash over (z, y, x) and works on
// float32 points only. Everything else (other NDim, other dtypes) falls
// back to the legacy implementation with identical semantics.
if (NDim != 3 || points.scalar_type() != at::kFloat) {
return HardVoxelizeForwardV1CUDAKernelLauncher(
points, voxels, coors, num_points_per_voxel, voxel_size, coors_range,
max_points, max_voxels, NDim);
}
const int64_t grid_total = (int64_t)grid_x * grid_y * grid_z;
TORCH_CHECK(grid_total <= INT32_MAX,
"hard_voxelize: grid dims overflow int32 lin index.");
const auto opts = points.options();
at::Tensor points_contig = points.contiguous();
// K1: per-point linear voxel id (out-of-range -> -1). Float32 arithmetic
// is identical to dynamic_voxelize_kernel, so valid lin values match the
// legacy coors exactly.
at::Tensor lin = at::empty({num_points}, opts.dtype(at::kInt));
{
const int blocks =
std::min(at::cuda::ATenCeilDiv(num_points, kV2Threads), kV2MaxBlocks);
voxel_index_kernel<int><<<blocks, kV2Threads, 0, stream>>>(
points_contig.data_ptr<float>(), num_points, num_features, voxel_x,
voxel_y, voxel_z, coors_x_min, coors_y_min, coors_z_min, grid_x,
grid_y, grid_z, lin.data_ptr<int>());
AT_CUDA_CHECK(cudaGetLastError());
}
// K2a: stable sort by lin. Equal lin keeps the original relative order,
// so the first point of each group is the one that appeared first in the
// input, which is what the legacy determin_voxel_num numbering depends on.
at::Tensor order =
at::argsort(lin, /*stable=*/true, /*dim=*/0, /*descending=*/false);
at::Tensor lin_sorted_all = lin.index({order});
const int64_t num_invalid = (lin_sorted_all < 0).sum().item<int64_t>();
const int64_t num_valid = num_points - num_invalid;
if (num_valid == 0) return 0;
at::Tensor lin_sorted =
lin_sorted_all.slice(0, num_invalid, num_points).contiguous();
at::Tensor orig_idx = order.slice(0, num_invalid, num_points).contiguous();
// K2b: group boundaries (is_start) -> group id via cumsum.
at::Tensor is_start = at::ones({num_valid}, opts.dtype(at::kBool));
if (num_valid > 1) {
is_start.slice(0, 1, num_valid)
.copy_(lin_sorted.slice(0, 1, num_valid) !=
lin_sorted.slice(0, 0, num_valid - 1));
}
at::Tensor group_id = at::cumsum(is_start.to(at::kLong), 0).sub(1);
const int64_t num_groups = is_start.sum().item<int64_t>();

// original-row index of each group head.
at::Tensor group_start = at::empty({num_groups + 1}, opts.dtype(at::kInt));
at::Tensor leader_orig = at::empty({num_groups}, opts.dtype(at::kLong));
{

at::cuda::ATenCeilDiv(static_cast<int>(num_valid) + 1, kV2Threads);
group_meta_lead_kernel<int><<<blocks, kV2Threads, 0, stream>>>(
group_id.data_ptr<int64_t>(), is_start.data_ptr<bool>(),
orig_idx.data_ptr<int64_t>(), (int)num_valid, (int)num_groups,
group_start.data_ptr<int>(), leader_orig.data_ptr<int64_t>());
AT_CUDA_CHECK(cudaGetLastError());
}
// K2d: number voxels by first-occurrence order (ascending leader row id),
// identical to the legacy single-threaded numbering.
at::Tensor vorder = at::argsort(leader_orig, /*stable=*/true, /*dim=*/0,
/*descending=*/false);
at::Tensor voxel_id = at::empty({num_groups}, opts.dtype(at::kLong));
voxel_id.index_copy_(0, vorder,
at::arange(num_groups, opts.dtype(at::kLong)));
// K3: scatter into the pre-allocated (max_voxels-sized) buffers.
const int64_t voxel_num = std::min<int64_t>(num_groups, max_voxels);
if (voxel_num > 0) {
const int blocks = std::min(
at::cuda::ATenCeilDiv(static_cast<int>(num_valid), kV2Threads),
kV2MaxBlocks);
scatter_kernel<int><<<blocks, kV2Threads, 0, stream>>>(
points_contig.data_ptr<float>(), num_features,
orig_idx.data_ptr<int64_t>(), group_id.data_ptr<int64_t>(),
lin_sorted.data_ptr<int>(), group_start.data_ptr<int>(),
voxel_id.data_ptr<int64_t>(), (int)num_valid, max_points, max_voxels,
grid_x, grid_y, voxels.data_ptr<float>(), coors.data_ptr<int>(),
num_points_per_voxel.data_ptr<int>());
AT_CUDA_CHECK(cudaGetLastError());
}
return (int)voxel_num;
}
