339-461: Consider extracting a shared base to reduce near-identical scoped-override classes
ScopedObbDepthOverride and ScopedShadowmaskDepthOverride share ~95% of their logic — the only differences are the PS slot number (17 vs 2), the SRV fallback order, and the "should override" predicate. A single parameterized ScopedDepthSrvOverride class could eliminate the duplication.
♻️ Sketch of a unified scoped override
```
class ScopedDepthSrvOverride
{
public:
    ScopedDepthSrvOverride(ID3D11DeviceContext* ctx, UINT slot,
        bool (*shouldOverride)(),
        ID3D11ShaderResourceView* (*pickSrv)())
        : context(ctx), slot(slot)
    {
        if (!shouldOverride()) return;
        auto* overrideSrv = pickSrv();
        if (!overrideSrv) return;
        context->PSGetShaderResources(slot, 1, &previousSrv);
        if (previousSrv == overrideSrv) {
            if (previousSrv) { previousSrv->Release(); previousSrv = nullptr; }
            return;
        }
        context->PSSetShaderResources(slot, 1, &overrideSrv);
        active = true;
    }
    ~ScopedDepthSrvOverride()
    {
        if (active) context->PSSetShaderResources(slot, 1, &previousSrv);
        if (previousSrv) previousSrv->Release();
    }
    ScopedDepthSrvOverride(const ScopedDepthSrvOverride&) = delete;
    ScopedDepthSrvOverride& operator=(const ScopedDepthSrvOverride&) = delete;
private:
    ID3D11DeviceContext* context = nullptr;
    UINT slot = 0;
    ID3D11ShaderResourceView* previousSrv = nullptr;
    bool active = false;
};
```
