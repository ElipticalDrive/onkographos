I convert your code to SimpleITK and timed it in python:
```
import SimpleITK as sitk
import numpy as np

def rotate_volume(image, angle, view):
    deg_to_rad = np.pi / 180.0

    theta_x = theta_y = theta_z = 0.0
    if view == "Sagittal":
        theta_x = (360 - angle) * deg_to_rad
    elif view == "Coronal":
        theta_y = angle * deg_to_rad
    else:
        theta_z = angle * deg_to_rad

    # Center of rotation (in physical space)
    size = image.GetSize()  # (x, y, z)
    spacing = image.GetSpacing()
    origin = image.GetOrigin()
    direction = image.GetDirection()

    center_index = [s // 2 for s in size]
    center_physical = image.TransformIndexToPhysicalPoint(center_index)

    # Set up the Euler 3D transform
    transform = sitk.Euler3DTransform()
    transform.SetCenter(center_physical)
    transform.SetRotation(theta_x, theta_y, theta_z)

    # Resample
    resampler = sitk.ResampleImageFilter()
    resampler.SetReferenceImage(image)
    resampler.SetTransform(transform)
    resampler.SetDefaultPixelValue(-32767)
    resampler.SetInterpolator(sitk.sitkLinear)
    rotated_image = resampler.Execute(image)
    return rotated_image

# Create a random 512x512x551 sitkInt16 image
np_img = np.random.randint(-32768, 32767, size=(551, 512, 512), dtype=np.int16)
sitk_img = sitk.GetImageFromArray(np_img)
sitk_img = sitk.Cast(sitk_img, sitk.sitkInt16)

# Example usage
%time rotated = rotate_volume(sitk_img, angle=30, view="Coronal")
```

```
CPU times: user 5.26 s, sys: 30.5 ms, total: 5.29 s
Wall time: 642 ms
```
