# Helper function for manual rendering (for small cases)
def manual_render(alpha, rgbs, shift=1, eps=1e-10):
    N = alpha.shape[-2]
    trans = torch.ones_like(alpha)
    for i in range(N):
        prod = 1.0
        for j in range(i):
            prod *= (1 - alpha[j, 0] + eps)
        idx = (i - shift) % N
        if idx < shift:
            trans[i, 0] = 1.0
        else:
            trans[i, 0] = prod
    weights = trans * alpha
    return torch.sum(weights * rgbs, dim=-2)
