44-58: Consider extracting mean calculation to a shared utility.
This manual grid iteration duplicates the pattern used in grid_data_range. Consider extending grid_data_range to optionally return the mean, or create a companion grid_data_stats utility that computes min/max/mean in a single pass.
♻️ Suggested approach
In src/lib/isosurface/types.ts, modify grid_data_range or create a new function:
```
export function grid_data_stats(grid: number[][][]): {
  min: number
  max: number
  mean: number
  abs_max: number
} {
  let min = Infinity
  let max = -Infinity
  let sum = 0
  let count = 0
  for (const plane of grid) {
    for (const row of plane) {
      for (const val of row) {
        if (val < min) min = val
        if (val > max) max = val
        sum += val
        count++
      }
    }
  }
  return {
    min,
    max,
    mean: count > 0 ? sum / count : 0,
    abs_max: Math.max(Math.abs(min), Math.abs(max)),
  }
}
```
