はじめに
前回は空っぽの空をレンダリングした。
今回は球を表示してみよう。
球の数学
球は「中心からの距離が一定」という条件で定義される。
```
球の方程式:
(x - cx)² + (y - cy)² + (z - cz)² = r²

ベクトル表記:
(P - C) · (P - C) = r²

P: 球面上の点
C: 中心
r: 半径
```
レイとの交差
レイの方程式 $P(t) = O + tD$ を球の方程式に代入。
```
(O + tD - C) · (O + tD - C) = r²
```
展開すると2次方程式になる：
```
t²(D·D) + 2t(D·(O-C)) + (O-C)·(O-C) - r² = 0

a = D·D
b = 2 * D·(O-C)
c = (O-C)·(O-C) - r²

判別式: b² - 4ac
```
```
判別式 > 0 → 2点で交差
判別式 = 0 → 1点で接する
判別式 < 0 → 交差しない
```
球クラスの実装
```
struct HitRecord {
    Point3 p;          // 交点
    Vec3 normal;       // 法線
    double t;          // レイのパラメータ
    bool front_face;   // 表面から当たったか
    
    // 法線の向きを設定
    inline void set_face_normal(const Ray& r, const Vec3& outward_normal) {
        front_face = dot(r.direction, outward_normal) < 0;
        normal = front_face ? outward_normal : -outward_normal;
    }
};

class Sphere {
public:
    Point3 center;
    double radius;
    
    Sphere() {}
    Sphere(const Point3& center, double radius) 
        : center(center), radius(radius) {}
    
    bool hit(const Ray& r, double t_min, double t_max, HitRecord& rec) const {
        Vec3 oc = r.origin - center;
        
        auto a = r.direction.length_squared();
        auto half_b = dot(oc, r.direction);
        auto c = oc.length_squared() - radius * radius;
        
        auto discriminant = half_b * half_b - a * c;
        
        if (discriminant < 0) {
            return false;
        }
        
        auto sqrtd = std::sqrt(discriminant);
        
        // 近いほうの交点を探す
        auto root = (-half_b - sqrtd) / a;
        if (root < t_min || t_max < root) {
            root = (-half_b + sqrtd) / a;
            if (root < t_min || t_max < root) {
                return false;
            }
        }
        
        rec.t = root;
        rec.p = r.at(rec.t);
        Vec3 outward_normal = (rec.p - center) / radius;
        rec.set_face_normal(r, outward_normal);
        
        return true;
    }
};
```
なぜ half_b を使う？
普通の判別式は $b^2 - 4ac$ だけど、$b = 2h$ と置くと：
```
(2h)² - 4ac = 4h² - 4ac = 4(h² - ac)
```
4で割れるから、計算が少し効率化できる。
法線の可視化
ヒットした点の法線ベクトルを色として表示してみよう。
```
Color ray_color(const Ray& r) {
    Sphere sphere(Point3(0, 0, -1), 0.5);
    HitRecord rec;
    
    if (sphere.hit(r, 0, infinity, rec)) {
        // 法線を [-1,1] から [0,1] にマッピングして色にする
        return 0.5 * Color(rec.normal.x + 1, rec.normal.y + 1, rec.normal.z + 1);
    }
    
    // 背景
    Vec3 unit_direction = normalize(r.direction);
    auto t = 0.5 * (unit_direction.y + 1.0);
    return (1.0 - t) * Color(1.0, 1.0, 1.0) + t * Color(0.5, 0.7, 1.0);
}
```
これで法線マップ的な画像が出る。カラフルな球が表示されるはず。
平面の交差判定
床や壁を表現するには平面が必要。
平面は「法線と平面上の1点」で定義できる。
```
平面の方程式:
(P - P₀) · N = 0

P: 平面上の任意の点
P₀: 平面上の既知の点
N: 法線ベクトル
```
レイとの交差
```
class Plane {
public:
    Point3 point;   // 平面上の点
    Vec3 normal;    // 法線
    
    Plane(const Point3& point, const Vec3& normal)
        : point(point), normal(normalize(normal)) {}
    
    bool hit(const Ray& r, double t_min, double t_max, HitRecord& rec) const {
        auto denom = dot(normal, r.direction);
        
        // レイが平面と平行な場合
        if (std::abs(denom) < 1e-8) {
            return false;
        }
        
        auto t = dot(point - r.origin, normal) / denom;
        
        if (t < t_min || t > t_max) {
            return false;
        }
        
        rec.t = t;
        rec.p = r.at(t);
        rec.set_face_normal(r, normal);
        
        return true;
    }
};
```
複数オブジェクトの管理
球と平面を同じリストで扱いたい。抽象基底クラスを作ろう。
```
class Hittable {
public:
    virtual ~Hittable() = default;
    
    virtual bool hit(const Ray& r, double t_min, double t_max, 
                     HitRecord& rec) const = 0;
};

class HittableList : public Hittable {
public:
    std::vector<std::shared_ptr<Hittable>> objects;
    
    void add(std::shared_ptr<Hittable> object) {
        objects.push_back(object);
    }
    
    void clear() {
        objects.clear();
    }
    
    bool hit(const Ray& r, double t_min, double t_max, 
             HitRecord& rec) const override {
        HitRecord temp_rec;
        bool hit_anything = false;
        auto closest_so_far = t_max;
        
        for (const auto& object : objects) {
            if (object->hit(r, t_min, closest_so_far, temp_rec)) {
                hit_anything = true;
                closest_so_far = temp_rec.t;
                rec = temp_rec;
            }
        }
        
        return hit_anything;
    }
};
```

```
int main() {
```

```
void write_color(std::ostream& out, Color pixel_color, int samples) {
    auto scale = 1.0 / samples;
    auto r = pixel_color.x * scale;
    auto g = pixel_color.y * scale;
    auto b = pixel_color.z * scale;
    
    int ir = static_cast<int>(256 * clamp(r, 0.0, 0.999));
    int ig = static_cast<int>(256 * clamp(g, 0.0, 0.999));
    int ib = static_cast<int>(256 * clamp(b, 0.0, 0.999));
    
    out << ir << " " << ig << " " << ib << "\n";
}
```

ガンマ補正
人間の目は暗い部分の変化に敏感。
```
void write_color(std::ostream& out, Color pixel_color, int samples) {
    auto scale = 1.0 / samples;
    
    // ガンマ補正（γ=2.0 → sqrtを取る）
    auto r = std::sqrt(pixel_color.x * scale);
    auto g = std::sqrt(pixel_color.y * scale);
    auto b = std::sqrt(pixel_color.z * scale);
    
    int ir = static_cast<int>(256 * clamp(r, 0.0, 0.999));
    int ig = static_cast<int>(256 * clamp(g, 0.0, 0.999));
    int ib = static_cast<int>(256 * clamp(b, 0.0, 0.999));
    
    out << ir << " " << ig << " " << ib << "\n";
}
```
```
ガンマ補正なし: 暗い部分が潰れる
ガンマ補正あり: より自然な見た目
```
三角形の交差判定
より複雑な形状を作るには三角形が必要。
Möller-Trumbore法
効率的な三角形-レイ交差判定アルゴリズム。
```
class Triangle : public Hittable {
public:
    Point3 v0, v1, v2;  // 頂点
    Vec3 normal;
    
    Triangle(const Point3& v0, const Point3& v1, const Point3& v2)
        : v0(v0), v1(v1), v2(v2) 
    {
        // 法線を計算
        normal = normalize(cross(v1 - v0, v2 - v0));
    }
    
    bool hit(const Ray& r, double t_min, double t_max, 
             HitRecord& rec) const override {
        const double EPSILON = 1e-8;
        
        Vec3 edge1 = v1 - v0;
        Vec3 edge2 = v2 - v0;
        Vec3 h = cross(r.direction, edge2);
        double a = dot(edge1, h);
        
        // レイが三角形と平行
        if (std::abs(a) < EPSILON) {
            return false;
        }
        
        double f = 1.0 / a;
        Vec3 s = r.origin - v0;
        double u = f * dot(s, h);
        
        if (u < 0.0 || u > 1.0) {
            return false;
        }
        
        Vec3 q = cross(s, edge1);
        double v = f * dot(r.direction, q);
        
        if (v < 0.0 || u + v > 1.0) {
            return false;
        }
        
        double t = f * dot(edge2, q);
        
        if (t < t_min || t > t_max) {
            return false;
        }
        
        rec.t = t;
        rec.p = r.at(t);
        rec.set_face_normal(r, normal);
        
        return true;
    }
};
```
重心座標 (u, v)
Möller-Trumbore法で計算されるuとvは重心座標。
```
P = (1-u-v) * v0 + u * v1 + v * v2

三角形の内部: u >= 0, v >= 0, u + v <= 1
```
これは後でテクスチャマッピングに使える。
シーンを作る
ここまでの機能で、簡単なシーンを作ってみよう。
```
HittableList random_scene() {
    HittableList world;
    
    // 床
    world.add(std::make_shared<Sphere>(Point3(0, -1000, 0), 1000));
    
    // ランダムな小さい球をたくさん
    for (int a = -11; a < 11; ++a) {
        for (int b = -11; b < 11; ++b) {
            auto center = Point3(
                a + 0.9 * random_double(),
                0.2,
                b + 0.9 * random_double()
            );
            
            world.add(std::make_shared<Sphere>(center, 0.2));
        }
    }
    
    // 大きな球を3つ
    world.add(std::make_shared<Sphere>(Point3(0, 1, 0), 1.0));
    world.add(std::make_shared<Sphere>(Point3(-4, 1, 0), 1.0));
    world.add(std::make_shared<Sphere>(Point3(4, 1, 0), 1.0));
    
    return world;
}
```
