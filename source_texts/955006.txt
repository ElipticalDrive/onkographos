```
// Compute AABB  // ref issue15. a=sum(f*T3*T3) b=-2sum(f*T0*T3) c=sum(f*T0*T0) f=(R**2,R**2,-1) for pix R=3sigma=3
float3 temp_point = {cutoff * cutoff, cutoff * cutoff, -1.0f};
float distance = sumf3(T3 * T3 * temp_point);// a
float3 f = (1 / distance) * temp_point; // f/a
if (distance == 0.0) return false;
point_image = { //-b/2a
		sumf3(f * T0 * T3), // x
		sumf3(f * T1 * T3) // y
	};  
float2 temp = { // c/a
		sumf3(f * T0 * T0),
		sumf3(f * T1 * T1)
	};
float2 half_extend = point_image * point_image - temp;// |x1-x2|/2 = sqrt(b**2-4ac)/2a
extent = sqrtf2(maxf2(1e-4, half_extend));
```

你好，(x1 x2 y1 y2) 就是 AABB , (x1+x2) / 2 就是 x 轴上的中心，|x1-x2| / 2 是 x 轴上的半边长。代码就是对应了数学计算。
