 I copied and transpiled the same sdf code, and tried to replicate just the tree part on my end. With this sphere tracing code: 
 pub fn trace_sdf(
    start: &Vec3,
    ray: &Vec3,
    mut sdf: impl FnMut(&Vec3, usize, f32) -> f32,
    min_step: f32,
    max_step: f32,
    cutoff: f32,
    iter_max: usize,
    mut termination: impl FnMut(&Vec3, f32) -> bool,
) -> (bool, Vec3, usize, bool, f32)
{
    let mut iter_count = 0;
    let mut hit_sdf = false;
    let mut current_pos = start.clone();
    let mut prior_d = f32::MAX;
    let mut terminated = false;
    let mut distance = 0.0001;

    let mut small_step_count = 0;
    while iter_count < iter_max
    {
        current_pos = *start + distance * ray;
        let increment = sdf(&current_pos, iter_count, prior_d);
        distance += increment.clamp(min_step, max_step);
        if increment <= cutoff
        {
            hit_sdf = true;
            break;
        }

        iter_count += 1
    }

    (hit_sdf, current_pos, iter_count, terminated, distance)
}
