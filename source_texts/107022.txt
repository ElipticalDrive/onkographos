High-level description
rendering a volume with outline thickness > 0 freezes the browser on Windows.
Steps to reproduce
Live example: https://codesandbox.io/s/random-noise-outline-mpr-slice-ir58s
Create image data
```
import vtkDataArray from 'vtk.js/Sources/Common/Core/DataArray';
import vtkImageData from 'vtk.js/Sources/Common/DataModel/ImageData';
import vtkVolume from 'vtk.js/Sources/Rendering/Core/Volume';
import vtkVolumeMapper from 'vtk.js/Sources/Rendering/Core/VolumeMapper';
import vtkInteractorStyleMPRSlice from 'vtk.js/Sources/Interaction/Style/InteractorStyleMPRSlice';
import vtkGenericRenderWindow from 'vtk.js/Sources/Rendering/Misc/GenericRenderWindow'
import vtkColorTransferFunction from 'vtk.js/Sources/Rendering/Core/ColorTransferFunction';
import vtkPiecewiseFunction from 'vtk.js/Sources/Common/DataModel/PiecewiseFunction';

function createImageData() {
  const [i,j,k] = [512, 512, 256];

  const imageData = vtkImageData.newInstance();
  const pixelData = new Uint8Array(i*j*k);
  
  // Add noise as labels
  _addRandomLabels(pixelData);
  
  const dataArray = vtkDataArray.newInstance({
    numberOfComponents: 1,
    values: pixelData,
  });

  imageData.getPointData().setScalars(dataArray);
  imageData.setDimensions(i, j, k);
  imageData.setSpacing(1,1,1);

  const mapper = vtkVolumeMapper.newInstance();
  const actor = vtkVolume.newInstance();

  mapper.setInputData(imageData);
  actor.setMapper(mapper);

  return actor;
}

function _addRandomLabels(pixelData) {
  /** Adds random integers 1 to 3 to pixelData array */
  
  for (let i = 0; i < pixelData.length; i++) {
    if (Math.random() > 0.8) {
      const label =  Math.round(Math.random() * 3);
      pixelData[i] = label;
    }
  }
}

function applyLabelRenderOptions(volume) {

  const COLOR_LUT = [
    [1, 1, 0, 0], // label '1' will be red
    [2, 0, 1, 0], // label '2' will be green
    [3, 0, 0, 1], // label '3' will be blue
  ]

  const cfun = vtkColorTransferFunction.newInstance();
  const ofun = vtkPiecewiseFunction.newInstance();

  // Black transparent, and labels 1, 2 and 3 opaque
  ofun.addPointLong(0, 0, 0.5, 1.0);
  COLOR_LUT.forEach(([label, r, g, b]) => {
    cfun.addRGBPoint(label, r, g, b);
    ofun.addPointLong(label, 1, 0.5, 1.0);
  });

  volume.getProperty().setRGBTransferFunction(0, cfun);
  volume.getProperty().setScalarOpacity(0, ofun);
  volume.getProperty().setInterpolationTypeToNearest();
  
  // These lines cause issues.
  volume.getProperty().setUseLabelOutline(true);
  volume.getProperty().setLabelOutlineThickness(1);
}
```
Setting up the render window
```
function createRenderWindow(container) {
  const genericRenderWindow = vtkGenericRenderWindow.newInstance({
    background: [0, 0, 0]
  });

  genericRenderWindow.setContainer(container);
  genericRenderWindow.resize();
  const renderer = genericRenderWindow.getRenderer();
  const renderWindow = genericRenderWindow.getRenderWindow();
  
  return { renderer, renderWindow, genericRenderWindow }
}

function scaleImage(volume, camera) {
  const bounds = volume.getMapper().getBounds();
  const width = bounds[1] - bounds[0];
  camera.setParallelScale(width / 2.0);
}

/** Setup html */
const root = document.getElementById("root");
const div = document.createElement("div");  
div.style = 'width:640px;height:480px';
root.appendChild(div);

/** Prepare vtk imageData */
const volume = createImageData();
applyLabelRenderOptions(volume);

/** Create render pipeline */
const { renderer, renderWindow } = createRenderWindow(div);
renderer.addVolume(volume);

const interactor = renderWindow.getInteractor();
const istyle = vtkInteractorStyleMPRSlice.newInstance();    
interactor.setInteractorStyle(istyle);  
istyle.setVolumeMapper(volume.getMapper());

const camera = renderer.getActiveCamera();
camera.setParallelProjection(true);

scaleImage(volume, camera);
renderWindow.render();
```
Detailed behavior
When rendering a volume with outline thickness > 0, and using vtkInteractorStyleMPRSlice as interactor style, after a minute or so of interacting with the viewport (reslicing and scrolling), only on Windows, RAM usage goes up (I observed up to 2Gbs added) until either the browser freezes, or until closing the browser tab freezes the whole browser (this takes about two minutes until the browser tab is finally closed).
These two lines make the difference. When commented out or when outline thickness is set to 0, no issues occur.
```
volume.getProperty().setUseLabelOutline(true);
volume.getProperty().setLabelOutlineThickness(1);
```
Expected behavior
Adding an outline thickness to labels does not affect RAM usage and/or the ability to interact with the canvas.
Adding an outline thickness does not affect the ability to close the browser tab.
Environment
vtk.js: 17.15.0
OS: Windows 10
Browser: Chrome 90.0.4430.212
