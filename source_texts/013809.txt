The plane intersection is a simple linear problem that doesn't have a super stable solution, so a fallback method such as "use the closest point to the shape from the cell center, clamped to the cell" is necessary. mat3 yaw_pitch_roll(float yaw, float pitch, float roll)
{
    mat3 R = mat3(vec3(cos(yaw), sin(yaw), 0.0), vec3(-sin(yaw), cos(yaw), 0.0), vec3(0.0, 0.0, 1.0));
    mat3 S = mat3(vec3(1.0, 0.0, 0.0), vec3(0.0, cos(pitch), sin(pitch)), vec3(0.0, -sin(pitch), cos(pitch)));
    mat3 T = mat3(vec3(cos(roll), 0.0, sin(roll)), vec3(0.0, 1.0, 0.0), vec3(-sin(roll), 0.0, cos(roll)));

    return R * S * T;
}

#define DIST_MAX 1000.0

vec4 triangle_intersection(vec3 v0, vec3 v1, vec3 v2, vec3 ro, vec3 rd) // based on https://iquilezles.org/articles/intersectors/
{
    vec3 v1v0 = v1 - v0;
    vec3 v2v0 = v2 - v0;
    vec3 rov0 = ro - v0;
    vec3  n = cross(v1v0, v2v0);
    vec3  q = cross(rov0, rd);
    float d = 1.0/dot(rd, n);
    float u = d*dot(-q, v2v0);
    float v = d*dot( q, v1v0);
    float t = d*dot(-n, rov0);
    if(u<0.0 || v<0.0 || (u+v)>1.0) t = DIST_MAX;
    return vec4(normalize(n), t);
}

float torus(vec2 t, vec3 p)
{
    return length(vec2(length(p.xy)-t.x,p.z))-t.y;
}

    //return length(p.xzy*vec3(1,1,2))-5.0; // sphere (anisotropic scale)
    //return torus(vec2(7.0,1.5),p);
    
    //return exp(-dot(p.xy/5.0,p.xy/5.0))-p.z/5.0; // bell curve
/*
    // Lemniscate
    {
        float x = p.x / 5.0;
        float y = length(p.yz) / 5.0;
        //float x = length(p.xy) / 5.0;
        //float y = p.z / 5.0;
        float a = 0.7;

    }

    return p.x*p.x+p.y*p.y-p.z*p.z-25.0*(0.5+0.5*cos(tau*iTime/10.0)); // hyperboloid
    //return min(min(length(p.xy)-2.5, length(p.yz)-2.5), length(p.zx)-2.5); // crossing tubes
    //return sin(p.x)*cos(p.y)+sin(p.y)*cos(p.z)+sin(p.z)*cos(p.x); // gyroid
}

vec4 pass_through(ivec3 v)
{
    return vec4(v, field(vec3(v)));
}

// Linearly interpolate the position where an isosurface cuts
// an edge between two vertices, each with their own scalar value.
vec3 vertex_lerp(vec4 p0, vec4 p1, float isolevel) // adapted from http://paulbourke.net/geometry/polygonise/
{
    float mu = (isolevel - p0.w) / (p1.w - p0.w);
    //return mix(p0.xyz,p1.xyz,mu); // XXX glitches XXX
    if (abs(isolevel - p0.w) < 0.00001)
        return vec3(p0);

    if (abs(isolevel - p1.w) < 0.00001)
        return vec3(p1);

    if (abs(p0.w - p1.w) < 0.00001)
        return vec3(p0);

    return vec3(
        p0.x + mu * (p1.x - p0.x),
        p0.y + mu * (p1.y - p0.y),
        p0.z + mu * (p1.z - p0.z));
}

vec3 field_normal(vec3 p)
{
    //float e = 0.0001;
    float e = 1.0; // very robust for smooth surfaces
    return // symmetric normal -1,+1 (better approximation that the asymmetric variant 0,1)
        normalize
        (
            vec3
            (
                field(p - vec3(e,0,0)) - field(p + vec3(e,0,0)),
                field(p - vec3(0,e,0)) - field(p + vec3(0,e,0)),
                field(p - vec3(0,0,e)) - field(p + vec3(0,0,e))
            )
        );
}

// Source: Polygonising a Scalar Field Using Tetrahedrons
//         Also known as: "3D Contouring", "Marching Tetrahedrons", "Surface Reconstruction"
//         Written by Paul Bourke, May 1994
//         http://paulbourke.net/geometry/polygonise/

/*
   Polygonise a tetrahedron given its vertices within a cube
   This is an alternative algorithm to polygonisegrid.
   It results in a smoother surface but more triangular facets.

                      0
                     /|\
                    3---1
                     \|/
                      2

*/

vec4 combine(vec4 a, vec4 b)
{
    if(a.w < b.w) return a;
    return b;
}

// adds none, one or two triangles to distance-dield
vec4 polygonise(vec4[8] voxels, float isolevel, ivec4 index, vec3 ro, vec3 rd) // adapted from http://paulbourke.net/geometry/polygonise/
{
    // Determine which of the 16 cases we have given which vertices
    // are above or below the isosurface
    uint triindex = 0u;
    if (voxels[index[0]].w < isolevel) triindex |= 1u;
    if (voxels[index[1]].w < isolevel) triindex |= 2u;
    if (voxels[index[2]].w < isolevel) triindex |= 4u;
    if (voxels[index[3]].w < isolevel) triindex |= 8u;

    // TODO: fix CW/CCW triangles
    switch (triindex)
    {
    case 0x00u:
    case 0x0Fu:
        return vec4(vec3(0),DIST_MAX);
    case 0x0Eu:
    case 0x01u:
        return
            triangle_intersection
            (
                vertex_lerp(voxels[index[0]], voxels[index[1]], isolevel),
                vertex_lerp(voxels[index[0]], voxels[index[2]], isolevel),
                vertex_lerp(voxels[index[0]], voxels[index[3]], isolevel),
                ro,
                rd
            );
    case 0x0Du:
    case 0x02u:
        return
            triangle_intersection
            (
                vertex_lerp(voxels[index[1]], voxels[index[0]], isolevel),
                vertex_lerp(voxels[index[1]], voxels[index[3]], isolevel),
                vertex_lerp(voxels[index[1]], voxels[index[2]], isolevel),
                ro,
                rd
            );
    case 0x0Cu:
    case 0x03u:
        return
            combine
            (
                triangle_intersection
                (
                    vertex_lerp(voxels[index[0]], voxels[index[3]], isolevel),
                    vertex_lerp(voxels[index[0]], voxels[index[2]], isolevel),
                    vertex_lerp(voxels[index[1]], voxels[index[3]], isolevel),
                    ro,
                    rd
                ),
                triangle_intersection
                (
                    vertex_lerp(voxels[index[1]], voxels[index[3]], isolevel),
                    vertex_lerp(voxels[index[1]], voxels[index[2]], isolevel),
                    vertex_lerp(voxels[index[0]], voxels[index[2]], isolevel),
                    ro,
                    rd
                )
            );
    case 0x0Bu:
    case 0x04u:
        return
            triangle_intersection
            (
                vertex_lerp(voxels[index[2]], voxels[index[0]], isolevel),
                vertex_lerp(voxels[index[2]], voxels[index[1]], isolevel),
                vertex_lerp(voxels[index[2]], voxels[index[3]], isolevel),
                ro,
                rd
            );
    case 0x0Au:
    case 0x05u:
        return
            combine
            (
                triangle_intersection
                (
                    vertex_lerp(voxels[index[0]], voxels[index[1]], isolevel),
                    vertex_lerp(voxels[index[2]], voxels[index[3]], isolevel),
                    vertex_lerp(voxels[index[0]], voxels[index[3]], isolevel),
                    ro,
                    rd
                ),
                triangle_intersection
                (
                    vertex_lerp(voxels[index[0]], voxels[index[1]], isolevel),
                    vertex_lerp(voxels[index[1]], voxels[index[2]], isolevel),
                    vertex_lerp(voxels[index[2]], voxels[index[3]], isolevel),
                    ro,
                    rd
                )
            );
    case 0x09u:
    case 0x06u:
        return
            combine
            (
                triangle_intersection
                (
                    vertex_lerp(voxels[index[0]], voxels[index[1]], isolevel),
                    vertex_lerp(voxels[index[1]], voxels[index[3]], isolevel),
                    vertex_lerp(voxels[index[2]], voxels[index[3]], isolevel),
                    ro,
                    rd
                ),
                triangle_intersection
                (
                    vertex_lerp(voxels[index[0]], voxels[index[1]], isolevel),
                    vertex_lerp(voxels[index[0]], voxels[index[2]], isolevel),
                    vertex_lerp(voxels[index[2]], voxels[index[3]], isolevel),
                    ro,
                    rd
                )
            );
    case 0x07u:
    case 0x08u:
        return
            triangle_intersection
            (
                vertex_lerp(voxels[index[3]], voxels[index[0]], isolevel),
                vertex_lerp(voxels[index[3]], voxels[index[2]], isolevel),
                vertex_lerp(voxels[index[3]], voxels[index[1]], isolevel),
                ro,
                rd
            );
    }

    return vec4(vec3(0),DIST_MAX);
}

// assuming ro at start of voxel
vec4 triangles(ivec3 v0, vec3 ro, vec3 rd) // adapted from http://paulbourke.net/geometry/polygonise/
{
    vec4 voxels[8];
    ivec3 v1 = v0 + 1;
/*
    for(int k = 0; k < voxels.length(); k++)
    {
        // NOTE: simplified voxel-numbering (binary) compared to article
        ivec3 offset = v0+ivec3((k>>0)&1,(k>>1)&1,(k>>2)&1);
        voxels[k] = vec4(offset, field(vec3(offset)));
    }
*/
    // NOTE: simplified voxel-numbering compared to article
    voxels[0] = pass_through(v0+ivec3(0,0,0));
    voxels[1] = pass_through(v0+ivec3(1,0,0));
    voxels[2] = pass_through(v0+ivec3(0,1,0));
    voxels[3] = pass_through(v0+ivec3(1,1,0));
    voxels[4] = pass_through(v0+ivec3(0,0,1));
    voxels[5] = pass_through(v0+ivec3(1,0,1));
    voxels[6] = pass_through(v0+ivec3(0,1,1));
    voxels[7] = pass_through(v0+ivec3(1,1,1));

    float isolevel = 0.0;

    // speedup skipping full voxels
    if(max(max(max(voxels[0].w,voxels[1].w),max(voxels[2].w,voxels[3].w)),max(max(voxels[4].w,voxels[5].w),max(voxels[6].w,voxels[7].w)))<isolevel)
        return vec4(vec3(0),DIST_MAX);

    // speedup skipping empty voxels
    if(min(min(min(voxels[0].w,voxels[1].w),min(voxels[2].w,voxels[3].w)),min(min(voxels[4].w,voxels[5].w),min(voxels[6].w,voxels[7].w)))>isolevel)
        return vec4(vec3(0),DIST_MAX);

    vec4 result = // NOTE: indices adjusted - the original indices don't work here due to different voxel numbering! (jt)
        combine
        (
            combine
            (
                polygonise(voxels, isolevel, ivec4(0u, 3u, 2u, 6u), ro, rd),
                polygonise(voxels, isolevel, ivec4(0u, 3u, 7u, 6u), ro, rd)
            ),
            combine
            (
                combine
                (
                    polygonise(voxels, isolevel, ivec4(0u, 4u, 7u, 6u), ro, rd),
                    polygonise(voxels, isolevel, ivec4(0u, 7u, 1u, 3u), ro, rd)
                ),
                combine
                (
                    polygonise(voxels, isolevel, ivec4(0u, 7u, 1u, 4u), ro, rd),
                    polygonise(voxels, isolevel, ivec4(5u, 7u, 1u, 4u), ro, rd)
                )
            )
        );

    // return result; // XXX normal signs flipped randomly XXX
    // brute-force fix: flip normal if it points away from viewer
    return dot(rd, result.xyz) > 0.0 ? vec4(-result.xyz,result.w) : result;
}

struct result
{
    vec3 norm;
    float dist;
    vec3 col;
};

result combine(result a, result b)
{
    if(a.dist < b.dist) return a;
    return b;
}

// returns normal & distance
result traverse(vec3 ray_pos, vec3 ray_dir)
{
    result res = result(-ray_dir, DIST_MAX, vec3(0)); // sky

    // dda initialization (four lines) adapted from
    // https://www.shadertoy.com/view/4dX3zl Branchless Voxel Raycasting by fb39ca4
    // implementing https://lodev.org/cgtutor/raycasting.html
    ivec3 voxel_pos = ivec3(floor(ray_pos)); // round down position to get tile position
    vec3 delta_dist = 1.0 / abs(ray_dir); // reciprocal absolute ray direction
    ivec3 ray_step = ivec3(sign(ray_dir)); // sign of ray direction
    vec3 side_dist = (sign(ray_dir) * (vec3(voxel_pos) - ray_pos) + (sign(ray_dir) * 0.5) + 0.5) * delta_dist; // initial distances to nearest walls by dimension
    bvec3 nearest = bvec3(0);

    for(int i = 0; i < MAX_VOXEL_STEPS; i++)
    {
        // dda core (three lines) adapted from
        // https://www.shadertoy.com/view/4dX3zl Branchless Voxel Raycasting by fb39ca4
        // implementing https://lodev.org/cgtutor/raycasting.html
        bvec3 nearest_next = lessThanEqual(side_dist.xyz, min(side_dist.yzx, side_dist.zxy)); // component is true if nearest wall in that dimension
        vec3 side_dist_next = side_dist + vec3(nearest_next) * delta_dist; // sign of ray direction
        ivec3 voxel_pos_next = voxel_pos + ivec3(vec3(nearest_next)) * ray_step; // step to next vpxel in direction of nearest wall

        //if(all(lessThan(abs(voxel_pos), ivec3(SIZE))))
        if(all(greaterThanEqual(voxel_pos,-ivec3(SIZE))) && all(lessThan(voxel_pos,+ivec3(SIZE))))
        {
            vec4 enter = vec4(vec3(nearest)*vec3(-ray_step), length(vec3(nearest) * (side_dist - delta_dist)));
            vec4 leave = vec4(vec3(nearest_next)*vec3(-ray_step), length(vec3(nearest_next) * (side_dist_next - delta_dist)));

            result current = res;
            {
#ifndef BISECTION
                vec4 normdist = triangles(voxel_pos, ray_pos, ray_dir);
#else
                vec4 normdist = vec4(-ray_dir, DIST_MAX);
                {
                    float t0 = enter.w;
                    float t1 = leave.w;
                    float v0 = field(ray_pos + ray_dir * t0);
                    float v1 = field(ray_pos + ray_dir * t1);
                    if(v0*v1 < 0.0)
                    {
                        for(int i = 0; i < 10; i++)
                        {
                            float t = (t0+t1)/2.0;
                            float v = field(ray_pos + ray_dir * t);
                            if(v0*v >= 0.0) { t0 = t; v0 = v; }
                            else { t1 = t; v1 = v; }
                        }
                        normdist.w = (t0+t1)/2.0;
                    }
                }
#endif
#ifdef SMOOTH_NORMALS
                {
                    normdist.xyz = field_normal(ray_pos + ray_dir * normdist.w);
                    // brute-force fix: flip normal if it points away from viewer
                    normdist.xyz = dot(ray_dir, normdist.xyz) > 0.0 ? -normdist.xyz : normdist.xyz;
                }
#endif
                current = result(normdist.xyz, normdist.w, vec3(1));
            }

            if(current.dist > 0.0 && current.dist < DIST_MAX)
            {
/*
                // constrain sub-object to voxel - is this still necessary?
                if(current.dist < enter.w)
                    current = result(enter.xyz, enter.w, current.col);
                else if(current.dist > leave.w)
                    current = result(leave.xyz, leave.w, current.col);
                //else
                //    current = current;
*/
                res = current; // can just assign here for opaque objects...
                //res = combine(res, current); // ...but e.g. transparent objects would need this

                break;
            }
        }

        nearest = nearest_next;
        side_dist = side_dist_next;
        voxel_pos = voxel_pos_next;
    }

    return res;
}

void mainImage(out vec4 o, in vec2 I)
{
    bool demo = all(lessThan(iMouse.xy, vec2(10.0)));
    vec2 R = iResolution.xy;
    I = (I + I - R) / R.y; // concise pixel-position mapping thanks to Fabrice
    float yaw = 2.0 * pi * float(iMouse.x) / float(R.x);
    //float pitch = pi - pi * float(iMouse.y) / float(R.y);
    //float pitch = pi - pi / 2.0 * float(iMouse.y) / float(R.y);
    float pitch = pi - pi * float(iMouse.y) / float(R.y);
    //float pitch = -pi*1.0 - 0.5 * pi * float(iMouse.y) / float(R.y);
    yaw = !demo ? yaw : 2.0 * pi * fract(iTime * 0.01);
    //pitch = !demo ? pitch : 1.0 * pi/2.0;
    pitch = !demo ? pitch : 4.0/3.0 * pi/2.0;

    //vec3 ray_pos = vec3(0.0, 0.0, -12.5);
    vec3 ray_pos = vec3(0.0, 0.0, -25.0);
    vec3 ray_dir = normalize(vec3(I.x, I.y, 2.0));

    mat3 M = yaw_pitch_roll(yaw, pitch, 0.0);

    ray_pos = M * ray_pos;
    ray_dir = M * ray_dir;

    result res = traverse(ray_pos, ray_dir);
    vec3 color = res.col;
    if(res.dist >= 0.0 && res.dist < DIST_MAX)
    {
        vec3 normal = res.norm;
        float dist = res.dist;

        vec3 dst = ray_pos + ray_dir * dist;
        //color *= texture(iChannel0, dst).xyz;
        //color *= vec3(0.5 + 0.5 * checker(dst));
        //color *= 0.5 + 0.5 * normal; // debug normals
        //color *= dot(ray_dir,normal) < 0.0 ? vec3(0,1,0) : vec3(1,0,0); // debug normals

        vec3 light_dir = normalize(vec3(1, 2, 3));
        //vec3 light_dir = normalize(vec3(cos(iTime/5.0), sin(iTime/5.0), 1));
        float diffuse = max(0.0, dot(light_dir, normal));
        //float diffuse = abs(dot(light_dir, normal)); // WORKAROUND for flipped normals
#ifdef SHADOWS
        float brightness = diffuse > 0.0 ? traverse(dst + normal * 0.001, light_dir).dist < DIST_MAX ? 0.0 : 1.0 : 0.0;
#else
        float brightness = 1.0;
#endif
        float specular = pow(max(0.0, dot(normal, normalize(-ray_dir + light_dir))), 50.0);
        vec3 ambient = vec3(0.1);

        color *= ambient + (diffuse + specular) * brightness * 2.0;

        //vec3 fog_color = vec3(0.25, 0.4, 0.5); // fog
        vec3 fog_color = vec3(0); // darken with distance
        color = mix(fog_color, color, exp(-pow(dist / 40.0, 2.0))); // fog for depth impression & to suppress flickering
    }
    else
    {
        color = vec3(0);
    }

    color = tanh(color); // roll-off / soft-clamp overly bright colors
    color = sqrt(color); // approximate gamma
    o = vec4(color, 1);
}
