Hi, I am not sure what you have implemented. if you have modified render_view, you have to then load rgb and depth from the saved file. Anyway if it is not clear, these are my updates in gs2d_mesh_utils.py
```
class GS2DMeshUtils:
    @classmethod
    @torch.no_grad()
    def render_views(cls, model, renderer, cameras: Iterable, bg_color: torch.Tensor,tmp_path):
        rgbmaps = []
        depthmaps = []

        for i, viewpoint_cam in tqdm(enumerate(cameras), total=len(cameras), desc="Rendering RGB and depth maps"):
            render_pkg = renderer(viewpoint_cam, model, bg_color)
            rgb = render_pkg['render']
            # alpha = render_pkg['rend_alpha']
            # normal = torch.nn.functional.normalize(render_pkg['rend_normal'], dim=0)
            depth = render_pkg['surf_depth']
            # depth_normal = render_pkg['surf_normal']
            
            rgb.cpu()
            depth.cpu()
            rgb_path = os.path.join(tmp_path, f"rgb_{i:05d}.pt")
            depth_path = os.path.join(tmp_path, f"depth_{i:05d}.pt")
            torch.save(rgb, rgb_path)
            torch.save(depth, depth_path)
            
            # rgbmaps.append(rgb.cpu())
            # depthmaps.append(depth.cpu())
            # self.alphamaps.append(alpha.cpu())
            # self.normals.append(normal.cpu())
            # self.depth_normals.append(depth_normal.cpu())
        
        # return empty rgbmaps and depthmaps
        return rgbmaps, depthmaps
```
```
@classmethod
    @torch.no_grad()
    def extract_mesh_bounded(
        cls,
        maps,
        cameras,
        tmp_file_path,
        voxel_size=0.004,
        sdf_trunc=0.02,
        depth_trunc=3,
        mask_backgrond=False,
    ):
        """
        Perform TSDF fusion given a fixed depth range, used in the paper.

        voxel_size: the voxel size of the volume
        sdf_trunc: truncation value
        depth_trunc: maximum depth range, should depended on the scene's scales
        mask_backgrond: whether to mask backgroud, only works when the dataset have masks

        return o3d.mesh
        """
        print("Running tsdf volume integration ...")
        print(f'voxel_size: {voxel_size}')
        print(f'sdf_trunc: {sdf_trunc}')
        print(f'depth_truc: {depth_trunc}')

        # rgbmaps, depthmaps = maps

        volume = o3d.pipelines.integration.ScalableTSDFVolume(
            voxel_length=voxel_size,
            sdf_trunc=sdf_trunc,
            color_type=o3d.pipelines.integration.TSDFVolumeColorType.RGB8
        )

        # tmp_file_path = "outputs/tmp"
        
        with tqdm(enumerate(cls.to_cam_open3d(cameras)), total=len(cameras), desc="TSDF integration progress") as t:
            for i, cam_o3d in t:
                rgb = torch.load(os.path.join(tmp_file_path, f"rgb_{i:05d}.pt"))
                depth = torch.load(os.path.join(tmp_file_path, f"depth_{i:05d}.pt"))

                # if we have mask provided, use it
                assert mask_backgrond is False
                # if mask_backgrond and (self.viewpoint_stack[i].gt_alpha_mask is not None):
                #     depth[(self.viewpoint_stack[i].gt_alpha_mask < 0.5)] = 0

                # make open3d rgbd
                rgbd = o3d.geometry.RGBDImage.create_from_color_and_depth(
                    o3d.geometry.Image(np.asarray(np.clip(rgb.permute(1, 2, 0).cpu().numpy(), 0.0, 1.0) * 255, order="C", dtype=np.uint8)),
                    o3d.geometry.Image(np.asarray(depth.permute(1, 2, 0).cpu().numpy(), order="C")),
                    depth_trunc=depth_trunc, convert_rgb_to_intensity=False,
                    depth_scale=1.0
                )

                volume.integrate(rgbd, intrinsic=cam_o3d.intrinsic, extrinsic=cam_o3d.extrinsic)

        mesh = volume.extract_triangle_mesh()
        return mesh
```
Similar change for unbounded mesh extraction also if you are doing that. Make sure that when you call it from internal/entrypoints/gs2d_mesh_extraction, you pass the additional tmp_file_path to the method.
