import java.io.*;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.*;
import javafx.scene.control.Slider;
import javafx.scene.image.ImageView;
import javafx.scene.image.WritableImage;
import javafx.scene.image.PixelWriter;
import javafx.scene.paint.Color;
import javafx.stage.Stage;

public class Main extends Application {
  short cthead[][][]; //store the 3D volume data set
  float grey[][][]; //store the 3D volume data set converted to 0-1 ready to copy to the image
  short min, max; //min/max value in the 3D volume data set
  int currZSlice = 128;
  int currYSlice = 128;
  int currXSlice = 128;

  @Override
  public void start(Stage stage) throws FileNotFoundException {
    stage.setTitle("CThead Viewer");

    try {
      ReadData();
    } catch (IOException e) {
      System.out.println("Error: CThead file not found in the working directory");
      return;
    }

    WritableImage sliceZImage = new WritableImage(256, 256);
    WritableImage sliceYImage = new WritableImage(256, 256);
    WritableImage sliceXImage = new WritableImage(256, 256);

    GetZSlice(currZSlice, sliceZImage);
    GetYSlice(currYSlice, sliceYImage);
    GetXSlice(currXSlice, sliceXImage);

    ImageView viewZ = new ImageView(sliceZImage);
    ImageView viewY = new ImageView(sliceYImage);
    ImageView viewX = new ImageView(sliceXImage);

    WritableImage volumeRenderX = new WritableImage(256, 256);
    WritableImage volumeRenderY = new WritableImage(256, 256);
    WritableImage volumeRenderZ = new WritableImage(256, 256);

    GetXVolumeRender(volumeRenderX);
    GetYVolumeRender(volumeRenderY);
    GetZVolumeRender(volumeRenderZ);

    ImageView volumeXView = new ImageView(volumeRenderX);
    ImageView volumeYView = new ImageView(volumeRenderY);
    ImageView volumeZView = new ImageView(volumeRenderZ);


    // now we will do the same for MIP
    WritableImage MIPZImage = new WritableImage(256, 256);
    GetZMIP(currZSlice, MIPZImage);
    ImageView MIPZView = new ImageView(MIPZImage);

    WritableImage MIPYImage = new WritableImage(256, 256);
    GetYMIP(currYSlice, MIPYImage);
    ImageView MIPYView = new ImageView(MIPYImage);

    WritableImage MIPXImage = new WritableImage(256, 256);
    GetXMIP(currXSlice, MIPXImage);
    ImageView MIPXView = new ImageView(MIPXImage);

    // here we will create sliders for Z, Y, and X slices
    Slider sliderZ = new Slider(0, 255, currZSlice);
    Slider sliderY = new Slider(0, 255, currYSlice);
    Slider sliderX = new Slider(0, 255, currXSlice);

    // here we will create an opacity slider for volume rendering
    Slider opacitySlider = new Slider(0, 100, 100); // min 0, max 100, default 100
    opacitySlider.setShowTickLabels(true);
    opacitySlider.setShowTickMarks(true);
    opacitySlider.setMajorTickUnit(50);
    opacitySlider.setMinorTickCount(5);
    opacitySlider.setBlockIncrement(10);

    opacitySlider.valueProperty().addListener((observable, oldValue, newValue) -> {
      double opacity = newValue.doubleValue() / 100;

      // now we will apply opacity to volume render images
      volumeXView.setOpacity(opacity);
      volumeYView.setOpacity(opacity);
      volumeZView.setOpacity(opacity);
    });

    // listeners for the sliders
    sliderZ.valueProperty().addListener((obs, oldVal, newVal) -> {
      currZSlice = newVal.intValue();
      GetZSlice(currZSlice, sliceZImage);
      GetZMIP(currZSlice, MIPZImage); // Update MIP for Z
    });

    sliderY.valueProperty().addListener((obs, oldVal, newVal) -> {
      currYSlice = newVal.intValue();
      GetYSlice(currYSlice, sliceYImage);
      GetYMIP(currYSlice, MIPYImage); // Update MIP for Y
    });

    sliderX.valueProperty().addListener((obs, oldVal, newVal) -> {
      currXSlice = newVal.intValue();
      GetXSlice(currXSlice, sliceXImage);
      GetXMIP(currXSlice, MIPXImage); // Update MIP for X
    });


    GridPane grid = new GridPane();
    grid.setHgap(10);
    grid.setVgap(10);
    grid.add(sliderZ, 0, 0);
    grid.add(viewZ, 0, 1);
    grid.add(sliderY, 1, 0);
    grid.add(viewY, 1, 1);
    grid.add(sliderX, 2, 0);
    grid.add(viewX, 2, 1);
    grid.add(MIPZView, 0, 2);
    grid.add(MIPYView, 1, 2);
    grid.add(MIPXView, 2, 2);
    grid.add(volumeXView, 2, 3);
    grid.add(volumeYView, 1, 3);
    grid.add(volumeZView, 0, 3);
    grid.add(opacitySlider, 1, 4);

    Scene scene = new Scene(grid, 800, 840);
    stage.setTitle("CT Data Viewer");
    stage.setScene(scene);
    stage.show();

  }

  // update the Z slice
  public void GetZSlice(int slice, WritableImage image) {
    PixelWriter writer = image.getPixelWriter();
    for (int y = 0; y < 256; y++) {
      for (int x = 0; x < 256; x++) {
        float val = grey[slice][y][x];
        writer.setColor(x, y, Color.color(val, val, val));
      }
    }
  }

  // update the Y slice
  public void GetYSlice(int slice, WritableImage image) {
    PixelWriter writer = image.getPixelWriter();

      for (int x = 0; x < 256; x++) {
        float val = grey[z][slice][x];
        writer.setColor(x, z, Color.color(val, val, val));
      }
    }
  }

  // update the X slice

  public void ReadData() throws IOException {
    File file = new File("CThead-256cubed.bin");
    DataInputStream in = new DataInputStream(new BufferedInputStream(new FileInputStream(file)));

    int i, j, k;

    short read;
    int b1, b2;

    cthead = new short[256][256][256];
    grey = new float[256][256][256];
    min = Short.MAX_VALUE;
    max = Short.MIN_VALUE;


    for (k = 0; k < 256; k++) {  // loops over depth (Z-axis)
      for (j = 0; j < 256; j++) {  // loops over rows (Y-axis)
        for (i = 0; i < 256; i++) {  // loops over columns (X-axis)
          b1 = ((int) in.readByte()) & 0xff;
          b2 = ((int) in.readByte()) & 0xff;
          read = (short) ((b2 << 8) | b1);
          if (read < min) min = read;
          if (read > max) max = read;
          cthead[k][j][i] = read;
        }
      }
    }

    System.out.println(min + " " + max); //diagnostic - for CThead-256cubed.bin this should be -1897, 3029
    //(i.e. there are 4927 levels of grey, and now we will normalise them to 0-1 for display purposes
    //I know the min and max already, so I could have put the normalisation in the above loop, but I put it separate here
    for (k = 0; k < 256; k++) {
      for (j = 0; j < 256; j++) {
        for (i = 0; i < 256; i++) {
          grey[k][j][i] = ((float) cthead[k][j][i] - (float) min) / ((float) max - (float) min);
        }
      }
    }
  }

  public void GetZMIP(int slice, WritableImage image) {
    int width = (int) image.getWidth();
    int height = (int) image.getHeight();
    PixelWriter image_writer = image.getPixelWriter();

    for (int y = 0; y < height; y++) {
      for (int x = 0; x < width; x++) {
        float maxIntensity = 0.0f;
        for (int z = 0; z < 256; z++) {
          if (grey[z][y][x] > maxIntensity) {
            maxIntensity = grey[z][y][x];
          }
        }
        image_writer.setColor(x, y, Color.color(maxIntensity, maxIntensity, maxIntensity));
      }
    }
  }

  public void GetZVolumeRender(WritableImage image) {
    PixelWriter writer = image.getPixelWriter();
    for (int y = 0; y < 256; y++) {
      for (int x = 0; x < 256; x++) {
        double alpha_acc = 0.0;
        double r_acc = 0.0, g_acc = 0.0, b_acc = 0.0;

        for (int z = 255; z >= 0; z--) {
          short ctValue = cthead[z][y][x];
          Color color = applyTransferFunction(ctValue);

          double alpha = color.getOpacity();
          double r = color.getRed();
          double g = color.getGreen();
          double b = color.getBlue();

          // here we will apply compositing
          r_acc = r * alpha + r_acc * (1 - alpha);
          g_acc = g * alpha + g_acc * (1 - alpha);
          b_acc = b * alpha + b_acc * (1 - alpha);
          alpha_acc = alpha + alpha_acc * (1 - alpha);

          if (alpha_acc >= 1.0) break; // Early ray termination
        }

        if (alpha_acc > 0) {
          writer.setColor(x, y, new Color(r_acc, g_acc, b_acc, 1));
        } else {
          writer.setColor(x, y, Color.BLACK);
        }
      }
    }
  }

  private Color applyTransferFunction(short voxelValue) {
    if (voxelValue < -300) {
      return new Color(0, 0, 0, 0);
    } else if (voxelValue >= -300 && voxelValue <= 49) {
      return new Color(0.82, 0.49, 0.18, 0.12);
    } else if (voxelValue >= 50 && voxelValue <= 299) {
      return new Color(0, 0, 0, 0);
    } else {
      return new Color(1.0, 1.0, 1.0, 0.8);
    }
  }
