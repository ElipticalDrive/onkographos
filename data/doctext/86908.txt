```
void VoxelChunk::set_voxel(Vector3 voxel_position, Variant data) {
    _voxel_data[voxel_position] = data;

void VoxelChunk::generate_mesh() {
    // init in this method instead of _ready() to avoid interacting with the scene tree for threading purposes
    _init_mesh();

    // cut short if no voxels exist, avoids error
    if(_voxel_data.size() <= 0) {
        return;
    }

    // prevent size = 0
    _voxel_surface_data.resize(Mesh::ArrayType::ARRAY_MAX);

    Face face_list[] = {Face::LEFT, Face::RIGHT, Face::BOTTOM, Face::TOP, Face::BACK, Face::FRONT};
    for(Vector3 voxel_position : _voxel_data.keys()) {
        // checks if faces are covered based on what is in the _voxel_data array
        for(Face face : face_list) {
            if(!_face_has_neighbor(_voxel_data, face, voxel_position)) {
                _add_voxel_face(face, voxel_position, _voxel_data[voxel_position]);
            }
        }
    }

    _commit_array_mesh();
}

ArrayMesh * VoxelChunk::_get_voxel_array_mesh() {
    return cast_to<ArrayMesh>(*_voxel_mesh_instance->get_mesh());
}

void VoxelChunk::clear_voxels() {
    ArrayMesh * voxel_array_mesh = _get_voxel_array_mesh();
    voxel_array_mesh->clear_surfaces();

    // clear all arrays
    
    for(Variant data_arr : Array {_voxel_data, _voxel_surface_data, _voxel_vertices, _voxel_normals, _voxel_colors}) {
        data_arr.clear();
    }
}

void VoxelChunk::_init_mesh() {
    _voxel_mesh_instance = memnew(MeshInstance3D);
    _voxel_mesh_instance->set_mesh(memnew(ArrayMesh));

    // collision
    _voxel_collision_shape = memnew(CollisionShape3D);
    add_child(_voxel_collision_shape);

    // colors TEMP
    Ref<StandardMaterial3D> voxel_standard_material_3d(memnew(StandardMaterial3D));
    voxel_standard_material_3d->set_flag(BaseMaterial3D::Flags::FLAG_ALBEDO_FROM_VERTEX_COLOR, true);
    _voxel_material = voxel_standard_material_3d;

    add_child(_voxel_mesh_instance);
}

void VoxelChunk::_commit_array_mesh() {
    _voxel_surface_data[Mesh::ArrayType::ARRAY_VERTEX] = _voxel_vertices;
    _voxel_surface_data[Mesh::ArrayType::ARRAY_NORMAL] = _voxel_normals;
    _voxel_surface_data[Mesh::ArrayType::ARRAY_COLOR] = _voxel_colors;

    ArrayMesh * voxel_array_mesh = _get_voxel_array_mesh();
    voxel_array_mesh->add_surface_from_arrays(Mesh::PrimitiveType::PRIMITIVE_TRIANGLES, _voxel_surface_data);
    _voxel_mesh_instance->set_surface_override_material(0, _voxel_material);

    // collision
    _voxel_collision_shape->set_shape(_voxel_mesh_instance->get_mesh()->create_trimesh_shape());
}

void VoxelChunk::_add_voxel_face(Face face, Vector3 face_position, Color color) {
    
    Array face_vertex_indices = VOXEL_FACE_VERTEX_INDICES.at(face);
    for(Array triangle : face_vertex_indices) {
        for(int face_index : triangle) {
            Vector3 face_location = VOXEL_VERTEX_LOCATIONS[face_index];
            _voxel_vertices.append(face_location + face_position);
            _voxel_normals.append(VOXEL_FACE_NORMALS.at(face));
            _voxel_colors.append(color);
        }
    }
}

bool VoxelChunk::_face_has_neighbor(Dictionary data, Face face, Vector3 voxel_position) {
    // the function of adding the face normal is checking if there is a voxel adjacent to it
    Vector3 neighbor_position = voxel_position + VOXEL_FACE_NORMALS.at(face);
    if(data.has(neighbor_position)) {
        return true;
    } else {
        return false;
    }
}
```

```
    Dictionary surrounding_chunks = Dictionary();

void VoxelTerrain::_find_surrounding_chunks(Vector3 point, double radius, Dictionary result) {
    result.clear(); // make sure it's empty
    
	int render_diameter = int(radius * 2) / _chunk_size;
	int chunk_range_start = (render_diameter / 2) * -1;
	int chunk_range_end = render_diameter + chunk_range_start;

	// finds the chunks in the bounding sphere of a radius around the point
	for(int x = chunk_range_start; x < chunk_range_end; x += 1) {
		for(int y = chunk_range_start; y < chunk_range_end; y += 1) {
			for(int z = chunk_range_start; z < chunk_range_end; z += 1) {
				Vector3 chunk_possibly_in_radius = Vector3(x, y, z) + _get_chunk_from_point(point, _chunk_size);
				Vector3 chunk_location = chunk_possibly_in_radius * _chunk_size;
				if(chunk_location.distance_to(point) < radius) {
					result[chunk_possibly_in_radius] = 0; // editing result dictionary
				}
			} // for z
		} // for y
	} // for x
}

Vector3 VoxelTerrain::_get_chunk_from_point(Vector3 point, double chunk_size) {
	Vector3 chunk = Vector3(Vector3i(point / chunk_size));
	return chunk;
}

void VoxelTerrain::_find_chunks_to_load(Dictionary surrounding_chunks, Dictionary result) {
	// find chunks that haven't loaded yet inside the render distance
	Dictionary vec_chunks = Dictionary();
    _convert_to_vec_chunks(vec_chunks);
	// get anything in surrounding_chunks that is not in vec_chunks
    _find_chunks_in_a_not_in_b(surrounding_chunks, vec_chunks, result);
}

void VoxelTerrain::_find_chunks_to_remove(Dictionary surrounding_chunks, Dictionary result) {
	// removing chunks outside of render distance
	// format _chunks as dictionary of vectors
	Dictionary vec_chunks = Dictionary();
    _convert_to_vec_chunks(vec_chunks);
	// get anything in vec_chunks that is not in surrounding_chunks
    _find_chunks_in_a_not_in_b(vec_chunks, surrounding_chunks, result);
}

void VoxelTerrain::_find_chunks_in_a_not_in_b(Dictionary a, Dictionary b, Dictionary result) {
    result.clear(); // make sure it's empty
    
	// get anything in a that is not in b
	for(Vector3 chunk : a.keys()) {
		if(!b.has(chunk)) {
			result[chunk] = 0;
		}
	}
}

void VoxelTerrain::_convert_to_vec_chunks(Dictionary result) {
    result.clear(); // make sure it's empty
    
	// format _chunks as dictionary of vectors
	for(Object * chunk_object : _chunks) {
        VoxelChunk * voxel_chunk = cast_to<VoxelChunk>(chunk_object);
		if(voxel_chunk->is_inside_tree()) {
			// HACK getting the chunk index shouldn't be in _get_vec_chunks() and should instead be in
            // the VoxelChunk object itself, but oh whatever
			Vector3 vec_chunk = voxel_chunk->get_global_position() / _chunk_size;
			result[vec_chunk] = 0;
		}
	}
}

void VoxelTerrain::_update_loading_threshold(Node3D * player) {
	Area3D * loading_threshold = _get_player_data(player)->loading_threshold;
	if(!loading_threshold) { // if loading_threshold not instantiated yet
		// creates the threshold and connects its area_exited signal to _on_loading_threshold_exited()
		// creation part
		loading_threshold = memnew(Area3D);
		CollisionShape3D * collision_shape(memnew(CollisionShape3D));
        Ref<SphereShape3D> sphere_shape(memnew(SphereShape3D));
		collision_shape->set_shape(sphere_shape);
		loading_threshold->add_child(collision_shape);

		// connect signal
        Callable signal_callable = Callable(this, "_on_loading_threshold_exited");
		loading_threshold->connect("body_exited", signal_callable.bind(loading_threshold));

		// put into scene tree
		add_child(loading_threshold);
        
        // store in player_data->loading_threshold
        _get_player_data(player)->loading_threshold = loading_threshold;
	}

    // sets the radius
	// what the f**k is this piece of sh**
	cast_to<SphereShape3D>(*cast_to<CollisionShape3D>(loading_threshold->get_child(0))->get_shape())->set_radius(_render_distance * _CHUNK_ZONE_FACTOR);

	// brings to player
	loading_threshold->set_global_position(player->get_global_position());
}

void VoxelTerrain::generate_terrain() {
	_is_generating = true;

bool VoxelTerrain::_sort_by_distance(Vector3 vec1, Vector3 vec2, Vector3 player_chunk) {
    return vec1.distance_to(player_chunk) < vec2.distance_to(player_chunk);
}

void VoxelTerrain::_thread_generate_terrain(Ref<VoxelTerrainPlayerData> player_data) {
	Vector3 current_loading_position = player_data->last_loading_position;
	Array chunks_to_generate = player_data->chunks_to_load.keys();

	// sorts based on distance to player_chunk
	Vector3 player_chunk = _get_chunk_from_point(player_data->last_loading_position, _chunk_size);
    Callable sort_method = Callable(this, "_sort_by_distance");
	chunks_to_generate.sort_custom(sort_method.bind(player_chunk));
    // loops through chunks to generate
	for(Vector3 chunk : chunks_to_generate) {
		// should abort if the loading position changes since that's a new set of chunks to generate
		bool loading_position_changed = player_data->last_loading_position != current_loading_position;
		bool should_exit = !_is_generating || loading_position_changed;

		// abort if needed
		if(should_exit) {
			return ;
		}

		Vector3 chunk_position = chunk * _chunk_size;
		_generate_chunk(chunk_position);
	}
}
```

```
#pragma once

#include <godot_cpp/godot.hpp>
#include <godot_cpp/classes/ref_counted.hpp>
#include <godot_cpp/classes/area3d.hpp>
#include <godot_cpp/classes/shape3d.hpp>
#include <godot_cpp/classes/thread.hpp>

using namespace godot;

class VoxelTerrainPlayerData : public RefCounted {
    GDCLASS(VoxelTerrainPlayerData, RefCounted);
    
public:
    // if the player leaves this area, the chunks will update to follow the player.
    Area3D * loading_threshold;    

    // important so the threads can understand if their chunks to load have changed.
    // REFACTOR NOTICE: On the surface, this member could be replaced by loading_threshold.global_position.
    // However, last_loading_position is used by threads.
    // Replacing it with loading_threshold.global_position is NOT thread safe.
    // Do not repurpose last_loading_position using loading_threshold.global_position. It WILL cause an ERROR.
    Vector3 last_loading_position;

    // the chunks to load within the player's render distance.
    Dictionary chunks_to_load = Dictionary();
    // the chunks to remove outside the player's render distance.
    Dictionary chunks_to_remove = Dictionary();
    
    Ref<Thread> chunk_loading_thread = Ref<Thread>();
    Ref<Thread> chunk_removal_thread = Ref<Thread>();
    
    VoxelTerrainPlayerData();
    ~VoxelTerrainPlayerData();
    void _init();

    void set_threads(Ref<Thread> loading_thread, Ref<Thread> removal_thread);
    
    static void _bind_methods();
};
```
