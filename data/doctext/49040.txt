## 🚀 Key Features

### 1. WebGL Ray Marching & SDF (Signed Distance Fields) Core
- **Mathematical Primitives:** Sphere, Box, Rounded Box, Torus, Octahedron, Mandelbulb 3D Fractal, Infinite Ground Plane.
- **Constructive Solid Geometry (CSG):** Polynomial smooth minimum (`smin`), Boolean Subtraction, Intersection, and Modulo Repetition.
- **Lighting & Shading Pipeline:** Blinn-Phong specular reflection, Screen-Space / Distance-field Ambient Occlusion (AO), Soft Penumbra Shadows ($k \cdot h / t$), and Depth Exponential Fog.

### 2. Constraint-Based Verlet Physics Engine
- **Verlet Numerical Integration:** Time-reversible, unconditionally stable particle dynamics.
- **Elastic & Tearable Cloth:** Grid-based structural, shear, and bend constraints with tear threshold dynamics.
- **Collision Resolution:** Sphere obstacles, ground bounce damping, mouse constraint grabbing and slicing.

### 3. Developer Tooling & Zero-Dependency Architecture
- Pure TypeScript implementation with zero external runtime dependencies.
- Automated unit test suite with 100% pass rate.
- Modern WebGL2/Canvas UI with live performance HUD and parameter sliders.

```
import { RaymarchEngine } from './graphics/raymarching-engine.js';
import { PhysicsEngine } from './physics/physics-engine.js';
import { Vec3 } from './math/vec3.js';
import { VerletParticle } from './physics/verlet-particle.js';
import { DistanceConstraint } from './physics/constraint.js';

window.addEventListener('DOMContentLoaded', () => {
  const raymarchCanvas = document.getElementById('raymarch-canvas') as HTMLCanvasElement;
  const physicsCanvas = document.getElementById('physics-canvas') as HTMLCanvasElement;
  const fpsVal = document.getElementById('fps-val') as HTMLElement;
  const frameVal = document.getElementById('frame-val') as HTMLElement;
  const modeVal = document.getElementById('mode-val') as HTMLElement;

  // Initialize Raymarching Engine
  const raymarch = new RaymarchEngine(raymarchCanvas);
  raymarch.start();

  // Initialize Physics Engine
  const physics = new PhysicsEngine();
  const pCtx = physicsCanvas.getContext('2d');

  function resizePhysics() {
    physicsCanvas.width = physicsCanvas.clientWidth;
    physicsCanvas.height = physicsCanvas.clientHeight;
    physics.bounds.maxX = physicsCanvas.width;
    physics.bounds.maxY = physicsCanvas.height;
  }
  window.addEventListener('resize', resizePhysics);
  resizePhysics();

  function setupCloth() {
    physics.clear();
    const cols = 22;
    const rows = 16;
    const spacing = 22;
    const startX = Math.max(50, (physicsCanvas.width - cols * spacing) / 2);
    const startY = 60;
    physics.createCloth(startX, startY, cols, rows, spacing, true);
    physics.collisionSpheres = [
      { center: new Vec3(physicsCanvas.width / 2, startY + 180, 0), radius: 65 },
    ];
  }

  function setupRagdoll() {
    physics.clear();
    const cx = physicsCanvas.width / 2;
    let prev = new VerletParticle(new Vec3(cx, 80, 0), 1.0, true, 8, '#ff3366');
    physics.addParticle(prev);

    for (let i = 1; i <= 10; i++) {
      const p = new VerletParticle(new Vec3(cx + i * 25, 80, 0), 1.0, false, 6, '#1d9996');
      physics.addParticle(p);
      physics.addConstraint(new DistanceConstraint(prev, p, 25, false));
      prev = p;
    }
  }

  function setupParticles() {
    physics.clear();
    const cx = physicsCanvas.width / 2;
    for (let i = 0; i < 40; i++) {
      const p = new VerletParticle(
        new Vec3(cx + (Math.random() - 0.5) * 300, 100 + Math.random() * 200, 0),
        1.0,
        false,
        8 + Math.random() * 6,
        i % 2 === 0 ? '#1d9996' : '#38efeb'
      );
      p.oldPos.x += (Math.random() - 0.5) * 50;
      physics.addParticle(p);
    }
  }

  setupCloth();

  // Physics animation loop
  let physicsRunning = false;
  let animPhysicsId: number | null = null;

  function renderPhysics() {
    if (!pCtx) return;
    physics.step(0.016);

    pCtx.clearRect(0, 0, physicsCanvas.width, physicsCanvas.height);

    // Draw Obstacles
    for (const sphere of physics.collisionSpheres) {
      pCtx.beginPath();
      pCtx.arc(sphere.center.x, sphere.center.y, sphere.radius, 0, Math.PI * 2);
      pCtx.fillStyle = 'rgba(56, 239, 235, 0.15)';
      pCtx.fill();
      pCtx.strokeStyle = '#38efeb';
      pCtx.lineWidth = 2;
      pCtx.stroke();
    }

    // Draw Constraints (springs/cloth)
    pCtx.beginPath();
    pCtx.strokeStyle = 'rgba(29, 153, 150, 0.6)';
    pCtx.lineWidth = 1.5;
    for (const c of physics.constraints) {
      if (c.broken) continue;
      pCtx.moveTo(c.p1.pos.x, c.p1.pos.y);
      pCtx.lineTo(c.p2.pos.x, c.p2.pos.y);
    }
    pCtx.stroke();

    // Draw Particles
    for (const p of physics.particles) {
      pCtx.beginPath();
      pCtx.arc(p.pos.x, p.pos.y, p.radius, 0, Math.PI * 2);
      pCtx.fillStyle = p.color;
      pCtx.fill();
    }

    animPhysicsId = requestAnimationFrame(renderPhysics);
  }

  // Mouse physics interaction
  let draggedParticle: VerletParticle | null = null;
  physicsCanvas.addEventListener('mousedown', (e) => {
    const rect = physicsCanvas.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;
    const mPos = new Vec3(mx, my, 0);

    if (e.button === 2) {
      // Right click: tear constraints near cursor
      for (const c of physics.constraints) {
        if (c.p1.pos.distanceTo(mPos) < 25 || c.p2.pos.distanceTo(mPos) < 25) {
          c.broken = true;
        }
      }
      return;
    }

    for (const p of physics.particles) {
      if (p.pos.distanceTo(mPos) < 30) {
        draggedParticle = p;
        break;
      }
    }
  });

  physicsCanvas.addEventListener('contextmenu', (e) => e.preventDefault());

  physicsCanvas.addEventListener('mousemove', (e) => {
    if (!draggedParticle) return;
    const rect = physicsCanvas.getBoundingClientRect();
    draggedParticle.pos.x = e.clientX - rect.left;
    draggedParticle.pos.y = e.clientY - rect.top;
    draggedParticle.oldPos.copy(draggedParticle.pos);
  });

  window.addEventListener('mouseup', () => {
    draggedParticle = null;
  });

  // Tab switching
  const tabBtns = document.querySelectorAll('.tab-btn');
  tabBtns.forEach((btn) => {
    btn.addEventListener('click', () => {
      tabBtns.forEach((b) => b.classList.remove('active'));
      document.querySelectorAll('.tab-content').forEach((c) => c.classList.remove('active'));

      btn.classList.add('active');
      const tabKey = (btn as HTMLElement).dataset.tab;
      document.getElementById(`tab-${tabKey}`)?.classList.add('active');

      if (tabKey === 'physics') {
        physicsCanvas.style.display = 'block';
        raymarchCanvas.style.display = 'none';
        modeVal.textContent = 'Verlet Physics Engine (Canvas 2D/3D)';
        physicsRunning = true;
        renderPhysics();
      } else {
        physicsCanvas.style.display = 'none';
        raymarchCanvas.style.display = 'block';
        modeVal.textContent = 'Raymarching WebGL (GLSL Shaders)';
        if (physicsRunning) {
          physicsRunning = false;
          if (animPhysicsId) cancelAnimationFrame(animPhysicsId);
        }
      }
    });
  });

  // Controls bindings
  const sceneSelect = document.getElementById('scene-select') as HTMLSelectElement;
  sceneSelect.addEventListener('change', () => {
    raymarch.params.sceneId = parseInt(sceneSelect.value);
  });

  const stepsSlider = document.getElementById('steps-slider') as HTMLInputElement;
  const stepsLabel = document.getElementById('steps-label') as HTMLElement;
  stepsSlider.addEventListener('input', () => {
    raymarch.params.maxSteps = parseInt(stepsSlider.value);
    stepsLabel.textContent = stepsSlider.value;
  });

  const morphSlider = document.getElementById('morph-slider') as HTMLInputElement;
  const morphLabel = document.getElementById('morph-label') as HTMLElement;
  morphSlider.addEventListener('input', () => {
    raymarch.params.morphParam = parseFloat(morphSlider.value);
    morphLabel.textContent = parseFloat(morphSlider.value).toFixed(2);
  });

  const aoSlider = document.getElementById('ao-slider') as HTMLInputElement;
  const aoLabel = document.getElementById('ao-label') as HTMLElement;
  aoSlider.addEventListener('input', () => {
    raymarch.params.ambientOcclusion = parseFloat(aoSlider.value);
    aoLabel.textContent = parseFloat(aoSlider.value).toFixed(2);
  });

  const colorSelect = document.getElementById('color-select') as HTMLSelectElement;
  colorSelect.addEventListener('change', () => {
    raymarch.params.colorScheme = parseInt(colorSelect.value);
  });

  document.getElementById('reset-cam-btn')?.addEventListener('click', () => {
    raymarch.camera.yaw = 0.4;
    raymarch.camera.pitch = 0.2;
    raymarch.camera.distance = 4.5;
  });

  // Physics presets bindings
  const physPreset = document.getElementById('physics-preset') as HTMLSelectElement;
  physPreset?.addEventListener('change', () => {
    if (physPreset.value === 'cloth') setupCloth();
    else if (physPreset.value === 'ragdoll') setupRagdoll();
    else setupParticles();
  });

  const gravitySlider = document.getElementById('gravity-slider') as HTMLInputElement;
  const gravityLabel = document.getElementById('gravity-label') as HTMLElement;
  gravitySlider?.addEventListener('input', () => {
    physics.gravity.y = parseFloat(gravitySlider.value);
    gravityLabel.textContent = gravitySlider.value;
  });

  const iterSlider = document.getElementById('iter-slider') as HTMLInputElement;
  const iterLabel = document.getElementById('iter-label') as HTMLElement;
  iterSlider?.addEventListener('input', () => {
    physics.constraintIterations = parseInt(iterSlider.value);
    iterLabel.textContent = iterSlider.value;
  });

  const dampingSlider = document.getElementById('damping-slider') as HTMLInputElement;
  const dampingLabel = document.getElementById('damping-label') as HTMLElement;
  dampingSlider?.addEventListener('input', () => {
    physics.damping = parseFloat(dampingSlider.value);
    dampingLabel.textContent = parseFloat(dampingSlider.value).toFixed(3);
  });

  document.getElementById('spawn-obstacle-btn')?.addEventListener('click', () => {
    physics.collisionSpheres.push({
      center: new Vec3(
        100 + Math.random() * (physicsCanvas.width - 200),
        150 + Math.random() * (physicsCanvas.height - 300),
        0
```

### src/graphics/raymarching-engine.ts
```
export interface RaymarchParams {
  sceneId: number;
  maxSteps: number;
  surfDist: number;
  maxDist: number;
  lightPos: [number, number, number];
  ambientOcclusion: number;
  softShadows: boolean;
  roughness: number;
  morphParam: number;
  colorScheme: number;
}

export class RaymarchEngine {
  private canvas: HTMLCanvasElement;
  private gl: WebGLRenderingContext | null = null;
  private program: WebGLProgram | null = null;
  private animFrameId: number | null = null;
  private startTime: number = Date.now();

  public params: RaymarchParams = {
    sceneId: 0,
    maxSteps: 96,
    surfDist: 0.001,
    maxDist: 40.0,
    lightPos: [3.0, 5.0, -3.0],
    ambientOcclusion: 0.7,
    softShadows: true,
    roughness: 0.4,
    morphParam: 1.0,
    colorScheme: 0,
  };

  public camera = {
    yaw: 0.4,
    pitch: 0.2,
    distance: 4.5,
    target: [0, 0, 0],
  };

  public metrics = {
    fps: 0,
    frameTimeMs: 0,
    stepCountAvg: 0,
  };

  private lastTime: number = performance.now();
  private frameCount: number = 0;
  private fpsTimer: number = 0;

  constructor(canvas: HTMLCanvasElement) {
    this.canvas = canvas;
    this.initGL();
    this.setupEvents();
  }

  private initGL(): void {
    const gl = this.canvas.getContext('webgl', { powerPreference: 'high-performance' });
    if (!gl) {
      console.warn('WebGL not supported');
      return;
    }
    this.gl = gl;

    const vsSource = `
      attribute vec2 a_position;
      varying vec2 v_uv;
      void main() {
        v_uv = a_position * 0.5 + 0.5;
        gl_Position = vec4(a_position, 0.0, 1.0);
      }
    `;

    const fsSource = `
      precision highp float;
      varying vec2 v_uv;
      uniform vec2 u_resolution;
      uniform float u_time;
      uniform int u_scene;
      uniform int u_max_steps;
      uniform float u_surf_dist;
      uniform float u_max_dist;
      uniform vec3 u_light_pos;
      uniform float u_ao;
      uniform float u_morph;
      uniform int u_color_scheme;
      uniform vec3 u_cam_pos;
      uniform vec3 u_cam_target;

      // SDF Primitives
      float sdSphere(vec3 p, float r) {
        return length(p) - r;
      }

      float sdBox(vec3 p, vec3 b) {
        vec3 q = abs(p) - b;
        return length(max(q, 0.0)) + min(max(q.x, max(q.y, q.z)), 0.0);
      }

      float sdTorus(vec3 p, vec2 t) {
        vec2 q = vec2(length(p.xz) - t.x, p.y);
        return length(q) - t.y;
      }

      float sdOctahedron(vec3 p, float s) {
        p = abs(p);
        return (p.x + p.y + p.z - s) * 0.57735027;
      }

      // Smooth operations
      float smin(float a, float b, float k) {
        float h = max(k - abs(a - b), 0.0) / k;
        return min(a, b) - h * h * h * (k * (1.0 / 6.0));
      }

      float smax(float a, float b, float k) {
        return -smin(-a, -b, k);
      }

      // Mandelbulb fractal
      float sdMandelbulb(vec3 pos) {
        vec3 z = pos;
        float dr = 1.0;
        float r = 0.0;
        int iter = 0;
        for (int i = 0; i < 6; i++) {
          r = length(z);
          if (r > 2.0) break;
          float theta = acos(z.z / r);
          float phi = atan(z.y, z.x);
          dr = pow(r, 7.0) * 8.0 * dr + 1.0;
          float zr = pow(r, 8.0);
          theta = theta * 8.0;
          phi = phi * 8.0;
          z = zr * vec3(sin(theta) * cos(phi), sin(phi) * sin(theta), cos(theta));
          z += pos;
          iter++;
        }
        return 0.5 * log(r) * r / dr;
      }

      // Map scene
      vec2 map(vec3 p) {
        float d = 1000.0;
        float matId = 0.0;

        if (u_scene == 0) {
          // Metaball Fusion
          float t = u_time * 0.8;
          vec3 p1 = p - vec3(sin(t) * 0.9, cos(t * 0.7) * 0.6, sin(t * 0.5) * 0.3);
          vec3 p2 = p - vec3(-cos(t * 0.6) * 0.8, -sin(t * 0.8) * 0.7, cos(t * 0.9) * 0.4);
          vec3 p3 = p - vec3(cos(t * 1.1) * 0.5, sin(t * 1.3) * 0.8, -sin(t * 0.6) * 0.5);

          float d1 = sdSphere(p1, 0.65 * u_morph);
          float d2 = sdSphere(p2, 0.55 * u_morph);
          float d3 = sdSphere(p3, 0.48 * u_morph);
          float d4 = sdTorus(p, vec2(1.4, 0.12));

          d = smin(d1, d2, 0.4);
          d = smin(d, d3, 0.4);
          d = smin(d, d4, 0.3);
          matId = 1.0;
        } else if (u_scene == 1) {
          // Mandelbulb Fractal
          d = sdMandelbulb(p * 0.85) / 0.85;
          matId = 2.0;
        } else if (u_scene == 2) {
          // CSG Sculptor
          float dSphere = sdSphere(p, 1.2 * u_morph);
          float dBox = sdBox(p, vec3(0.95));
          float dCross1 = sdBox(p, vec3(0.4, 0.4, 2.0));
          float dCross2 = sdBox(p, vec3(0.4, 2.0, 0.4));
          float dCross3 = sdBox(p, vec3(2.0, 0.4, 0.4));
          
          float crossHole = min(dCross1, min(dCross2, dCross3));
          float sculpted = max(dBox, dSphere);
          d = max(sculpted, -crossHole);
          matId = 3.0;
        } else if (u_scene == 3) {
          // Infinite Crystal Grid
          vec3 q = mod(p + vec3(1.5), 3.0) - vec3(1.5);
          float dOct = sdOctahedron(q, 0.75 * u_morph);
          float dSphere = sdSphere(q, 0.5);
          d = smin(dOct, dSphere, 0.2);
          matId = 4.0;
        } else {
          // Floating Torus & Plane
          float plane = p.y + 1.2;
          float torus = sdTorus(p, vec2(1.1, 0.35 * u_morph));
          float sphere = sdSphere(p - vec3(0.0, sin(u_time * 2.0) * 0.3 + 0.2, 0.0), 0.5);
          d = min(plane, min(torus, sphere));
          matId = p.y < -1.18 ? 5.0 : 6.0;
        }

        return vec2(d, matId);
      }

      vec3 calcNormal(vec3 p) {
        vec2 e = vec2(0.001, 0.0);
        return normalize(vec3(
          map(p + e.xyy).x - map(p - e.xyy).x,
          map(p + e.yxy).x - map(p - e.yxy).x,
          map(p + e.yyx).x - map(p - e.yyx).x
        ));
      }

      float calcAO(vec3 p, vec3 n) {
        float occ = 0.0;
        float sca = 1.0;
        for (int i = 0; i < 5; i++) {
          float h = 0.01 + 0.12 * float(i) / 4.0;
          float d = map(p + h * n).x;
          occ += (h - d) * sca;
          sca *= 0.92;
        }
        return clamp(1.0 - 3.0 * occ * u_ao, 0.0, 1.0);
      }

      float calcSoftShadow(vec3 ro, vec3 rd, float mint, float maxt, float k) {
        float res = 1.0;
        float t = mint;
        for (int i = 0; i < 32; i++) {
          if (t >= maxt) break;
          float h = map(ro + rd * t).x;
          if (h < 0.001) return 0.0;
          res = min(res, k * h / t);
          t += clamp(h, 0.02, 0.25);
        }
        return clamp(res, 0.0, 1.0);
      }

      vec3 getBaseColor(float matId, vec3 p, vec3 n) {
        if (u_color_scheme == 0) {
          // Delos Teal Cyberpunk
          if (matId == 1.0) return vec3(0.11, 0.60, 0.59) + 0.15 * sin(p * 2.0);
          if (matId == 2.0) return vec3(0.15, 0.75, 0.70) * (0.8 + 0.4 * cos(length(p) * 4.0 + u_time));
          if (matId == 3.0) return vec3(0.05, 0.85, 0.75);
          if (matId == 4.0) return vec3(0.2, 0.65, 0.9);
          if (matId == 5.0) return mod(floor(p.x) + floor(p.z), 2.0) == 0.0 ? vec3(0.1, 0.12, 0.15) : vec3(0.18, 0.22, 0.28);
          return vec3(0.95, 0.4, 0.6);
        } else {
          // Sunset Neon / Plasma
          return 0.5 + 0.5 * cos(vec3(0.0, 0.33, 0.67) * 6.28318 + length(p) * 0.8 + u_time * 0.5);
        }
      }

      void main() {
        vec2 uv = (gl_FragCoord.xy - 0.5 * u_resolution.xy) / u_resolution.y;

        // Camera setup
        vec3 ro = u_cam_pos;
        vec3 ta = u_cam_target;
        vec3 ww = normalize(ta - ro);
        vec3 uu = normalize(cross(ww, vec3(0.0, 1.0, 0.0)));
        vec3 vv = normalize(cross(uu, ww));
        vec3 rd = normalize(uv.x * uu + uv.y * vv + 1.5 * ww);

        // Raymarching loop
        float t = 0.01;
        float matId = 0.0;
        int stepsTaken = 0;

        for (int i = 0; i < 128; i++) {
          if (i >= u_max_steps || t > u_max_dist) break;
          vec3 p = ro + rd * t;
          vec2 res = map(p);
          float d = res.x;
          matId = res.y;
          stepsTaken = i;
          if (d < u_surf_dist) break;
          t += d * 0.85; // slightly relaxed for convergence
        }

        vec3 col = vec3(0.03, 0.04, 0.06); // Dark background

        if (t < u_max_dist) {
          vec3 p = ro + rd * t;
          vec3 n = calcNormal(p);
          vec3 l = normalize(u_light_pos - p);
          vec3 v = -rd;
          vec3 h = normalize(l + v);

          vec3 baseCol = getBaseColor(matId, p, n);

          // Diffuse + Specular
          float diff = max(dot(n, l), 0.0);
          float spec = pow(max(dot(n, h), 0.0), 32.0);
          float ao = calcAO(p, n);
          float shadow = calcSoftShadow(p + n * 0.02, l, 0.02, 10.0, 16.0);

          vec3 ambient = vec3(0.08, 0.12, 0.15) * baseCol * ao;
          vec3 diffuse = baseCol * diff * shadow;
          vec3 specular = vec3(1.0) * spec * shadow * 0.8;

          col = ambient + diffuse + specular;

          // Depth fog
          col = mix(col, vec3(0.03, 0.04, 0.06), 1.0 - exp(-0.02 * t * t));
        }

        // Vignette & tone curve
        col *= 0.5 + 0.5 * pow(16.0 * v_uv.x * v_uv.y * (1.0 - v_uv.x) * (1.0 - v_uv.y), 0.25);
        gl_FragColor = vec4(col, 1.0);
      }
    `;

    const vs = this.compileShader(gl.VERTEX_SHADER, vsSource);
    const fs = this.compileShader(gl.FRAGMENT_SHADER, fsSource);
    if (!vs || !fs) return;

    const program = gl.createProgram();
    if (!program) return;
    gl.attachShader(program, vs);
    gl.attachShader(program, fs);
    gl.linkProgram(program);

    if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
      console.error('Program link failed:', gl.getProgramInfoLog(program));
      return;
    }
    this.program = program;

    // Fullscreen triangle/quad buffer
    const posBuffer = gl.createBuffer();
    gl.bindBuffer(gl.ARRAY_BUFFER, posBuffer);
    const positions = new Float32Array([-1, -1, 3, -1, -1, 3]);
    gl.bufferData(gl.ARRAY_BUFFER, positions, gl.STATIC_DRAW);

    const aPos = gl.getAttribLocation(program, 'a_position');
    gl.enableVertexAttribArray(aPos);
    gl.vertexAttribPointer(aPos, 2, gl.FLOAT, false, 0, 0);
  }

  private compileShader(type: number, source: string): WebGLShader | null {
    if (!this.gl) return null;
    const shader = this.gl.createShader(type);
    if (!shader) return null;
    this.gl.shaderSource(shader, source);
    this.gl.compileShader(shader);
    if (!this.gl.getShaderParameter(shader, this.gl.COMPILE_STATUS)) {
      console.error('Shader compile error:', this.gl.getShaderInfoLog(shader));
      this.gl.deleteShader(shader);
      return null;
    }
    return shader;
  }

  private setupEvents(): void {
    let isDragging = false;
    let lastX = 0;
    let lastY = 0;

    this.canvas.addEventListener('mousedown', (e) => {
      isDragging = true;
      lastX = e.clientX;
      lastY = e.clientY;
    });

    window.addEventListener('mouseup', () => {
      isDragging = false;
    });

    this.canvas.addEventListener('mousemove', (e) => {
      if (!isDragging) return;
      const dx = e.clientX - lastX;
      const dy = e.clientY - lastY;
      lastX = e.clientX;
      lastY = e.clientY;

      this.camera.yaw += dx * 0.008;
      this.camera.pitch = Math.max(-1.4, Math.min(1.4, this.camera.pitch + dy * 0.008));
    });

    this.canvas.addEventListener('wheel', (e) => {
      e.preventDefault();
      this.camera.distance = Math.max(1.5, Math.min(20.0, this.camera.distance + e.deltaY * 0.005));
    });
  }

  public render(): void {
    if (!this.gl || !this.program) return;
    const gl = this.gl;

    // Update canvas resolution
    const width = this.canvas.clientWidth;
    const height = this.canvas.clientHeight;
    if (this.canvas.width !== width || this.canvas.height !== height) {
      this.canvas.width = width;
      this.canvas.height = height;
      gl.viewport(0, 0, width, height);
    }

    gl.useProgram(this.program);

    const now = performance.now();
    const dt = now - this.lastTime;
    this.lastTime = now;
    this.frameCount++;
    this.fpsTimer += dt;
    if (this.fpsTimer >= 500) {
      this.metrics.fps = Math.round((this.frameCount * 1000) / this.fpsTimer);
      this.metrics.frameTimeMs = parseFloat((this.fpsTimer / this.frameCount).toFixed(2));
      this.frameCount = 0;
      this.fpsTimer = 0;
    }

    const elapsed = (Date.now() - this.startTime) / 1000.0;

    // Calculate camera position from yaw, pitch, distance
    const cx = this.camera.distance * Math.cos(this.camera.pitch) * Math.sin(this.camera.yaw);
    const cy = this.camera.distance * Math.sin(this.camera.pitch);
    const cz = this.camera.distance * Math.cos(this.camera.pitch) * Math.cos(this.camera.yaw);

    // Uniforms
    gl.uniform2f(gl.getUniformLocation(this.program, 'u_resolution'), width, height);
    gl.uniform1f(gl.getUniformLocation(this.program, 'u_time'), elapsed);
    gl.uniform1i(gl.getUniformLocation(this.program, 'u_scene'), this.params.sceneId);
    gl.uniform1i(gl.getUniformLocation(this.program, 'u_max_steps'), this.params.maxSteps);
    gl.uniform1f(gl.getUniformLocation(this.program, 'u_surf_dist'), this.params.surfDist);
    gl.uniform1f(gl.getUniformLocation(this.program, 'u_max_dist'), this.params.maxDist);
    gl.uniform3fv(gl.getUniformLocation(this.program, 'u_light_pos'), this.params.lightPos);
    gl.uniform1f(gl.getUniformLocation(this.program, 'u_ao'), this.params.ambientOcclusion);
    gl.uniform1f(gl.getUniformLocation(this.program, 'u_morph'), this.params.morphParam);
    gl.uniform1i(gl.getUniformLocation(this.program, 'u_color_scheme'), this.params.colorScheme);
    gl.uniform3f(gl.getUniformLocation(this.program, 'u_cam_pos'), cx, cy, cz);
    gl.uniform3fv(gl.getUniformLocation(this.program, 'u_cam_target'), this.camera.target);

    gl.drawArrays(gl.TRIANGLES, 0, 3);
  }

  public start(): void {
    const loop = () => {
      this.render();
      this.animFrameId = requestAnimationFrame(loop);
    };
    loop();
  }

  public stop(): void {
    if (this.animFrameId !== null) {
      cancelAnimationFrame(this.animFrameId);
      this.animFrameId = null;
    }
  }
}
```

```
import { Vec3 } from '../math/vec3.js';

export interface SDFObject {
  distance(p: Vec3): number;
}

export class SDFSphere implements SDFObject {
  constructor(public center: Vec3, public radius: number) {}

  distance(p: Vec3): number {
    return p.sub(this.center).length() - this.radius;
  }
}

export class SDFBox implements SDFObject {
  constructor(public center: Vec3, public size: Vec3) {}

  distance(p: Vec3): number {
    const q = new Vec3(
      Math.abs(p.x - this.center.x) - this.size.x,
      Math.abs(p.y - this.center.y) - this.size.y,
      Math.abs(p.z - this.center.z) - this.size.z
    );
    const outside = new Vec3(
      Math.max(q.x, 0),
      Math.max(q.y, 0),
      Math.max(q.z, 0)
    ).length();
    const inside = Math.min(Math.max(q.x, Math.max(q.y, q.z)), 0);
    return outside + inside;
  }
}

export class SDFTorus implements SDFObject {
  constructor(public center: Vec3, public r1: number, public r2: number) {}

  distance(p: Vec3): number {
    const local = p.sub(this.center);
    const qx = Math.sqrt(local.x * local.x + local.z * local.z) - this.r1;
    const qy = local.y;
    return Math.sqrt(qx * qx + qy * qy) - this.r2;
  }
}

export class SDFPlane implements SDFObject {
  constructor(public normal: Vec3 = new Vec3(0, 1, 0), public h: number = 0) {}

  distance(p: Vec3): number {
    return p.dot(this.normal) + this.h;
  }
}

export function smin(a: number, b: number, k: number = 0.2): number {
  const h = Math.max(k - Math.abs(a - b), 0) / k;
  return Math.min(a, b) - h * h * h * (k * (1.0 / 6.0));
}

export function smax(a: number, b: number, k: number = 0.2): number {
  return -smin(-a, -b, k);
}

export function sdfUnion(d1: number, d2: number): number {
  return Math.min(d1, d2);
}

export function sdfSubtraction(d1: number, d2: number): number {
  return Math.max(-d1, d2);
}

export function sdfIntersection(d1: number, d2: number): number {
  return Math.max(d1, d2);
}
```

```
export class Vec3 {
  constructor(public x: number = 0, public y: number = 0, public z: number = 0) {}

  static create(x: number = 0, y: number = 0, z: number = 0): Vec3 {
    return new Vec3(x, y, z);
  }

  clone(): Vec3 {
    return new Vec3(this.x, this.y, this.z);
  }

  set(x: number, y: number, z: number): this {
    this.x = x;
    this.y = y;
    this.z = z;
    return this;
  }

  copy(v: Vec3): this {
    this.x = v.x;
    this.y = v.y;
    this.z = v.z;
    return this;
  }

  add(v: Vec3): Vec3 {
    return new Vec3(this.x + v.x, this.y + v.y, this.z + v.z);
  }

  addScaled(v: Vec3, s: number): Vec3 {
    return new Vec3(this.x + v.x * s, this.y + v.y * s, this.z + v.z * s);
  }

  sub(v: Vec3): Vec3 {
    return new Vec3(this.x - v.x, this.y - v.y, this.z - v.z);
  }

  scale(s: number): Vec3 {
    return new Vec3(this.x * s, this.y * s, this.z * s);
  }

  dot(v: Vec3): number {
    return this.x * v.x + this.y * v.y + this.z * v.z;
  }

  cross(v: Vec3): Vec3 {
    return new Vec3(
      this.y * v.z - this.z * v.y,
      this.z * v.x - this.x * v.z,
      this.x * v.y - this.y * v.x
    );
  }

  lengthSq(): number {
    return this.x * this.x + this.y * this.y + this.z * this.z;
  }

  length(): number {
    return Math.sqrt(this.lengthSq());
  }

  normalize(): Vec3 {
    const len = this.length();
    if (len > 0.00001) {
      return this.scale(1 / len);
    }
    return new Vec3(0, 0, 0);
  }

  distanceTo(v: Vec3): number {
    return this.sub(v).length();
  }

  lerp(v: Vec3, t: number): Vec3 {
    return new Vec3(
      this.x + (v.x - this.x) * t,
      this.y + (v.y - this.y) * t,
      this.z + (v.z - this.z) * t
    );
  }

  toArray(): [number, number, number] {
    return [this.x, this.y, this.z];
  }
}
```

### src/physics/constraint.ts
```
import { VerletParticle } from './verlet-particle.js';

export class DistanceConstraint {
  public length: number;
  public broken: boolean = false;
  public tearable: boolean = false;
  public maxStretch: number = 2.5;

  constructor(
    public p1: VerletParticle,
    public p2: VerletParticle,
    length?: number,
    tearable: boolean = false,
    public stiffness: number = 1.0
  ) {
    this.length = length !== undefined ? length : p1.pos.distanceTo(p2.pos);
    this.tearable = tearable;
  }

  solve(): void {
    if (this.broken) return;

    const delta = this.p2.pos.sub(this.p1.pos);
    const dist = delta.length();

    if (dist < 0.0001) return;

    if (this.tearable && dist > this.length * this.maxStretch) {
      this.broken = true;
      return;
    }

    const diff = (dist - this.length) / dist;
    const totalMass = (this.p1.pinned ? 0 : 1 / this.p1.mass) + (this.p2.pinned ? 0 : 1 / this.p2.mass);
    if (totalMass <= 0) return;

    const correction = delta.scale(diff * this.stiffness);

    if (!this.p1.pinned) {
      const scalar1 = (1 / this.p1.mass) / totalMass;
      this.p1.pos = this.p1.pos.addScaled(correction, scalar1);
    }
    if (!this.p2.pinned) {
      const scalar2 = (1 / this.p2.mass) / totalMass;
      this.p2.pos = this.p2.pos.addScaled(correction, -scalar2);
    }
  }
}
```

```
import { Vec3 } from '../math/vec3.js';
import { VerletParticle } from './verlet-particle.js';
import { DistanceConstraint } from './constraint.js';

export class PhysicsEngine {
  public particles: VerletParticle[] = [];
  public constraints: DistanceConstraint[] = [];
  public gravity: Vec3 = new Vec3(0, 980, 0);
  public damping: number = 0.99;
  public constraintIterations: number = 5;
  public bounds: { minX: number; maxX: number; minY: number; maxY: number } = {
    minX: 0,
    maxX: 800,
    minY: 0,
    maxY: 600,
  };
  public collisionSpheres: { center: Vec3; radius: number }[] = [];

  constructor() {}

  addParticle(p: VerletParticle): VerletParticle {
    this.particles.push(p);
    return p;
  }

  addConstraint(c: DistanceConstraint): DistanceConstraint {
    this.constraints.push(c);
    return c;
  }

  createCloth(
    originX: number,
    originY: number,
    cols: number,
    rows: number,
    spacing: number,
    pinRow: boolean = true
  ): void {
    const grid: VerletParticle[][] = [];

    for (let r = 0; r < rows; r++) {
      grid[r] = [];
      for (let c = 0; c < cols; c++) {
        const isPinned = pinRow && r === 0 && (c % 3 === 0 || c === cols - 1);
        const p = new VerletParticle(
          new Vec3(originX + c * spacing, originY + r * spacing, 0),
          1.0,
          isPinned,
          3,
          isPinned ? '#ff3366' : '#1d9996'
        );
        this.addParticle(p);
        grid[r][c] = p;

        // Structural horizontal
        if (c > 0) {
          this.addConstraint(new DistanceConstraint(grid[r][c - 1], p, spacing, true));
        }
        // Structural vertical
        if (r > 0) {
          this.addConstraint(new DistanceConstraint(grid[r - 1][c], p, spacing, true));
        }
        // Shear diagonal
        if (c > 0 && r > 0) {
          this.addConstraint(new DistanceConstraint(grid[r - 1][c - 1], p, spacing * 1.414, true, 0.5));
          this.addConstraint(new DistanceConstraint(grid[r - 1][c], grid[r][c - 1], spacing * 1.414, true, 0.5));
        }
      }
    }
  }

  step(dt: number): void {
    // Apply environmental forces
    for (const p of this.particles) {
      p.applyForce(this.gravity.scale(p.mass));
      p.update(dt, this.damping);

      // Boundary collision
      if (p.pos.y > this.bounds.maxY - p.radius) {
        p.pos.y = this.bounds.maxY - p.radius;
        p.oldPos.y = p.pos.y;
      }
      if (p.pos.y < this.bounds.minY + p.radius) {
        p.pos.y = this.bounds.minY + p.radius;
        p.oldPos.y = p.pos.y;
      }
      if (p.pos.x > this.bounds.maxX - p.radius) {
        p.pos.x = this.bounds.maxX - p.radius;
        p.oldPos.x = p.pos.x;
      }
      if (p.pos.x < this.bounds.minX + p.radius) {
        p.pos.x = this.bounds.minX + p.radius;
        p.oldPos.x = p.pos.x;
      }

      // Sphere obstacle collisions
      for (const sphere of this.collisionSpheres) {
        const d = p.pos.sub(sphere.center);
        const dist = d.length();
        const minDist = sphere.radius + p.radius;
        if (dist < minDist && dist > 0.0001) {
          const normal = d.normalize();
          p.pos = sphere.center.addScaled(normal, minDist);
        }
      }
    }

    // Solve constraints
    for (let i = 0; i < this.constraintIterations; i++) {
      for (const c of this.constraints) {
        c.solve();
      }
    }

    // Clean broken constraints
    this.constraints = this.constraints.filter((c) => !c.broken);
  }

  clear(): void {
    this.particles = [];
    this.constraints = [];
    this.collisionSpheres = [];
  }
}
```

```
import { Vec3 } from '../math/vec3.js';

export class VerletParticle {
  public pos: Vec3;
  public oldPos: Vec3;
  public acc: Vec3;
  public mass: number;
  public pinned: boolean;
  public radius: number;
  public color: string;

  constructor(pos: Vec3, mass: number = 1.0, pinned: boolean = false, radius: number = 4, color: string = '#1d9996') {
    this.pos = pos.clone();
    this.oldPos = pos.clone();
    this.acc = new Vec3(0, 0, 0);
    this.mass = mass;
    this.pinned = pinned;
    this.radius = radius;
    this.color = color;
  }

  applyForce(force: Vec3): void {
    if (this.pinned) return;
    this.acc = this.acc.addScaled(force, 1 / this.mass);
  }

  update(dt: number, damping: number = 0.99): void {
    if (this.pinned) return;
    const velocity = this.pos.sub(this.oldPos).scale(damping);
    this.oldPos = this.pos.clone();
    this.pos = this.pos.add(velocity).addScaled(this.acc, dt * dt);
    this.acc.set(0, 0, 0);
  }
}
```
