+import { invalidate, useThree } from '@react-three/fiber';
+import { useEffect, useRef } from 'react';
+import { MathUtils, PerspectiveCamera } from 'three';
+
+interface Props {
+  initialAzimuth?: number;
+  initialPolar?: number;
+  initialDistance?: number;
+  minDistance?: number;
+  maxDistance?: number;
+}
+
+const ROTATE_SPEED = 0.005;
+const ZOOM_SPEED = 0.001;
+
+function OrbitCamera(props: Props) {
+  const {
+    initialAzimuth = Math.PI / 4,
+    initialPolar = Math.PI / 2.6,
+    initialDistance = 2.4,
+    minDistance = 0.01,
+    maxDistance = 20,
+  } = props;
+
+  const { camera, gl } = useThree();
+  const azimuth = useRef(initialAzimuth);
+  const polar = useRef(initialPolar);
+  const distance = useRef(initialDistance);
+
+  useEffect(() => {
+    if (camera instanceof PerspectiveCamera) {
+      camera.near = 0.001;
+      camera.updateProjectionMatrix();
+    }
+  }, [camera]);
+
+  useEffect(() => {
+    function apply(): void {
+      const a = azimuth.current;
+      const p = polar.current;
+      const d = distance.current;
+      camera.position.set(
+        d * Math.sin(p) * Math.cos(a),
+        d * Math.cos(p),
+        d * Math.sin(p) * Math.sin(a),
+      );
+      camera.lookAt(0, 0, 0);
+      camera.updateProjectionMatrix();
+      invalidate();
+    }
+
+    apply();
+
+    const dom = gl.domElement;
+    let dragging = false;
+    let lastX = 0;
+    let lastY = 0;
+    let activePointerId: number | null = null;
+
+    function onPointerDown(e: PointerEvent): void {
+      if (e.button !== 0) {
+        return;
+      }
+      dragging = true;
+      lastX = e.clientX;
+      lastY = e.clientY;
+      activePointerId = e.pointerId;
+      dom.setPointerCapture(e.pointerId);
+    }
+
+    function onPointerUp(e: PointerEvent): void {
+      if (e.pointerId !== activePointerId) {
+        return;
+      }
+      dragging = false;
+      activePointerId = null;
+      try {
+        dom.releasePointerCapture(e.pointerId);
+      } catch {
+        /* pointer may already be released */
+      }
+    }
+
+    function onPointerMove(e: PointerEvent): void {
+      if (!dragging) {
+        return;
+      }
+      const dx = e.clientX - lastX;
+      const dy = e.clientY - lastY;
+      lastX = e.clientX;
+      lastY = e.clientY;
+      azimuth.current += dx * ROTATE_SPEED;
+      polar.current = MathUtils.clamp(
+        polar.current - dy * ROTATE_SPEED,
+        0.05,
+        Math.PI - 0.05,
+      );
+      apply();
+    }
+
+    function onWheel(e: WheelEvent): void {
+      e.preventDefault();
+      distance.current = MathUtils.clamp(
+        distance.current * Math.exp(e.deltaY * ZOOM_SPEED),
+        minDistance,
+        maxDistance,
+      );
+      apply();
+    }
+
+    dom.addEventListener('pointerdown', onPointerDown);
+    dom.addEventListener('pointerup', onPointerUp);
+    dom.addEventListener('pointercancel', onPointerUp);
+    dom.addEventListener('pointermove', onPointerMove);
+    dom.addEventListener('wheel', onWheel, { passive: false });
+
+    return () => {
+      dom.removeEventListener('pointerdown', onPointerDown);
+      dom.removeEventListener('pointerup', onPointerUp);
+      dom.removeEventListener('pointercancel', onPointerUp);
+      dom.removeEventListener('pointermove', onPointerMove);
+      dom.removeEventListener('wheel', onWheel);
+    };
+  }, [camera, gl, minDistance, maxDistance]);
+
+  return null;
+}
+
+export default OrbitCamera;

+import { type Domain, type ScaleType } from '@h5web/shared/vis-models';
+import { rgb } from 'd3-color';

+import { memo, useMemo } from 'react';

+  DataTexture,
+  GLSL3,
+  RGBAFormat,
+  UnsignedByteType,
+  Vector3,
+} from 'three';
+
+import { type ColorMap } from '../heatmap/models';
+import { getInterpolator, scaleDomain } from '../heatmap/utils';
+import { type VisScaleType } from '../models';
+import { getUniforms } from '../utils';
+import { VolumeRenderMode } from './models';
+import { getVolumeTexture, type VolumeSafeArray } from './utils';
+import {
+  buildVolumeFragmentShader,
+  VOLUME_VERTEX_SHADER,
+} from './volumeShaders';
+
+interface Props {
+  values: NdArray<VolumeSafeArray>;
+  domain: Domain;
+  scaleType: VisScaleType;
+  colorMap: ColorMap;
+  invertColorMap: boolean;
+  mode: VolumeRenderMode;
+  opacityScale: number;
+  steps: number;
+  localScale: [number, number, number];
+}
+
+const CMAP_SIZE = 256;
+
+function VolumeMaterial(props: Props) {
+  const {
+    values,
+    domain,
+    scaleType: visScaleType,
+    colorMap,
+    invertColorMap,
+    mode,
+    opacityScale,
+    steps,
+    localScale,
+  } = props;
+
+  const volumeTexture = useMemo(() => getVolumeTexture(values), [values]);
+
+  const colorMapTexture = useMemo(() => {
+    const interpolator = getInterpolator(colorMap, invertColorMap);
+    const colors = Array.from({ length: CMAP_SIZE }).flatMap((_, i) => {
+      const { r, g, b } = rgb(interpolator(i / (CMAP_SIZE - 1)));
+      return [r, g, b, 255];
+    });
+    const texture = new DataTexture(
+      Uint8Array.from(colors),
+      CMAP_SIZE,
+      1,
+      RGBAFormat,
+      UnsignedByteType,
+    );
+    texture.needsUpdate = true;
+    return texture;
+  }, [colorMap, invertColorMap]);
+
+  const [scaleType, gammaExponent] = Array.isArray(visScaleType)
+    ? visScaleType
+    : [visScaleType as ScaleType, 1];
+
+  const scaledDomain = scaleDomain(domain, scaleType);
+  const range = scaledDomain[1] - scaledDomain[0];
+
+  const fragmentShader = useMemo(
+    () => buildVolumeFragmentShader(scaleType),
+    [scaleType],
+  );
+
+  const args = {
+    uniforms: getUniforms({
+      u_volume: volumeTexture,
+      u_colorMap: colorMapTexture,
+      u_localScale: new Vector3(...localScale),
+      u_min: scaledDomain[0],
+      u_oneOverRange: range === 0 ? 0 : 1 / range,
+      u_gammaExponent: gammaExponent,
+      u_normRevertFactor:
+        values.dtype === 'uint8' || values.dtype === 'uint8_clamped' ? 255 : 1,
+      u_steps: steps,
+      u_mode: mode === VolumeRenderMode.Mip ? 0 : 1,
+      u_opacityScale: opacityScale,
+    }),
+    vertexShader: VOLUME_VERTEX_SHADER,
+    fragmentShader,
+    glslVersion: GLSL3,
+  };
+
+  return (
+    <shaderMaterial
+      args={[args]}
+      side={BackSide}
+      transparent
+      depthWrite={false}
+    />
+  );
+}
+
+export default memo(VolumeMaterial);

+import { type NumArray } from '@h5web/shared/vis-models';
+import { toTypedNdArray } from '@h5web/shared/vis-utils';
+import { type NdArray } from 'ndarray';
+import {
+  Data3DTexture,
+  FloatType,
+  HalfFloatType,
+  LinearFilter,
+  RedFormat,
+  type TextureDataType,
+  UnsignedByteType,
+} from 'three';
+
+export interface VolumeDims {
+  depth: number;
+  rows: number;
+  cols: number;
+}
+
+export type VolumeSafeArray =
+  | Float32Array
+  | Uint8Array
+  | Uint8ClampedArray
+  | Uint16Array;
+
+const TEXTURE_TYPES: Record<string, TextureDataType> = {
+  float32: FloatType,
+  uint16: HalfFloatType, // matches the heatmap convention: uint16 treated as half-float
+  uint8: UnsignedByteType,
+  uint8_clamped: UnsignedByteType,
+};
+
+export function getVolumeDims(arr: NdArray): VolumeDims {
+  if (arr.shape.length !== 3) {
+    throw new Error(
+      `Volume rendering expects a 3-D array (got ${arr.shape.length}-D)`,
+    );
+  }
+  const [depth, rows, cols] = arr.shape;
+  return { depth, rows, cols };
+}
+
+/**
+ * Returns the [x, y, z] mesh scale that preserves the dataset's voxel aspect ratio,
+ * normalized so the largest extent equals 1. Mapping: cols → x, rows → y, depth → z.
+ */
+export function getVolumeBoxScale(dims: VolumeDims): [number, number, number] {
+  const { depth, rows, cols } = dims;
+  const m = Math.max(depth, rows, cols);
+  return [cols / m, rows / m, depth / m];
+}
+
+export function toVolumeSafeNdArray(
+  arr: NdArray<NumArray>,
+): NdArray<VolumeSafeArray> {
+  if (
+    arr.dtype === 'float32' ||
+    arr.dtype === 'uint8' ||
+    arr.dtype === 'uint8_clamped' ||
+    arr.dtype === 'uint16'
+  ) {
+    return arr as NdArray<VolumeSafeArray>;
+  }
+  return toTypedNdArray(arr, Float32Array);
+}
+
+export function getVolumeTexture(arr: NdArray<VolumeSafeArray>): Data3DTexture {
+  const { depth, rows, cols } = getVolumeDims(arr);
+  const texture = new Data3DTexture(arr.data, cols, rows, depth);
+  texture.format = RedFormat;
+  texture.type = TEXTURE_TYPES[arr.dtype] ?? FloatType;
+  texture.minFilter = LinearFilter;
+  texture.magFilter = LinearFilter;
+  texture.unpackAlignment = 1;
+  texture.needsUpdate = true;
+  return texture;
+}

+import { ScaleType } from '@h5web/shared/vis-models';
+
+const IMPL_SCALE: Partial<Record<ScaleType, string>> = {
+  [ScaleType.Log]: 'log(value) * oneOverLog10',
+  [ScaleType.SymLog]: 'sign(value) * log(1. + abs(value)) * oneOverLog10',
+  [ScaleType.Sqrt]: 'sqrt(value)',
+};
+
+const IMPL_IS_SUPPORTED: Partial<Record<ScaleType, string>> = {
+  [ScaleType.Log]: 'value > 0.',
+  [ScaleType.Sqrt]: 'value >= 0.',
+};
+
+export const VOLUME_VERTEX_SHADER = `
+  out vec3 v_localPosition;
+
+  void main() {
+    v_localPosition = position;
+    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
+  }
+`;
+
+/* The shader caps total ray-march iterations to a constant for GLSL portability;
+ * the actual step count is set by the u_steps uniform and clamped by this cap. */
+export const MAX_RAY_STEPS = 512;
+
+export function buildVolumeFragmentShader(scaleType: ScaleType): string {
+  return `
+    precision highp float;
+    precision highp sampler3D;
+
+    uniform sampler3D u_volume;
+    uniform sampler2D u_colorMap;
+    uniform vec3 u_localScale;
+    uniform float u_min;
+    uniform float u_oneOverRange;
+    uniform float u_gammaExponent;
+    uniform float u_normRevertFactor;
+    uniform int u_steps;
+    uniform int u_mode;
+    uniform float u_opacityScale;
+
+    in vec3 v_localPosition;
+    out vec4 fragColor;
+
+    const float oneOverLog10 = 0.43429448190325176;
+
+    bool isSupported(float value) {
+      return ${IMPL_IS_SUPPORTED[scaleType] || 'true'};
+    }
+
+    float scaleVal(float value) {
+      return ${IMPL_SCALE[scaleType] || 'value'};
+    }
+
+    vec2 intersectUnitBox(vec3 origin, vec3 dir) {
+      vec3 invDir = 1.0 / dir;
+      vec3 t0 = (vec3(0.0) - origin) * invDir;
+      vec3 t1 = (vec3(1.0) - origin) * invDir;
+      vec3 tMin = min(t0, t1);
+      vec3 tMax = max(t0, t1);
+      return vec2(
+        max(max(tMin.x, tMin.y), tMin.z),
+        min(min(tMax.x, tMax.y), tMax.z)
+      );
+    }
+
+    void main() {
+      // Local-space (cube-aligned) coordinates in [0, 1]^3 used to sample the 3D texture.
+      vec3 cameraLocal = cameraPosition / u_localScale;
+      vec3 entryUVW = v_localPosition + 0.5;
+      vec3 cameraUVW = cameraLocal + 0.5;
+      vec3 rayDir = normalize(entryUVW - cameraUVW);
+
+      vec2 tHit = intersectUnitBox(cameraUVW, rayDir);
+      float tNear = max(tHit.x, 0.0);
+      float tFar = tHit.y;
+      if (tFar <= tNear) {
+        discard;
+      }
+
+      vec3 start = cameraUVW + rayDir * tNear;
+      float pathLen = tFar - tNear;
+      float dt = pathLen / float(u_steps);
+
+      vec4 acc = vec4(0.0);
+      float maxNorm = -1.0;
+
+      for (int i = 0; i < ${MAX_RAY_STEPS}; ++i) {
+        if (i >= u_steps) {
+          break;
+        }
+        vec3 p = start + rayDir * (dt * (float(i) + 0.5));
+        float raw = texture(u_volume, p).r * u_normRevertFactor;
+        if (isnan(raw) || !isSupported(raw)) {
+          continue;
+        }
+        float n = clamp(u_oneOverRange * (scaleVal(raw) - u_min), 0.0, 1.0);
+        n = pow(n, u_gammaExponent);
+
+        if (u_mode == 0) {
+          if (n > maxNorm) {
+            maxNorm = n;
+          }
+        } else {
+          vec4 src = texture(u_colorMap, vec2(n, 0.5));
+          float alpha = clamp(n * u_opacityScale, 0.0, 1.0);
+          src.rgb *= alpha;
+          src.a = alpha;
+          acc += (1.0 - acc.a) * src;
+          if (acc.a > 0.99) {
+            break;
+          }
+        }
+      }
+
+      if (u_mode == 0) {
+        if (maxNorm < 0.0) {
+          discard;
+        }
+        fragColor = texture(u_colorMap, vec2(maxNorm, 0.5));
+      } else {
+        fragColor = acc;
+      }
+    }
+  `;
+}
