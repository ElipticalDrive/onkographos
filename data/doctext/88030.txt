# VoxelCoreEngine
# 🌍 TerraVox-GL (Custom Voxel Terrain Engine)

A high-performance, custom-built voxel-based terrain engine developed from scratch in **C#** and **(OpenGL)**. This project focuses on exploring modern graphics architecture, efficient GPU-CPU data synchronization, and procedural world-building techniques.

---
## 📖3D Voxel Terrain Generator
* **3D Voxel Based Render : ** this is just an practice using OpenGL to Developing a 3D Terrain Generator for Learning and etc... purpose and you can built your own Terrain



## 🚀 Key Features

* **Procedural Terrain Generation:** Dynamic heightmap and density calculation powered by custom Perlin noise algorithms, featuring adjustable terrain roughness and depth intensity.
* **Two-Pass Biome & Vegetation System:** Intelligent world population pipeline that generates solid terrain foundations (stone, dirt, grass) first, followed by safe structural placement for tree trunks and leaf canopies.
* **Optimized Graphics Pipeline (`std140` UBOs):** Efficient GPU-CPU data synchronization using Uniform Buffer Objects (`std140` layout) to manage Projection and View matrices with minimal overhead.
* **Modern Optimized Architecture(Frustom Culling):** Designed a built-in frustom culling Technique to handle 3D voxel storage efficiently utilizing.
* **Cubemap Texturing:** Implemented multi-face cubemap loading and rendering pipelines for environmental materials and skyboxes.
* **Camera & View Management:** Stable, free-roaming 3D camera controller with real-time VRAM matrix updates during the render loop.

 ## 🧠 Core Methods & Techniques

* **Procedural Noise Pipeline:** Generates organic, continuous terrain heights and structures using custom Perlin noise algorithms.
* **Two-Pass Vegetation System:** Decouples terrain creation (Pass 1: stone/dirt/grass foundation) from decoration (Pass 2: secure tree trunk and leaf placement), preventing chunk boundary corruptions.
* **`std140` Uniform Buffer Objects (UBOs):** Manages GPU-CPU synchronization efficiently by batching Projection and View matrices into aligned memory blocks.

## 🎛️ Key Control Variables

You can tweak the procedural generation and world-building behavior using the following core parameters in the engine:

* **`TERRAIN_ROUGHNESS`**: Controls the overall variance, steepness, and ruggedness of the terrain mountains. Higher values create sharp, dramatic peaks while lower values produce flat plains.
* **`row` & `col`**: Represent the 2D grid coordinates/indices of the chunk generation loops, mapping local block placements across the world space.
* **`depth`**: Determines the height intensity/elevation values derived from noise functions or texture data, acting as the foundation for surface height calculations and safe tree height bounds.

        protected override void OnLoad()
        {
            if (!File.Exists("Textures\\perlin_output.png"))

            Random randomTreeHeightGenerator = new Random();
            Random randomTreeLeavesGenerator = new Random();
            Random randomTreeGenerator = new Random();

            Image<Gray, byte> Image = CvInvoke.Imread("Textures\\perlin_output.png").ToImage<Gray, byte>();

            GL.Enable(EnableCap.DepthTest);
            GL.Enable(EnableCap.CullFace);

                {
                    int column = Image.Height / this._col;
                    int row = Image.Width / this._row;

                    int surfaceHeight = (int)((Image[j * row, i * column].Intensity / 255.0) * TERRAIN_ROUGHNESS);

                    for (int k = 0; k <= surfaceHeight - 1; k++)
                    {
                        Vector3 blockTerrainVector = new(i, k, j);
                        Matrix4 terrainMatrix = Matrix4.CreateScale(1f) *
                                                Matrix4.CreateTranslation(blockTerrainVector);
                        _groundVector3List.Add(blockTerrainVector);
                        _groundTranslationList.Add(terrainMatrix);
                    }

                    Vector3 surfaceVector = new(i, surfaceHeight, j);
                    Matrix4 groundMatrix = Matrix4.CreateScale(1f) *
                                           Matrix4.CreateTranslation(surfaceVector);
                    _surfaceTranslationList.Add(groundMatrix);
                    _surfaceVector3List.Add(surfaceVector);

                    bool canSpawnTree = (i % 7 == 0 && j % 7 == 0);
                    int treeCreatingProbability = randomTreeGenerator.Next(0, 10);

                    if (canSpawnTree && treeCreatingProbability <= 4)
                    {
                        int minHeight = 3;
                        int maxHeight = Math.Clamp(4 + (int)(TERRAIN_ROUGHNESS * 0.1f), 4, 9);

                        if (maxHeight <= minHeight) maxHeight = minHeight + 1;

                        int randomHeight = randomTreeHeightGenerator.Next(minHeight, maxHeight);


                        for (int x = 0; x < randomHeight; x++)
                        {
                            int trunkY = surfaceHeight + 1 + x;
                            Vector3 treeBodyVector = new Vector3(i, trunkY, j);
                            Matrix4 trunkMatrix = Matrix4.CreateScale(1f) *
                                                  Matrix4.CreateTranslation(treeBodyVector);
                            _blockTreeBodiesList.Add(trunkMatrix);
                            _treeBodyVector3List.Add(treeBodyVector);
                        }

                        int leavesStartY = surfaceHeight + 1 + randomHeight;

                        for (int ly = 0; ly < randomTreeLeavesGenerator.Next(1, 4); ly++)
                        {
                            for (int lx = -1; lx <= 1; lx++)
                            {
                                for (int lz = -1; lz <= 1; lz++)
                                {
                                    Vector3 leavesVector = new(i + lx, leavesStartY + ly, j + lz);
                                    Matrix4 leafMatrix = Matrix4.CreateScale(1f) *
                                                         Matrix4.CreateTranslation(leavesVector);
                                    _blockLeavesList.Add(leafMatrix);
                                    _leavesVector3List.Add(leavesVector);

                        }
                    }

                groundBuffer = FilterFrustomView(_groundVector3List);
                surfaceBuffer = FilterFrustomView(_surfaceVector3List);
                leavesBuffer = FilterFrustomView(_leavesVector3List);
                treeBodiesBuffer = FilterFrustomView(_treeBodyVector3List);
                _blockTerrainModel.BindVAO();
                _blockTerrainModel.SetMatrixBuffer(groundBuffer.ToArray(), 4, 3, 4, 1, "ModelMatrices", 4 * Vector4.SizeInBytes, Vector4.SizeInBytes);
                _groundTerrainModel.BindVAO();
                _groundTerrainModel.SetMatrixBuffer(surfaceBuffer.ToArray(), 4, 3, 4, 1, "ModelMatrices", 4 * Vector4.SizeInBytes, Vector4.SizeInBytes);
                _blockLeavesModel.BindVAO();
                _blockLeavesModel.SetMatrixBuffer(leavesBuffer.ToArray(), 4, 3, 4, 1, "ModelMatrices", 4 * Vector4.SizeInBytes, Vector4.SizeInBytes);
                _blockTreeBodiesModel.BindVAO();
                _blockTreeBodiesModel.SetMatrixBuffer(treeBodiesBuffer.ToArray(), 4, 3, 4, 1, "ModelMatrices", 4 * Vector4.SizeInBytes, Vector4.SizeInBytes);
            }

            GL.ClearColor(.78f, .86f, 1.0f, 0f);
            GL.Clear(ClearBufferMask.ColorBufferBit | ClearBufferMask.DepthBufferBit);

            Vector3 cameraPosition = _camera.Position;
            Matrix4 viewMatrix = _camera.GetViewMatrix();
            _projectionBuffer.SetMatrixBuffer(viewMatrix, Vector4.SizeInBytes * 4);
            _projectionBuffer.SetVector3Buffer(_lightDir, Vector4.SizeInBytes * 8);
            _projectionBuffer.SetVector3Buffer(cameraPosition, Vector4.SizeInBytes * 9);

            _modelShader.Use();
            _groundTexture.UseTexture2D();

            _blockTerrainModel.BindVAO();
            GL.DrawArraysInstanced(PrimitiveType.Triangles, 0, _vertices.Length, groundBuffer.Count);


            _groundTerrainModel.BindVAO();
            _terrainTexture.UseCubemap();
            _groundlShader.Use();
            GL.DrawArraysInstanced(PrimitiveType.Triangles, 0, _vertices.Length, surfaceBuffer.Count);


            _blockLeavesModel.BindVAO();
            _grassShader.Use();
            _leavesTexture.UseTexture2D();
            GL.DrawArraysInstanced(PrimitiveType.Triangles, 0, _vertices.Length, leavesBuffer.Count);


            _blockTreeBodiesModel.BindVAO();
            _treeBodyShader.Use();
            _treeBodyTexture.UseTexture2D();

            GL.DrawArraysInstanced(PrimitiveType.Triangles, 0, _vertices.Length, treeBodiesBuffer.Count);

            _lightTranslation = Matrix4.CreateTranslation(_lightDir);
            _lightShader.Use();
            _lightShader.UniformMatrix4("model", _lightTranslation);
            GL.DrawArrays(PrimitiveType.Triangles, 0, _vertices.Length);



            SwapBuffers();

        }

        private List<Matrix4> FilterFrustomView(List<Vector3> translationArray)
        {
            List<Matrix4> translationMatrices = new List<Matrix4>();
            for (int i = 0; i < translationArray.Count; i++)
            {


                if (_camera.CheckFrustomView(translationArray[i]))
                {
                    translationMatrices.Add(Matrix4.CreateTranslation(translationArray[i]));
                }


            }

            return translationMatrices;
        }

                _lightDir += Vector3.UnitX * 5 * (float)args.Time;
            }
            if (keyboardInput.IsKeyDown(OpenTK.Windowing.GraphicsLibraryFramework.Keys.Right))
            {
                _lightDir -= Vector3.UnitX * 5 * (float)args.Time;
            }


        }

            if (input.IsKeyDown(Keys.W) || input.IsKeyDown(Keys.A) || input.IsKeyDown(Keys.D) || input.IsKeyDown(Keys.S) || input.IsKeyDown(Keys.E) || input.IsKeyDown(Keys.Q) || mouse.IsButtonDown(MouseButton.Left))
            {
                CameraMovingState = true;
                CreateFrustumFromCamera();
            }
            else
            {
                CameraMovingState = false;
            }




            if (input.IsKeyDown(Keys.W))
            {
                Position += _front * SPEED * (float)e.Time;

            }
            if (input.IsKeyDown(Keys.A))
            {
                Position -= _right * SPEED * (float)e.Time;

            }
            if (input.IsKeyDown(Keys.D))
            {
                Position += _right * SPEED * (float)e.Time;


            }
            if (input.IsKeyDown(Keys.S))
            {
                Position -= _front * SPEED * (float)e.Time;


            }
            if (input.IsKeyDown(Keys.E))
            {
                Position += _up * SPEED * (float)e.Time;

            }
            if (input.IsKeyDown(Keys.Q))
            {
                Position -= _up * SPEED * (float)e.Time;

            }

            if (mouse.IsButtonDown(MouseButton.Left))
            {

                if (firstMove)
                {
                    lastPos = new Vector2(mouse.X, mouse.Y);
                    firstMove = false;
                }
                else
                {



                    float xDelta = mouse.X - lastPos.X;
                    float yDelta = mouse.Y - lastPos.Y;
                    lastPos = new Vector2(mouse.X, mouse.Y);


                    yaw += xDelta * SENSITIVITY * (float)e.Time;
                    Pitch -= yDelta * SENSITIVITY * (float)e.Time;

                }


            }





        }

        private void UpdateVectors()
        {

            _front.X = (float)(Math.Cos(MathHelper.DegreesToRadians(Pitch)) * Math.Cos(MathHelper.DegreesToRadians(yaw)));
            _front.Y = (float)(Math.Sin(MathHelper.DegreesToRadians(Pitch)));
            _front.Z = (float)(Math.Cos(MathHelper.DegreesToRadians(Pitch)) * Math.Sin(MathHelper.DegreesToRadians(yaw)));

            _front.Normalize();

            _right = Vector3.Cross(_front, Vector3.UnitY);
            _right.Normalize();
            _up = Vector3.Cross(_right, _front);
            _up.Normalize();
        }

        public void Update(KeyboardState input, MouseState mouse, FrameEventArgs e)
        {
            InputController(input, mouse, e);
        }
        void CreateFrustumFromCamera()
        {
            float halfVSide = _far * (float)Math.Tan(MathHelper.DegreesToRadians(_fieldOfView) * .5f);
            float halfHSide = halfVSide * _aspect;
            Vector3 frontMultFar = _far * _front;

            _nearFace = new Face() { Point = (Position + _near * _front), Normal = _front };
            _farFace = new Face()
            {
                Point = Position + frontMultFar,
                Normal = -_front
            };

            _rightFace = new Face()
            {
                Point = Position,
                Normal = Vector3.Cross(frontMultFar - _right * halfHSide, _up)
            };
            _leftFace = new Face()
            {
                Point = Position,
                Normal = Vector3.Cross(_up, frontMultFar + _right * halfHSide)
            };
            _topFace = new Face()
            {
                Point = Position,
                Normal = Vector3.Cross(_right, frontMultFar - _up * halfVSide)
            };
            _bottomFace = new Face()
            {
                Point = Position,
                Normal = Vector3.Cross(frontMultFar + _up * halfVSide, _right)
            };


        }

        public bool CheckFrustomView(Vector3 objectTranslation)
        {

            farDot = Vector3.Dot(_farFace.Normal, objectTranslation - _farFace.Point);
            nearDot = Vector3.Dot(_nearFace.Normal, objectTranslation - _nearFace.Point);
            rightDot = Vector3.Dot(_rightFace.Normal, objectTranslation - _rightFace.Point);
            leftDot = Vector3.Dot(_leftFace.Normal, objectTranslation - _leftFace.Point);
            topDot = Vector3.Dot(_topFace.Normal, objectTranslation - _topFace.Point);
            bottomDot = Vector3.Dot(_bottomFace.Normal, objectTranslation - _bottomFace.Point);
            if (farDot > 0 && nearDot > 0 && rightDot > 0 && leftDot > 0 && topDot > 0 && bottomDot > 0)
            {
                return true;
            }


            return false;
        }
    }

            GL.BufferData(BufferTarget.ArrayBuffer, sizeof(float) * _vertexes.Length, _vertexes, BufferUsageHint.StreamDraw);
            GL.VertexAttribPointer(index, _dimension, VertexAttribPointerType.Float, false, _sizeOfJumpToNextPerMember, _offsetToFirstDimension);
            GL.EnableVertexAttribArray(index);

        }
        public void SetNewData<T2>(T2[] vertices, int index, int dimensionOfVector, int divisorValue, string BufferName, int stride = 0, int offset = 0) where T2 : struct
        {
            if (!_vbos.ContainsKey(BufferName))
            {
                _vbos.Add(BufferName, 0);
                _vbos[BufferName] = GL.GenBuffer();
            }
            GL.BindBuffer(BufferTarget.ArrayBuffer, _vbos[BufferName]);
            GL.BufferData(BufferTarget.ArrayBuffer, sizeof(float) * vertices.Length, vertices, BufferUsageHint.StreamDraw);
            GL.BindVertexArray(_vao);

            SetAttrib(index, dimensionOfVector, stride, offset);

        }
        public void SetMatrixBuffer(Matrix4[] vertices, int divisorLoop, int index, int dimensionOfVector, int divisorValue, string BufferName, int stride = 0, int offset = 0)
        {
            if (!_vbos.ContainsKey(BufferName))
            {
                _vbos.Add(BufferName, 0);
                _vbos[BufferName] = GL.GenBuffer();
            }

            GL.BindBuffer(BufferTarget.ArrayBuffer, _vbos[BufferName]);
            GL.BufferData(BufferTarget.ArrayBuffer, vertices.Length * Vector4.SizeInBytes * 4, vertices, BufferUsageHint.StreamDraw);
            GL.BindVertexArray(_vao);
            for (int i = 0; i < divisorLoop; i++)
            {
                SetAttrib(index + i, dimensionOfVector, stride, offset * i);
                SetDivisor(index + i, divisorValue);

            }

        }

        public void SetAttrib(int index, int dimensionOfVector, int stride, int offset)
        {
            GL.VertexAttribPointer(index, dimensionOfVector, VertexAttribPointerType.Float, false, stride, offset);
            GL.EnableVertexAttribArray(index);
        }

            GL.BindVertexArray(_vao);
            GL.VertexAttribDivisor(index, divisor);
        }
        public void BindVAO()
        {
            GL.BindVertexArray(_vao);
        }

﻿using OpenTK.Graphics.OpenGL4;
using OpenTK.Mathematics;
namespace VoxelCoreEngine.RenderPipeline
{
    internal class Shader
    {
        public int programLocation;

        public Shader(string vertexShaderPath, string fragmentShaderPath)
        {

            int vertexShaderLocation = GL.CreateShader(ShaderType.VertexShader);
            GL.ShaderSource(vertexShaderLocation, ReadShaderSource(vertexShaderPath));
            GL.CompileShader(vertexShaderLocation);
            int fragmentShaderLocation = GL.CreateShader(ShaderType.FragmentShader);
            GL.ShaderSource(fragmentShaderLocation, ReadShaderSource(fragmentShaderPath));

            programLocation = GL.CreateProgram();
            GL.AttachShader(programLocation, vertexShaderLocation);
            GL.AttachShader(programLocation, fragmentShaderLocation);
            GL.LinkProgram(programLocation);
            Use();



            GL.DetachShader(programLocation, vertexShaderLocation);
            GL.DetachShader(programLocation, fragmentShaderLocation);
            GL.DeleteShader(fragmentShaderLocation);
            GL.DeleteShader(vertexShaderLocation);

        }
        public void UniformMatrix4(string name, Matrix4 matrix)
        {
            int index = ExtractLocation(name);
            GL.UniformMatrix4(index, false, ref matrix);
        }

        private int ExtractLocation(string name)
        {
            return GL.GetUniformLocation(programLocation, name);
        }

        public void UniformFloat(string name, float magnitude)
        {
            int index = ExtractLocation(name);
            GL.Uniform1(index, magnitude);
        }
        private string ReadShaderSource(string filePath)
        {
            return File.ReadAllText(filePath);
        }

        internal void Use()
        {
            GL.UseProgram(programLocation);
        }

        internal void UniformVector(string name, Vector3 vector3)
        {
            int index = ExtractLocation(name);
            GL.Uniform3(index, ref vector3);
        }
        public void SetBindingPoint(string uniformMemoryName, int bindingPoint)
        {
            int index_modelShader = GL.GetUniformBlockIndex(programLocation, uniformMemoryName);
            GL.UniformBlockBinding(programLocation, index_modelShader, bindingPoint);

        }
    }
}

        public ImageLoader(string filePath, int width, int height, double scale, int seed = 123)
        {


            Image<Gray, byte> noiseImage = new Image<Gray, byte>(width, height);

            for (int y = 0; y < height; y++)
            {
                for (int x = 0; x < width; x++)
                {
                    double nx = x * scale;
                    double ny = y * scale;

                    double n = (perlin.Compute(nx, ny) + 1.0) * 0.5;

                    byte pixelValue = (byte)(n * 255);

                    noiseImage[y, x] = new Gray(pixelValue);
                }
            }

            noiseImage.Save(filePath);

        }
    }

### NoiseForge/StoragePipeline/PerlinNoise.cs
```
﻿namespace NoiseForge.StoragePipeline
{


    public class PerlinNoise
    {
        private readonly int[] p = new int[512];

        public PerlinNoise(int seed = 0)
        {
            Random rand = new Random(seed);
            int[] permutation = new int[256];
            for (int i = 0; i < 256; i++) permutation[i] = i;

            // به‌هم‌ریختگی جدول با توجه به Seed
            for (int i = 0; i < 256; i++)
            {
                int j = rand.Next(256);
                int temp = permutation[i];
                permutation[i] = permutation[j];
                permutation[j] = temp;
            }

            for (int i = 0; i < 256; i++)
            {
                p[i] = permutation[i];
                p[256 + i] = permutation[i];
            }
        }

        private double Fade(double t) => t * t * t * (t * (t * 6 - 15) + 10);
        private double Lerp(double t, double a, double b) => a + t * (b - a);

        private double Grad(int hash, double x, double y)
        {
            int h = hash & 7;
            double u = h < 4 ? x : y;
            double v = h < 4 ? y : x;
            return ((h & 1) == 0 ? u : -u) + ((h & 2) == 0 ? v : -v);
        }

        public double Compute(double x, double y)
        {
            int X = (int)Math.Floor(x) & 255;
            int Y = (int)Math.Floor(y) & 255;

            x -= Math.Floor(x);
            y -= Math.Floor(y);

            double u = Fade(x);
            double v = Fade(y);

            int A = p[X] + Y;
            int B = p[X + 1] + Y;

            return Lerp(v, Lerp(u, Grad(p[A], x, y), Grad(p[B], x - 1, y)),
                           Lerp(u, Grad(p[A + 1], x, y - 1), Grad(p[B + 1], x - 1, y - 1)));
        }
    }
}
