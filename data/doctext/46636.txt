## GPU Voxelization Algorithm

This repository provides an implementation of the GPU-based voxelization algorithm introduced in:

[**"Fast parallel surface and solid voxelization on GPUs"**](https://doi.org/10.1145/1882261.1866201)

The paper discusses both surface and bulk voxelization. Currently, only surface voxelization is fully implemented,
while bulk voxelization is still a WIP.

The algorithm is implemented in CUDA/C++ and can be found in the ```/mesh_voxelization directory```.

### Clone the Repository

To clone the repository, use the following commands:

```bash
git clone git@gitlab.com:ivandiegorodriguez/gpuvoxelization.git
cd gpuvoxelization
git submodule update --init --recursive
```

### Dependencies

The following dependencies are required to build and run this project:

- **CUDA Toolkit:**
- **GCC Compiler:**

> - Linux with an NVIDIA RTX 4060 GPU. However, it is likely to work on any recent NVIDIA GPU.
> - CUDA Version 12.2

### Build Instructions

Create a build directory and navigate into it:

```bash
mkdir build
cd build
cmake ..
make
```

This process will generate an executable named ```voxelize```, located in the ```build/bin``` directory.


### Usage

You can explore the available options by running:

```bash
build/bin/voxelize --help
```

**Example Command**:

To voxelize a mesh using a specific voxel size (e.g., 0.2), use the following command:

```bash
build/bin/voxelize -s 0.2
```

By default, this will voxelize the ```Turbine.stl``` mesh located at: ```data/InputData/Turbine.stl```
and the voxelized model will be saved to: ```/data/Output```

import json
from enum import Enum
from pathlib import Path
from typing import Optional
from pydantic import BaseModel, Field


class JobStatus(str, Enum):
    PENDING = "pending"
    PROCESSING = "processing"
    COMPLETED = "completed"
    FAILED = "failed"


class VoxelizationJobMetadata(BaseModel):
    """Internal model for job metadata"""
    job_id: Optional[str] = None
    mesh_file_path: Optional[Path] = None
    voxel_file_name: Optional[Path] = None
    voxel_size: Optional[float] = Field(None, gt=0)
    status: JobStatus = JobStatus.PENDING
    error_message: Optional[str] = None
    output_voxel_dir: Optional[Path] = None
    log_info: Optional[str] = None

    class Config:
        from_attributes = True  # Allows conversion from ORM objects

    def to_dict(self) -> dict:
        """Convert to dictionary with string paths"""
        return {
            "job_id": self.job_id,
            "mesh_file_path": str(self.mesh_file_path) if self.mesh_file_path else None,
            "voxel_file_name": str(self.voxel_file_name) if self.voxel_file_name else None,
            "voxel_size": self.voxel_size,
            "status": self.status.value,
            "error_message": self.error_message,
            "output_voxel_dir": str(self.output_voxel_dir) if self.output_voxel_dir else None,
            "log_info": self.log_info,
        }

    @classmethod
    def from_json(cls, json_input: str | dict) -> "VoxelizationJobMetadata":
        """Create from JSON string or dict"""
        if isinstance(json_input, str):
            data = json.loads(json_input)
        elif isinstance(json_input, dict):
            data = json_input
        else:
            raise ValueError("Input must be a JSON string or a dictionary")

        # Convert string paths to Path objects
        if data.get("mesh_file_path"):
            data["mesh_file_path"] = Path(data["mesh_file_path"])
        if data.get("voxel_file_name"):
            data["voxel_file_name"] = Path(data["voxel_file_name"])
        if data.get("output_voxel_dir"):
            data["output_voxel_dir"] = Path(data["output_voxel_dir"])

        return cls(**data)
```

```
import logging
import subprocess
from abc import ABC, abstractmethod
from datetime import datetime
from pathlib import Path
from typing import List

from end_points_voxelizer.config import BINARY_VOXELIZER_EXEC
from shared_utils import VoxelizationJobMetadata
from shared_utils.VoxelizationJobMetadata import JobStatus

logging.basicConfig(
    level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s"
)


class VoxelizationJobStorage(ABC):
    """
    Interface for job storage, e.g. local storage or DB storage.
    """

    @abstractmethod
    def save_job(self, job: VoxelizationJobMetadata):
        pass

    @abstractmethod
    def get_job(self, job_id: str) -> VoxelizationJobMetadata:
        pass

    @abstractmethod
    def get_job_list(self) -> List[str]:
        pass


class CreateVoxelizationJob(VoxelizationJobStorage, ABC):
    """
    Create a voxelization job with a custom VoxelizationJobStorage storage
    """

    @staticmethod
    def _run_voxelization(
        input_stl_path: Path,
        voxel_size: float,
        voxel_file_name: Path,
        output_voxel_dir: Path,
        log_voxelizer_path: Path,
        job: VoxelizationJobMetadata,
    ) -> subprocess.CompletedProcess[str]:
        command = [
            BINARY_VOXELIZER_EXEC,
            "-i",
            str(input_stl_path),
            "-o",
            str(output_voxel_dir) + "/",
            "-s",
            str(voxel_size),
            "-n",
            str(voxel_file_name),
            "-l",
            str(log_voxelizer_path) + "/",
        ]

        try:
            job.status = JobStatus.PROCESSING

            result = subprocess.run(command, capture_output=True, text=True, check=True)

            # Example code
            logging.info("Output:\n%s", result.stdout)
            if result.stderr:
                logging.error("Error:\n%s", result.stderr)

            if result.returncode == 0:
                job.status = JobStatus.COMPLETED
            else:
                job.status = JobStatus.FAILED
                job.error_message = result.stderr

            return result

        except subprocess.CalledProcessError as e:
            job.status = JobStatus.FAILED
            job.error_message = str(e)
            logging.info("Error occurred while running the C++ executable:", e.stderr)

        except FileNotFoundError:
            print("Executable not found. Please check the path and file name.")

    @staticmethod
    def _create_job_id() -> str:
        current_datetime = datetime.now().strftime("%Y%m%d_%H%M%S")
        return f"{current_datetime}"

    @abstractmethod
    def run_voxelization_job(
        self,
        input_stl_path: Path | str,
        voxel_size: float,
        voxel_file_name: Path | str,
    ) -> VoxelizationJobMetadata:
        pass
```

import json
import logging
from pathlib import Path
from typing import List

from end_points_voxelizer.config import VOXELIZER_DATA_PATH
from end_points_voxelizer.create_voxelization_job.create_voxelization_job import (
    CreateVoxelizationJob,
)
from end_points_voxelizer.create_voxelization_job.utils import parse_log_file
from gui_components.controller import LOG_FILE_NAME
from shared_utils.VoxelizationJobMetadata import VoxelizationJobMetadata, JobStatus

logging.basicConfig(level=logging.INFO)


class CreateVoxelizationJobLocalStorage(CreateVoxelizationJob):
    def __init__(self):
        self.job_id = self._create_job_id()
        self.output_voxel_dir_job = self.get_job_folder(self.job_id)
        self.job = VoxelizationJobMetadata()
        self.job.job_id = str(self.job_id)
        self.job.output_voxel_dir = self.output_voxel_dir_job

    def run_voxelization_job(
        self,
        input_stl_path: Path | str,
        voxel_size: float,
        voxel_file_name: Path | str,
    ) -> str:
        input_stl_path = (
            Path(input_stl_path) if isinstance(input_stl_path, str) else input_stl_path
        )
        voxel_file_name = (
            Path(voxel_file_name)
            if isinstance(voxel_file_name, str)
            else voxel_file_name
        )

        self.job.mesh_file_path = input_stl_path
        self.job.voxel_size = voxel_size
        self.job.voxel_file_name = voxel_file_name
        self.job.status = JobStatus.PROCESSING

        result = self._run_voxelization(
            input_stl_path=input_stl_path,
            voxel_size=voxel_size,
            voxel_file_name=voxel_file_name,
            output_voxel_dir=self.output_voxel_dir_job,
            log_voxelizer_path=self.output_voxel_dir_job,
            job=self.job,
        )

        # Store log_info
        log_file_path = str(self.output_voxel_dir_job / LOG_FILE_NAME)
        log_info = parse_log_file(log_file_path)
        self.job.log_info = log_info
        self.save_job(self.job)

        return self.job.job_id

    def save_job(self, job: VoxelizationJobMetadata):
        """Serialize job and save to file as json in local storage."""

        job_dict = job.to_dict()
        job_id = job.job_id

        job_file = self.get_job_folder(job_id) / Path(f"{job_id}.json")

        with job_file.open("w") as file:
            json.dump(job_dict, file, indent=4)
        logging.info(f"Job {job.job_id} saved at {job_file}")

    @staticmethod
    def get_job_folder(job_id: str) -> Path:
        return VOXELIZER_DATA_PATH / Path(f"Job_{job_id}")

    def get_job(self, job_id: str) -> VoxelizationJobMetadata | None:
        """Retrieves a job from a JSON file in the local storage."""

        job_file = self.get_job_folder(job_id) / Path(f"{job_id}.json")

        if not job_file.exists():
            return None

        with job_file.open("r") as file:
            job_dict = json.load(file)

        job_json_string = json.dumps(job_dict)

        return VoxelizationJobMetadata.from_json(job_json_string)

    def get_job_list(self) -> List[str]:
        pass
```

filter_lines_by_words = (
    "Running",
    "Log",
    "Device heap",
    "GPU to CPU VOXELIZATION COPY finalized",
)
bold_lines_by_words = (
    "MESH PREPROCESSING",
    "VOXELIZATION INFORMATION",
    "GPU to CPU VOXELIZATION COPY",
    "CPU MESH PREPORCESSING",
)


def parse_log_file(file_path):
    log_text = ""

    with open(file_path, "r") as file:
        lines = file.readlines()
        for line in lines:
            line_accepted = True
            bold_flag = False
            for filter_word in filter_lines_by_words:
                if filter_word in line:
                    line_accepted = False
                    break

            for bold_word in bold_lines_by_words:
                if bold_word in line:
                    bold_flag = True

            if line_accepted:
                if not bold_flag:
                    log_text += f"- {line.split(']')[-1]}"
                else:
                    log_text += f"\n {line.split(']')[-1]} \n"

    return log_text

"""FastAPI endpoints for the voxelization service.

This module provides REST API endpoints for mesh voxelization operations including:
- Uploading and voxelizing mesh files
- Checking job status
- Retrieving voxelization results

The API uses FastAPI for async operations and background task processing.
"""

import logging
import time
from pathlib import Path
from typing import Dict

from fastapi import (
    File,
    UploadFile,
    HTTPException,
    BackgroundTasks,
    FastAPI,
    Depends,
    Form,
)
from fastapi.responses import FileResponse

from end_points_voxelizer.create_voxelization_job.create_voxelization_job_local_storage import (
    CreateVoxelizationJobLocalStorage,
)
from shared_utils.VoxelizationJobMetadata import VoxelizationJobMetadata, JobStatus


ALLOWED_EXTENSIONS = {'.stl', '.obj'}
MAX_FILE_SIZE = 100 * 1024 * 1024  # 100MB
CHUNK_SIZE = 8192  # 8KB chunks for file reading

logger = logging.getLogger(__name__)


app = FastAPI()


# ToDo: This should be a factory for fastAPI dependency injection
def get_create_voxelization_job() -> "CreateVoxelizationJob":
    return CreateVoxelizationJobLocalStorage()


@app.post("/voxelize/", response_model=VoxelizationJobMetadata)
async def voxelize_mesh(
    background_tasks: BackgroundTasks,
    mesh_file: UploadFile = File(...),
    voxel_size: float | None = Form(None),
    voxel_file_name: str | None = Form(None),
    create_voxelization_job: "CreateVoxelizationJob" = Depends(
        get_create_voxelization_job
    ),
) -> Dict[str, str]:
    """Start a voxelization job for the uploaded mesh file.
    
    @param background_tasks: FastAPI background tasks handler
    @type background_tasks: BackgroundTasks
    
    @param mesh_file: The mesh file to be voxelized (.stl or .obj)
    @type mesh_file: UploadFile
    
    @param voxel_size: The size of each voxel in the output
    @type voxel_size: float or None
    
    @param voxel_file_name: Optional custom name for the output file
    @type voxel_file_name: str or None
    
    @param create_voxelization_job: Job creation service (injected)
    @type create_voxelization_job: CreateVoxelizationJob
    
    @return: The job ID for tracking the voxelization process
    @rtype: str
    
    @raise HTTPException: If file type is invalid, file is too large, or processing fails
    """

    mesh_full_path = None
    try:
        file_extension = Path(mesh_file.filename).suffix.lower()
        if file_extension not in ALLOWED_EXTENSIONS:
            raise HTTPException(
                status_code=400,
                detail=f"Invalid file type. Allowed types: {ALLOWED_EXTENSIONS}"
            )

        if not voxel_size or voxel_size <= 0:
            raise HTTPException(
                status_code=400,
                detail="Voxel size must be a positive number"
            )

        file_content = bytearray()
        file_size = 0
        while chunk := await mesh_file.read(CHUNK_SIZE):
            file_size += len(chunk)
            if file_size > MAX_FILE_SIZE:
                raise HTTPException(
                    status_code=400,
                    detail=f"File too large. Max size: {MAX_FILE_SIZE/1024/1024}MB"
                )
            file_content.extend(chunk)

        path_to_save_received_mesh = create_voxelization_job.output_voxel_dir_job
        mesh_file_name = Path(mesh_file.filename).stem
        mesh_tmp_full_name = f"{mesh_file_name}_server{file_extension}"
        mesh_full_path = path_to_save_received_mesh / mesh_tmp_full_name
        mesh_full_path.parent.mkdir(parents=True, exist_ok=True)
        mesh_full_path.write_bytes(file_content)

        params = {
            "input_stl_path": str(mesh_full_path),
            "voxel_file_name": str(voxel_file_name or f"voxelized_{mesh_file_name}.obj"),
            "voxel_size": float(voxel_size),
        }

        logger.info(f"Starting voxelization job for {mesh_file.filename}")
        background_tasks.add_task(
            create_voxelization_job.run_voxelization_job, 
            **params
        )


        #return {"job_id": create_voxelization_job.job.job_id}
        return create_voxelization_job.job
    except Exception as e:
        logger.error(f"Error processing voxelization request: {str(e)}")
        if mesh_full_path and mesh_full_path.exists():
            mesh_full_path.unlink()
        raise HTTPException(
            status_code=500,
            detail=f"Error processing upload: {str(e)}"
        )

def _check_job_availability(
    job_id: str,
    create_voxelization_job: "CreateVoxelizationJob",
    timeout: float = 300,
    poll_interval: float = 0.0,
) -> VoxelizationJobMetadata:
    """
    Check if the job is already available at intervals of 5 seconds. If the checking time is higher that 'timeout'
    stop the program.
    """
    start_time = time.time()

    while True:
        try:
            voxelization_job_metadata = create_voxelization_job.get_job(job_id=job_id)
            if voxelization_job_metadata is not None:
                if voxelization_job_metadata.status == JobStatus.COMPLETED:
                    return voxelization_job_metadata
        except Exception as e:
            print(f"Error fetching job status: {e}")

        elapsed_time = time.time() - start_time
        if elapsed_time > timeout:
            raise TimeoutError(
                f"Job {job_id} did not complete within {timeout} seconds."
            )

        time.sleep(poll_interval)


@app.get("/result/{job_id}")
async def get_job_result(
    job_id: str,
    create_voxelization_job: "CreateVoxelizationJob" = Depends(
        get_create_voxelization_job
    ),
) -> FileResponse:
    """Get the voxelized mesh file for a completed job.
    
    @param job_id: The ID of the completed job
    @type job_id: str
    
    @return: The voxelized mesh file
    @rtype: FileResponse
    
    @raise HTTPException: 
        - 404 if job or file not found
        - 400 if job not completed
        - 500 for other errors
    """
    job = _check_job_availability(job_id, create_voxelization_job)

    if not job:
        raise HTTPException(status_code=404, detail="Job not found")

    if job.status != JobStatus.COMPLETED:
        raise HTTPException(status_code=400, detail="Job not completed")

    voxel_file_path = str(job.output_voxel_dir / job.voxel_file_name)

    try:
        file_path = Path(voxel_file_path)
        if not file_path.exists():
            raise HTTPException(status_code=404, detail="File not found")

        return FileResponse(
            path=file_path,
            media_type="application/octet-stream",
            filename=file_path.name,
        )

    except Exception as e:
        logging.error(f"Error reading file for job {job_id}: {e}")
        raise HTTPException(status_code=500, detail=str(e))

    std::string voxel_file_name = "";
    std::string log_path_voxelizer = getAbsolutePathRelativeToExecutable(LOG_PATH_VOXELIZER);
    float gridVoxelSize = 1.0f;

    std::cout<<output_path<<std::endl;
    std::cout<<log_path_voxelizer<<std::endl;

    cxxopts::Options options(argv[0], "Mesh Voxelization Application");
    
    options.add_options()("i,input", "Input STL file path", cxxopts::value<std::string>()->default_value(input_stl_file))
                            ("o,output", "Output directory path for voxelized mesh", cxxopts::value<std::string>()->default_value(output_path))
                            ("n,name", "File name for voxelized mesh", cxxopts::value<std::string>()->default_value(voxel_file_name))
                            ("s, size", "Voxel size", cxxopts::value<float>()->default_value(std::to_string(gridVoxelSize)))
                            ("l,log_path", "Logging directory ", cxxopts::value<std::string>()->default_value(log_path_voxelizer))
                            ("h,help", "Print usage");

    auto result = options.parse(argc, argv);

    if (result.count("help"))
    {
        std::cout << options.help() << std::endl;
        return 0;
    }

    if (!result.count("input"))
    {
        std::cerr << "Error: Input STL file path is required.\n";
        std::cerr << options.help() << std::endl;
        return 1;
    }

    input_stl_file = result["input"].as<std::string>();
    output_path = result["output"].as<std::string>();
    gridVoxelSize = result["size"].as<float>();
    voxel_file_name = result["name"].as<std::string>();
    log_path_voxelizer = result["log_path"].as<std::string>();


    LoggingConfig logging_config(
    LOG_NAME,           // Logger name
    log_path_voxelizer, // Log directory
    google::INFO,       // Log level
    false               // Log to console
    );
    LoggerConfig logger(logging_config);

    cudaHeapMemory(MALLOC_LIMIT);

    // Preprocess input mesh file
    std::shared_ptr<mesh::IPreprocessingMesh> preprocess4Voxel =
        std::make_shared<mesh::PreprocessingMesh4Gpu>();
    preprocess4Voxel->preprocess(input_stl_file);

    auto meshData =
        preprocess4Voxel->getMeshesData()->meshArray[0]; // Change it UGLY!

    // Create Voxel model
    std::shared_ptr<voxelization::IVoxelization> voxelization =
        std::make_shared<voxelization::SurfaceVoxelization>();

    // create object to copy mesh from CPU to GPU
    std::shared_ptr<mesh::ICopyMeshData2Device> copyMesh2Gpu =
        std::make_shared<mesh::CopyMeshDataHost2Device>();

    // generate voxelization
    voxelization->create(meshData, gridVoxelSize, copyMesh2Gpu);

    cudaDeviceSynchronize();

    std::shared_ptr<voxelization::exporter::IVoxel2Mesh> voxel_to_mesh = std::make_shared<voxelization::exporter::Voxel2Mesh>();
    
    if (voxel_file_name=="")
        voxel_to_mesh->convertVoxel2Mesh(voxelization->getGrid(), voxelization->getMeshTranslation(), output_path);
    else
        voxel_to_mesh->convertVoxel2Mesh(voxelization->getGrid(), voxelization->getMeshTranslation(), output_path, voxel_file_name);


    voxelization->freeGpuData();

    return 0;
}

// Function to initialize CUDA heap memory
void cudaHeapMemory(float malloc_memory_limit)
{
    cudaSetDevice(0); // Set to device 0
    size_t *malloc_limit = new size_t;
    *malloc_limit = malloc_memory_limit;
    cudaDeviceSetLimit(cudaLimitMallocHeapSize, *malloc_limit);
    size_t *new_limit = new size_t();
    cudaDeviceGetLimit(new_limit, cudaLimitMallocHeapSize);
    LOG(INFO) << "Device heap memory limit: " << *new_limit / 1e9 << " GB" << std::endl;
}


// Function to create directories recursively
// Function to create directories recursively
bool createDirectoryV2(const std::string &path) {
    size_t pos = 0;
    std::string currentPath;

    // Traverse the path and create directories incrementally
    while ((pos = path.find('/', pos)) != std::string::npos) {
        currentPath = path.substr(0, pos++);
        if (currentPath.empty()) {
            continue; // Skip empty segments
        }

        if (mkdir(currentPath.c_str(), 0777) == -1 && errno != EEXIST) {
            std::cerr << "Error creating directory '" << currentPath << "': "
                      << strerror(errno) << std::endl;
            return false;
        }
    }

    // Create the final directory (full path)
    if (mkdir(path.c_str(), 0777) == -1 && errno != EEXIST) {
        std::cerr << "Error creating directory '" << path << "': "
                  << strerror(errno) << std::endl;
        return false;
    }

    return true;
}

// Function to get the directory of the executable
std::string getExecutablePath() {
    char result[PATH_MAX];
    ssize_t count = readlink("/proc/self/exe", result, PATH_MAX);
    if (count == -1) {
        throw std::runtime_error("Error determining executable path");
    }
    std::string fullPath(result, count);
    return fullPath.substr(0, fullPath.find_last_of('/'));
}

// Function to create an absolute path relative to the executable's directory
std::string getAbsolutePathRelativeToExecutable(const std::string &relativePath) {
    std::string basePath = getExecutablePath(); // Get the executable's directory
    std::string absolutePath = basePath + "/" + relativePath;

    if (!createDirectoryV2(absolutePath)) {
        throw std::runtime_error("Failed to create directory: " + absolutePath);
    }

    return absolutePath;
}

#include "CopyMeshDataHost2Device.h"
#include "Geometry.h"
#include "MeshStructs.h"
#include "LoggingConfig.h"

using namespace meshVoxelization;
using namespace mesh;

void mesh::CopyMeshDataHost2Device::copyData(MeshDataDevice* deviceData, std::shared_ptr<MeshData> hostData)
{
	MeshDataDevice* deviceMeshData = deviceData;
	MeshData* hostMeshData = hostData.get();

	MeshDataDevice *h_surf_gpu = (MeshDataDevice *)malloc(sizeof(MeshDataDevice));

	Triangle *h_workPieceTriangles = hostMeshData->triangles.data();

	int h_workPieceTrianglesNo = hostMeshData->triangles.size();

	Vertex *h_workPieceVertices = new Vertex[hostMeshData->vertices.size()];

	for (int i = 0; i < hostMeshData->vertices.size(); ++i)
	{
		h_workPieceVertices[i].x = hostMeshData->vertices[i].x;
		h_workPieceVertices[i].y = hostMeshData->vertices[i].y;
		h_workPieceVertices[i].z = hostMeshData->vertices[i].z;
	}

	float *h_workPieceBbox = hostMeshData->bbox.data();

	Triangle *workPieceTriangles = NULL;
	cudaError_t err;
	err = cudaMalloc((void **)&workPieceTriangles, hostMeshData->triangles.size() * sizeof(Triangle));
	if (err != cudaSuccess) {
		LOG(ERROR) << "CUDA malloc failed: " << cudaGetErrorString(err);
		throw std::runtime_error("CUDA memory allocation failed");
	}

	err = cudaMemcpy(workPieceTriangles, h_workPieceTriangles, hostMeshData->triangles.size() * sizeof(Triangle), cudaMemcpyHostToDevice);
	if (err != cudaSuccess) {
		LOG(ERROR) << "CUDA memcpy failed: " << cudaGetErrorString(err);
		cudaFree(workPieceTriangles);
		throw std::runtime_error("CUDA memory copy failed");
	}

	int *workPieceTrianglesNo = NULL;
	cudaMalloc((void **)&workPieceTrianglesNo, sizeof(int));
	cudaMemcpy(workPieceTrianglesNo, &h_workPieceTrianglesNo, sizeof(int), cudaMemcpyHostToDevice);

	Vertex *workPieceVertices = NULL;
	cudaMalloc((void **)&workPieceVertices, hostMeshData->vertices.size() * sizeof(Vertex));
	cudaMemcpy(workPieceVertices, h_workPieceVertices, hostMeshData->vertices.size() * sizeof(Vertex), cudaMemcpyHostToDevice);

	float *workPieceBbox = NULL;
	cudaMalloc((void **)&workPieceBbox, 6 * sizeof(float));
	cudaMemcpy(workPieceBbox, h_workPieceBbox, 6 * sizeof(float), cudaMemcpyHostToDevice);

	h_surf_gpu->triangles = workPieceTriangles;
	h_surf_gpu->trianglesNo = workPieceTrianglesNo;
	h_surf_gpu->vertices = workPieceVertices;
	h_surf_gpu->bbox = workPieceBbox;

	cudaMemcpy(deviceMeshData, h_surf_gpu, sizeof(MeshDataDevice), cudaMemcpyHostToDevice);

	return;
};

void mesh::CopyMeshDataHost2Device::copyDataArray(MeshDataArrayDevice *gpuData, std::vector<std::shared_ptr<MeshData>> cpuData)
{
	MeshDataArrayDevice *d_surf_gpu = gpuData;

	MeshDataArrayDevice *h_surf_gpu = (MeshDataArrayDevice *)malloc(sizeof(MeshDataArrayDevice));

	int dimArray = cpuData.size();
	int *dimArrayDevice;
	cudaMalloc((void **)&dimArrayDevice, sizeof(int));

	cudaMemcpy(dimArrayDevice, &dimArray, sizeof(int), cudaMemcpyHostToDevice);

	MeshDataDevice *meshData;
	cudaMalloc((void **)&meshData, dimArray * sizeof(MeshDataDevice));

	for (int i = 0; i < dimArray; ++i)
		copyData(&meshData[i], cpuData[i]);

	h_surf_gpu->meshArrayDevice = meshData;
	h_surf_gpu->dimArray = dimArrayDevice;

	cudaMemcpy(d_surf_gpu, h_surf_gpu, sizeof(MeshDataArrayDevice), cudaMemcpyHostToDevice);

	return;
}

void mesh::CopyMeshDataHost2Device::deleteMeshDataDevice(MeshDataDevice *gpuData, int *numMeshes)
{

	int num;
	if (numMeshes == nullptr)
		num = 1;
	else
		num = *numMeshes;

	MeshDataDevice *h_surf_gpu_copy = (MeshDataDevice *)malloc(num * sizeof(MeshDataDevice));
	cudaMemcpy(h_surf_gpu_copy, gpuData, num * sizeof(MeshDataDevice), cudaMemcpyDeviceToHost);

	for (auto i = 0; i < num; ++i)
	{
		cudaFree(h_surf_gpu_copy[i].triangles);
		cudaFree(h_surf_gpu_copy[i].trianglesNo);
		cudaFree(h_surf_gpu_copy[i].vertices);
		cudaFree(h_surf_gpu_copy[i].bbox);
	}

	cudaFree(gpuData);
	free(h_surf_gpu_copy);
}

```
#include <unordered_map>
#include <iostream>
#include <fstream>
#include <algorithm>
#include <map>
#include <set>
#include "LoggingConfig.h"
#include "PreprocessingMesh4Gpu.h"

using namespace meshVoxelization;
using namespace mesh;

bool equalPoint(Vertex p1, Vertex p2)
{
    if (p1.x == p2.x && p1.y == p2.y && p1.z == p2.z)
        return true;
    return false;
}

bool comparePoint(Vertex p1, Vertex p2)
{
    if (p1.x != p2.x)
        return p1.x > p2.x;
    else if (p1.y != p2.y)
        return p1.y > p2.y;
    else
        return p1.z > p2.z;
}

struct hashFunc
{
    size_t operator()(const Vertex &k) const
    {

        size_t h1 = std::hash<double>()(k.x);
        size_t h2 = std::hash<double>()(k.y);
        size_t h3 = std::hash<double>()(k.z);
        return (h1 ^ (h2 << 1)) ^ h3;
    }
};

struct equalsFunc
{
    bool operator()(const Vertex &lhs, const Vertex &rhs) const
    {
        return (lhs.x == rhs.x) && (lhs.y == rhs.y) && (lhs.z == rhs.z);
    }
};

typedef std::unordered_map<Vertex, int, hashFunc, equalsFunc> TPoint3DMap;

void mesh::PreprocessingMesh4Gpu::preprocess(std::string path2Wpiece)
{

    auto start_time = std::chrono::high_resolution_clock::now();

    _surfsDataWp->meshArray.resize(1);

    _surfsDataWp->meshArray[0] = std::make_shared<MeshData>();

    auto meshData = _surfsDataWp->meshArray[0];

    openmesh_app::MyMesh mesh;

    if (!OpenMesh::IO::read_mesh(mesh, path2Wpiece))
    {
        LOG(ERROR) << "Failed to read mesh from path: " << path2Wpiece;
        throw std::runtime_error("Mesh reading failed.");
    }

    // Bounding box
    typename openmesh_app::MyMesh::VertexIter vit(mesh.vertices_begin()),
        vend(mesh.vertices_end());

    openmesh_app::MyMesh::Point bbMin = mesh.point(vit);
    openmesh_app::MyMesh::Point bbMax = mesh.point(vit);

    for (++vit; vit != vend; ++vit)
    {
        bbMin.minimize(mesh.point(vit));
        bbMax.maximize(mesh.point(vit));
    }

    for (auto i = 0; i < 3; ++i)
        meshData->bbox.push_back(bbMin[i]);
    for (auto i = 0; i < 3; ++i)
        meshData->bbox.push_back(bbMax[i]);

    std::vector<Vertex> verticesTri{3};

    for (openmesh_app::MyMesh::FaceIter f_it = mesh.faces_begin(); f_it != mesh.faces_end(); ++f_it)
    {
        int idx = 0;
        for (openmesh_app::MyMesh::FaceVertexIter f_it2 = mesh.fv_begin(*f_it); f_it2.is_valid(); ++f_it2)
        {
            verticesTri[idx].x = mesh.point(*f_it2)[0];
            verticesTri[idx].y = mesh.point(*f_it2)[1];
            verticesTri[idx].z = mesh.point(*f_it2)[2];
            meshData->vertices.push_back(verticesTri[idx]);

            idx += 1;
        }

        // Triangle Bbox
        Triangle singleTri;
        float xCoord[3] = {verticesTri[0].x, verticesTri[1].x, verticesTri[2].x};
        float yCoord[3] = {verticesTri[0].y, verticesTri[1].y, verticesTri[2].y};
        float zCoord[3] = {verticesTri[0].z, verticesTri[1].z, verticesTri[2].z};

        singleTri._bottom.x = *std::min_element(xCoord, xCoord + 3);
        singleTri._top.x = *std::max_element(xCoord, xCoord + 3);
        singleTri._bottom.y = *std::min_element(yCoord, yCoord + 3);
        singleTri._top.y = *std::max_element(yCoord, yCoord + 3);
        singleTri._bottom.z = *std::min_element(zCoord, zCoord + 3);
        singleTri._top.z = *std::max_element(zCoord, zCoord + 3);

        meshData->triangles.push_back(singleTri);
    }

    std::vector<Vertex> vecAux(meshData->vertices.begin(), meshData->vertices.end());

    std::sort(vecAux.begin(), vecAux.end(), comparePoint);
    auto unique_end = std::unique(vecAux.begin(), vecAux.end(), equalPoint);
    vecAux.erase(unique_end, vecAux.end());

    TPoint3DMap myMap;

    for (int i = 0; i < vecAux.size(); ++i)
        myMap[vecAux[i]] = i;

    int numTriangles = mesh.n_faces();
    for (int i = 0; i < numTriangles; ++i)
    {
        meshData->triangles[i]._idx1 = myMap[meshData->vertices[3 * i]];
        meshData->triangles[i]._idx2 = myMap[meshData->vertices[3 * i + 1]];
        meshData->triangles[i]._idx3 = myMap[meshData->vertices[3 * i + 2]];
    }

    meshData->vertices = vecAux;

    auto end_time = std::chrono::high_resolution_clock::now();
    auto time = end_time - start_time;

    LOG(INFO) << "CPU MESH PREPROCESSING:";

    for (auto elem : _surfsDataWp->meshArray)
        LOG(INFO) << "Vertices in mesh: " << elem->vertices.size();

    LOG(INFO) << "Mesh bbox lengths (x, y, z): " << meshData->bbox[3] - meshData->bbox[0] << ", " << meshData->bbox[4] - meshData->bbox[1]
            << ", " << meshData->bbox[5] - meshData->bbox[2];

    LOG(INFO)
        << "CPU mesh preprocessing took: "
        << std::chrono::duration_cast<std::chrono::microseconds>(time).count() /
               1000.0
        << " milliseconds.\n";

    LOG(INFO) << "Processing mesh with " << mesh.n_vertices() << " vertices and "
              << mesh.n_faces() << " faces";
    VLOG(2) << "Detailed mesh processing information...";
}

### mesh_voxelization/voxelization/interfaces/IVoxelization.h
```
#pragma once

#include "Grid.h"
#include "ICopyMeshData2Device.h"

namespace meshVoxelization
{

  namespace voxelization
  {

    /**
     * Voxelization Interface. The idea is that given a mesh we will compute its bounding box and divide it first in 
     * a grained voxelization with voxels, L1, of size gridSize**3. Then we will divide these grained voxels further in smaller 
     * ones, L0, of size gridVoxelSize**3, so gridVoxelSize will be the tolerance (or pixel size) of our voxelization.
     * This information is saved in:
     * 
     * @code
     * struct Grid { 
     * unsigned int *vTable1;
     * int gridSize;
     * float gridVoxelSize;
     * ...
     * }
     * @endcode
     * 
     * where vTable1 is a bit-map with the occupancy of the smaller voxels. Basically it is a plain array with a 1 if at
     * 
     * loc = x + gridSize * y + gridSize * gridSize * z    (with (x,y,z) the position within the L0 voxels)
     * 
     * the L0 voxel is full and 0 otherwise.
     */

    class IVoxelization
    {
    public:
      ~IVoxelization() {};

    
      virtual void
      create(std::shared_ptr<mesh::MeshData> meshData, float gridVoxelSize,
             std::shared_ptr<mesh::ICopyMeshData2Device> copyMeshData2Device) = 0;

      /**
       * @brief get the voxel data. 
       * 
       * @return Grid* 
       */
      virtual Grid *getGrid() = 0;

      /**
       * @brief copy the Grid data from device to cpu.
       * 
       * @return std::shared_ptr<Grid> 
       */

### mesh_voxelization/voxelization/src/BulkVoxelization.cu
```
#include "BulkVoxelization.h"
#include <glog/logging.h>
#include "LoggingConfig.h"

namespace meshVoxelization
{
  namespace voxelization
  {

    __device__ void voxelizeTriangleSolidGpu(Grid *grid, float *meshBbox,
                                             float *triBbox, float *V0, float *V1,
                                             float *V2, bool morton_order,
                                             int i_thr)
    {
      float voxelSize = grid->gridVoxelSize;
      int gridSizeX = grid->gridSize;
      int gridSizeY = grid->gridSize;
      int gridSizeZ = grid->gridSize;

      maths::Vector3Df delta_p(voxelSize, voxelSize, voxelSize);
      maths::Vector3Df c(0.0f, 0.0f, 0.0f);

      maths::Vector3Df v0(V0[0], V0[1], V0[2]);
      maths::Vector3Df v1(V1[0], V1[1], V1[2]);
      maths::Vector3Df v2(V2[0], V2[1], V2[2]);

      maths::Vector3Df e0(v1 - v0);
      maths::Vector3Df e1(v2 - v1);
      maths::Vector3Df e2(v0 - v2);

      maths::Vector3Df t_bbox_world_min(
          triBbox[0] / voxelSize, triBbox[1] / voxelSize, triBbox[2] / voxelSize);
      maths::Vector3Df t_bbox_world_max(
          triBbox[3] / voxelSize, triBbox[4] / voxelSize, triBbox[5] / voxelSize);

      maths::Vector3Df voxelOrigin(0.0f, 0.0f, 0.0f);
      maths::Vector3Df voxelEnd(gridSizeX - 1, gridSizeY - 1, gridSizeZ - 1);

      maths::Vector3Df t_bbox_grid_min =
          clamp3Df(t_bbox_world_min, voxelOrigin, voxelEnd);
      maths::Vector3Df t_bbox_grid_max =
          clamp3Df(t_bbox_world_max, voxelOrigin, voxelEnd);

      maths::Vector3Df n(cross3Df(e0, e1));
      // PREPARE PLANE TEST PROPERTIES
      if (n.x > 0.0f)
      {
        c.x = delta_p.x;
      }
      if (n.y > 0.0f)
      {
        c.y = delta_p.y;
      }
      if (n.z > 0.0f)
      {
        c.z = delta_p.z;
      }

      float d1 = dot(n, (c - v0));
      float d2 = dot(n, ((delta_p - c) - v0));

      // YZ plane
      maths::Vector2Df n_yz_e0(-1.0f * e0.z, e0.y);
      maths::Vector2Df n_yz_e1(-1.0f * e1.z, e1.y);
      maths::Vector2Df n_yz_e2(-1.0f * e2.z, e2.y);
      if (n.x < 0.0f)
      {
        n_yz_e0 = n_yz_e0 * (-1.0f);
        n_yz_e1 = n_yz_e1 * (-1.0f);
        n_yz_e2 = n_yz_e2 * (-1.0f);
      }

      float d_yz_e0 =
          (-1.0f *
           maths::dot2D(
               n_yz_e0,
               maths::Vector2Df(v0.y, v0.z))); // + max(0.0f, delta_p.y*n_yz_e0.x) +
                                               // max(0.0f, delta_p.z*n_yz_e0.y);
      float d_yz_e1 =
          (-1.0f *
           maths::dot2D(
               n_yz_e1,
               maths::Vector2Df(v1.y, v1.z))); // + max(0.0f, delta_p.y*n_yz_e1.x) +
                                               // max(0.0f, delta_p.z*n_yz_e1.y);
      float d_yz_e2 =
          (-1.0f *
           maths::dot2D(
               n_yz_e2,
               maths::Vector2Df(v2.y, v2.z))); // + max(0.0f, delta_p.y*n_yz_e2.x) +
                                               // max(0.0f, delta_p.z*n_yz_e2.y);

      // XY plane
      maths::Vector2Df n_xy_e0(-1.0f * e0.y, e0.x);
      maths::Vector2Df n_xy_e1(-1.0f * e1.y, e1.x);
      maths::Vector2Df n_xy_e2(-1.0f * e2.y, e2.x);
      if (n.z < 0.0f)
      {
        n_xy_e0 = n_xy_e0 * (-1.0f);
        n_xy_e1 = n_xy_e1 * (-1.0f);
        n_xy_e2 = n_xy_e2 * (-1.0f);
      }
      float d_xy_e0 =
          (-1.0f * maths::dot2D(n_xy_e0, maths::Vector2Df(v0.x, v0.y))) +
          max(0.0f, delta_p.x * n_xy_e0.x) + max(0.0f, delta_p.y * n_xy_e0.y);
      float d_xy_e1 =
          (-1.0f * maths::dot2D(n_xy_e1, maths::Vector2Df(v1.x, v1.y))) +
          max(0.0f, delta_p.x * n_xy_e1.x) + max(0.0f, delta_p.y * n_xy_e1.y);
      float d_xy_e2 =
          (-1.0f * maths::dot2D(n_xy_e2, maths::Vector2Df(v2.x, v2.y))) +
          max(0.0f, delta_p.x * n_xy_e2.x) + max(0.0f, delta_p.y * n_xy_e2.y);

      // ZX plane
      maths::Vector2Df n_zx_e0(-1.0f * e0.x, e0.z);
      maths::Vector2Df n_zx_e1(-1.0f * e1.x, e1.z);
      maths::Vector2Df n_zx_e2(-1.0f * e2.x, e2.z);
      if (n.y < 0.0f)
      {
        n_zx_e0 = n_zx_e0 * (-1.0f);
        n_zx_e1 = n_zx_e1 * (-1.0f);
        n_zx_e2 = n_zx_e2 * (-1.0f);
      }
      float d_xz_e0 =
          (-1.0f * maths::dot2D(n_zx_e0, maths::Vector2Df(v0.z, v0.x))) +
          max(0.0f, delta_p.x * n_zx_e0.x) + max(0.0f, delta_p.z * n_zx_e0.y);
      float d_xz_e1 =
          (-1.0f * maths::dot2D(n_zx_e1, maths::Vector2Df(v1.z, v1.x))) +
          max(0.0f, delta_p.x * n_zx_e1.x) + max(0.0f, delta_p.z * n_zx_e1.y);
      float d_xz_e2 =
          (-1.0f * maths::dot2D(n_zx_e2, maths::Vector2Df(v2.z, v2.x))) +
          max(0.0f, delta_p.x * n_zx_e2.x) + max(0.0f, delta_p.z * n_zx_e2.y);

      bool cond1, cond2, cond3;

      for (int z = t_bbox_grid_min.z; z <= t_bbox_grid_max.z; z++)
        for (int y = t_bbox_grid_min.y; y <= t_bbox_grid_max.y; y++)
        {
          // YZ
          maths::Vector2Df p_yz((y + 0.5f) * voxelSize, (z + 0.5f) * voxelSize);

          cond1 =
              ((maths::dot2D(n_yz_e0, p_yz) + d_yz_e0) > 0.0f ||
               ((maths::dot2D(n_yz_e0, p_yz) + d_yz_e0) == 0.0f) &&
                   (n_yz_e0.x > 0.0f || (n_yz_e0.x == 0.0f && n_yz_e0.y < 0.0f)));

          cond2 =
              ((maths::dot2D(n_yz_e1, p_yz) + d_yz_e1) > 0.0f ||
               ((maths::dot2D(n_yz_e1, p_yz) + d_yz_e1) == 0.0f) &&
                   (n_yz_e1.x > 0.0f || (n_yz_e1.x == 0.0f && n_yz_e1.y < 0.0f)));

          cond3 =
              ((maths::dot2D(n_yz_e2, p_yz) + d_yz_e2) > 0.0f ||
               ((maths::dot2D(n_yz_e2, p_yz) + d_yz_e2) == 0.0f) &&
                   (n_yz_e2.x > 0.0f || (n_yz_e2.x == 0.0f && n_yz_e2.y < 0.0f)));

          if (!(cond1 && cond2 && cond3))
            continue;

          for (int x = t_bbox_grid_min.x; x <= t_bbox_grid_max.x; x++)
          {
            // XY
            maths::Vector2Df p_xy((x)*voxelSize, (y)*voxelSize);
            if ((maths::dot2D(n_xy_e0, p_xy) + d_xy_e0) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_xy_e1, p_xy) + d_xy_e1) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_xy_e2, p_xy) + d_xy_e2) < 0.0f)
            {
              continue;
            }

            // XZ
            maths::Vector2Df p_zx((z)*voxelSize, (x)*voxelSize);
            if ((maths::dot2D(n_zx_e0, p_zx) + d_xz_e0) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_zx_e1, p_zx) + d_xz_e1) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_zx_e2, p_zx) + d_xz_e2) < 0.0f)
            {
              continue;
            }

            maths::Vector3Df p((x)*voxelSize, (y)*voxelSize, (z)*voxelSize);
            // TRIANGLE PLANE THROUGH BOX TEST
            float nDOTp = dot(n, p);
            if (((nDOTp + d1) * (nDOTp + d2)) > 0.0f)
            {
              continue;
            }

            size_t location = x + (y * gridSizeY) + (z * gridSizeY * gridSizeZ);

            ops::flipBit(grid->vTable1, location);

            break;
          }
        }
    };

    __device__ void fillBulkGpu(size_t i_thr, int &gridSize,
                                unsigned int *vTableBulk)
    {


      int numPoints = 0;

      int gridSizeX = gridSize;
      int gridSizeY = gridSize;

      for (size_t i = 0; i < (gridSizeX); ++i)
      {
        size_t location = (size_t)(i + gridSizeY * i_thr);
        if (ops::checkBit(vTableBulk, location))
          numPoints += 1;
      }

      size_t *locRastPoints = new size_t[numPoints];

      int count = 0;
      for (size_t i = 0; i < (gridSizeX); ++i)
      {
        size_t location = (size_t)(i + gridSizeY * i_thr);
        if (ops::checkBit(vTableBulk, location))
        {
          locRastPoints[count] = i;
          count += 1;
        }
      }

      if (numPoints > 1)
        for (int m = 0; m < numPoints; ++m)
        {
          for (size_t i = locRastPoints[m]; i < gridSizeX; ++i)
          {
            size_t location = i + gridSizeY * i_thr;
            ops::flipBit(vTableBulk, location);
          }
        }

      delete[] locRastPoints;
    }

    __global__ void fillBulkVoxelGpu(Grid *grid, int totNumThreads)
    {

      size_t i_thr = threadIdx.x + blockIdx.x * blockDim.x;

      if (i_thr < totNumThreads)
      {
        int gridSize = grid->gridSize;

        if (i_thr < gridSize * gridSize)
          fillBulkGpu(i_thr, gridSize, grid->vTable1);
      }

      return;
    }

    __global__ void solidVoxelGpu(mesh::MeshDataDevice *GPUMeshData, Grid *grid,
                                  int totNumThreads)
    {
      size_t thread_id = threadIdx.x + blockIdx.x * blockDim.x;

      if (thread_id < totNumThreads)
      {

        float meshBbox[6] = {GPUMeshData->bbox[0], GPUMeshData->bbox[1],
                             GPUMeshData->bbox[2], GPUMeshData->bbox[3],
                             GPUMeshData->bbox[4], GPUMeshData->bbox[5]};

        mesh::Triangle triThread = GPUMeshData->triangles[thread_id];

        int index1 = 3 * triThread._idx1;
        int index2 = 3 * triThread._idx2;
        int index3 = 3 * triThread._idx3;

        float V0[3] = {GPUMeshData->vertices[index1].x - meshBbox[0],
                       GPUMeshData->vertices[index1].y - meshBbox[1],
                       GPUMeshData->vertices[index1].z - meshBbox[2]};
        float V1[3] = {GPUMeshData->vertices[index2].x - meshBbox[0],
                       GPUMeshData->vertices[index2].y - meshBbox[1],
                       GPUMeshData->vertices[index2].z - meshBbox[2]};
        float V2[3] = {GPUMeshData->vertices[index3].x - meshBbox[0],
                       GPUMeshData->vertices[index3].y - meshBbox[1],
                       GPUMeshData->vertices[index3].z - meshBbox[2]};

        float triBbox[6] = {
            triThread._bottom.x - meshBbox[0], triThread._bottom.y - meshBbox[1],
            triThread._bottom.z - meshBbox[2], triThread._top.x - meshBbox[0],
            triThread._top.y - meshBbox[1], triThread._top.z - meshBbox[2]};

        if (thread_id < *GPUMeshData->trianglesNo)
          voxelizeTriangleSolidGpu(grid, meshBbox, triBbox, V0, V1, V2, true,
                                   thread_id);
      }

      return;
    }

    void SolidVoxelization::create(

        std::shared_ptr<mesh::ICopyMeshData2Device> copyMeshData2Device)
    {
      LoggerConfig logger;

      _globalTranslation.resize(3);
      _globalTranslation[0] = meshData->bbox[0];
      _globalTranslation[1] = meshData->bbox[1];
      _globalTranslation[2] = meshData->bbox[2];

      int gridSize = ops::computeTwoLevelSizes(meshData->bbox, gridVoxelSize);

      ////// Compute grid-1 Voxel size (_gridVoxelSize)
      float meshBboxSize[3] = {meshData->bbox[3] - meshData->bbox[0],
                               meshData->bbox[4] - meshData->bbox[1],
                               meshData->bbox[5] - meshData->bbox[2]};

      LOG(INFO) << "############################################################";
      LOG(INFO) << "#######  INFORMATION ABOUT TWO LEVEL MODEL ";
      LOG(INFO) << "#######  grid Size:  " << gridSize
                << ",  grid Voxel Size: " << gridVoxelSize;
      LOG(INFO) << "#######  W-piece bbox dimensions:  " << meshBboxSize[0]
                << "  " << meshBboxSize[1] << "  " << meshBboxSize[2];
      LOG(INFO) << "############################################################";

      ///// Create _grid and _grid0
      //_grid = constructorgrid(gridSize, gridVoxelSize);
      _grid = constructorgridGpu(gridSize, gridVoxelSize);

      // voxelize(info, meshData, _grid);

      ///// 1st Copy data to GPU /////
      mesh::MeshDataDevice *gpuData = NULL;
      cudaMalloc((void **)&gpuData, sizeof(mesh::MeshDataDevice));

      copyMeshData2Device->copyData(gpuData, meshData);

      int totNumThreads = meshData->triangles.size();
      int numThreads = 256;
      int numBlocks = ceilf(1.0f * totNumThreads / (1.0f * numThreads));

      auto start_time = std::chrono::high_resolution_clock::now();

      solidVoxelGpu<<<numBlocks, numThreads>>>(gpuData, _grid, totNumThreads);

      totNumThreads = gridSize * gridSize;
      numThreads = 256;
      numBlocks = ceilf(1.0f * totNumThreads / (1.0f * numThreads));

      fillBulkVoxelGpu<<<numBlocks, numThreads>>>(_grid, totNumThreads);

      cudaDeviceSynchronize();

      auto end_time = std::chrono::high_resolution_clock::now();
      auto time = end_time - start_time;

      LOG(INFO)
          << "The BULK-VOXELIZATION L1 algorithm took "
          << std::chrono::duration_cast<std::chrono::microseconds>(time).count() /
                 1000.0
          << "milliseconds to run.\n";

      copyMeshData2Device->deleteMeshDataDevice(gpuData);

      return;
    }

      int dimvTable = vTableDimension(linearGridDim);

#include "MeshStructs.h"
#include "SurfaceVoxelization.h"
#include <memory>
#include <glog/logging.h>

        __device__ void voxelizeTriangle(Grid *grid, float *meshBbox, float *triBbox,
                                     float *V0, float *V1, float *V2)
    {

      ////// Triangle-Voxelization Data //////
      float voxelSize = grid->gridVoxelSize;
      int gridSizeX = grid->gridSize;
      int gridSizeY = grid->gridSize;
      int gridSizeZ = grid->gridSize;

      maths::Vector3Df delta_p(voxelSize, voxelSize, voxelSize);
      maths::Vector3Df c(0.0f, 0.0f, 0.0f);

      maths::Vector3Df v0(V0[0], V0[1], V0[2]);
      maths::Vector3Df v1(V1[0], V1[1], V1[2]);
      maths::Vector3Df v2(V2[0], V2[1], V2[2]);

      maths::Vector3Df e0(v1 - v0);
      maths::Vector3Df e1(v2 - v1);
      maths::Vector3Df e2(v0 - v2);

      maths::Vector3Df t_bbox_world_min(
          triBbox[0] / voxelSize, triBbox[1] / voxelSize, triBbox[2] / voxelSize);
      maths::Vector3Df t_bbox_world_max(
          triBbox[3] / voxelSize, triBbox[4] / voxelSize, triBbox[5] / voxelSize);

      maths::Vector3Df voxelOrigin(0.0f, 0.0f, 0.0f);
      maths::Vector3Df voxelEnd(gridSizeX - 1, gridSizeY - 1, gridSizeZ - 1);

      maths::Vector3Df t_bbox_grid_min =
          clamp3Df(t_bbox_world_min, voxelOrigin, voxelEnd);
      maths::Vector3Df t_bbox_grid_max =
          clamp3Df(t_bbox_world_max, voxelOrigin, voxelEnd);

      // int numZ = 0;
      maths::Vector3Df n(cross3Df(e0, e1));
      // PREPARE PLANE TEST PROPERTIES
      if (n.x > 0.0f)
      {
        c.x = delta_p.x;
      }
      if (n.y > 0.0f)
      {
        c.y = delta_p.y;
      }
      if (n.z > 0.0f)
      {
        c.z = delta_p.z;
      }

      float d1 = dot(n, (c - v0));
      float d2 = dot(n, ((delta_p - c) - v0));

      /////////////////////////////////////////

      // YZ plane
      maths::Vector2Df n_yz_e0(-1.0f * e0.z, e0.y);
      maths::Vector2Df n_yz_e1(-1.0f * e1.z, e1.y);
      maths::Vector2Df n_yz_e2(-1.0f * e2.z, e2.y);
      if (n.x < 0.0f)
      {
        n_yz_e0 = n_yz_e0 * (-1.0f);
        n_yz_e1 = n_yz_e1 * (-1.0f);
        n_yz_e2 = n_yz_e2 * (-1.0f);
      }
      float d_yz_e0 = (-1.0f * maths::dot2D(n_yz_e0, maths::Vector2Df(v0.y, v0.z))) +
                      max(0.0f, delta_p.y * n_yz_e0.x) +
                      max(0.0f, delta_p.z * n_yz_e0.y);
      float d_yz_e1 = (-1.0f * maths::dot2D(n_yz_e1, maths::Vector2Df(v1.y, v1.z))) +
                      max(0.0f, delta_p.y * n_yz_e1.x) +
                      max(0.0f, delta_p.z * n_yz_e1.y);
      float d_yz_e2 = (-1.0f * maths::dot2D(n_yz_e2, maths::Vector2Df(v2.y, v2.z))) +
                      max(0.0f, delta_p.y * n_yz_e2.x) +
                      max(0.0f, delta_p.z * n_yz_e2.y);

      // PREPARE PROJECTION TEST PROPERTIES
      // XY plane
      maths::Vector2Df n_xy_e0(-1.0f * e0.y, e0.x);
      maths::Vector2Df n_xy_e1(-1.0f * e1.y, e1.x);
      maths::Vector2Df n_xy_e2(-1.0f * e2.y, e2.x);
      if (n.z < 0.0f)
      {
        n_xy_e0 = n_xy_e0 * (-1.0f);
        n_xy_e1 = n_xy_e1 * (-1.0f);
        n_xy_e2 = n_xy_e2 * (-1.0f);
      }
      float d_xy_e0 = (-1.0f * maths::dot2D(n_xy_e0, maths::Vector2Df(v0.x, v0.y))) +
                      max(0.0f, delta_p.x * n_xy_e0.x) +
                      max(0.0f, delta_p.y * n_xy_e0.y);
      float d_xy_e1 = (-1.0f * maths::dot2D(n_xy_e1, maths::Vector2Df(v1.x, v1.y))) +
                      max(0.0f, delta_p.x * n_xy_e1.x) +
                      max(0.0f, delta_p.y * n_xy_e1.y);
      float d_xy_e2 = (-1.0f * maths::dot2D(n_xy_e2, maths::Vector2Df(v2.x, v2.y))) +
                      max(0.0f, delta_p.x * n_xy_e2.x) +
                      max(0.0f, delta_p.y * n_xy_e2.y);

      // ZX plane
      maths::Vector2Df n_zx_e0(-1.0f * e0.x, e0.z);
      maths::Vector2Df n_zx_e1(-1.0f * e1.x, e1.z);
      maths::Vector2Df n_zx_e2(-1.0f * e2.x, e2.z);
      if (n.y < 0.0f)
      {
        n_zx_e0 = n_zx_e0 * (-1.0f);
        n_zx_e1 = n_zx_e1 * (-1.0f);
        n_zx_e2 = n_zx_e2 * (-1.0f);
      }
      float d_xz_e0 = (-1.0f * maths::dot2D(n_zx_e0, maths::Vector2Df(v0.z, v0.x))) +
                      max(0.0f, delta_p.x * n_zx_e0.x) +
                      max(0.0f, delta_p.z * n_zx_e0.y);
      float d_xz_e1 = (-1.0f * maths::dot2D(n_zx_e1, maths::Vector2Df(v1.z, v1.x))) +
                      max(0.0f, delta_p.x * n_zx_e1.x) +
                      max(0.0f, delta_p.z * n_zx_e1.y);
      float d_xz_e2 = (-1.0f * maths::dot2D(n_zx_e2, maths::Vector2Df(v2.z, v2.x))) +
                      max(0.0f, delta_p.x * n_zx_e2.x) +
                      max(0.0f, delta_p.z * n_zx_e2.y);

      for (int z = t_bbox_grid_min.z; z <= t_bbox_grid_max.z; z++)
        for (int y = t_bbox_grid_min.y; y <= t_bbox_grid_max.y; y++)
        {

          // YZ
          maths::Vector2Df p_yz(y * voxelSize, z * voxelSize);
          if ((maths::dot2D(n_yz_e0, p_yz) + d_yz_e0) < 0.0f)
          {
            continue;
          }
          if ((maths::dot2D(n_yz_e1, p_yz) + d_yz_e1) < 0.0f)
          {
            continue;
          }
          if ((maths::dot2D(n_yz_e2, p_yz) + d_yz_e2) < 0.0f)
          {
            continue;
          }

          for (int x = t_bbox_grid_min.x; x <= t_bbox_grid_max.x; x++)
          {
            // XY
            maths::Vector2Df p_xy(x * voxelSize, y * voxelSize);
            if ((maths::dot2D(n_xy_e0, p_xy) + d_xy_e0) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_xy_e1, p_xy) + d_xy_e1) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_xy_e2, p_xy) + d_xy_e2) < 0.0f)
            {
              continue;
            }

            // XZ
            maths::Vector2Df p_zx(z * voxelSize, x * voxelSize);
            if ((maths::dot2D(n_zx_e0, p_zx) + d_xz_e0) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_zx_e1, p_zx) + d_xz_e1) < 0.0f)
            {
              continue;
            }
            if ((maths::dot2D(n_zx_e2, p_zx) + d_xz_e2) < 0.0f)
            {
              continue;
            }

            maths::Vector3Df p(x * voxelSize, y * voxelSize, z * voxelSize);
            // TRIANGLE PLANE THROUGH BOX TEST
            float nDOTp = dot(n, p);
            if (((nDOTp + d1) * (nDOTp + d2)) > 0.0f)
            {
              continue;
            }

            size_t location = x + (y * gridSizeY) + (z * gridSizeY * gridSizeZ);

            ops::setBit(grid->vTable1, location);
            continue;
          }
        }

      return;
    }

    __global__ void voxelGPU(Grid *grid, mesh::MeshDataDevice *GPUMeshData,
                             int totNumThreads)
    {

        size_t thread_id = threadIdx.x + blockIdx.x * blockDim.x;

        if (thread_id < totNumThreads)
        {
            float meshBbox[6] = {GPUMeshData->bbox[0], GPUMeshData->bbox[1],
                                 GPUMeshData->bbox[2], GPUMeshData->bbox[3],
                                 GPUMeshData->bbox[4], GPUMeshData->bbox[5]};

            mesh::Triangle triThread = GPUMeshData->triangles[thread_id];

            int index1 = triThread._idx1;
            int index2 = triThread._idx2;
            int index3 = triThread._idx3;

            float V0[3] = {GPUMeshData->vertices[index1].x - meshBbox[0],
                           GPUMeshData->vertices[index1].y - meshBbox[1],
                           GPUMeshData->vertices[index1].z - meshBbox[2]};
            float V1[3] = {GPUMeshData->vertices[index2].x - meshBbox[0],
                           GPUMeshData->vertices[index2].y - meshBbox[1],
                           GPUMeshData->vertices[index2].z - meshBbox[2]};
            float V2[3] = {GPUMeshData->vertices[index3].x - meshBbox[0],
                           GPUMeshData->vertices[index3].y - meshBbox[1],
                           GPUMeshData->vertices[index3].z - meshBbox[2]};

            float triBbox[6] = {
                triThread._bottom.x - meshBbox[0], triThread._bottom.y - meshBbox[1],
                triThread._bottom.z - meshBbox[2], triThread._top.x - meshBbox[0],
                triThread._top.y - meshBbox[1], triThread._top.z - meshBbox[2]};

            voxelizeTriangle(grid, meshBbox, triBbox, V0, V1, V2);
        }

        return;
    };

        Grid *grid = constructorgrid(gridSize, gridVoxelSize);

    __global__ void moveVtableToPlainArray(unsigned int *array, int dimArray,
                                           Grid *grid)
    {
        memcpy(array, grid->vTable1, dimArray * sizeof(unsigned int));

    std::shared_ptr<Grid> voxelization::SurfaceVoxelization::copyGridToCpu()
    {

        int gridSize;
        float gridVoxelSize;
        cudaMemcpy(&gridSize, &(_grid->gridSize), sizeof(int),
                   cudaMemcpyDeviceToHost);
        cudaMemcpy(&gridVoxelSize, &(_grid->gridVoxelSize), sizeof(float),
                   cudaMemcpyDeviceToHost);

        std::shared_ptr<Grid> gridCpu = std::make_shared<Grid>();
        gridCpu->gridSize = gridSize;
        gridCpu->gridVoxelSize = gridVoxelSize;

        unsigned int *vTablePlainArray;
        int dimPlainArray = vTableDimension(gridSize);
        cudaMalloc((void **)&vTablePlainArray, dimPlainArray * sizeof(unsigned int));
        moveVtableToPlainArray<<<1, 1>>>(vTablePlainArray, dimPlainArray, _grid);

        gridCpu->vTable1 = new unsigned int[dimPlainArray]();

        cudaMemcpy(gridCpu->vTable1, vTablePlainArray,
                   dimPlainArray * sizeof(unsigned int), cudaMemcpyDeviceToHost);

        cudaFree(vTablePlainArray);

        return gridCpu;
    }

    void voxelization::SurfaceVoxelization::create(
        std::shared_ptr<mesh::MeshData> meshData, float gridVoxelSize,
        std::shared_ptr<mesh::ICopyMeshData2Device> copyMeshData2Device)
    {

        //LoggerConfig logger;

        _globalTranslation.resize(3);
        _globalTranslation[0] = meshData->bbox[0];
        _globalTranslation[1] = meshData->bbox[1];
        _globalTranslation[2] = meshData->bbox[2];

        int gridSize = ops::computeTwoLevelSizes(meshData->bbox, gridVoxelSize);

        // Calculate required GPU memory
        size_t requiredMemory = sizeof(Grid);  // Base structure
        requiredMemory += vTableDimension(gridSize) * sizeof(unsigned int);  // vTable size

        // Check available GPU memory
        size_t free_memory, total_memory;
        CUDA_CHECK(cudaMemGetInfo(&free_memory, &total_memory));

        if (requiredMemory > free_memory) {
            LOG(ERROR) << "Required memory (" << requiredMemory / (1024*1024)
                      << "MB) exceeds available GPU memory ("
                      << free_memory / (1024*1024) << "MB)";
            throw std::runtime_error("Insufficient GPU memory - try increasing voxel size");
        }

        ////// Compute grid-1 Voxel size (_gridVoxelSize)
        float meshBboxSize[3] = {meshData->bbox[3] - meshData->bbox[0],
                                 meshData->bbox[4] - meshData->bbox[1],
                                 meshData->bbox[5] - meshData->bbox[2]};

        LOG(INFO) << "GPU VOXELIZATION INFORMATION:";
        LOG(INFO) << "Grid Size: " << gridSize;
        LOG(INFO) << "Voxel Size: " << gridVoxelSize;

        int gridSizedims[3] = {gridSize, gridSize,
                               gridSize}; //// Include more general case with
                                          /// gridSizedims all different --> TODO

        ///// Create _grid
        //_grid = constructorgrid(gridSize, gridVoxelSize);
        _grid = constructorgridGpu(gridSize, gridVoxelSize);
        cudaDeviceSynchronize();

        mesh::MeshDataDevice *meshDataGpu;

        CUDA_CHECK(cudaMalloc((void **)&meshDataGpu, sizeof(mesh::MeshDataDevice)));

        try {
            copyMeshData2Device->copyData(meshDataGpu, meshData);
            CUDA_CHECK(cudaGetLastError()); // Check kernel launch
        } catch (const std::exception& e) {
            cudaError_t err = cudaFree(meshDataGpu);
            if (err != cudaSuccess) {
                LOG(WARNING) << "Failed to free CUDA memory during error cleanup: "
                            << cudaGetErrorString(err);
            }
            throw;  // Re-throw the original exception
        }

        int totNumThreads = meshData->triangles.size();
        int numThreads = 256;
        int numBlocks = ceilf(1.0f * totNumThreads / (1.0f * numThreads));

        //// 2nd launch kernels for voxelization /////
        auto start_time = std::chrono::high_resolution_clock::now();

        voxelGPU<<<numBlocks, numThreads>>>(_grid, meshDataGpu, totNumThreads);

        cudaDeviceSynchronize();

        auto end_time = std::chrono::high_resolution_clock::now();
        auto time = end_time - start_time;

        LOG(INFO)
            << "GPU surface-voxelization took: "
            << std::chrono::duration_cast<std::chrono::microseconds>(time).count() /
                   1000.0
            << " milliseconds.\n";

        copyMeshData2Device->deleteMeshDataDevice(meshDataGpu);
    }

  }

bool createDirectory(const std::string &path)
{
  // Attempt to create the directory, with recursive option
  if (mkdir(path.c_str(), 0777) == -1)
  {
    if (errno != EEXIST)
    {
      std::cerr << "Error creating directory: " << strerror(errno) << std::endl;
      return false;
    }
  }
  return true;
}

void Voxel2Mesh::convertVoxel2Mesh(Grid *grid,
                                   std::vector<float> globalTranslation,
                                   std::string path,
                                   std::string nameVoxelizedMesh)
{
  auto start_time = std::chrono::high_resolution_clock::now();

  // Check file extension
  if (nameVoxelizedMesh.length() < 4 ||
      nameVoxelizedMesh.substr(nameVoxelizedMesh.length() - 4) != ".obj") {
    LOG(WARNING) << "File extension missing or not .obj, appending .obj extension";
    nameVoxelizedMesh += ".obj";
  }

  Grid *gridCpu = new Grid();

  LOG(INFO) << "GPU to CPU VOXELIZATION COPY" << std::endl;

  ICopyVoxelDataGpu2Cpu *copyTest = new CopyVoxelDataGpu2Cpu();
  copyTest->copyDataL1(gridCpu, grid);

  LOG(INFO) << "GPU to CPU VOXELIZATION COPY finalized" << std::endl;

  std::vector<float> voxelPtsL1;
  auto gridSize = gridCpu->gridSize;
  auto gridVoxelSize = gridCpu->gridVoxelSize;

  LOG(INFO) << "Creating CPU mesh of voxelization" << std::endl;

  for (size_t loc1 = 0; loc1 < gridSize * gridSize * gridSize; ++loc1)
  {

    int3 posXYZL1;
    if (checkBit(gridCpu->vTable1, loc1))
    {

      posXYZL1 = linearTo3dIndex(loc1, gridSize);

      voxelPtsL1.push_back(posXYZL1.x * gridVoxelSize + globalTranslation[0]);
      voxelPtsL1.push_back(posXYZL1.y * gridVoxelSize + globalTranslation[1]);
      voxelPtsL1.push_back(posXYZL1.z * gridVoxelSize + globalTranslation[2]);
    }
  }

  std::string file = path + nameVoxelizedMesh;
  createDirectory(path);

  openmesh_app::VoxelToMesh voxelMeshL1(gridVoxelSize);

  try {
    voxelMeshL1.createMesh(voxelPtsL1, file);
    LOG(INFO) << "Mesh saved to: " << file;
  } catch (const std::exception& e) {
    LOG(ERROR) << "Failed to save mesh to " << file << ": " << e.what();
    throw;
  }

  auto end_time = std::chrono::high_resolution_clock::now();
  auto time = end_time - start_time;

  LOG(INFO)
      << "GPU-CPU voxelization copy plus CPU triangulation took: "
      << std::chrono::duration_cast<std::chrono::microseconds>(time).count() /
             1000.0
      << " milliseconds.\n";

  return;
}

      __host__ __device__ __inline__ bool checkBit(unsigned int *voxel_table,
                                                   size_t &index)
      {
        size_t int_location = index / size_t(32);
        unsigned int bit_pos = size_t(31) - (index % size_t(32));
        return ((voxel_table[int_location]) & (1 << bit_pos));
      };

      __device__ __inline__ void killBit(unsigned int *voxel_table, size_t &index)
      {
        size_t int_location = index / size_t(32);
        unsigned int bit_pos = size_t(31) - (index % size_t(32));
        unsigned int mask = 0 << bit_pos;
        atomicAnd(&(voxel_table[int_location]), mask);
      };

      __device__ __inline__ void setBit(unsigned int *voxel_table, size_t &index)
      {
        size_t int_location = index / size_t(32);
        unsigned int bit_pos = size_t(31) - (index % size_t(32));
        unsigned int mask = 1 << bit_pos;
        atomicOr(&(voxel_table[int_location]), mask);
        return;
      };

      __device__ __inline__ void flipBit(unsigned int *voxel_table, size_t &index)
      {
        size_t int_location = index / size_t(32);
        unsigned int bit_pos = size_t(31) - (index % size_t(32));
        unsigned int mask = 1 << bit_pos;
        atomicXor(&(voxel_table[int_location]), mask);
        return;
      };

      // given the linear voxel coordinate returns the voxel coordinates in x,y,z
      __host__ __device__ inline int3 linearTo3dIndex(size_t loc, float gridSize)
      {

        int a = int(loc / gridSize);
        int b = int(1.0 * loc / (1.0 * powf(gridSize, 2.0f)));
        int3 pos3D;
        pos3D.x = int(loc - a * gridSize);
        pos3D.y = int(a - b * gridSize);
        pos3D.z = b;
        return pos3D;
      };

      // given the voxel coordinates in x,y,z returns the linear voxel coordinate
      __host__ __device__ inline size_t index3DTolinear(int x, int y, int z,
                                                        float gridSize)
      {

        size_t loc;

        loc = x + gridSize * y + gridSize * gridSize * z;

        return loc;
      };

      // Convert a bitMap to an array of integres:0,1
      inline void bitMapTointeger(unsigned int *bitMap, int size,
                                  std::vector<int> &vectorMap)
      {

        if (bitMap != NULL)
          for (size_t loc = 0; loc < size; ++loc)
          {
            if (checkBit(bitMap, loc))
              vectorMap.push_back(loc);
          }
      };

      /**
       * @file
       */

      /**
       * @brief Given the size of a voxel and the mesh bounding box computes the
       * number of 1-d voxels
       *
       * @param[in] bboxMeshMinMax mesh bounding box in min max parameterization
       * @param[out] gridVoxelSize linear size of the voxels
       * @return number of voxels in one dimension (i.e. voxelVolume =
       * gridVoxelSize**3)
       * @exceptsafe This function does not throw exceptions.
       */
      inline float computeTwoLevelSizes(std::vector<float> bboxMeshMinMax,
                                        float gridVoxelSize)
      {

        float meshDimensions[3] = {bboxMeshMinMax[3] - bboxMeshMinMax[0],
                                   bboxMeshMinMax[4] - bboxMeshMinMax[1],
                                   bboxMeshMinMax[5] - bboxMeshMinMax[2]};

        int posLarge = std::distance(
            meshDimensions, std::max_element(meshDimensions, meshDimensions + 3));
        float largestSize = meshDimensions[posLarge];

        int gridSize = ceilf(largestSize / gridVoxelSize);

        return gridSize;
      };

    }

  }

import base64
import logging
from pathlib import Path
from typing import Dict, Any

import requests

from config import UPLOADED_DATA_STREAMLIT_APP
from connector.connector import Connector
from shared_utils.VoxelizationJobMetadata import VoxelizationJobMetadata

POST_ENDPOINT = "http://127.0.0.1:8001/voxelize/"
GET_FILE_ENDPOINT = "http://127.0.0.1:8001/result/"
GET_JOB_METADATA_ENDPOINT = "http://127.0.0.1:8001/status/"

logging.basicConfig(
    level=logging.INFO, format="%(asctime)s - %(levelname)s - %(message)s"
)


class ConnectorRestApi(Connector):
    def run_voxelization_job(
        self, voxel_size: float, mesh_file_path: Path, voxel_file_name: Path
    ) -> str:
        with open(mesh_file_path, "rb") as file:
            # Send the POST request

            params = {
                "voxel_size": str(voxel_size),
                "mesh_file_path": str(mesh_file_path),
                "voxel_file_name": str(voxel_file_name),
            }

            response = requests.post(
                POST_ENDPOINT,
                files={
                    "mesh_file": (str(mesh_file_path), file, "application/octet-stream")
                },
                data=params,
            )

        return response.json()["job_id"]

    @staticmethod
    def _save_file_sync(content: bytes, file_path: str):
        try:
            path = Path(file_path)
            path.parent.mkdir(parents=True, exist_ok=True)  # Ensure directories exist
            with open(file_path, "wb") as f:
                f.write(content)
            print(f"File saved to {file_path}")
        except Exception as e:
            print(f"Error saving file: {e}")
            raise e

    def get_voxelization_job(self, job_id: str) -> VoxelizationJobMetadata | None:
        get_file_url = GET_FILE_ENDPOINT + job_id
        file_response = requests.get(get_file_url)
        get_job_status_url = GET_JOB_METADATA_ENDPOINT + job_id
        job_response = requests.get(get_job_status_url)

        if file_response.status_code == 200:
            voxelization_job_metadata = VoxelizationJobMetadata.from_json(
                job_response.json()
            )
            logging.info(f"Job Status: {voxelization_job_metadata.status}")
            voxelization_file_path = (
                UPLOADED_DATA_STREAMLIT_APP / voxelization_job_metadata.voxel_file_name
            )

            self._save_file_sync(
                content=file_response.content, file_path=voxelization_file_path
            )

            return voxelization_job_metadata
        else:
            logging.info(f"Error: {file_response.status_code}, {file_response.text}")
            return None

from pathlib import Path

from config import UPLOADED_DATA_STREAMLIT_APP
from gui_components.model import Model
from gui_components.streamlit_view import StreamlitView

VOXELIZATION_FILE_NAME_PREFIX = "voxelization"
LOG_FILE_NAME = "gpu_voxelization.INFO"


class Controller:
    def __init__(self, model: Model, view: StreamlitView, app_title: str):
        self.model = model
        self.view = view
        self.app_title = app_title

    @staticmethod
    def _file_name_validation(file_name: str) -> bool:
        if not (file_name.endswith(".stl") or file_name.endswith(".obj")):
            raise ValueError(f"So far only .stl or .obj files supported.")
        else:
            return True

    def run_api(self):
        self.view.app_title(self.app_title)

        # upload file button
        uploaded_file = self.view.upload_file()

        # creates sidebar menu with color changer
        self.view.create_sidebar()

        if uploaded_file:
            self._file_name_validation(uploaded_file.name)
            # Creates temp copy of mesh. Needed because streamlit load the mesh file in cash. See create_temp_mesh docs.
            self.view.create_temp_mesh(uploaded_file)

            # Create plotter for pyvista and perform stpyvista rendering
            plotter, _ = self.model.create_rendering_plotter_and_actor(
                mesh_file_path=self.view.view_setup.mesh_file,
                color=self.view.view_setup.mesh_color,
            )
            self.view.rendering(plotter=plotter)

            # create voxelization button
            self.view.voxelization_button()

            if self.view.view_setup.voxel_button_press:
                mesh_file = self.view.view_setup.mesh_file

                (
                    original_file_name,
                    extension,
                ) = self.view.view_setup.mesh_file.name.split("local")
                voxel_file_name = Path(
                    original_file_name + VOXELIZATION_FILE_NAME_PREFIX + ".obj"
                )

                voxelization_job = self.model.voxelization(
                    mesh_file=mesh_file,
                    voxel_size=self.view.view_setup.voxel_size,
                    file_name=voxel_file_name,
                )

                self.view.view_setup.voxel_file_path = UPLOADED_DATA_STREAMLIT_APP
                self.view.view_setup.voxel_file_name = voxel_file_name

            if (
                self.view.view_setup.voxel_button_press
                or self.view.view_setup.voxel_file_path
            ):
                full_path = self.view.view_setup.voxel_file_path / Path(
                    self.view.view_setup.voxel_file_name
                )
                plotter, _ = self.model.create_rendering_plotter_and_actor(
                    mesh_file_path=full_path, color=self.view.view_setup.mesh_color
                )
                self.view.rendering(plotter=plotter)

                self.view.show_voxelizer_info(
                    self.model.model_setup.voxelization_job_metadata.log_info

import time
from dataclasses import dataclass
from pathlib import Path
from typing import Tuple
import pyvista as pv

from connector.connector import Connector
from shared_utils.VoxelizationJobMetadata import VoxelizationJobMetadata, JobStatus


@dataclass
class ModelSetup:
    camera_position: str = "xy"
    camera_azimuth: float = 15
    camera_elevation: float = -50
    plotter_screen_size: Tuple[int, int] = (200, 500)
    pv_plotter: pv.Plotter | None = None
    voxelization_job_metadata: VoxelizationJobMetadata | None = None


class Model:
    def __init__(self, connector: Connector, model_setup: ModelSetup) -> None:
        self.model_setup = model_setup
        self.model_setup.pv_plotter = pv.Plotter(
            border=True, window_size=self.model_setup.plotter_screen_size
        )
        self.connector = connector

    def voxelization(
        self, mesh_file: Path, voxel_size: float, file_name: Path
    ) -> VoxelizationJobMetadata:
        # POST: Will run the voxelization on the server side
        job_id = self.connector.run_voxelization_job(
            mesh_file_path=mesh_file,
            voxel_size=voxel_size,
            voxel_file_name=file_name,
        )

        # GET: Check if the job is done
        voxelization_job_metadata = self.connector.get_voxelization_job(job_id=job_id)

        # GET: voxel file saved to mesh_file_path
        self.model_setup.voxelization_job_metadata = voxelization_job_metadata
        return voxelization_job_metadata

    def create_rendering_plotter_and_actor(
        self,
        mesh_file_path: Path | None = None,
        color: str | None = "#BDA0A0",
        opacity: float | None = 1.0,
    ) -> Tuple[pv.Plotter, pv.Actor]:
        if mesh_file_path is None:
            raise ValueError(f"The Mesh file was not provided")

        file_extension = "." + mesh_file_path.name.split(".")[-1]
        if file_extension == ".stl":
            reader = pv.STLReader(str(mesh_file_path))
        elif file_extension == ".obj":
            reader = pv.OBJReader(str(mesh_file_path))
        else:
            raise ValueError(f"Unsupported file extension {file_extension}")

        plotter = self.model_setup.pv_plotter

        mesh = reader.read()
        actor = plotter.add_mesh(
            mesh, color=color, opacity=opacity if opacity is None else opacity
        )

        plotter.camera_position = self.model_setup.camera_position
        plotter.camera.azimuth = self.model_setup.camera_azimuth
        plotter.camera.elevation = self.model_setup.camera_elevation
        return plotter, actor
```

import pyvista as pv
import streamlit as st
from dataclasses import dataclass
from pathlib import Path
from typing import Tuple, Any, Literal
from stpyvista import stpyvista
from streamlit.runtime.uploaded_file_manager import UploadedFile

from config import UPLOADED_DATA_STREAMLIT_APP


@dataclass
class AppState:
    mesh_color: str = "#BDA0A0"
    opacity: float = 1.0
    screen_size: Tuple[int, int] = (200, 500)
    mesh_file: Path | None = None
    voxel_file_path: Path | None = None
    voxel_file_name: str | None = None
    voxel_button_press: bool = False
    voxel_size: float | None = None


class StreamlitView:
    def __init__(self, app_state: AppState):
        self.view_setup = app_state

    def rendering(self, plotter: pv.Plotter | Any):
        stpyvista(plotter)
        # Show voxel button as soon mesh rendering is visualized
        self.view_setup.voxel_button_press = True

    def create_temp_mesh(self, uploaded_file: UploadedFile):
        """
        This is needed in order to access the mesh file and manipulate it with the Model, like changing color, opacity,
        etc. This trick must be done because streamlit create a copy of the upload file in UploadedFile.
        :param uploaded_file: usually the file uploaded from the 'upload_file' widget member function.
        :return:
        """

        file_name, file_extension = uploaded_file.name.split(".")
        file_extension = "." + file_extension
        tmp_file_name = file_name + "_local" + file_extension

        temp_mesh_path = UPLOADED_DATA_STREAMLIT_APP / tmp_file_name
        self.view_setup.mesh_file = temp_mesh_path

        temp_mesh_path.parent.mkdir(parents=True, exist_ok=True)

        if not temp_mesh_path.exists():
            with open(temp_mesh_path, "wb") as temp_file:
                temp_file.write(uploaded_file.read())

    def create_sidebar(self):
        with st.sidebar:
            # side_bar_img = ""
            # st.sidebar.image(side_bar_img)

            st.title("OPTIONS")

            color = st.color_picker(
                "Color mesh",
                value=self.view_setup.mesh_color,
                key="color_mesh",
            )

            self.view_setup.mesh_color = color

    def voxelization_button(self):
        with st.sidebar:
            voxel_size = st.number_input(
                "Enter the voxel size:",
                value=0.7,
                min_value=0.2,
                max_value=2.0,
                step=0.1,
                format="%.2f",
            )

            self.view_setup.voxel_button_press = st.button("Start voxelization")
            self.view_setup.voxel_size = voxel_size

    @staticmethod
    def start_streamlit_app(layout: Literal["centered", "wide"] = "centered"):
        st.set_page_config(layout=layout)

    @staticmethod
    def app_title(title: str):
        st.title(title)

    @staticmethod
    def upload_file() -> UploadedFile:
        uploaded_file = st.file_uploader("Choose a file", type=["stl", "obj"])
        return uploaded_file

    @staticmethod
    def show_voxelizer_info(log_data: str):
        with st.sidebar:
            if log_data:
                st.header("Voxelizer information")
                with st.expander("", expanded=True):
                    st.markdown(log_data, unsafe_allow_html=True)
