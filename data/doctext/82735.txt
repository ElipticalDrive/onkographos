# Neural Volume Rendering and Surface Rendering

Implementation for the CMU *Learning for 3D Vision* assignment on neural volume rendering and surface rendering with implicit representations.

This repo covers:

- **SDF → density** volume rendering, and fitting analytic SDFs from multi-view images
- **NeRF** training and surround-view rendering
- **Sphere tracing** for analytic / neural SDF surfaces
- Fitting a neural SDF from point clouds (with Eikonal regularization)
- **VolSDF**-style training from multi-view images (geometry + appearance)

## Setup

Dependencies are listed in [`environment.yml`](environment.yml) (Python 3.9, PyTorch, PyTorch3D, Hydra, etc.). Create the conda env with:

```bash
conda env create -f environment.yml
conda activate l3d
```

Before running scripts:

1. **Set `PYTHONPATH`** so imports resolve from `src/`:

   ```bash
   # Linux / macOS
   export PYTHONPATH="$PWD/src:$PYTHONPATH"

   # Windows PowerShell
   $env:PYTHONPATH = "$PWD\src;$env:PYTHONPATH"
   ```

   In Cursor / VS Code you can also use the existing debug config (`.vscode/launch.json` already sets `PYTHONPATH`).

2. **Set the workspace path**  
   Edit [`src/Pathes.py`](src/Pathes.py) and point `wsf` to this repository’s absolute path on your machine.

3. **Prepare data**  
   `data/` is gitignored by default. Scripts expect assets such as:

   | Path | Use |
   | --- | --- |
   | `data/box/` | Multi-view box images (`.npy` / `.png`) |
   | `data/lego/` | Lego scene for NeRF / VolSDF |
   | `data/bunny_pointcloud.npz` | Point cloud for neural SDF fitting |

   NeRF datasets can also be downloaded on demand via [`src/utils/dataset.py`](src/utils/dataset.py) (`lego`, `fern`, etc.).

## Config ↔ module map

| Kind | Config field | Implementation |
| --- | --- | --- |
| SDF volume | `implicit_function.type: sdf_volume` | `SDFVolume` |
| NeRF | `nerf` | `NeuralRadianceField` |
| Analytic surface | `sdf_surface` | `SDFSurface` (Sphere / Box / Torus) |
| Neural surface | `neural_surface` | `NeuralSurface` |
| Volume renderer | `renderer.type: volume` | `VolumeRendererChunk` |
| Sphere tracing | `sphere_tracing` | `SphereTracingRendererChunk` |
| VolSDF renderer | `volume_sdf` | `VolumeSDFRendererChunk` |
| Sampler | `sampler.type: stratified` | Stratified sampling |

Learning rate, epochs, samples per ray, `depth_range`, Eikonal weights, etc. are all set in the corresponding YAML files.

## Notes

- Training and rendering expect **CUDA**.
- Set `training.resume` in the config to continue from a checkpoint.
- Hydra writes run logs under `outputs/` (gitignored).
- If mesh extraction fails, scripts may print `Empty mesh` and skip the geometry GIF — that is intentional fallback behavior.

### src/losses.py
```
import torch


def eikonal_loss(gradients: torch.Tensor):
    # TODO (Q6): Implement eikonal loss
    eikonal_loss = ((gradients.norm(2, dim=1) - 1.0) ** 2).mean()
    return eikonal_loss


def sphere_loss(signed_distance, points, radius=1.0):
    return torch.square(signed_distance[..., 0] - (torch.norm(points, dim=-1) - radius)).mean()


def get_random_points(num_points, bounds, device):
    min_bound = torch.tensor(bounds[0], device=device).unsqueeze(0)
    max_bound = torch.tensor(bounds[1], device=device).unsqueeze(0)
    return torch.rand((num_points, 3), device=device) * (max_bound - min_bound) + min_bound


def select_random_points(points, n_points):
    points_sub = points[torch.randperm(points.shape[0])]
    return points_sub.reshape(-1, 3)[:n_points]
```

### Scripts/1 Volume/2 Train.py
```
import imageio
import numpy as np
import torch
import os
import hydra
import tqdm
from omegaconf import DictConfig

from pipeline.volume import VolumeModel, render_imagesV, render_imagesV_
from utils.data_utils import create_surround_cameras, dataset_from_config
from ray.utils import (
    sample_images_at_xy,
    get_random_pixels_from_image,
    get_rays_from_pixels,
)
from Pathes import P


@hydra.main(config_path=f"{P.wsf}/configs/Volume", config_name="train_box")
def main(cfg: DictConfig):
    torch.manual_seed(42)
    os.chdir(P.wsf)
    # Create model
    model = VolumeModel(cfg)
    model = model.cuda()
    model.train()
    # Create dataset
    train_dataset = dataset_from_config(cfg.data)
    train_dataloader = torch.utils.data.DataLoader(
        train_dataset,
        batch_size=1,
        shuffle=True,
        num_workers=0,
        collate_fn=lambda batch: batch,
    )
    image_size = cfg.data.image_size
    optimizer = torch.optim.Adam(model.parameters(), lr=cfg.training.lr)
    # Render images before training for Comparison
    cameras = [item["camera"] for item in train_dataset]
    render_imagesV_(model, cameras, image_size, save=True, file_prefix="images/part_2/before")
    # Train
    t_range = tqdm.tqdm(range(cfg.training.num_epochs))
    torch.autograd.set_detect_anomaly(True)
    for epoch in t_range:
        for iteration, batch in enumerate(train_dataloader):
            image, camera, camera_idx = batch[0].values()
            image = image.cuda()
            camera = camera.cuda()
            # Sample rays
            xy_grid = get_random_pixels_from_image(cfg.training.batch_size, image_size, camera)  
            ray_bundle = get_rays_from_pixels(xy_grid, camera)
            rgb_gt = sample_images_at_xy(image, xy_grid)

            feature, depth = model(ray_bundle)
            loss = torch.nn.functional.mse_loss(rgb_gt, feature)

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

        if (epoch % 10) == 0:
            t_range.set_description(f"Epoch: {epoch:04d}, Loss: {loss:.06f}")
            t_range.refresh()
    print("Box center:", tuple(np.array(model.implicit_fn.sdf.center.data.detach().cpu()).tolist()[0]))
    print("Box side lengths:", tuple(np.array(model.implicit_fn.sdf.side_lengths.data.detach().cpu()).tolist()[0]))
    # Render images after training
    render_imagesV_(model, cameras, image_size, save=True, file_prefix="images/part_2/after")
    all_images = render_imagesV_(
        model,
        create_surround_cameras(3.0, n_poses=20),
        image_size,
        file_prefix="part_2",
    )
    imageio.mimsave("images/part_2/part_2.gif", [np.uint8(im * 255) for im in all_images])

if __name__ == "__main__":
    main()
```

```
import os
import hydra
import numpy as np
import torch
import tqdm
import imageio
from omegaconf import DictConfig

from pipeline.volume import render_imagesV_, render_imagesV
from ray.utils import (
    sample_images_at_xy,

    model, optimizer, lr_scheduler, start_epoch, checkpoint_path = create_model(cfg)

    for epoch in t_range:
        for batch in train_dataloader:
            image, camera, camera_idx = batch[0].values()

            camera = camera.cuda()

            xy_grid = get_random_pixels_from_image(cfg.training.batch_size, cfg.data.image_size, camera)
            ray_bundle = get_rays_from_pixels(xy_grid, camera)
            rgb_gt = sample_images_at_xy(image, xy_grid)

            feature, depth = model(ray_bundle)
            loss = torch.nn.functional.mse_loss(rgb_gt, feature)

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            t_range.set_description(f"Epoch: {epoch:04d}, Loss: {loss:.06f}")
            t_range.refresh()

        if (
            epoch % cfg.training.checkpoint_interval == 0
            and len(cfg.training.checkpoint_path) > 0
            and epoch > 0

            print(f"Storing checkpoint {checkpoint_path}.")

                "model": model.state_dict(),
                "optimizer": optimizer.state_dict(),
                "epoch": epoch,

        if epoch % cfg.training.render_interval == 0 and epoch > 0:
            with torch.no_grad():
                cameras = create_surround_cameras(4.0, n_poses=20, up=(0.0, 0.0, 1.0), focal_length=2.0)
```

### Scripts/1 Volume/util.py
```
import os
import torch
from pipeline.volume import VolumeModel
from Pathes import P


def create_model(cfg):
    model = VolumeModel(cfg)
    model.cuda()
    model.train()
    # Load checkpoints
    optimizer_state_dict = None
    start_epoch = 0
    checkpoint_path = os.path.join(P.wsf, cfg.training.checkpoint_path)
    if len(cfg.training.checkpoint_path) > 0:
        # Make the root of the experiment directory.
        checkpoint_dir = os.path.split(checkpoint_path)[0]
        os.makedirs(checkpoint_dir, exist_ok=True)

        # Resume training if requested.
        if cfg.training.resume and os.path.isfile(checkpoint_path):
            print(f"Resuming from checkpoint {checkpoint_path}.")
            loaded_data = torch.load(checkpoint_path)
            model.load_state_dict(loaded_data["model"])
            start_epoch = loaded_data["epoch"]

            print(f"   => resuming from epoch {start_epoch}.")
            optimizer_state_dict = loaded_data["optimizer"]
    optimizer = torch.optim.Adam(model.parameters(), lr=cfg.training.lr)
    # Load the optimizer state dict in case we are resuming.
    if optimizer_state_dict is not None:
        optimizer.load_state_dict(optimizer_state_dict)
        optimizer.last_epoch = start_epoch

    # The learning rate scheduling is implemented with LambdaLR PyTorch scheduler.
    def lr_lambda(epoch):
        gamma = cfg.training.lr_scheduler_gamma
        step_size = cfg.training.lr_scheduler_step_size
        return gamma ** (epoch / step_size)

    lr_scheduler = torch.optim.lr_scheduler.LambdaLR(
        optimizer, lr_lambda, last_epoch=start_epoch - 1, verbose=False
    )
    return model, optimizer, lr_scheduler, start_epoch, checkpoint_path
```

### Scripts/2 Surface/2 Train_Points.py
```
import hydra
import numpy as np
import torch
import tqdm
import imageio

from losses import eikonal_loss, get_random_points, select_random_points
from utils.data_utils import create_surround_cameras
from utils.render_functions import render_geometry, render_points_with_save
from util import create_model, pretrain_sdf
from Pathes import P


@hydra.main(config_path=f"{P.wsf}/configs/Surface", config_name="points_surface")
def main(cfg):
    model, optimizer, lr_scheduler, start_epoch, checkpoint_path = create_model(cfg)
    pretrain_sdf(cfg, model)
    # Load pointcloud
    point_cloud = np.load(f"{P.wsf}/{cfg.data.point_cloud_path}")
    all_points = torch.Tensor(point_cloud["verts"][::2]).cuda().view(-1, 3)
    all_points = all_points - torch.mean(all_points, dim=0).unsqueeze(0)

    point_images = render_points_with_save(
        all_points.unsqueeze(0),
        create_surround_cameras(3.0, n_poses=20, up=(0.0, 1.0, 0.0), focal_length=2.0),
        cfg.data.image_size,
        file_prefix="points",
    )
    imageio.mimsave(f"{P.wsf}/images/part_6_input.gif", [np.uint8(im * 255) for im in point_images])

    bs = cfg.training.batch_size
    t_range = tqdm.tqdm(range(start_epoch, cfg.training.num_epochs))
    for epoch in t_range:
        for idx in range(0, all_points.shape[0], bs):
            points = select_random_points(all_points, bs)
            distances = model.implicit_fn.get_distance(points)
            loss = distances.norm(dim=-1).mean()
            point_loss = loss

            eikonal_points = get_random_points(bs, cfg.training.bounds, "cuda")
            eikonal_distances, eikonal_gradients = model.implicit_fn.get_distance_and_gradient(eikonal_points)
            loss += torch.exp(-1e2 * torch.abs(eikonal_distances)).mean() * cfg.training.inter_weight
            loss += eikonal_loss(eikonal_gradients) * cfg.training.eikonal_weight

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
        t_range.set_description(f"Epoch: {epoch:04d}, Loss: {point_loss:.06f}")
        t_range.refresh()

        if (
            epoch % cfg.training.checkpoint_interval == 0
            and len(cfg.training.checkpoint_path) > 0
            and epoch > 0
        ):
            print(f"Storing checkpoint {checkpoint_path}.")
            data_to_store = {
                "model": model.state_dict(),
                "optimizer": optimizer.state_dict(),
                "epoch": epoch,
            }
            torch.save(data_to_store, checkpoint_path)

        if epoch % cfg.training.render_interval == 0 and epoch > 0:
            try:
                test_images = render_geometry(
                    model,
                    create_surround_cameras(3.0, n_poses=20, up=(0.0, 1.0, 0.0), focal_length=2.0),
                    cfg.data.image_size,
                    file_prefix="eikonal",
                    thresh=0.002,
                )
                imageio.mimsave("images/part_6.gif", [np.uint8(im * 255) for im in test_images])
            except Exception as e:
                print("Empty mesh")
                pass


if __name__ == "__main__":
    main()
```

```
from losses import get_random_points, sphere_loss

def create_model(cfg):
    model = SurfaceModel(cfg)
    model.cuda()
    model.train()

    optimizer_state_dict = None
    start_epoch = 0

    checkpoint_path = os.path.join(P.wsf, cfg.training.checkpoint_path)

    if len(cfg.training.checkpoint_path) > 0:
        # Make the root of the experiment directory.
        checkpoint_dir = os.path.split(checkpoint_path)[0]
        os.makedirs(checkpoint_dir, exist_ok=True)

        # Resume training if requested.
        if cfg.training.resume and os.path.isfile(checkpoint_path):
            print(f"Resuming from checkpoint {checkpoint_path}.")
            loaded_data = torch.load(checkpoint_path)
            model.load_state_dict(loaded_data["model"])
            start_epoch = loaded_data["epoch"]

            print(f"   => resuming from epoch {start_epoch}.")
            optimizer_state_dict = loaded_data["optimizer"]

    # Initialize the optimizer.
    optimizer = torch.optim.Adam(
        model.parameters(),
        lr=cfg.training.lr,
    )

    # Load the optimizer state dict in case we are resuming.
    if optimizer_state_dict is not None:
        optimizer.load_state_dict(optimizer_state_dict)
        optimizer.last_epoch = start_epoch

    # The learning rate scheduling is implemented with LambdaLR PyTorch scheduler.
    def lr_lambda(epoch):
        return cfg.training.lr_scheduler_gamma ** (
            epoch / cfg.training.lr_scheduler_step_size
        )

    lr_scheduler = torch.optim.lr_scheduler.LambdaLR(
        optimizer, lr_lambda, last_epoch=start_epoch - 1, verbose=False
    )

    return model, optimizer, lr_scheduler, start_epoch, checkpoint_path


def pretrain_sdf(cfg, model):
    optimizer = torch.optim.Adam(
        model.parameters(),
        lr=cfg.training.lr,
    )

    # Run the main training loop.
    for iter in range(0, cfg.training.pretrain_iters):
        points = get_random_points(cfg.training.batch_size, cfg.training.bounds, "cuda")

        # Run model forward
        distances = model.implicit_fn.get_distance(points)
        loss = sphere_loss(distances, points, 1.0)

        # Take the training step
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
```

```
import torch
import torch.nn.functional as F


class HarmonicEmbedding(torch.nn.Module):
    def __init__(
        self,
        in_channels: int = 3,
        n_harmonic_functions: int = 6,
        omega0: float = 1.0,
        logspace: bool = True,
        include_input: bool = True,
    ) -> None:
        super().__init__()

        if logspace:
            frequencies = 2.0 ** torch.arange(n_harmonic_functions, dtype=torch.float32)
        else:
            frequencies = torch.linspace(
                1.0,
                2.0 ** (n_harmonic_functions - 1),
                n_harmonic_functions,
                dtype=torch.float32,
            )

        self.register_buffer("_frequencies", omega0 * frequencies, persistent=False)
        self.include_input = include_input
        self.output_dim = n_harmonic_functions * 2 * in_channels

        if self.include_input:
            self.output_dim += in_channels

    def forward(self, x: torch.Tensor):
        embed = (x[..., None] * self._frequencies).view(*x.shape[:-1], -1)

        if self.include_input:
            return torch.cat((embed.sin(), embed.cos(), x), dim=-1)
        else:
            return torch.cat((embed.sin(), embed.cos()), dim=-1)
```

### src/ray/RayBundle.py

```
from torch import Tensor

class RayBundle:

    def __init__(self, origins: Tensor, directions: Tensor, sample_points: Tensor, sample_lengths: Tensor):
        self.origins = origins                  # (N, 3), actually have the same value, just for batch processing
        self.directions = directions            # (N, 3)
        self.sample_points = sample_points      # (N, n_pts_per_ray, 3)
        self.sample_lengths = sample_lengths    # (N, n_pts_per_ray, 1), distance of sample points from ray origin

    def __getitem__(self, idx):
        return RayBundle(
            self.origins[idx],
            self.directions[idx],
            self.sample_points[idx],
            self.sample_lengths[idx],
        )

    @property
    def shape(self):
        return self.origins.shape[:-1]

    @property
    def sample_shape(self):
        return self.sample_points.shape[:-1]

    def reshape(self, *args):
        return RayBundle(
            self.origins.reshape(*args, 3),
            self.directions.reshape(*args, 3),
            self.sample_points.reshape(*args, self.sample_points.shape[-2], 3),
            self.sample_lengths.reshape(*args, self.sample_lengths.shape[-2], 3),
        )

    def view(self, *args):
        return RayBundle(
            self.origins.view(*args, 3),
            self.directions.view(*args, 3),
            self.sample_points.view(*args, self.sample_points.shape[-2], 3),
            self.sample_lengths.view(*args, self.sample_lengths.shape[-2], 3),
        )

    def _replace(self, **kwargs):
        for key in kwargs.keys():
            setattr(self, key, kwargs[key])
        return self

    @property
    def device(self):
        return self.sample_points.device
```

```
# Sample image colors from pixel values
def sample_images_at_xy(images, xy_grid):
    batch_size = images.shape[0]
    spatial_size = images.shape[1:-1]
    xy_grid = -xy_grid.view(batch_size, -1, 1, 2)
    images_sampled = torch.nn.functional.grid_sample(
        images.permute(0, 3, 1, 2),
        xy_grid,
        align_corners=True,
        mode="bilinear",
    )
    return images_sampled.permute(0, 2, 3, 1).view(-1, images.shape[-1])


def get_normalized_image_plane_coordinates(image_size):
    W, H = image_size
    x = torch.linspace(-1, 1, steps=W)                              # (W)
    y = torch.linspace(-1, 1, steps=H)                              # (H)
    grid_y, grid_x = torch.meshgrid(y, x)                           # (H, W)
    xy_grid = torch.stack((grid_x, grid_y), dim=-1).view(W * H, 2)  # (H*W, 2)
    return -xy_grid


def get_random_pixels_from_image(n_pixels, image_size, camera):
    device = camera.device
    xy_grid = get_normalized_image_plane_coordinates(image_size).to(device)
    HW = xy_grid.shape[0]
    idx = torch.randperm(HW, device=device)[:n_pixels]
    return xy_grid[idx]


def get_rays_from_pixels(xy_grid, camera: CamerasBase):
    N = xy_grid.shape[0]
    xy_grid_copy = xy_grid.clone().to(camera.device)                                            # (N, 2)
    # ip : image plane
    ndc_pts_ip = torch.cat([xy_grid_copy, torch.ones([N, 1], device=camera.device)], dim=-1)    # (N, 3)
    w_pts_ip = camera.unproject_points(ndc_pts_ip, from_ndc=True, world_coordinates=True)       # (N, 3)
    w_rays_origins = camera.get_camera_center().repeat(N, 1)                                    # (N, 3)
    w_rays_directions = F.normalize(w_pts_ip - w_rays_origins)                                  # (N, 3)
    return RayBundle(
        w_rays_origins,
        w_rays_directions,
        torch.zeros_like(w_rays_origins).unsqueeze(1),
        torch.zeros_like(w_rays_origins).unsqueeze(1),
    )
```

### src/renderer/sphere_tracing.py
```
import torch
from implicit.function.surface import ImplicitSurface
from ray.RayBundle import RayBundle


class SphereTracingRenderer(torch.nn.Module):
    """沿射线用 SDF 逐步走到表面，再在交点取颜色"""
    def __init__(self, cfg):
        super().__init__()
        self.near = cfg.near
        self.far = cfg.far
        self.max_iters = cfg.max_iters
        self.mask_threshold = cfg.mask_threshold

    def sphere_tracing(self, implicit_fn: ImplicitSurface, ray_bundle: RayBundle):
        # 找射线与表面的交点
        pts = ray_bundle.origins.clone()      # (N, 3)
        for _ in range(self.max_iters):
            sdf = implicit_fn(pts)
            pts = pts + ray_bundle.directions * sdf
        sdf = implicit_fn(pts)                # (N, 1)
        mask_hit = sdf <= self.mask_threshold # (N, 1)
        return pts, mask_hit

    def forward(
        self,
        sampler,
        implicit_fn: ImplicitSurface,
        ray_bundle: RayBundle,
        light_dir=None,
    ):
        points, mask = self.sphere_tracing(implicit_fn, ray_bundle)
        mask = mask.repeat(1, 3)
        isect_points = points[mask].view(-1, 3)
        isect_color = implicit_fn.get_color(isect_points)
        color = torch.zeros_like(ray_bundle.origins)
        color[mask] = isect_color.view(-1)
        return color.view(-1, 3)


class SphereTracingRendererChunk(SphereTracingRenderer):

    def __init__(self, cfg):
        super().__init__(cfg)
        self._chunk_size = cfg.chunk_size

    def forward(
        self,
        sampler,
        implicit_fn: ImplicitSurface,
        ray_bundle: RayBundle,
        light_dir=None,
    ):
        N = ray_bundle.origins.shape[0]
        chunk_colors = []
        for chunk_start in range(0, N, self._chunk_size):
            cur_ray_bundle = ray_bundle[chunk_start : chunk_start + self._chunk_size]
            cur_color = super().forward(sampler, implicit_fn, cur_ray_bundle)
            chunk_colors.append(cur_color)
        color = torch.cat(chunk_colors, dim=0)
        return color
```

### src/renderer/utils.py
```
import torch

def get_deltas_from_depth(depth):
    _deltas = depth[..., 1:] - depth[..., :-1]       # (N, n_pts_per_ray - 1)
    deltas_ = 1e10 * torch.ones_like(depth[..., :1]) # (N, 1), last delta is infinity
    deltas = torch.cat((_deltas, deltas_), dim=-1)   # (N, n_pts_per_ray)
    return deltas.unsqueeze(-1)                      # (N, n_pts_per_ray, 1)


def compute_transmittance(deltas, density):
    N, n_pts_per_ray = deltas.shape[0:2]
    Ts = [None] * n_pts_per_ray
    Ts[0] = torch.ones((N, 1), dtype=deltas.dtype, device=deltas.device)
    for i in range(1, n_pts_per_ray):
        Ts[i] = Ts[i - 1] * torch.exp(-density[:, i - 1] * deltas[:, i - 1] + 1e-10)
    T = torch.stack(Ts, dim=1)
    return T


def aggregate(weights, features):
    L = weights * features
    feature = torch.sum(L, dim=1)
    return feature
```

```
import torch
from sampler import StratifiedRaysampler
from implicit.function.volume import ImplicitVolume
from ray.RayBundle import RayBundle
from .utils import get_deltas_from_depth, compute_transmittance, aggregate


class VolumeRenderer(torch.nn.Module):

    def __init__(self):
        super().__init__()

    def forward(  # each sample point is both absorber and emitter
        self,
        sampler: StratifiedRaysampler,
        implicit_fn: ImplicitVolume,
        ray_bundle: RayBundle,
    ):
        ray_bundle = sampler(ray_bundle)
        depth = ray_bundle.sample_lengths[..., 0]              # (N, ppr)
        deltas = get_deltas_from_depth(depth)                  # (N, ppr, 1)
        density, feature = implicit_fn(ray_bundle)
        density = density.view(-1, sampler.n_pts_per_ray, 1)   # (N, ppr, 1)
        feature = feature.view(-1, sampler.n_pts_per_ray, 3)   # (N, ppr, 3)
        T = compute_transmittance(deltas, density)             # (N, ppr, 1), how much light can be transmitted to camera origin
        alpha = 1.0 - torch.exp(-density * deltas + 1e-10)     # (N, ppr, 1)
        weights = T * alpha                                    # (N, ppr, 1)
        plane_feature = aggregate(weights, feature)            # (N, 3)
        plane_depth = aggregate(weights, depth.unsqueeze(-1))  # (N, 1)
        return plane_feature, plane_depth


class VolumeRendererChunk(VolumeRenderer):

    def __init__(self, cfg):
        super().__init__()
        self.chunk_size = cfg.chunk_size

    def forward(
        self,
        sampler: StratifiedRaysampler,
        implicit_fn: ImplicitVolume,
        ray_bundle: RayBundle,
    ):
        chunk_features = []
        chunk_depths = []
        N = ray_bundle.origins.shape[0]
        for chunk_start in range(0, N, self.chunk_size):
            cur_ray_bundle = ray_bundle[chunk_start : chunk_start + self.chunk_size]
            cur_feature, cur_depth = super().forward(sampler, implicit_fn, cur_ray_bundle)
            chunk_features.append(cur_feature)
            chunk_depths.append(cur_depth)
        # Concatenate chunk outputs
        feature = torch.cat(chunk_features, dim=0)
        depth = torch.cat(chunk_depths, dim=0)
        return feature, depth
```

### src/sampler/stratified.py
```
import torch
from ray.utils import RayBundle

# Stratified = Uniform
class StratifiedRaysampler(torch.nn.Module):

    def __init__(self, cfg):
        super().__init__()
        self.n_pts_per_ray = cfg.n_pts_per_ray
        self.depth_range = cfg.depth_range

    def forward(self, ray_bundle: RayBundle):
        o = ray_bundle.origins                              # (N, 3)
        d = ray_bundle.directions                           # (N, 3)
        o = o.unsqueeze(1).repeat(1, self.n_pts_per_ray, 1) # (N, n_pts_per_ray, 3)
        d = d.unsqueeze(1).repeat(1, self.n_pts_per_ray, 1) # (N, n_pts_per_ray, 3)
        N = o.shape[0]

        z_vals = torch.linspace(self.depth_range[0], self.depth_range[1], self.n_pts_per_ray)\
            .to(ray_bundle.device)                 # (n_pts_per_ray)
        z_vals = z_vals.unsqueeze(-1).unsqueeze(0) # (1, n_pts_per_ray, 1)
        ray_bundle.sample_points = o + d * z_vals  # (N, n_pts_per_ray, 3)
        ray_bundle.sample_lengths = z_vals * torch.ones([N, self.n_pts_per_ray, 1])\
            .to(ray_bundle.device)                 # (N, n_pts_per_ray, 1)
        return ray_bundle
```

```
def create_surround_cameras(radius, n_poses=20, up=(0.0, 1.0, 0.0), focal_length=1.0):
    cameras = []

            focal_length=torch.tensor([focal_length])[None],

            R=R,
            T=T,
        )
        cameras.append(camera)
    return cameras


def vis_grid(xy_grid, image_size):

    xy_vis = xy_vis.view(image_size[1], image_size[0], 3)
    xy_vis = np.array(xy_vis.detach().cpu())

    rays = rays.view(image_size[1], image_size[0], 3)
```

```
class ListDataset(Dataset):
    """
    A simple dataset made of a list of entries.
    """

    def __init__(self, entries: List) -> None:
        """
        Args:
            entries: The list of dataset entries.
        """
        self._entries = entries

    def __len__(self) -> int:
        return len(self._entries)

    def __getitem__(self, index):
        return self._entries[index]


def get_nerf_datasets(
    dataset_name: str,  # 'lego | fern'
    image_size: Tuple[int, int],
    data_root: str = P.data_dir,
    autodownload: bool = True,
) -> Tuple[Dataset, Dataset, Dataset]:
    """
    Obtains the training and validation dataset object for a dataset specified
    with the `dataset_name` argument.

    Args:
        dataset_name: The name of the dataset to load.
        image_size: A tuple (height, width) denoting the sizes of the loaded dataset images.
        data_root: The root folder at which the data is stored.
        autodownload: Auto-download the dataset files in case they are missing.

    Returns:
        train_dataset: The training dataset object.
        val_dataset: The validation dataset object.
        test_dataset: The testing dataset object.
    """

    if dataset_name not in ALL_DATASETS:
        raise ValueError(f"'{dataset_name}'' does not refer to a known dataset.")

    print(f"Loading dataset {dataset_name}, image size={str(image_size)} ...")

    cameras_path = os.path.join(data_root, dataset_name, dataset_name + ".pth")
    image_path = cameras_path.replace(".pth", ".png")

    if autodownload and any(not os.path.isfile(p) for p in (cameras_path, image_path)):
        # Automatically download the data files if missing.
        download_data((dataset_name,), data_root=data_root)

    train_data = torch.load(cameras_path)
    n_cameras = train_data["cameras"]["R"].shape[0]

    _image_max_image_pixels = Image.MAX_IMAGE_PIXELS
    Image.MAX_IMAGE_PIXELS = None  # The dataset image is very large ...
    images = torch.FloatTensor(np.array(Image.open(image_path))) / 255.0
    images = torch.stack(torch.chunk(images, n_cameras, dim=0))[..., :3]
    Image.MAX_IMAGE_PIXELS = _image_max_image_pixels

    scale_factors = [s_new / s for s, s_new in zip(images.shape[1:3], image_size)]

    if abs(scale_factors[0] - scale_factors[1]) > 1e-3:
        raise ValueError("Non-isotropic scaling is not allowed. Consider changing the 'image_size' argument.")
    scale_factor = sum(scale_factors) * 0.5

    if scale_factor != 1.0:
        print(f"Rescaling dataset (factor={scale_factor})")
        images = torch.nn.functional.interpolate(
            images.permute(0, 3, 1, 2),
            size=tuple(image_size),
            mode="bilinear",
        ).permute(0, 2, 3, 1)

    cameras = [
        PerspectiveCameras(**{k: v[cami][None] for k, v in train_data["cameras"].items()}).to("cpu")
        for cami in range(n_cameras)
    ]

    train_idx, val_idx, test_idx = train_data["split"]

    train_dataset, val_dataset, test_dataset = [
        ListDataset(
            [
                {"image": images[i], "camera": cameras[i], "camera_idx": int(i)}
                for i in idx
            ]
        )
        for idx in [train_idx, val_idx, test_idx]
    ]

    return train_dataset, val_dataset, test_dataset


def download_data(
    dataset_names: Optional[List[str]] = None,
    data_root: str = P.data_dir,
    url_root: str = DEFAULT_URL_ROOT,
) -> None:
    """
    Downloads the relevant dataset files.

    Args:
        dataset_names: A list of the names of datasets to download. If `None`,
            downloads all available datasets.
    """

    if dataset_names is None:
        dataset_names = ALL_DATASETS

    os.makedirs(data_root, exist_ok=True)

    for dataset_name in dataset_names:
        cameras_file = dataset_name + ".pth"
        images_file = cameras_file.replace(".pth", ".png")
        license_file = cameras_file.replace(".pth", "_license.txt")

        for fl in (cameras_file, images_file, license_file):
            local_fl = os.path.join(data_root, fl)
            remote_fl = os.path.join(url_root, fl)

            print(f"Downloading dataset {dataset_name} from {remote_fl} to {local_fl}.")

            r = requests.get(remote_fl)

            with open(local_fl, "wb") as f:
                f.write(r.content)
```

```
def implicit_to_mesh(implicit_fn, scale=0.5, grid_size=128, device='cpu', color=[0.7, 0.7, 1], chunk_size=262144, thresh=0):
    Xs = torch.linspace(-1*scale, scale, grid_size+1).to(device)
    Ys = torch.linspace(-1*scale, scale, grid_size+1).to(device)
    Zs = torch.linspace(-1*scale, scale, grid_size+1).to(device)
    grid = torch.stack(torch.meshgrid(Xs, Ys, Zs), dim=-1)

    grid = grid.view(-1, 3)
    num_points = grid.shape[0]
    sdfs = torch.zeros(num_points)

    with torch.no_grad():
        for chunk_start in range(0, num_points, chunk_size):
            torch.cuda.empty_cache()
            chunk_end = min(num_points, chunk_start+chunk_size)
            sdfs[chunk_start:chunk_end] = implicit_fn.get_distance(grid[chunk_start:chunk_end,:]).view(-1)

        sdfs = sdfs.view(grid_size+1, grid_size+1, grid_size+1)

    vertices, triangles = mcubes.marching_cubes(sdfs.cpu().numpy(), thresh)
    # normalize to [-scale, scale]
    vertices = (vertices / grid_size - 0.5) * 2 * scale

    vertices = torch.from_numpy(vertices).unsqueeze(0).float()
    faces = torch.from_numpy(triangles.astype(np.int64)).unsqueeze(0)

    textures = torch.ones_like(vertices) * torch.tensor(color)
    mesh = pytorch3d.structures.Meshes(
        verts=vertices,
        faces=faces,
        textures=pytorch3d.renderer.TexturesVertex(textures),
    )
    mesh = mesh.to(device)
    return mesh
```

```
def sdf_to_density(signed_distance, alpha, beta):
    s = -signed_distance
    psi = torch.where(
        s <= 0.0,
        0.5 * torch.exp(s / beta),
        1 - 0.5 * torch.exp(-s / beta),
    )
    density = alpha * psi
    return density
```

```
                    )
                )
                for i in range(self.n_layers_rgb - 1)
            ]
        )

    def get_distance(self, points):
        points = points.view(-1, 3)
        emb_x = self.harmonic_embedding_xyz(points)
```

```
                )
                for i in range(self.n_layers_xyz - 1)
            ]

            [torch.nn.Linear(self.h_size_xyz + embedding_dim_dir, self.h_size_dir)]
            + [
                torch.nn.Linear(self.h_size_dir, self.h_size_dir)
                for _ in range(self.n_layers_dir - 1)
            ]
        )
        self.linear_density = torch.nn.Linear(self.h_size_xyz, 1)
        self.linear_feat = torch.nn.Linear(self.h_size_xyz, self.h_size_xyz)
        self.linear_rgb = torch.nn.Linear(self.h_size_dir, 3)

        x = self.harmonic_embedding_xyz(x)  # [N, ppr, 39]
        emb_x = x.clone()
        for i in range(self.n_layers_xyz):
            x = F.relu(self.linears_xyz[i](x))
            if i in self.skips:
                x = torch.cat([x, emb_x], dim=-1)
        density = F.relu(self.linear_density(x))

        rgb = F.sigmoid(self.linear_rgb(x))
```

```
    def forward(self, ray_bundle: RayBundle):
        sample_points = ray_bundle.sample_points.view(-1, 3)
        depth_values = ray_bundle.sample_lengths[..., 0]
        deltas = torch.cat(
            (
                depth_values[..., 1:] - depth_values[..., :-1],
                1e10 * torch.ones_like(depth_values[..., :1]),
            ),
            dim=-1,
        ).view(-1, 1)
        # Transform SDF to density
        signed_distance = self.sdf(ray_bundle.sample_points)
        density_ = sdf_to_density(signed_distance, self.alpha, self.beta)
        # Outputs
        if self.rainbow:
            base_color = torch.abs(sample_points - self.sdf.center)
            base_color = torch.clamp(base_color, 0.02, 0.98)
        else:
            base_color = 1.0
        density = -torch.log(1.0 - density_) / deltas
        feature = base_color * self.feature * density_.new_ones(sample_points.shape[0], 1)
        return density, feature
```

```
import torch
from torch.linalg import norm


class BoxSDF(torch.nn.Module):

    def __init__(self, cfg):
        super().__init__()
        center = torch.tensor(cfg.center.val).float().unsqueeze(0)
        side_lengths = torch.tensor(cfg.side_lengths.val).float().unsqueeze(0)
        self.center = torch.nn.Parameter(center, requires_grad=cfg.center.opt)
        self.side_lengths = torch.nn.Parameter(side_lengths, requires_grad=cfg.side_lengths.opt)

    def forward(self, points):
        points = points.view(-1, 3)
        diff = torch.abs(points - self.center) - self.side_lengths / 2.0
        signed_distance = norm(torch.maximum(diff, torch.zeros_like(diff)), dim=-1) + torch.minimum(torch.max(diff, dim=-1)[0], torch.zeros_like(diff[..., 0]))
        return signed_distance.unsqueeze(-1)
```

```
import torch
from torch.linalg import norm

class TorusSDF(torch.nn.Module):

    def __init__(self, cfg):
        super().__init__()
        self.center = torch.nn.Parameter(
            torch.tensor(cfg.center.val).float().unsqueeze(0),
            requires_grad=cfg.center.opt,
        )
        self.radii = torch.nn.Parameter(
            torch.tensor(cfg.radii.val).float().unsqueeze(0),
            requires_grad=cfg.radii.opt,
        )

    def forward(self, points):
        points = points.view(-1, 3)
        diff = points - self.center
        q = torch.stack(
            [
                norm(diff[..., :2], dim=-1) - self.radii[..., 0],
                diff[..., -1],
            ],
            dim=-1
        )
        return (norm(q, dim=-1) - self.radii[..., 1]).unsqueeze(-1)
```
