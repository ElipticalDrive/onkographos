# VisualNifty

VisualNifty is an interactive volumetric image viewer developed for medical imaging and scientific visualization. It is designed to efficiently render three-dimensional scalar fields from NifTi-format datasets. The system utilizes a ray casting-based Direct Volume Rendering (DVR) pipeline to transform raw intensity data into detailed visualizations. Advanced rendering techniques - such as gradient-based shading and ambient occlusion — are integrated to improve the perception of depth and surface detail, resulting in photorealistic images that depict internal structures. High performance enable real-time interactivity even on complex and high-resolution datasets. VisualNifty also offers an intuitive user interface that allows fine-grained adjustments of visualization parameters, like transfer functions, lighting configurations, and view manipulations.

### shaders/raycast.frag
```
#version 410 core
in vec3 vertexPosition;
uniform sampler3D volumeTexture;
uniform sampler1D colorMap;
uniform vec3 cameraWorldPosition;
uniform float stepSize = 0.001;
uniform float growthFactor = 0.00001;
uniform int renderType = 0; // 0 = MIP, 1 = First Contact, 2 = Mean Density, 3 = Compositing, 4 = Full Opacity
uniform bool exactRayCasting = false;
uniform float minDensityThreshold = 0.15; // Minimum threshold for all rendering modes
uniform float maxDensityThreshold = 1.0;  // Maximum threshold for all rendering modes
uniform vec3 minSlices;
uniform vec3 maxSlices;
uniform float alphaFactor = 20.0f;
uniform vec3 volumeDimensions = vec3(1.0, 1.0, 1.0);

uniform vec3 lightPosition = vec3(1.0, 1.0, 2.0);
uniform float ambientStrength = 0.3;
uniform float diffuseStrength = 0.7;
uniform float specularStrength = 0.8;
uniform float shininess = 32.0;
uniform bool enableLighting = true;
uniform bool enableAO = false;
uniform bool enableShadow = false;
uniform vec3 lightColor = vec3(1.0, 1.0, 1.0);

out vec4 FragColor;

float rand(vec2 seed) {
    return fract(sin(dot(seed, vec2(12.9898, 78.233))) * 43758.5453);
}

/*****************************
* Lighting functions
******************************/

vec3 calculateNormal(vec3 pos) {
    float delta = 0.01;
    
    // Sample neighboring points for central difference
    float x1 = texture(volumeTexture, pos - vec3(delta, 0.0, 0.0)).r;
    float x2 = texture(volumeTexture, pos + vec3(delta, 0.0, 0.0)).r;
    float y1 = texture(volumeTexture, pos - vec3(0.0, delta, 0.0)).r;
    float y2 = texture(volumeTexture, pos + vec3(0.0, delta, 0.0)).r;
    float z1 = texture(volumeTexture, pos - vec3(0.0, 0.0, delta)).r;
    float z2 = texture(volumeTexture, pos + vec3(0.0, 0.0, delta)).r;
    
    // Calculate gradient using central difference
    vec3 grad = vec3(x2 - x1, y2 - y1, z2 - z1);
    
    // Avoid normalizing very small gradients which could lead to artifacts
    float magnitude = length(grad);
    if(magnitude > 0.0001) {
        // Surface normals are defined as the negative gradient
        return - normalize(grad); 
    } else {
        return vec3(0.0, 0.0, 1.0); // Default normal if gradient is too small
    }
}

vec3 calculateLighting(vec3 normal, vec3 viewDir, vec3 pos, vec3 baseColor) {
    // Convert sample position from [0,1] to [-1,1] for lighting calculation
    vec3 worldPos = pos * 2.0 - 1.0;
    
    // Ambient
    vec3 ambient = ambientStrength * baseColor;
    
    // Diffuse
    vec3 lightDir = normalize(lightPosition - worldPos);
    float diff = max(dot(normal, lightDir), 0.0);
    vec3 diffuse = diffuseStrength * diff * lightColor * baseColor;
    
    // Specular
    vec3 halfwayDir = normalize(lightDir + viewDir);
    float spec = pow(max(dot(normal, halfwayDir), 0.0), shininess);
    vec3 specular = specularStrength * spec * lightColor;
    
    // Combine lighting components
    return ambient + diffuse + specular;
}


vec3 sampleUniformSphere(int i) {
    float theta = 2.0 * 3.14 * rand(i * gl_FragCoord.xy);
    float phi = acos(1.0 - 2.0 * rand(i * gl_FragCoord.zz)); 

    float x = cos(theta) * sin(phi);
    float y = sin(theta) * sin(phi);
    float z = cos(phi);

    return vec3(x, y, z);
}

float caculateOcclusion(vec3 pos) {
    float occlusion = 0.0;
    int sampleCount = 8;

    const int NUM_RAYS = 10;
    const int NUM_SAMPLES_PER_RAY = 5;

    for (int i = 0; i < NUM_RAYS; i++) {
        vec3 offset = sampleUniformSphere(i) * 0.01;
        // vec3 offset = directions[i] * 0.01;
        vec3 samplePos = pos + offset;

        for (int j = 0; j < NUM_SAMPLES_PER_RAY; j++) {
            samplePos += offset;
            float neighborIntensity = texture(volumeTexture, samplePos).r;
            if(any(lessThan(samplePos, minSlices)) || any(greaterThan(samplePos, maxSlices)))
                break;
            if (neighborIntensity > minDensityThreshold && neighborIntensity < maxDensityThreshold) 
                occlusion += -7.0*neighborIntensity / (NUM_RAYS * NUM_SAMPLES_PER_RAY);
        }
    }
    return clamp(2*exp(occlusion),0.0f,1.0f);
}

float calculateShadow(vec3 pos) {
    float shadowFactor = 1.0;
    float shadowStepSize = 0.01;
    int maxShadowSteps = 20;
    vec3 lightDir = normalize((lightPosition * 0.5 + 0.5) - pos);
    vec3 samplePos = pos;

    for (int i = 0; i < maxShadowSteps; i++) {
        samplePos += lightDir * shadowStepSize;
    
        float sampleDensity = texture(volumeTexture, samplePos).r;
    
        shadowFactor *= exp(-sampleDensity / maxShadowSteps * 5);
    
        if (shadowFactor < 0.01) break;
    }

    return shadowFactor;
}
/******************************
* Projection functions with sampling propogation
******************************/
void projectionMIP(vec3 startPos, vec3 rayDir, float tNear, float tFar) {

    float maxVal = -1.0;
    vec3 maxValPos;
    vec3 samplePos;
    float localGrowthFactor = growthFactor;

    for (float t = tNear; t < tFar; t += stepSize) {
        t += localGrowthFactor;
        localGrowthFactor += growthFactor;

        samplePos = startPos + rayDir * t;
        
        float val = texture(volumeTexture, samplePos).r;
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);
        
        if (isInRange && val > maxVal) {
            maxVal = val;
            maxValPos = samplePos;
        }
    
    }

    if (maxVal < 0.0f){
        FragColor = vec4(0.0f);
        return;
    }

    vec4 baseColor = texture(colorMap, maxVal);
    vec3 shadedColor = enableLighting ? calculateLighting(calculateNormal(maxValPos), -rayDir, maxValPos, baseColor.rgb) : baseColor.rgb;
    FragColor = vec4(shadedColor, baseColor.a);
}

void projectionFirstContact(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
        
    float firstContactVal = 0.0;
    vec3 samplePos;
    float localGrowthFactor = growthFactor;

    for (float t = tNear; t < tFar; t += stepSize) {
        t += localGrowthFactor;
        localGrowthFactor += growthFactor;

        // Move along ray
        samplePos = startPos + rayDir * t;
        
        float val = texture(volumeTexture, samplePos).r;
        
        // Check if value is within threshold range
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);
        

        // First contact rendering
        if (isInRange) {
            firstContactVal = val;
            break; // Stop once we've found first contact
        }

    }

    if (firstContactVal <= 0.0f){
        FragColor = vec4(0.0f);
        return;
    }

    vec4 baseColor = texture(colorMap, firstContactVal);
    vec3 shadedColor = enableLighting ? calculateLighting(calculateNormal(samplePos), -rayDir, samplePos, baseColor.rgb) : baseColor.rgb;
    float occlusion = enableAO ? caculateOcclusion(samplePos) : 1.0f;
    float shadow = enableShadow ? calculateShadow(samplePos) : 1.0f;
    FragColor = vec4(shadedColor*occlusion*shadow, 1.0f);
}

void projectionMeanDensity(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
    float meanSum = 0.0;
    int sampleCount = 0;     
    int validSamples = 0;
    vec3 samplePos;
    vec3 meanPos = vec3(0.0); // Track the average position for normal calculation
    // float localGrowthFactor = growthFactor;

    for (float t = tNear; t < tFar; t += stepSize) {

        // Move along ray
        samplePos = startPos + rayDir * t;
        
        float val = texture(volumeTexture, samplePos).r;
        
        // Check if value is within threshold range
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);
        
        // Mean density rendering (only consider values within range)
        if (isInRange) {
            meanSum += val;
            meanPos += samplePos; // Accumulate positions for lighting calculation
            validSamples++;
        }
        sampleCount++;
    }
    
    // Calculate final color
    if (validSamples > 0) {
        float meanValue = meanSum / float(validSamples);
        vec4 color = texture(colorMap, meanValue);
        
        if (enableLighting) {
            // Calculate mean position for normal calculation
            meanPos /= float(validSamples);
            vec3 normal = calculateNormal(meanPos);
            vec3 litColor = calculateLighting(normal, -rayDir, meanPos, color.rgb);
            FragColor = vec4(litColor, 1.0);
        } else {
            FragColor = color;
        }
    } else {
        FragColor = vec4(0.0);
    }
}

void projectionCompositing(vec3 startPos, vec3 rayDir, float tNear, float tFar){

    vec3 samplePos;
    float adjustedAlphaFactor = alphaFactor * stepSize;
    float accumulatedAlpha = 0.0;
    vec3 accumulatedColor = vec3(0.0);
    // float localGrowthFactor = growthFactor;

    for (float t = tNear; t < tFar; t += stepSize) {
        
        samplePos = startPos + rayDir * t;
        
        float val = texture(volumeTexture, samplePos).r;
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);

        if (isInRange){
            vec4 baseColor = texture(colorMap, val);
            vec3 shadedColor = enableLighting ? calculateLighting(calculateNormal(samplePos), -rayDir, samplePos, baseColor.rgb) : baseColor.rgb;
            float shadow = enableShadow ? calculateShadow(samplePos) : 1.0f;

            float alpha = baseColor.a * adjustedAlphaFactor;
            accumulatedColor += (1.0 - accumulatedAlpha) * shadedColor * shadow * alpha;
            accumulatedAlpha += (1.0 - accumulatedAlpha) * alpha;
        }

        if (accumulatedAlpha >= 1.0) break;
    }
    FragColor = vec4(accumulatedColor, accumulatedAlpha);
}

void projectionFullOpacity(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
    vec3 samplePos;
    float localGrowthFactor = growthFactor;
    
    // Accumulation variables
    vec3 accumulatedColor = vec3(0.0);
    float accumulatedAlpha = 0.0;
    
    // For storing primary visible object (front object with sufficient density)
    bool foundPrimaryObject = false;
    vec3 primaryObjectPos = vec3(0.0);
    vec4 primaryObjectColor = vec4(0.0);
    float primaryObjectDistance = 0.0;
    float primaryObjectDensity = 0.0;
    
    // For storing high-density background object
    bool foundBackgroundObject = false;
    vec3 backgroundObjectPos = vec3(0.0);
    vec4 backgroundObjectColor = vec4(0.0);
    float backgroundObjectDistance = 0.0;
    
    // Constants to control visibility behavior
    const float HIGH_DENSITY_THRESHOLD = 0.7;    // Density above which front objects obscure back objects
    const float DISTANCE_THRESHOLD = 0.5;        // Distance beyond which back objects are ignored
    
    // Front-to-back ray traversal (reversed from typical compositing which is back-to-front)
    for (float t = tNear; t < tFar; t += stepSize) {
        t += localGrowthFactor;
        localGrowthFactor += growthFactor;

        samplePos = startPos + rayDir * t;
        
        float val = texture(volumeTexture, samplePos).r;
        if (val < minDensityThreshold || val > maxDensityThreshold) continue;
        
        vec4 color = texture(colorMap, val);
        
        // Skip very low opacity samples
        if (color.a < 0.05) continue;
        
        // If we haven't found a primary object yet
        if (!foundPrimaryObject) {
            // Found our first meaningful object - store it as primary
            foundPrimaryObject = true;
            primaryObjectPos = samplePos;
            primaryObjectColor = color;
            primaryObjectDistance = t;
            primaryObjectDensity = val;
            
            // If this primary object has very high density or opacity, we can stop
            // No need to look for background objects
            if (val > HIGH_DENSITY_THRESHOLD || color.a > 0.9) {
                break;
            }
            
            // Otherwise continue looking for background objects
            continue;
        }
        
        // We already have a primary object, now looking for background objects
        // Only consider significant background objects (high opacity)
        if (!foundBackgroundObject && color.a > 0.6) {
            foundBackgroundObject = true;
            backgroundObjectPos = samplePos;
            backgroundObjectColor = color;
            backgroundObjectDistance = t;
            
            // Check if the distance between primary and background object is significant
            float distanceBetweenObjects = backgroundObjectDistance - primaryObjectDistance;
            
            // If primary object has moderate-high density or objects are far apart,
            // ignore the background object
            if (primaryObjectDensity > 0.5 || distanceBetweenObjects > DISTANCE_THRESHOLD) {
                foundBackgroundObject = false;  // Effectively ignore the background object
            }
            
            // We found what we needed, no need to continue ray traversal
            break;
        }
    }
    
    // If we didn't find any objects
    if (!foundPrimaryObject) {
        FragColor = vec4(0.0);
        return;
    }
    
    // Apply lighting to primary object
    vec3 primaryColor;
    if (enableLighting) {
        vec3 normal = calculateNormal(primaryObjectPos);
        primaryColor = calculateLighting(normal, -rayDir, primaryObjectPos, primaryObjectColor.rgb);
    } else {
        primaryColor = primaryObjectColor.rgb;
    }
    
    // Start with the primary object
    accumulatedColor = primaryColor;
    accumulatedAlpha = primaryObjectColor.a;
    
    // If we have a relevant background object, blend it
    if (foundBackgroundObject) {
        // Calculate how much of the background should show through based on:
        // 1. Distance between objects
        // 2. Primary object's density/opacity
        float distanceFactor = clamp((backgroundObjectDistance - primaryObjectDistance) / DISTANCE_THRESHOLD, 0.0, 1.0);
        float densityFactor = clamp(1.0 - primaryObjectDensity / HIGH_DENSITY_THRESHOLD, 0.0, 1.0);
        
        // Combined factor determines background contribution
        float backgroundContribution = min(distanceFactor, densityFactor) * 0.7;
        
        // Apply lighting to background object
        vec3 backgroundLitColor;
        if (enableLighting) {
            vec3 normal = calculateNormal(backgroundObjectPos);
            backgroundLitColor = calculateLighting(normal, -rayDir, backgroundObjectPos, backgroundObjectColor.rgb);
        } else {
            backgroundLitColor = backgroundObjectColor.rgb;
        }
        
        // Blend background with accumulated color
        accumulatedColor = mix(accumulatedColor, backgroundLitColor, backgroundContribution * (1.0 - accumulatedAlpha));
        accumulatedAlpha = accumulatedAlpha + backgroundContribution * (1.0 - accumulatedAlpha);
    }
    
    FragColor = vec4(accumulatedColor, max(accumulatedAlpha, 0.5));  // Ensure minimum opacity
}

/******************************
* Projection functions with exact propogation
******************************/
void exactProjectionMIP(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
    float maxVal = -1.0;
    vec3 maxValPos = startPos;
    vec3 samplePos;
    float t = tNear;
    const int MAX_ITERATIONS = 2000;
    int iterations = 0;

    while (t < tFar && iterations < MAX_ITERATIONS) {
        samplePos = startPos + rayDir * t;

        // Check if we're still inside volume bounds
        if (any(lessThan(samplePos, vec3(0.0))) || any(greaterThan(samplePos, vec3(1.0)))) {
            t += stepSize;
            iterations++;
            continue;
        }

        // Calculate current voxel coordinates
        vec3 voxelPos = samplePos * volumeDimensions;

        // Determine next voxel boundaries differently for positive/negative ray directions
        vec3 nextVoxel;
        nextVoxel.x = rayDir.x > 0.0 ? floor(voxelPos.x) + 1.0 : ceil(voxelPos.x) - 1.0;
        nextVoxel.y = rayDir.y > 0.0 ? floor(voxelPos.y) + 1.0 : ceil(voxelPos.y) - 1.0; 
        nextVoxel.z = rayDir.z > 0.0 ? floor(voxelPos.z) + 1.0 : ceil(voxelPos.z) - 1.0;

        // Compute dt per axis using exact boundaries with explicit safety checks:
        vec3 dtVec = vec3(tFar);
        if (abs(rayDir.x) > 0.0001) dtVec.x = (nextVoxel.x - voxelPos.x) / (rayDir.x * volumeDimensions.x);
        if (abs(rayDir.y) > 0.0001) dtVec.y = (nextVoxel.y - voxelPos.y) / (rayDir.y * volumeDimensions.y);
        if (abs(rayDir.z) > 0.0001) dtVec.z = (nextVoxel.z - voxelPos.z) / (rayDir.z * volumeDimensions.z);

        float dt = min(min(dtVec.x, dtVec.y), dtVec.z);

        // Ensure dt is positive and not too small
        dt = max(dt, 0.0001);

        float val = texture(volumeTexture, samplePos).r;
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);

        if (isInRange && val > maxVal) {
            maxVal = val;
            maxValPos = samplePos;
        }

        // Advance to the next voxel boundary (adding small epsilon)
        t += dt;
        iterations++;
    }

    if (maxVal < 0.0f){
        FragColor = vec4(0.0f);
        return;
    }

    vec4 baseColor = texture(colorMap, maxVal);
    vec3 shadedColor = enableLighting ? calculateLighting(calculateNormal(maxValPos), -rayDir, maxValPos, baseColor.rgb) : baseColor.rgb;
    FragColor = vec4(shadedColor, baseColor.a);
}

void exactProjectionCompositing(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
    vec3 samplePos;
    float accumulatedAlpha = 0.0;
    vec3 accumulatedColor = vec3(0.0);

    // Start at the near intersection point
    float t = tNear;

    // Safety counter to prevent infinite loops
    const int MAX_ITERATIONS = 2000;
    int iterations = 0;
    int voxelCounter = 0;

    // Calculate alpha adjustment factor based on expected sampling density
    float avgStep = 1.0 / float(volumeDimensions.x + volumeDimensions.y + volumeDimensions.z) / 3.0;
    float adjustedAlphaFactor = alphaFactor * avgStep;

    while (t < tFar && iterations < MAX_ITERATIONS && accumulatedAlpha < 1.0) {
        samplePos = startPos + rayDir * t;

        // Check if we're still inside volume bounds
        if (any(lessThan(samplePos, vec3(0.0))) || any(greaterThan(samplePos, vec3(1.0)))) {
            t += stepSize;
            iterations++;
            continue;
        }

        // Calculate current voxel coordinates
        vec3 voxelPos = samplePos * volumeDimensions;

        // Determine next voxel boundaries differently for positive/negative ray directions
        vec3 nextVoxel;
        nextVoxel.x = rayDir.x > 0.0 ? floor(voxelPos.x) + 1.0 : ceil(voxelPos.x) - 1.0;
        nextVoxel.y = rayDir.y > 0.0 ? floor(voxelPos.y) + 1.0 : ceil(voxelPos.y) - 1.0; 
        nextVoxel.z = rayDir.z > 0.0 ? floor(voxelPos.z) + 1.0 : ceil(voxelPos.z) - 1.0;

        // Compute dt per axis using exact boundaries with explicit safety checks:
        vec3 dtVec = vec3(tFar);
        if (abs(rayDir.x) > 0.0001) dtVec.x = (nextVoxel.x - voxelPos.x) / (rayDir.x * volumeDimensions.x);
        if (abs(rayDir.y) > 0.0001) dtVec.y = (nextVoxel.y - voxelPos.y) / (rayDir.y * volumeDimensions.y);
        if (abs(rayDir.z) > 0.0001) dtVec.z = (nextVoxel.z - voxelPos.z) / (rayDir.z * volumeDimensions.z);

        float dt = min(min(dtVec.x, dtVec.y), dtVec.z);

        // Ensure dt is positive and not too small
        dt = max(dt, 0.0001);

        float val = texture(volumeTexture, samplePos).r;
        bool isInRange = (val >= minDensityThreshold && val <= maxDensityThreshold);

        // For compositing, accumulate color and alpha
        if (isInRange) {
            vec4 baseColor = texture(colorMap, val);
            vec3 shadedColor = enableLighting ? calculateLighting(calculateNormal(samplePos), -rayDir, samplePos, baseColor.rgb) : baseColor.rgb;
            float shadow = enableShadow ? calculateShadow(samplePos) : 1.0f;

            float alpha = baseColor.a * adjustedAlphaFactor;
            accumulatedColor += (1.0 - accumulatedAlpha) * shadedColor * shadow * alpha;
            accumulatedAlpha += (1.0 - accumulatedAlpha) * alpha;
        }

        // Advance to the next voxel boundary (adding small epsilon)
        t += dt;
        iterations++;
    }

    FragColor = vec4(accumulatedColor, accumulatedAlpha);
}

void exactProjectionFullOpacity(vec3 startPos, vec3 rayDir, float tNear, float tFar) {
    vec3 samplePos;
    
    // Accumulation variables
    vec3 accumulatedColor = vec3(0.0);
    float accumulatedAlpha = 0.0;
    
    // For storing primary visible object (front object with sufficient density)
    bool foundPrimaryObject = false;
    vec3 primaryObjectPos = vec3(0.0);
    vec4 primaryObjectColor = vec4(0.0);
    float primaryObjectDistance = 0.0;
    float primaryObjectDensity = 0.0;
    
    // For storing high-density background object
    bool foundBackgroundObject = false;
    vec3 backgroundObjectPos = vec3(0.0);
    vec4 backgroundObjectColor = vec4(0.0);
    float backgroundObjectDistance = 0.0;
    
    // Constants to control visibility behavior
    const float HIGH_DENSITY_THRESHOLD = 0.7;    // Density above which front objects obscure back objects
    const float DISTANCE_THRESHOLD = 0.5;        // Distance beyond which back objects are ignored
    
    // Start at the near intersection point
    float t = tNear;
    
    // Safety counter to prevent infinite loops
    const int MAX_ITERATIONS = 2000;
    int iterations = 0;
    
    // Front-to-back ray traversal with exact steps
    while (t < tFar && iterations < MAX_ITERATIONS) {
        samplePos = startPos + rayDir * t;
        
        // Check if we're still inside volume bounds
        if (any(lessThan(samplePos, vec3(0.0))) || any(greaterThan(samplePos, vec3(1.0)))) {
            t += stepSize;
            iterations++;
            continue;
        }
        
        // Calculate current voxel coordinates
        vec3 voxelPos = samplePos * volumeDimensions;
        
        // Determine next voxel boundaries differently for positive/negative ray directions
        vec3 nextVoxel;
        nextVoxel.x = rayDir.x > 0.0 ? floor(voxelPos.x) + 1.0 : ceil(voxelPos.x) - 1.0;
        nextVoxel.y = rayDir.y > 0.0 ? floor(voxelPos.y) + 1.0 : ceil(voxelPos.y) - 1.0; 
        nextVoxel.z = rayDir.z > 0.0 ? floor(voxelPos.z) + 1.0 : ceil(voxelPos.z) - 1.0;
        
        // Compute dt per axis using exact boundaries with explicit safety checks
        vec3 dtVec = vec3(tFar);
        if (abs(rayDir.x) > 0.0001) dtVec.x = (nextVoxel.x - voxelPos.x) / (rayDir.x * volumeDimensions.x);
        if (abs(rayDir.y) > 0.0001) dtVec.y = (nextVoxel.y - voxelPos.y) / (rayDir.y * volumeDimensions.y);
        if (abs(rayDir.z) > 0.0001) dtVec.z = (nextVoxel.z - voxelPos.z) / (rayDir.z * volumeDimensions.z);
        
        float dt = min(min(dtVec.x, dtVec.y), dtVec.z);
        
        // Ensure dt is positive and not too small
        dt = max(dt, 0.0001);
        
        // Sample the current position
        float val = texture(volumeTexture, samplePos).r;
        if (val < minDensityThreshold || val > maxDensityThreshold) {
            t += dt;
            iterations++;
            continue;
        }
        
        vec4 color = texture(colorMap, val);
        
        // Skip very low opacity samples
        if (color.a < 0.05) {
            t += dt;
            iterations++;
            continue;
        }
        
        // If we haven't found a primary object yet
        if (!foundPrimaryObject) {
            // Found our first meaningful object - store it as primary
            foundPrimaryObject = true;
            primaryObjectPos = samplePos;
            primaryObjectColor = color;
            primaryObjectDistance = t;
            primaryObjectDensity = val;
            
            // If this primary object has very high density or opacity, we can stop
            // No need to look for background objects
            if (val > HIGH_DENSITY_THRESHOLD || color.a > 0.9) {
                break;
            }
            
            // Otherwise continue looking for background objects
            t += dt;
            iterations++;
            continue;
        }
        
        // We already have a primary object, now looking for background objects
        // Only consider significant background objects (high opacity)
        if (!foundBackgroundObject && color.a > 0.6) {
            foundBackgroundObject = true;
            backgroundObjectPos = samplePos;
            backgroundObjectColor = color;
            backgroundObjectDistance = t;
            
            // Check if the distance between primary and background object is significant
            float distanceBetweenObjects = backgroundObjectDistance - primaryObjectDistance;
            
            // If primary object has moderate-high density or objects are far apart,
            // ignore the background object
            if (primaryObjectDensity > 0.5 || distanceBetweenObjects > DISTANCE_THRESHOLD) {
                foundBackgroundObject = false;  // Effectively ignore the background object
            }
            
            // We found what we needed, no need to continue ray traversal
            break;
        }
        
        // Advance to the next voxel boundary
        t += dt;
        iterations++;
    }
    
    // If we didn't find any objects
    if (!foundPrimaryObject) {
        FragColor = vec4(0.0);
        return;
    }
    
    // Apply lighting to primary object
    vec3 primaryColor;
    if (enableLighting) {
        vec3 normal = calculateNormal(primaryObjectPos);
        primaryColor = calculateLighting(normal, -rayDir, primaryObjectPos, primaryObjectColor.rgb);
    } else {
        primaryColor = primaryObjectColor.rgb;
    }
    
    // Start with the primary object
    accumulatedColor = primaryColor;
    accumulatedAlpha = primaryObjectColor.a;
    
    // If we have a relevant background object, blend it
    if (foundBackgroundObject) {
        // Calculate how much of the background should show through based on:
        // 1. Distance between objects
        // 2. Primary object's density/opacity
        float distanceFactor = clamp((backgroundObjectDistance - primaryObjectDistance) / DISTANCE_THRESHOLD, 0.0, 1.0);
        float densityFactor = clamp(1.0 - primaryObjectDensity / HIGH_DENSITY_THRESHOLD, 0.0, 1.0);
        
        // Combined factor determines background contribution
        float backgroundContribution = min(distanceFactor, densityFactor) * 0.7;
        
        // Apply lighting to background object
        vec3 backgroundLitColor;
        if (enableLighting) {
            vec3 normal = calculateNormal(backgroundObjectPos);
            backgroundLitColor = calculateLighting(normal, -rayDir, backgroundObjectPos, backgroundObjectColor.rgb);
        } else {
            backgroundLitColor = backgroundObjectColor.rgb;
        }
        
        // Blend background with accumulated color
        accumulatedColor = mix(accumulatedColor, backgroundLitColor, backgroundContribution * (1.0 - accumulatedAlpha));
        accumulatedAlpha = accumulatedAlpha + backgroundContribution * (1.0 - accumulatedAlpha);
    }
    
    FragColor = vec4(accumulatedColor, max(accumulatedAlpha, 0.5));  // Ensure minimum opacity
}

/******************************
* Main function
******************************/
void main() {
    // Normalized vertex position (in cube from -1 to 1)
    vec3 pos = vertexPosition;
    
    // Calculate ray direction in model space
    vec3 rayDir = normalize(pos - cameraWorldPosition);
    
    // Ray-box intersection for unit cube [-1,1]
    vec3 invRayDir = 1.0 / rayDir;
    vec3 tMin = (minSlices - pos) * invRayDir;
    vec3 tMax = (maxSlices - pos) * invRayDir;
    vec3 t1 = min(tMin, tMax);
    vec3 t2 = max(tMin, tMax);
    float tNear = max(max(t1.x, t1.y), t1.z);
    float tFar = min(min(t2.x, t2.y), t2.z);
    
    // Skip if ray doesn't intersect box
    if(tNear > tFar || tFar < 0.0) {
        discard;
        return;
    }

    // Adjust to texture coordinates [0,1]
    float jitter = rand(vertexPosition.xy*vertexPosition.zz);
    
    vec3 startPos = pos * 0.5 + 0.5 + jitter * stepSize * 0.3;

    rayDir = rayDir * 0.5;

    if (exactRayCasting) {
        
        if (renderType == 0) {
            exactProjectionMIP(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 1) {
            exactProjectionFirstContact(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 2) {
            exactProjectionMeanDensity(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 3) {
            exactProjectionCompositing(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 4) {
            exactProjectionFullOpacity(startPos, rayDir, tNear, tFar);
        }    
    } else {
        
        if (renderType == 0) {
            projectionMIP(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 1) {
            projectionFirstContact(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 2) {
            projectionMeanDensity(startPos, rayDir, tNear, tFar);
        }
        else if (renderType == 3) {
            projectionCompositing(startPos, rayDir, tNear, tFar);
        } 
        else if (renderType == 4) {
            projectionFullOpacity(startPos, rayDir, tNear, tFar);
        } 
    }
}
```

```
NiftiDataManager::NiftiDataManager() = default;

NiftiDataManager::NiftiDataManager(const RNifti::NiftiImage& image) {
    loadFromNiftiImage(image);
}

NiftiDataManager::~NiftiDataManager() = default;

bool NiftiDataManager::loadFromNiftiImage(const RNifti::NiftiImage& image) {
    // Clear old data
    m_volumeData.clear();
    m_width = 0;
    m_height = 0;
    m_depth = 0;
    
    if (image.isNull()) {
        qWarning() << "Cannot load from null NiftiImage";
        return false;
    }
    
    try {
        // Extract dimensions
        m_width = image.dim()[0];
        m_height = image.dim()[1];
        m_depth = image.dim()[2];
        
        qDebug() << "NiftiDataManager: Loading data with dimensions:" 
                 << m_width << "x" << m_height << "x" << m_depth;
        
        if (m_width <= 0 || m_height <= 0 || m_depth <= 0) {
            qWarning() << "Invalid NifTI dimensions:" << m_width << "x" << m_height << "x" << m_depth;
            return false;
        }
        
        // Extract actual volume data
        m_volumeData = image.getData<float>();
        if (m_volumeData.empty()) {
            qWarning() << "Failed to get NifTI data as float vector";
            return false;
        }
        
        qDebug() << "NiftiDataManager: Loaded volume data with size:" << m_volumeData.size();
        
        // Calculate min/max values
        auto [minIt, maxIt] = std::minmax_element(m_volumeData.begin(), m_volumeData.end());
        m_minValue = *minIt;
        m_maxValue = *maxIt;
        
        qDebug() << "NiftiDataManager: Value range:" << m_minValue << "to" << m_maxValue;
        
        return true;
    }
    catch (const std::exception& e) {
        qWarning() << "Exception in loadFromNiftiImage:" << e.what();
        m_volumeData.clear();
        m_width = m_height = m_depth = 0;
        return false;
    }
}

void NiftiDataManager::getDimensions(int dimensions[3]) const {
    dimensions[0] = m_width;
    dimensions[1] = m_height;
    dimensions[2] = m_depth;
}

std::vector<float> NiftiDataManager::getNormalizedVolumeData() const {
    std::vector<float> normalizedData = m_volumeData;
    
    // Check if normalization is needed (same min/max)
    if (m_minValue == m_maxValue) {
        qWarning() << "NiftiDataManager: Data has uniform values, normalization skipped";
        return normalizedData;
    }
    
    // Normalize values to 0-1 range
    qDebug() << "NiftiDataManager: Normalizing volume data";
    for (auto& val : normalizedData) {
        val = (val - m_minValue) / (m_maxValue - m_minValue);
    }
    
    return normalizedData;
}
```

```
void GradientEditor::openColorPicker() {
  if (m_selectedStop == nullptr) return;
  
  QColorDialog colorDialog(this);
  colorDialog.setCurrentColor((*m_selectedStop).second);
  colorDialog.setOption(QColorDialog::ShowAlphaChannel, true);
  
  if (colorDialog.exec() == QDialog::Accepted) {
    QColor selectedColor = colorDialog.selectedColor();
    (*m_selectedStop).second = selectedColor;
    
    setPresetSelectorItem(GradientPresets::CUSTOM);
    update();
    emitGradientChange();
  }
}

void GradientEditor::paintEvent(QPaintEvent *event) {
  QPainter painter(this);

  // Create a linear gradient
  QLinearGradient gradient(0, 0, width(), 0);
  for (const auto &stop : m_stops) {
    gradient.setColorAt(stop.first, stop.second);
  }

  // Draw the gradient background
  QRect gradientArea(m_gradientMargin, 0, width() - 2 * m_gradientMargin, m_gradientHeight);
  painter.fillRect(gradientArea, gradient);

  // Draw control points
  for (const auto &stop : m_stops) {
    int x = m_gradientMargin + stop.first * (width() - 2 * m_gradientMargin);
    painter.setBrush(m_selectedStop != nullptr && *m_selectedStop == stop ? Qt::red : Qt::black);
    painter.drawEllipse(QPointF(x, m_gradientHeight / 2), 5, 5);
  }

  QFontMetrics fm(painter.font());

  for (int i = 0; i <= m_numTicks; ++i) {
    double value = static_cast<double>(i) / m_numTicks;
    int x = m_gradientMargin + value * (width() - 2 * m_gradientMargin);

    // Draw tick
    painter.drawLine(x, m_gradientHeight, x, m_gradientHeight + m_tickHeight);

    // Draw label
    QString label = QString::number(value, 'f', 1);
    int textWidth = fm.horizontalAdvance(label);
    painter.drawText(x - textWidth / 2, m_gradientHeight + m_tickHeight + m_tickTextHeight, label);
  }
}

void GradientEditor::setStops(QGradientStops stops) {
  m_stops = stops;
  m_selectedStop = &m_stops[0];

  update();
  emitGradientChange();
}

void GradientEditor::addStop() {
  QGradientStop newStop(0.5, Qt::white);
  m_stops.append(newStop);
  m_selectedStop = &m_stops[m_stops.size() - 1];

  update();
  emitGradientChange();
}

void GradientEditor::removeStop() {
  if (m_selectedStop == nullptr || m_stops.size() == 1) return;
  m_stops.removeOne(*m_selectedStop);
  m_selectedStop = nullptr;

  update();
  emitGradientChange();
}

void GradientEditor::mousePressEvent(QMouseEvent *event) {
  float hitMargin = 0.05;
  float x = (event->pos().rx() - m_gradientMargin) / (float)(width() - 2 * m_gradientMargin);

  if (event->button() == Qt::LeftButton) {
    for (auto &stop : m_stops) {
      if (stop.first - hitMargin < x && x < stop.first + hitMargin) {
        if (m_selectedStop == nullptr || *m_selectedStop != stop)
          m_selectedStop = &stop;
      }
    }
  }

  update();
}

void GradientEditor::mouseMoveEvent(QMouseEvent *event) {
  if (m_selectedStop == nullptr) return;

  float x = (event->pos().rx() - m_gradientMargin) / (float)(width() - 2 * m_gradientMargin);
  x = qBound(0.0, x, 1.0);
  (*m_selectedStop).first = x;

  setPresetSelectorItem(GradientPresets::CUSTOM);
  update();
  emitGradientChange();
}

void GradientEditor::changeColor(const QString &hex) {
  if (QColor::isValidColorName(hex) && m_selectedStop != nullptr) {
    (*m_selectedStop).second = QColor(hex);

    setPresetSelectorItem(GradientPresets::CUSTOM);
    update();
    emitGradientChange();
  }
}

QColor GradientEditor::interoplateColor(const QColor &c1, const QColor &c2, double t) {
  int r = (1 - t) * c1.red() + t * c2.red();
  int g = (1 - t) * c1.green() + t * c2.green();
  int b = (1 - t) * c1.blue() + t * c2.blue();
  int a = (1 - t) * c1.alpha() + t * c2.alpha();
  return QColor(r, g, b, a);
}

QVector<QVector4D> GradientEditor::interpolateGradient(QGradientStops stops, int numSamples) {
  QVector<QVector4D> gradient(numSamples);

  std::sort(stops.begin(), stops.end(), [](const QGradientStop &a, const QGradientStop &b) {
    return a.first < b.first;
  });

  int j = 0;

  for (int i = 0; i < numSamples; i++) {
    double pos = double(i) / (numSamples - 1);  // Normalize position between 0 and 1

    QColor finalColor;

    if (pos <= stops[0].first) {
      finalColor = stops[0].second;
    } else if (pos >= stops[stops.size() - 1].first) {
      finalColor = stops[stops.size() - 1].second;
    } else {
      if (pos >= stops[j + 1].first)
        j++;

      double startPos = stops[j].first, endPos = stops[j + 1].first;
      QColor startColor = stops[j].second, endColor = stops[j + 1].second;
      double t = (pos - startPos) / (endPos - startPos);
      finalColor = interoplateColor(startColor, endColor, t);
    }

    gradient[i] = QVector4D(finalColor.redF(), finalColor.greenF(), finalColor.blueF(), finalColor.alphaF());
  }

  return gradient;
}

void GradientEditor::setPreset(int index) {
  QGradientStops stops;

  GradientPresets preset = static_cast<GradientPresets>(m_presetSelector->itemData(index).toInt());

  switch (preset) {
  case GradientPresets::BLUE_TO_RED:
    stops << QGradientStop(0.0, QColor(0, 0, 255, 0)) << QGradientStop(1.0, QColor(255, 0, 0, 255));
    break;
  case GradientPresets::GRAYSCALE:
    stops << QGradientStop(0.0, QColor(0, 0, 0, 0)) << QGradientStop(1.0, QColor(255, 255, 255, 255));
    break;
  case GradientPresets::LAVA:
    stops << QGradientStop(0.0, QColor(0, 0, 0, 0)) << QGradientStop(0.3, QColor(255, 0, 0, 80)) << QGradientStop(0.7, QColor(255, 255, 0, 200)) << QGradientStop(1.0, QColor(255, 255, 255, 255));
    break;
  case GradientPresets::REALISTIC:
    stops << QGradientStop(0.0, QColor(0, 0, 0, 0)) << QGradientStop(0.25, QColor(100, 220, 240, 15)) << QGradientStop(0.37, QColor(100, 20, 20, 25)) << QGradientStop(0.45, QColor(240, 230, 130, 255)) << QGradientStop(0.85, QColor(205, 210, 240, 255));
    break;
  case GradientPresets::PHOTOREALISTIC:
    // Completely transparent for values below 0.12
    
    stops 
    << QGradientStop(0.0, QColor(0, 0, 0, 0)) 
    << QGradientStop(0.275, QColor(0, 0, 0, 5)) 
    << QGradientStop(0.295, QColor(255, 0, 0, 100)) 
    << QGradientStop(0.302, QColor(255, 0, 0, 100)) 
    << QGradientStop(0.31, QColor(240, 230, 130, 255)) 
    << QGradientStop(0.85, QColor(205, 210, 240, 255))
    << QGradientStop(1.0, QColor(255, 255, 255, 255)); 

    break;
  case GradientPresets::CUSTOM:
    break;
  }

  if (!stops.isEmpty()) {
    setStops(stops);
  }
}

void GradientEditor::setPresetSelectorItem(GradientPresets preset) {
  int index = m_presetSelector->findData(static_cast<int>(preset));
  m_presetSelector->setCurrentIndex(index);
}

void GradientEditor::emitGradientChange() {
  emit gradientChanged(interpolateGradient(m_stops, SAMPLING_SIZE));
}
```

```
            maxSlider->setValue(minSlider->value());
            maxSlider->blockSignals(false);
            value = minSlider->value();
        }
        
        updateLabels();
        emit maxValueChanged(value);
        emit rangeChanged(minSlider->value(), maxSlider->value());
    });
```

```
#include "Camera.h"
#include <cmath>
#include <algorithm>

Camera::Camera()
    : m_position(0.0f, 0.0f, 5.0f),
      m_center(0.0f, 0.0f, 0.0f),
      m_front(0.0f, 0.0f, -1.0f),
      m_up(0.0f, 1.0f, 0.0f),
      m_radius(5.0f),
      m_rotation()
{
    updateCameraVectors();
}

void Camera::setPosition(const QVector3D& position) {
    m_position = position;
    m_front = (m_center - m_position).normalized();
    m_radius = (m_center - m_position).length();
}

void Camera::setCenter(const QVector3D& center) {
    m_center = center;
    m_front = (m_center - m_position).normalized();
    m_radius = (m_center - m_position).length();
}

void Camera::setUpVector(const QVector3D& up) {
    m_up = up.normalized();
}

void Camera::setRadius(float radius) {
    m_radius = std::max(0.5f, std::min(20.0f, radius));
    updateCameraVectors();
}

void Camera::rotate(float yawDelta, float pitchDelta) {
  // Calculate camera's current right and up vectors dynamically
  QVector3D cameraRight = QVector3D::crossProduct(m_front, m_up).normalized();
  QVector3D cameraUp = m_up;

  // Create rotation quaternions for yaw (around the up vector) and pitch (around the right vector)
  QQuaternion yawRotation = QQuaternion::fromAxisAndAngle(cameraUp, -yawDelta);
  QQuaternion pitchRotation = QQuaternion::fromAxisAndAngle(cameraRight, -pitchDelta);

  // Combine rotations
  m_rotation = yawRotation * pitchRotation * m_rotation;
  m_rotation.normalize(); // Keep quaternion normalized

  // Update camera vectors
  updateCameraVectors();
}

void Camera::zoom(float zoomDelta) {
    // Update radius (zoom in/out)
    m_radius -= zoomDelta * 0.1f * m_radius;
    
    // Clamp radius to reasonable values
    m_radius = std::max(0.5f, std::min(20.0f, m_radius));
    
    updateCameraVectors();
}

void Camera::resetOrientation() {
    m_rotation = QQuaternion();
    updateCameraVectors();
}

void Camera::setViewAxis(const QString& axis) {
    // Reset rotation quaternion first
    m_rotation = QQuaternion();
    
    // Then set up standard views using quaternions
    if (axis.compare("axial", Qt::CaseInsensitive) == 0) {
        // Top-down view (axial)
        QQuaternion xRotation = QQuaternion::fromAxisAndAngle(QVector3D(1, 0, 0), -90.0f);
        QQuaternion yRotation = QQuaternion::fromAxisAndAngle(QVector3D(0, 1, 0), 180.0f);

        m_rotation = yRotation * xRotation;
    } else if (axis.compare("coronal", Qt::CaseInsensitive) == 0) {
        // Front view (coronal) - default orientation, no rotation needed
        m_rotation = QQuaternion();
    } else if (axis.compare("sagittal", Qt::CaseInsensitive) == 0) {
        // Side view (sagittal)
        m_rotation = QQuaternion::fromAxisAndAngle(QVector3D(0, 1, 0), -90.0f);
    } else if (axis.compare("oblique", Qt::CaseInsensitive) == 0) {
        // Oblique view (good for 3D visualization)
        QQuaternion yawRotation = QQuaternion::fromAxisAndAngle(QVector3D(0, 1, 0), -45.0f);
        QQuaternion pitchRotation = QQuaternion::fromAxisAndAngle(QVector3D(1, 0, 0), -30.0f);
        m_rotation = yawRotation * pitchRotation;
    } else if (axis.compare("bottom", Qt::CaseInsensitive) == 0) {
        // Bottom-up view
        m_rotation = QQuaternion::fromAxisAndAngle(QVector3D(1, 0, 0), 90.0f);
    }
    
    // Normalize the quaternion to ensure proper rotation
    m_rotation.normalize();
    
    // Update camera vectors
    updateCameraVectors();
}

QMatrix4x4 Camera::getViewMatrix() const {
    QMatrix4x4 viewMatrix;
    viewMatrix.lookAt(m_position, m_center, m_up);
    return viewMatrix;
}

void Camera::updateCameraVectors() {
  QVector3D basePosition(0, 0, m_radius);
  QVector3D rotatedPosition = m_rotation.rotatedVector(basePosition);

  m_position = rotatedPosition + m_center;
  m_front = (m_center - m_position).normalized();

  // Recalculate up vector dynamically from rotation to maintain intuitive orientation
  m_up = m_rotation.rotatedVector(QVector3D(0, 1, 0)).normalized();
}

void Camera::rotateViewAligned(float angle) {
  // Create a rotation quaternion around the front vector (view direction)
  QQuaternion viewRotation = QQuaternion::fromAxisAndAngle(m_front, angle);
  
  // Apply this rotation to the camera's orientation
  m_rotation = viewRotation * m_rotation;
  m_rotation.normalize(); // Keep quaternion normalized
  
  // Update camera vectors
  updateCameraVectors();
}
```

```
#include "ViewPort3D.h"

void ViewPort3D::mouseMoveEvent(QMouseEvent *event) {
    if (!m_isMousePressed && !m_isRightMousePressed) {
        QOpenGLWidget::mouseMoveEvent(event);
        return;
    }

    startInteractionTimer();

    // Compute the mouse movement deltas
    float dx = static_cast<float>(event->position().x() - m_lastMousePos.x());
    float dy = static_cast<float>(event->position().y() - m_lastMousePos.y());

    if (m_isMousePressed) {
        // Adjust sensitivity based on your application's feel
        const float sensitivity = 0.2f * (m_camera.getRadius() / 5.0f);

        // Rotate camera
        m_camera.rotate(dx * sensitivity, dy * sensitivity);
    } else if (m_isRightMousePressed) {
        // Adjust sensitivity for view-aligned rotation
        const float viewRotationSensitivity = 0.5f;

        // Use the new method to rotate around view direction
        m_camera.rotateViewAligned(dx * viewRotationSensitivity);
    }

    m_lastMousePos = event->pos();
    update();

    QOpenGLWidget::mouseMoveEvent(event);
}

void ViewPort3D::mousePressEvent(QMouseEvent *event) {
    if (event->button() == Qt::LeftButton) {
        m_isMousePressed = true;
        m_lastMousePos = event->pos();
    } else if (event->button() == Qt::RightButton) {
        m_isRightMousePressed = true;
        m_lastMousePos = event->pos();
    }

    QOpenGLWidget::mousePressEvent(event);
}

void ViewPort3D::mouseReleaseEvent(QMouseEvent *event) {
    if (event->button() == Qt::LeftButton) {
        m_isMousePressed = false;
    } else if (event->button() == Qt::RightButton) {
        m_isRightMousePressed = false;
    }

    QOpenGLWidget::mouseReleaseEvent(event);
}

void ViewPort3D::wheelEvent(QWheelEvent *event) {
    startInteractionTimer();

    float numDegrees = event->angleDelta().y() / 8.0f;
    float numSteps = numDegrees / 15.0f;

    // Zoom camera
    m_camera.zoom(numSteps);

    update();

    QOpenGLWidget::wheelEvent(event);
}

void ViewPort3D::startInteractionTimer(int time) {
    m_interactionTimer->start(time);
}

void ViewPort3D::interactionTimeout() {
    update();
```

```
void ViewPort3D::initializeGL()
{
    initializeOpenGLFunctions();

    // Create main shader program for volume rendering
    m_program = new QOpenGLShaderProgram(this);
    m_program->addShaderFromSourceFile(QOpenGLShader::Vertex, "shaders/raycast.vert");
    m_program->addShaderFromSourceFile(QOpenGLShader::Fragment, "shaders/raycast.frag");

    if (!m_program->link()) {
        qDebug() << "Main shader program failed to link!";
        qDebug() << m_program->log();  // Output log information for debugging
    } else {
        qDebug() << "Main shader program linked successfully!";
    }

    // Create shader program for axes
    m_axisProgram = new QOpenGLShaderProgram(this);
    m_axisProgram->addShaderFromSourceFile(QOpenGLShader::Vertex, "shaders/axis.vert");
    m_axisProgram->addShaderFromSourceFile(QOpenGLShader::Fragment, "shaders/axis.frag");

    if (!m_axisProgram->link()) {
        qDebug() << "Axis shader program failed to link!";
        qDebug() << m_axisProgram->log();  // Output log information for debugging
    } else {
        qDebug() << "Axis shader program linked successfully!";
    }

    // Create x-, y- and z-axis 
    createAxis();

    // Create bounding box for volume rendering
    createBoundingBox();

    // Create texture if we have NifTI data
    if (m_hasNiftiData) {
        createTexture();
    }

    // Enable alpha blending
    glEnable(GL_BLEND);
    glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);

    // Set OpenGL state
    glClearColor(0.4f, 0.4f, 0.4f, 1.0f);
    glEnable(GL_DEPTH_TEST);
    glEnable(GL_CULL_FACE);
}

void ViewPort3D::resizeGL(int w, int h)
{
    if (w <= 0 || h <= 0)
        return;

    m_projectionMatrix.setToIdentity();
    m_projectionMatrix.perspective(45.0f, static_cast<float>(w) / static_cast<float>(h), 0.1f, 100.0f);

    emit resized(w, h);
}

void ViewPort3D::paintGL()
{
    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

    // Set polygon draw mode
    glPolygonMode(GL_FRONT_AND_BACK, m_drawMode);

    // Get view matrix from camera
    m_viewMatrix = m_camera.getViewMatrix();

    // Draw coordinate axes if enabled
    if (m_showAxes)
        drawAxes();

    if (m_showBoundingBoxFrame)
        drawBoundingBoxFrame();

    if (m_showVolume)
        drawVolume();

    // Reset polygon mode
    glPolygonMode(GL_FRONT_AND_BACK, GL_FILL);
}


void ViewPort3D::adjustQuality() {
    // If user isn't interacting anymore when can reset the quality to default
    if (m_qualityReduced && !m_interactionTimer->isActive()) {
        m_adjustedStepSize = m_defaultStepSize;
        m_qualityReduced = false;
    }
    // If the quality already is reduced, but not sufficiently, reduce it even more
    else if (m_qualityReduced && m_timeElapsed > 1.2 * m_maxFrameRenderDuration) {
        m_adjustedStepSize = m_adjustedStepSize * m_timeElapsed / m_maxFrameRenderDuration;
    }
    // If the user begins to interact, we reduce the quality.
    else if (!m_qualityReduced && m_interactionTimer->isActive() && m_timeElapsed > m_maxFrameRenderDuration) {
        m_adjustedStepSize = m_adjustedStepSize * m_timeElapsed / m_maxFrameRenderDuration;
        m_qualityReduced = true;

    } else  if (!m_qualityReduced) {
        m_adjustedStepSize = m_defaultStepSize;
    }
}


void ViewPort3D::setVolumeData(const NiftiDataManager &dataManager)
{
    qDebug() << "ViewPort3D::setVolumeData - Starting";

    try {
        // Clean up OpenGL resources first
        if (isValid()) {
            qDebug() << "Making OpenGL context current";
            makeCurrent();

            cleanupTextures();

            doneCurrent();
            qDebug() << "OpenGL context released";
        }

        // Mark as having no data
        m_hasNiftiData = false;

        // If the data manager has no valid data, we're done
        if (!dataManager.isValid()) {
            qDebug() << "Input data manager contains no valid data";
            update();
            return;
        }

        // Store the data manager
        m_volumeData = dataManager;
        m_hasNiftiData = m_volumeData.isValid();

        qDebug() << "Volume data assigned with dimensions:"
            << m_volumeData.getWidth() << "x"
            << m_volumeData.getHeight() << "x"
            << m_volumeData.getDepth();

        // Reset view and slices
        qDebug() << "Resetting view parameters";
        m_sliceValues = SliceValues{};
        updateModelMatrix();

        // Now create OpenGL resources if the context is valid
        if (m_hasNiftiData && isValid()) {
            qDebug() << "Making OpenGL context current for texture creation";
            makeCurrent();

            try {
                qDebug() << "Creating 3D texture from volume data";
                createTexture();
                qDebug() << "Creating color map";
                createColorMapTexture();
                qDebug() << "Texture creation complete";
            }
            catch (const std::exception &e) {
                qWarning() << "Exception in texture creation:" << e.what();

                // Clean up if texture creation fails
                cleanupTextures();

                m_hasNiftiData = false;
            }

            doneCurrent();
            qDebug() << "OpenGL context released after texture creation";
        }

        // Reset camera view
        resetView();
        update();  // Request a redraw
        qDebug() << "ViewPort3D::setVolumeData - Completed successfully";
    }
    catch (const std::exception &e) {
        qWarning() << "Exception in setVolumeData:" << e.what();

        // Clean up in case of any exception
        if (isValid()) {
            qDebug() << "Making OpenGL context current for error cleanup";
            makeCurrent();

            cleanupTextures();

            doneCurrent();
            qDebug() << "OpenGL context released during error cleanup";
        }

        m_hasNiftiData = false;
        update(); // Request a redraw
        qDebug() << "ViewPort3D::setVolumeData - Failed with exception";
    }
}
```

```
void ViewPort3D::createBoundingBox()
{
    // Create a cube bounding box with vertices from -1 to 1
    VertexData axisVertices[] = {
        // Front face
        {{-1.0f, -1.0f,  1.0f}, {1.0f, 1.0f, 1.0f}},
        {{ 1.0f, -1.0f,  1.0f}, {1.0f, 1.0f, 1.0f}},
        {{ 1.0f,  1.0f,  1.0f}, {1.0f, 1.0f, 1.0f}},
        {{-1.0f,  1.0f,  1.0f}, {1.0f, 1.0f, 1.0f}},

        // Back face
        {{-1.0f, -1.0f, -1.0f}, {1.0f, 1.0f, 1.0f}},
        {{ 1.0f, -1.0f, -1.0f}, {1.0f, 1.0f, 1.0f}},
        {{ 1.0f,  1.0f, -1.0f}, {1.0f, 1.0f, 1.0f}},
        {{-1.0f,  1.0f, -1.0f}, {1.0f, 1.0f, 1.0f}}
    };

    unsigned int indices[] = {
        // Front face
        0, 1, 2,
        2, 3, 0,

        // Right face
        1, 5, 6,
        6, 2, 1,

        // Back face
        5, 4, 7,
        7, 6, 5,

        // Left face
        4, 0, 3,
        3, 7, 4,

        // Top face
        3, 2, 6,
        6, 7, 3,

        // Bottom face
        4, 5, 1,
        1, 0, 4
    };

    m_vao.create();
    m_vao.bind();

    m_vbo.create();
    m_vbo.bind();
    m_vbo.allocate(axisVertices, sizeof(axisVertices));

    m_ebo.create();
    m_ebo.bind();
    m_ebo.allocate(indices, sizeof(indices));

    // Position attribute
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, sizeof(VertexData), nullptr);

    // Color attribute
    glEnableVertexAttribArray(1);
    glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE, sizeof(VertexData),
        reinterpret_cast<void *>(sizeof(QVector3D)));

    m_vao.release();
}

void ViewPort3D::drawVolume() {
    if (!m_hasNiftiData || !m_program || !m_program->isLinked())
        return;

    adjustQuality();

    // Draw volume with raycasting
    m_program->bind();

    QMatrix4x4 mvpMatrix = m_projectionMatrix * m_viewMatrix * m_modelMatrix;

    // Calculate camera-relative light position
    QVector3D cameraPos = m_camera.getPosition();
    QVector3D cameraFront = m_camera.getFrontVector();
    QVector3D cameraRight = QVector3D::crossProduct(cameraFront, m_camera.getUpVector()).normalized();
    QVector3D cameraUp = m_camera.getUpVector();

    // Compute light position relative to camera
    QVector3D lightPos = cameraPos +
        (cameraRight * m_lightOffsetX) +
        (cameraUp * m_lightOffsetY) +
        (cameraFront * m_lightOffsetZ);

    m_program->setUniformValue("lightPosition", lightPos);

    m_program->setUniformValue("volumeDimensions", QVector3D(m_volumeData.getWidth(), m_volumeData.getHeight(), m_volumeData.getDepth()));
    m_program->setUniformValue("exactRayCasting", m_exactRayCasting);

    // Set uniforms for raycasting shader
    m_program->setUniformValue("mvp", mvpMatrix);
    m_program->setUniformValue("model", m_modelMatrix);
    m_program->setUniformValue("cameraWorldPosition", m_camera.getPosition());

    m_program->setUniformValue("aspectRatio",
        static_cast<float>(width()) / static_cast<float>(height()));

    // Pass step size to shader
    m_program->setUniformValue("stepSize", m_adjustedStepSize);

    // Pass render type and thresholds to shader
    m_program->setUniformValue("renderType", static_cast<int>(m_renderType));
    m_program->setUniformValue("minDensityThreshold", m_minDensityThreshold);
    m_program->setUniformValue("maxDensityThreshold", m_maxDensityThreshold);
    m_program->setUniformValue("minSlices", QVector3D(m_sliceValues.minSliceX * 2 - 1, m_sliceValues.minSliceY * 2 - 1, m_sliceValues.minSliceZ * 2 - 1));
    m_program->setUniformValue("maxSlices", QVector3D(m_sliceValues.maxSliceX * 2 - 1, m_sliceValues.maxSliceY * 2 - 1, m_sliceValues.maxSliceZ * 2 - 1));
    m_program->setUniformValue("alphaFactor", m_alphaFactor);

    m_program->setUniformValue("enableLighting", m_enableLighting);
    m_program->setUniformValue("enableAO", m_enableAO);
    m_program->setUniformValue("enableShadow", m_enableShadow);
    m_program->setUniformValue("ambientStrength", m_ambientStrength);
    m_program->setUniformValue("diffuseStrength", m_diffuseStrength);
    m_program->setUniformValue("specularStrength", m_specularStrength);
    m_program->setUniformValue("shininess", m_shininess);

    // Bind volume texture
    glActiveTexture(GL_TEXTURE0);
    glBindTexture(GL_TEXTURE_3D, m_volumeTexture);
    m_program->setUniformValue("volumeTexture", 0);

    // Bind color texture
    glActiveTexture(GL_TEXTURE1);
    glBindTexture(GL_TEXTURE_1D, m_colorTexture);
    m_program->setUniformValue("colorMap", 1);

    m_vao.bind();

    GLuint query;
    glGenQueries(1, &query);

    // Start query
    glBeginQuery(GL_TIME_ELAPSED, query);
    glDrawElements(GL_TRIANGLES, 36, GL_UNSIGNED_INT, 0);
    glEndQuery(GL_TIME_ELAPSED);

    // Retrieve time
    glGetQueryObjectuiv(query, GL_QUERY_RESULT, &m_timeElapsed);

    m_vao.release();

    // Unbind texture
    glBindTexture(GL_TEXTURE_3D, 0);
    m_program->release();
}

void ViewPort3D::drawBoundingBoxFrame() {
    if (!m_axisProgram || !m_axisProgram->isLinked())
        return;

    m_axisProgram->bind();

    QMatrix4x4 mvpMatrix = m_projectionMatrix * m_viewMatrix * m_modelMatrix;
    m_axisProgram->setUniformValue("mvp", mvpMatrix);

    m_vao.bind();
    glLineWidth(1.0f);
    glDrawElements(GL_LINE_STRIP, 24, GL_UNSIGNED_INT, 0);
    m_vao.release();

    m_axisProgram->release();
}

void ViewPort3D::drawAxes() {
    if (!m_axisProgram || !m_axisProgram->isLinked())
        return;

    m_axisProgram->bind();

    QMatrix4x4 mvpMatrix = m_projectionMatrix * m_viewMatrix;
    m_axisProgram->setUniformValue("mvp", mvpMatrix);

    // Draw axes
    m_axisVao.bind();
    glLineWidth(3.0f); // Make axes thicker for better visibility
    glDrawArrays(GL_LINES, 0, 6);
    m_axisVao.release();

    m_axisProgram->release();
}
```

```
void ViewPort3D::setSliceValue(float &sliceValue, float newValue) {
    sliceValue = newValue;
    updateModelMatrix();
    updateWithInteraction();
}


void ViewPort3D::updateModelMatrix() {
    m_modelMatrix.setToIdentity();

    // Apply object rotation first
    m_modelMatrix.rotate(m_objectRotation);

    // Then handle slicing
    float transX = ((m_sliceValues.maxSliceX + m_sliceValues.minSliceX) - 1) / (m_sliceValues.maxSliceX - m_sliceValues.minSliceX);
    float transY = ((m_sliceValues.maxSliceY + m_sliceValues.minSliceY) - 1) / (m_sliceValues.maxSliceY - m_sliceValues.minSliceY);
    float transZ = ((m_sliceValues.maxSliceZ + m_sliceValues.minSliceZ) - 1) / (m_sliceValues.maxSliceZ - m_sliceValues.minSliceZ);

    m_modelMatrix.scale(m_sliceValues.maxSliceX - m_sliceValues.minSliceX,
        m_sliceValues.maxSliceY - m_sliceValues.minSliceY,
        m_sliceValues.maxSliceZ - m_sliceValues.minSliceZ);
    m_modelMatrix.translate(transX, transY, transZ);
}
```

```
#include "ViewPort3D.h"


void ViewPort3D::createTexture()
{
    if (!m_hasNiftiData || !m_volumeData.isValid()) {
        qWarning() << "Cannot create texture: No valid volume data available";
        return;
    }

    try {
        if (m_volumeTexture != 0) {
            glDeleteTextures(1, &m_volumeTexture);
            m_volumeTexture = 0;
        }

        int width = m_volumeData.getWidth();
        int height = m_volumeData.getHeight();
        int depth = m_volumeData.getDepth();

        qDebug() << "Using volume dimensions:" << width << "x" << height << "x" << depth;

        if (width <= 0 || height <= 0 || depth <= 0) {
            qWarning() << "Invalid volume dimensions:" << width << "x" << height << "x" << depth;
            return;
        }

        // Get normalized data from manager
        std::vector<float> normalizedData = m_volumeData.getNormalizedVolumeData();
        qDebug() << "Creating 3D texture with data size:" << normalizedData.size();

        glGenTextures(1, &m_volumeTexture);
        glBindTexture(GL_TEXTURE_3D, m_volumeTexture);

        glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
        glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_BORDER);
        glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_BORDER);
        glTexParameteri(GL_TEXTURE_3D, GL_TEXTURE_WRAP_R, GL_CLAMP_TO_BORDER);

        glTexImage3D(GL_TEXTURE_3D, 0, GL_R32F, width, height, depth, 0, GL_RED, GL_FLOAT, normalizedData.data());
        glBindTexture(GL_TEXTURE_3D, 0);

    }
    catch (const std::exception &e) {
        qWarning() << "Exception in createTexture:" << e.what();
        if (m_volumeTexture != 0) {
            glDeleteTextures(1, &m_volumeTexture);
            m_volumeTexture = 0;
        }
    }
}


void ViewPort3D::cleanupTextures() {
    if (m_volumeTexture != 0) {
        qDebug() << "Deleting volume texture";
        glDeleteTextures(1, &m_volumeTexture);
        m_volumeTexture = 0;
    }

    if (m_colorTexture != 0) {
        qDebug() << "Deleting color texture";
        glDeleteTextures(1, &m_colorTexture);
        m_colorTexture = 0;
    }
}


void ViewPort3D::setColorMap(QVector<QVector4D> colorMap) {
    m_colorMap = colorMap;
    if (isValid()) {
        createColorMapTexture();
        updateWithInteraction();
    }
}

void ViewPort3D::createColorMapTexture() {
    if (m_colorMap.isEmpty()) {
        qDebug() << "Using fall back color map";
        setFallbackColorMap();
    }

    if (m_colorTexture != 0) {
        glDeleteTextures(1, &m_colorTexture);
        m_colorTexture = 0;
    }

    glGenTextures(1, &m_colorTexture);
    glBindTexture(GL_TEXTURE_1D, m_colorTexture);

    glTexParameteri(GL_TEXTURE_1D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_1D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
    glTexParameteri(GL_TEXTURE_1D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);

    glTexImage1D(GL_TEXTURE_1D, 0, GL_RGBA, m_colorMap.length(), 0, GL_RGBA, GL_FLOAT, m_colorMap.data());
    glBindTexture(GL_TEXTURE_1D, 0);

    update();
}

void ViewPort3D::setFallbackColorMap() {
    const int SIZE = 256;
    m_colorMap = QVector<QVector4D>(SIZE);
    for (int i = 0; i < SIZE; i++) {
        float f = i / static_cast<float>(SIZE - 1);
        m_colorMap[i] = QVector4D(f, f, f, 1.0f);
    }
}
```
